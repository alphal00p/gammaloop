1use serde::{Deserialize, Deserializer, Serialize, Serializer, de::Error};
2use symbolica::domains::{integer::Integer, rational::Rational};
3
4/// UFO 1.0 exports integral charges as numbers and fractional charges as strings.
5/// Keep them exact through model import/export; older numeric JSON remains valid.
6#[derive(Debug, Clone)]
7pub(super) struct SerializedCharge(pub Rational);
8
9impl Serialize for SerializedCharge {
10    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
11        if self.0.denominator_ref().is_one()
12            && let Some(value) = self.0.numerator_ref().to_i64()
13        {
14            serializer.serialize_i64(value)
15        } else {
16            serializer.serialize_str(&self.0.to_string())
17        }
18    }
19}
20
21impl<'de> Deserialize<'de> for SerializedCharge {
22    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
23        #[derive(Deserialize)]
24        #[serde(untagged)]
25        enum Input {
26            Rational(String),
27            Number(serde_json::Number),
28        }
29
30        let value = match Input::deserialize(deserializer)? {
31            Input::Rational(text) => {
32                let (num, den) = text.split_once('/').unwrap_or((&text, "1"));
33                let num = num.parse::<Integer>().map_err(D::Error::custom)?;
34                let den = den.parse::<Integer>().map_err(D::Error::custom)?;
35                if den.is_zero() {
36                    return Err(D::Error::custom("particle charge has a zero denominator"));
37                }
38                Rational::new(num, den)
39            }
40            Input::Number(number) => {
41                if let Some(integer) = number.as_i64() {
42                    integer.into()
43                } else if let Some(integer) = number.as_u64() {
44                    integer.into()
45                } else {
46                    // Older exports wrote fractions such as 2/3 as their nearest
47                    // binary64 value. Recover the simplest rational within that
48                    // rounding instead of keeping its binary expansion.
49                    Rational::try_from(number.as_f64().ok_or_else(|| {
50                        D::Error::custom("particle charge must be a finite number")
51                    })?)
52                    .map_err(D::Error::custom)?
53                    .round(&Rational::from((1, 1_i64 << (f64::MANTISSA_DIGITS - 1))))
54                }
55            }
56        };
57        Ok(Self(value))
58    }
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64
65    #[test]
66    fn exact_and_numeric_particle_charges_roundtrip() {
67        for (input, expected) in [
68            ("0", Rational::zero()),
69            ("-1", Rational::from(-1)),
70            ("0.5", Rational::from((1, 2))),
71            ("0.6666666666666666", Rational::from((2, 3))),
72            ("-0.3333333333333333", Rational::from((-1, 3))),
73            ("0.1", Rational::from((1, 10))),
74            (r#""2/3""#, Rational::from((2, 3))),
75            (r#""-4/3""#, Rational::from((-4, 3))),
76        ] {
77            let charge: SerializedCharge = serde_json::from_str(input).unwrap();
78            assert_eq!(charge.0, expected);
79            let encoded = serde_json::to_string(&charge).unwrap();
80            let decoded: SerializedCharge = serde_json::from_str(&encoded).unwrap();
81            assert_eq!(decoded.0, expected);
82        }
83        assert_eq!(
84            serde_json::to_string(&SerializedCharge(Rational::from((2, 3)))).unwrap(),
85            r#""2/3""#
86        );
87        for input in [r#""1/0""#, r#""Q""#, "null", "true"] {
88            assert!(
89                serde_json::from_str::<SerializedCharge>(input).is_err(),
90                "{input}"
91            );
92        }
93    }
