9enum BestAction {
10    Beam(usize),
11    Pair(usize, usize),
12}

14pub(crate) fn cluster_candidates<T: FloatLike>(
15    algorithm: JetAlgorithm,
16    d_r: f64,
17    min_jpt: f64,
18    mut active: SmallVec<[PseudoJet<T>; 8]>,
19) -> ClusteringResult<T> {
20    let mut jets = SmallVec::<[Jet<T>; 8]>::new();
21    if active.is_empty() {
22        return ClusteringResult { jets };
23    }
24
25    // Runtime settings currently store clustering parameters as `f64`, so this is
26    // the intentional configuration precision boundary for these two values.
27    let min_pt = F::from_f64(min_jpt);
28
29    if d_r <= 0.0 || !d_r.is_finite() {
30        jets.extend(
31            active
32                .drain(..)
33                .filter(|jet| jet.pt() >= min_pt)
34                .map(PseudoJet::jet),
35        );
36        sort_jets(&mut jets);
37        return ClusteringResult { jets };
38    }
39
40    let r = F::from_f64(d_r);
41    let inv_r2 = (r.clone() * r).inv();
42
43    while !active.is_empty() {
44        let mut best_action = BestAction::Beam(0);
45        let mut best_distance: Option<F<T>> = None;
46
47        for i in 0..active.len() {
48            update_best(
49                &mut best_action,
50                &mut best_distance,
51                BestAction::Beam(i),
52                beam_distance(&active[i], algorithm),
53            );
54
55            for j in (i + 1)..active.len() {
56                update_best(
57                    &mut best_action,
58                    &mut best_distance,
59                    BestAction::Pair(i, j),
60                    pair_distance(&active[i], &active[j], &inv_r2, algorithm),
61                );
62            }
63        }
64
65        match best_action {
66            BestAction::Beam(index) => {
67                let jet = active.remove(index);
68                if jet.pt() >= min_pt {
69                    jets.push(jet.jet());
70                }
71            }
72            BestAction::Pair(i, j) => {
73                let merged = PseudoJet::merged(&active[i], &active[j]);
74                active[i] = merged;
75                active.remove(j);
76            }
77        }
78    }
79
80    sort_jets(&mut jets);
81    ClusteringResult { jets }
82}

84fn update_best<T: FloatLike>(
85    best_action: &mut BestAction,
86    best_distance: &mut Option<F<T>>,
87    action: BestAction,
88    candidate_distance: F<T>,
89) {
90    let should_update = best_distance
91        .as_ref()
92        .is_none_or(|current| candidate_distance < current.clone());
93    if should_update {
94        *best_action = action;
95        *best_distance = Some(candidate_distance);
96    }
97}

99fn beam_distance<T: FloatLike>(jet: &PseudoJet<T>, algorithm: JetAlgorithm) -> F<T> {
100    scale_for_algorithm(jet.pt2(), algorithm)
101}
102
103fn pair_distance<T: FloatLike>(
104    left: &PseudoJet<T>,
105    right: &PseudoJet<T>,
106    inv_r2: &F<T>,
107    algorithm: JetAlgorithm,
108) -> F<T> {
109    let left_scale = scale_for_algorithm(left.pt2(), algorithm);
110    let right_scale = scale_for_algorithm(right.pt2(), algorithm);
111    let distance_scale = if left_scale <= right_scale {
112        left_scale
113    } else {
114        right_scale
115    };
116    distance_scale * delta_r2(left, right) * inv_r2.clone()
117}
118
119fn scale_for_algorithm<T: FloatLike>(pt2: F<T>, algorithm: JetAlgorithm) -> F<T> {
120    match algorithm {
121        JetAlgorithm::Kt => pt2,
122        JetAlgorithm::CambridgeAachen => pt2.one(),
123        JetAlgorithm::AntiKt => {
124            let tiny = pt2.epsilon();
125            if pt2 < tiny {
126                pt2.max_value()
127            } else {
128                pt2.inv()
129            }
130        }
131    }
132}
133
134fn delta_r2<T: FloatLike>(left: &PseudoJet<T>, right: &PseudoJet<T>) -> F<T> {
135    let delta_y = left.rapidity() - right.rapidity();
136    let delta_phi = wrapped_delta_phi(left.phi(), right.phi());
137    delta_y.square() + delta_phi.square()
138}
139
140fn wrapped_delta_phi<T: FloatLike>(lhs: F<T>, rhs: F<T>) -> F<T> {
141    let pi = lhs.PI();
142    let two_pi = lhs.TAU();
143    let delta = (lhs - rhs).norm();
144    if delta > pi { two_pi - delta } else { delta }
145}

147fn sort_jets<T: FloatLike>(jets: &mut SmallVec<[Jet<T>; 8]>) {
148    jets.sort_unstable_by(compare_jets_descending_pt);
149}
150
151fn compare_jets_descending_pt<T: FloatLike>(lhs: &Jet<T>, rhs: &Jet<T>) -> Ordering {
152    rhs.pt()
153        .partial_cmp(&lhs.pt())
154        .unwrap_or(Ordering::Equal)
155        .then_with(|| {
156            lhs.rapidity()
157                .partial_cmp(&rhs.rapidity())
158                .unwrap_or(Ordering::Equal)
159        })
160        .then_with(|| lhs.phi().partial_cmp(&rhs.phi()).unwrap_or(Ordering::Equal))
161        .then_with(|| lhs.constituents.cmp(&rhs.constituents))
162}
