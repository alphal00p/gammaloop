109pub(crate) struct PseudoJet<T: FloatLike> {
110    pub(crate) momentum: FourMomentum<F<T>>,
111    pub(crate) constituents: SmallVec<[usize; 8]>,
112    pt2: F<T>,
113    rapidity: F<T>,
114    phi: F<T>,
115}
116
117impl<T: FloatLike> PseudoJet<T> {
118    pub(crate) fn from_momentum(index: usize, momentum: FourMomentum<F<T>>) -> Self {
119        Self::new(momentum, smallvec![index])
120    }
121
122    pub(crate) fn new(momentum: FourMomentum<F<T>>, constituents: SmallVec<[usize; 8]>) -> Self {
123        let zero = momentum.temporal.value.zero();
124        let mut jet = Self {
125            momentum,
126            constituents,
127            pt2: zero.clone(),
128            rapidity: zero.clone(),
129            phi: zero,
130        };
131        jet.recompute_kinematics();
132        jet
133    }
134
135    pub(crate) fn merged(lhs: &Self, rhs: &Self) -> Self {
136        let mut constituents = lhs.constituents.clone();
137        constituents.extend_from_slice(&rhs.constituents);
138        constituents.sort_unstable();
139        Self::new(lhs.momentum.clone() + rhs.momentum.clone(), constituents)
140    }
141
142    pub(crate) fn jet(self) -> Jet<T> {
143        Jet::from_parts(
144            self.momentum,
145            self.constituents,
146            self.pt2,
147            self.rapidity,
148            self.phi,
149        )
150    }
151
152    pub(crate) fn pt(&self) -> F<T> {
153        self.pt2.clone().sqrt()
154    }
155
156    pub(crate) fn pt2(&self) -> F<T> {
157        self.pt2.clone()
158    }
159
160    pub(crate) fn rapidity(&self) -> F<T> {
161        self.rapidity.clone()
162    }
163
164    pub(crate) fn phi(&self) -> F<T> {
165        self.phi.clone()
166    }
167
168    fn recompute_kinematics(&mut self) {
169        self.pt2 = transverse_momentum_squared(&self.momentum);
170        self.phi = phi(&self.momentum);
171        self.rapidity = rapidity(&self.momentum, &self.pt2);
172    }
173}
174
175fn transverse_momentum_squared<T: FloatLike>(momentum: &FourMomentum<F<T>>) -> F<T> {
176    momentum.spatial.px.square() + momentum.spatial.py.square()
177}
178
179fn phi<T: FloatLike>(momentum: &FourMomentum<F<T>>) -> F<T> {
180    let zero = momentum.spatial.px.zero();
181    let two_pi = momentum.spatial.px.TAU();
182    let mut phi = momentum.spatial.py.atan2(&momentum.spatial.px);
183    if phi < zero {
184        phi += two_pi.clone();
185    }
186    if phi >= two_pi {
187        phi -= two_pi;
188    }
189    phi
190}
191
192fn rapidity<T: FloatLike>(momentum: &FourMomentum<F<T>>, pt2: &F<T>) -> F<T> {
193    momentum.rapidity_with_pt2(pt2)
