98        self == other

101                && self

113    /// Sampling may identify energies in an exact represented spatial frame.
114    /// This does not change structural routing equality used by raised edges.
115    pub(crate) fn spatial_equality_up_to_sign<T: FloatLike>(
116        &self,
117        other: &Self,
118        externals: &ExternalThreeMomenta<F<T>>,
119    ) -> eyre::Result<bool> {
120        use crate::integrands::process::sampling::maps::SamplingEvaluationError;
121        use rug::{Float, float::Round};
122        const PRECISION: u32 = 2048;
123        if self.internal.len() != other.internal.len()
124            || self.external.len() != other.external.len()
125            || self.external.len() != externals.len()
126        {
127            return Err(eyre::eyre!(
128                "spatial route comparison requires equal routing dimensions and the complete external frame"
129            ));
130        }
131        if externals
132            .iter()
133            .flat_map(|p| [&p.px, &p.py, &p.pz])
134            .any(|value| !value.0.is_finite())
135        {
136            return Err(SamplingEvaluationError::Unrepresentable {
137                operation: "spatial route comparison",
138                detail: "nonfinite original external component".into(),
139            }
140            .into());
141        }
142        if self.equality_up_to_sign(other) {
143            return Ok(true);
144        }
145        let mut uncertain = false;
146        for sign in [1, -1] {
147            if !self
148                .internal
149                .iter()
150                .zip(other.internal.iter())
151                .all(|(a, b)| *a == if sign == 1 { *b } else { -*b })
152            {
153                continue;
154            }
155            // Differences can be +/-2, which SignatureLike cannot represent.
156            let coefficients = self
157                .external
158                .to_momtrop_format()
159                .into_iter()
160                .zip(other.external.to_momtrop_format())
161                .map(|(a, b)| a - sign * b)
162                .collect::<Vec<_>>();
163            let mut sums: [[Float; 2]; 3] = std::array::from_fn(|_| {
164                [Float::with_val(PRECISION, 0), Float::with_val(PRECISION, 0)]
165            });
166            for (&coefficient, momentum) in coefficients.iter().zip(externals) {
167                if coefficient == 0 {
168                    continue;
169                }
170                for (sum, component) in
171                    sums.iter_mut()
172                        .zip([&momentum.px, &momentum.py, &momentum.pz])
173                {
174                    let (mut lo, mut hi) = component.0.mpfr_enclosure(PRECISION);
175                    if coefficient < 0 {
176                        std::mem::swap(&mut lo, &mut hi);
177                    }
178                    let lo = Float::with_val_round(PRECISION, &lo * coefficient, Round::Down).0;
179                    let hi = Float::with_val_round(PRECISION, &hi * coefficient, Round::Up).0;
180                    sum[0] = Float::with_val_round(PRECISION, &sum[0] + &lo, Round::Down).0;
181                    sum[1] = Float::with_val_round(PRECISION, &sum[1] + &hi, Round::Up).0;
182                }
183            }
184            if sums.iter().flatten().any(|value| !value.is_finite()) {
185                return Err(SamplingEvaluationError::Unrepresentable {
186                    operation: "spatial route comparison",
187                    detail: "nonfinite directed external sum".into(),
188                }
189                .into());
190            }
191            if sums.iter().any(|[lo, hi]| lo > &0 || hi < &0) {
192                continue;
193            }
194            if sums.iter().all(|[lo, hi]| lo == &0 && hi == &0) {
195                return Ok(true);
196            }
197            uncertain = true;
198        }
199        if uncertain {
200            return Err(SamplingEvaluationError::UncertainGeometry {
201                detail: format!("spatial routing equality is unresolved at {PRECISION}-bit directed precision: {self:?} versus {other:?}"),
202            }
203            .into());
204        }
205        Ok(false)
206    }
207}

443    pub(crate) fn try_apply<B>(&self, basis: &[B]) -> Option<B>
444    where
445        B: Clone + Neg<Output = B> + Add<B, Output = B>,
446    {
447        self.0

449            .zip(basis.iter().cloned())
450            .filter_map(|(sign, t)| match sign {
451                SignOrZero::Zero => None,

458    pub(crate) fn apply_typed<O, I, V>(&self, basis: &V) -> O
459    where
460        V: Index<I, Output = O>,
461        O: RefZero + Neg<Output = O> + AddAssign<O> + Clone,
462        I: From<usize>,
463        usize: From<I>,
464    {
465        let mut result = basis[I::from(0)].ref_zero();
466        for (&sign, i) in self.0.iter().zip(0..) {
467            result += sign * basis[I::from(i)].clone();
468        }
469
470        result

473    pub(crate) fn apply_iter<I, O>(&self, basis: I) -> Option<O>
474    where
475        I: IntoIterator,
476        I::Item: RefZero<O>,
477        O: Clone + SubAssign<I::Item> + AddAssign<I::Item>,
478    {
479        let mut basis_iter = basis.into_iter();
480        let mut signature_iter = self.into_iter();
481
482        while let (Some(sign), Some(item)) = (signature_iter.next(), basis_iter.next()) {
483            if sign.is_sign() {
484                // Initialize the result based on the first non-zero sign
485                let mut result = item.ref_zero();
486                match sign {
487                    SignOrZero::Zero => {
488                        panic!("unreachable");
489                        // return None;
490                    }
491                    SignOrZero::Plus => {
492                        result += item;
493                    }
494                    SignOrZero::Minus => {
495                        result -= item;
496                    }
497                }
498
499                // Continue processing the rest of the iterator
500                while let (Some(sign), Some(item)) = (signature_iter.next(), basis_iter.next()) {
501                    match sign {
502                        SignOrZero::Zero => {}
503                        SignOrZero::Plus => {
504                            result += item;
505                        }
506                        SignOrZero::Minus => {
507                            result -= item;
508                        }
509                    }
510                }
511
512                return Some(result);
513            }
514        }
515
516        // Return None if no non-zero sign was found
517        None
518    }

695        external_moms: &ExternalFourMomenta<F<T>>,
696    ) -> FourMomentum<F<T>> {
697        let loop_moms = loop_moms
698            .iter()
699            .map(|m| m.clone().into_on_shell_four_momentum(None))
700            .collect::<TiVec<LoopIndex, _>>();
701
702        self.compute_momentum(&loop_moms, external_moms)
703    }
704
705    pub(crate) fn compute_three_momentum_from_four<T: FloatLike>(
706        &self,
707        loop_moms: &LoopMomenta<F<T>>,
708        external_moms: &ExternalFourMomenta<F<T>>,
709    ) -> ThreeMomentum<F<T>> {
710        let external_moms: ExternalThreeMomenta<F<T>> =
711            external_moms.iter().map(|m| m.spatial.clone()).collect();
712        self.compute_momentum(loop_moms, &external_moms)
713    }

745    #[test]
746    fn spatial_routing_equality_certifies_native_external_frames() {
747        use crate::{
748            integrands::process::sampling::maps::SamplingEvaluationError,
749            momentum::{ThreeMomentum, sample::ExternalThreeMomenta},
750            utils::{ArbPrec, F, FloatLike, QuadFloat},
751        };
752        fn check<T: FloatLike>() {
753            let one = F::<T>::default().one();
754            let zero = one.zero();
755            let left = LoopExtSignature::from((vec![1], vec![0, 0]));
756            let right = LoopExtSignature::from((vec![1], vec![1, 1]));
757            let beam = ThreeMomentum::new(one.from_i64(2), zero.clone(), one.from_i64(3));
758            let cm = ExternalThreeMomenta::from_iter([beam.clone(), -beam]);
759            assert!(!left.equality_up_to_sign(&right));
760            assert!(left.spatial_equality_up_to_sign(&right, &cm).unwrap());
761            let reversed = LoopExtSignature::from((vec![-1], vec![-1, -1]));
762            assert!(left.spatial_equality_up_to_sign(&reversed, &cm).unwrap());
763
764            // A common boost and even a tiny nonzero represented displacement
765            // break the fixed-frame identity; neither has a norm tolerance.
766            for displacement in [one.clone(), one.epsilon().square()] {
767                let mut shifted = cm.clone();
768                for momentum in &mut shifted {
769                    // Preserve the tiny original input on a zero-valued axis.
770                    momentum.py += &displacement;
771                }
772                assert!(!left.spatial_equality_up_to_sign(&right, &shifted).unwrap());
773            }
774            let a = LoopExtSignature::from((vec![1], vec![1, 0]));
775            let wrong_sign = LoopExtSignature::from((vec![-1], vec![1, 0]));
776            assert!(!a.spatial_equality_up_to_sign(&wrong_sign, &cm).unwrap());
777            // With matching internal signs, the external difference is +2.
778            let doubled = LoopExtSignature::from((vec![1], vec![-1, 0]));
779            assert!(!a.spatial_equality_up_to_sign(&doubled, &cm).unwrap());
780
781            let big = &one / one.epsilon().square();
782            let medium = big.sqrt();
783            let values = [big.clone(), medium.clone(), one.clone(), -medium, -big];
784            assert_eq!(
785                values.iter().fold(zero.clone(), |sum, value| sum + value),
786                zero
787            );
788            let cancelled = ExternalThreeMomenta::from_iter(
789                values.map(|x| ThreeMomentum::new(x, zero.clone(), zero.clone())),
790            );
791            let empty = LoopExtSignature::from((vec![1], vec![0; 5]));
792            let sum = LoopExtSignature::from((vec![1], vec![1; 5]));
793            assert!(!empty.spatial_equality_up_to_sign(&sum, &cancelled).unwrap());
794
795            // These original binary64 values span more than the fixed 2048-bit
796            // certificate budget, also when represented exactly in Quad/Arb.
797            let huge = F(T::from_f64_exact_binary(2.0_f64.powi(1023)));
798            let tiny = F(T::from_f64_exact_binary(f64::from_bits(1)));
799            let unresolved = ExternalThreeMomenta::from_iter(
800                [huge.clone(), tiny, -huge]
801                    .map(|x| ThreeMomentum::new(x, zero.clone(), zero.clone())),
802            );
803            let empty = LoopExtSignature::from((vec![1], vec![0; 3]));
804            let sum = LoopExtSignature::from((vec![1], vec![1; 3]));
805            let error = empty
806                .spatial_equality_up_to_sign(&sum, &unresolved)
807                .unwrap_err();
808            assert!(matches!(
809                error.downcast_ref::<SamplingEvaluationError>(),
810                Some(SamplingEvaluationError::UncertainGeometry { .. })
811            ));
812
813            let mut nonfinite = cm.clone();
814            nonfinite.iter_mut().next().unwrap().px = F(T::from_f64_exact_binary(f64::INFINITY));
815            // Input validation precedes even the structural fast path.
816            let error = left
817                .spatial_equality_up_to_sign(&left, &nonfinite)
818                .unwrap_err();
819            assert!(matches!(
820                error.downcast_ref::<SamplingEvaluationError>(),
821                Some(SamplingEvaluationError::Unrepresentable { .. })
822            ));
823            assert!(
824                left.spatial_equality_up_to_sign(&right, &ExternalThreeMomenta::<F<T>>::default())
825                    .is_err()
826            );
827        }
828        check::<f64>();
829        check::<QuadFloat>();
830        check::<ArbPrec>();
831    }

834    fn raised_edge_comparison_preserves_complete_affine_routing() {
835        use crate::{
836            dot,
837            graph::{Graph, parse::from_dot::IntoGraph},
838            initialisation::test_initialise,
839        };
840        use linnet::half_edge::involution::EdgeIndex;
841        test_initialise().unwrap();
842        let graph: Graph = dot!(digraph raised_affine_signature {
843            ext [style=invis]
844            edge [num=1 mass=0]
845            node [num=1]
846            ext -> a [id=0]
847            a -> b [id=1]
848            b -> c [id=2]
849            c -> a [id=3]
850            ext -> c [id=4]
851        })
852        .unwrap();
853        let groups = graph.get_raised_edge_groups();
854        assert!(groups.iter().any(|group| group.len() >= 2));
855        for group in groups {
856            for pair in group.windows(2) {
857                assert!(graph.loop_momentum_basis.edges_are_raised(pair[0], pair[1]));
858            }
859        }
860        // Deliberately supplied affine rows isolate the caller contract; they
861        // are not a new physical routing assignment for this graph.
862        let mut basis = graph.loop_momentum_basis.clone();
863        basis.edge_signatures[EdgeIndex(1)] = LoopExtSignature::from((vec![1], vec![1, 0]));
864        basis.edge_signatures[EdgeIndex(2)] = LoopExtSignature::from((vec![-1], vec![1, 0]));
865        assert!(!basis.edges_are_raised(EdgeIndex(1), EdgeIndex(2)));
866        basis.edge_signatures[EdgeIndex(2)] = LoopExtSignature::from((vec![-1], vec![-1, 0]));
867        assert!(basis.edges_are_raised(EdgeIndex(1), EdgeIndex(2)));
868    }
