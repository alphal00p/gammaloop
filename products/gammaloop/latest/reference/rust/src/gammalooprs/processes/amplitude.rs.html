126    pub fn validate_graph_group_selection_plan(
127        &self,
128        plan: &GraphGroupSelectionPlan,
129    ) -> Result<()> {
130        if !self.group_derived_data.is_empty()
131            && self.group_derived_data.len() != self.graph_group_structure.len()
132        {
133            return Err(eyre!(
134                "Amplitude '{}' has {} group-derived-data entries for {} graph groups.",
135                self.name,
136                self.group_derived_data.len(),
137                self.graph_group_structure.len()
138            ));
139        }
140
141        for &old_group_id in plan.retained_group_ids() {
142            plan.new_group_id_for_old(old_group_id).ok_or_else(|| {
143                eyre!(
144                    "Selection plan is missing compact group id for old group {}.",
145                    old_group_id.0
146                )
147            })?;
148            if old_group_id.0 >= self.graph_group_structure.len() {
149                return Err(eyre!(
150                    "Selection plan refers to missing graph group {}.",
151                    old_group_id.0
152                ));
153            }
154            let group = &self.graph_group_structure[old_group_id];
155            for old_graph_id in group {
156                if old_graph_id >= self.graphs.len() {
157                    return Err(eyre!(
158                        "Graph group {} refers to missing graph id {}.",
159                        old_group_id.0,
160                        old_graph_id
161                    ));
162                }
163            }
164            let master = group.master();
165            if master >= self.graphs.len() {
166                return Err(eyre!(
167                    "Graph group {} refers to missing master graph id {}.",
168                    old_group_id.0,
169                    master
170                ));
171            }
172        }
173
174        Ok(())
175    }

177    pub fn apply_graph_group_selection(&mut self, plan: &GraphGroupSelectionPlan) -> Result<()> {
178        self.validate_graph_group_selection_plan(plan)?;
179
180        let mut old_graph_to_new_group = vec![None; self.graphs.len()];
181        let mut old_graph_is_master = vec![false; self.graphs.len()];
182        for &old_group_id in plan.retained_group_ids() {
183            let new_group_id = plan.new_group_id_for_old(old_group_id).ok_or_else(|| {
184                eyre!(
185                    "Selection plan is missing compact group id for old group {}.",
186                    old_group_id.0
187                )
188            })?;
189            let group = &self.graph_group_structure[old_group_id];
190            for old_graph_id in group {
191                old_graph_to_new_group[old_graph_id] = Some(new_group_id);
192            }
193            old_graph_is_master[group.master()] = true;
194        }
195
196        let mut new_graphs = self
197            .graphs
198            .iter()
199            .cloned()
200            .enumerate()
201            .filter_map(|(old_graph_id, mut graph)| {
202                let new_group_id = old_graph_to_new_group[old_graph_id]?;
203                graph.graph.group_id = Some(new_group_id);
204                graph.graph.is_group_master = old_graph_is_master[old_graph_id];
205                if let Some(multi_channeling_setup) = &mut graph.derived_data.multi_channeling_setup
206                {
207                    multi_channeling_setup.graph.group_id = Some(new_group_id);
208                    multi_channeling_setup.graph.is_group_master =
209                        old_graph_is_master[old_graph_id];
210                }
211                Some(graph)
212            })
213            .collect::<Vec<_>>();
214
215        let mut parsed_graphs = new_graphs
216            .iter()
217            .map(|graph| graph.graph.clone())
218            .collect::<Vec<_>>();
219        let new_graph_group_structure = complete_group_parsing(&mut parsed_graphs)?;
220        for (graph, parsed_graph) in new_graphs.iter_mut().zip(parsed_graphs) {
221            graph.graph.group_id = parsed_graph.group_id;
222            graph.graph.is_group_master = parsed_graph.is_group_master;
223            if let Some(multi_channeling_setup) = &mut graph.derived_data.multi_channeling_setup {
224                multi_channeling_setup.graph.group_id = graph.graph.group_id;
225                multi_channeling_setup.graph.is_group_master = graph.graph.is_group_master;
226            }
227        }
228
229        let new_group_derived_data = if self.group_derived_data.is_empty() {
230            TiVec::new()
231        } else {
232            plan.retained_group_ids()
233                .iter()
234                .map(|&old_group_id| self.group_derived_data[old_group_id].clone())
235                .collect::<TiVec<GroupId, _>>()
236        };
237
238        self.graphs = new_graphs;
239        self.graph_group_structure = new_graph_group_structure;
240        self.group_derived_data = new_group_derived_data;
241        self.integrand = None;
242        Ok(())
243    }

383    pub fn preprocess(
384        &mut self,
385        model: &Model,
386        settings: &GenerationSettings,
387        locked_runtime_settings: &LockedRuntimeSettings,
388        thread_pool: &ThreadPool,
389    ) -> Result<Vec<NamedGraphGenerationReport>> {
390        // preprocess each graph individually
391        // Threshold directives of every member use the master's edge IDs and topology.
392        // The supplied graph group is responsible for aligning their physical meaning.
393        for group in &self.graph_group_structure {
394            if group.into_iter().all(|id| {
395                self.graphs[id]
396                    .graph
397                    .threshold_counterterms
398                    .is_legacy_equivalent()
399            }) {
400                // Metadata-free groups retain their native member geometry and shared full-space solve.
401                for graph_id in group {
402                    self.graphs[graph_id].derived_data.threshold_topology = None;
403                }
404                continue;
405            }
406            let master = &self.graphs[group.master()].graph;
407            let topology = (
408                master.clone(),
409                master.generate_loop_momentum_bases_of(&master.no_dummy()),
410            );
411            for graph_id in group {
412                self.graphs[graph_id]
413                    .graph
414                    .threshold_counterterms
415                    .validate_for_graph(&topology.0)?;
416                self.graphs[graph_id].derived_data.threshold_topology = Some(topology.clone());
417            }
418            let mut explicit = BTreeMap::new();
419            let all_lmbs = &self.graphs[group.master()]
420                .derived_data
421                .threshold_topology
422                .as_ref()
423                .unwrap()
424                .1;
425            for id in group {
426                for (label, subspace, description) in
427                    self.graphs[id].explicit_threshold_group_members()?
428                {
429                    explicit
430                        .entry(label)
431                        .or_insert_with(Vec::new)
432                        .push((subspace.solve_signature(all_lmbs), description));
433                }
434            }
435            let conflicts = explicit
436                .into_iter()
437                .filter_map(|(label, members)| {
438                    (members.iter().map(|(key, _)| key).unique().count() > 1).then(|| {
439                        format!(
440                            "group_id={label}: {}",
441                            members
442                                .into_iter()
443                                .map(|(key, description)| format!(
444                                    "{description}, signed cycles {key:?}"
445                                ))
446                                .join("; ")
447                        )
448                    })
449                })
450                .collect_vec();
451            if !conflicts.is_empty() {
452                return Err(eyre!(
453                    "Amplitude group master '{}' has incompatible threshold solve groups:\n{}",
454                    self.graphs[group.master()].graph.name,
455                    conflicts.join("\n")
456                ));
457            }
458        }
459        let integrand_name = self.name.clone();
460
461        let preprocess_span = if generation_progress::detailed_progress_enabled() {
462            let span = info_span!("Preprocessing graphs", indicatif.pb_show = true);
463            span.pb_set_style(&ProgressStyle::with_template(
464                "{wide_bar} {pos}/{len} {msg}",
465            )?);
466            span.pb_set_length(self.graphs.len() as u64);
467            span.pb_set_message("Preprocessing graphs");
468            Some(span)
469        } else {
470            None
471        };
472        let preprocess_span_enter = preprocess_span.as_ref().map(|span| span.enter());
473
474        let preprocess_reports = thread_pool.install(|| {
475            let parent = preprocess_span.clone();
476            self.graphs
477                .par_iter_mut()
478                .map(|amplitude_graph| {
479                    if crate::is_interrupted() {
480                        return Err(eyre!("Generation interrupted by user"));
481                    }
482                    let graph_name = amplitude_graph.graph.name.clone();
483                    generation_progress::graph_started(
484                        GenerationProcessKind::Amplitude,
485                        &integrand_name,
486                        &graph_name,
487                        None,
488                    );
489                    let _guard = parent.as_ref().map(|span| span.enter());
490                    let stats =
491                        amplitude_graph.preprocess(model, settings, locked_runtime_settings);
492                    if let Some(span) = &parent {
493                        span.pb_inc(1);
494                    }
495
496                    let stats = stats?;
497                    if crate::is_interrupted() {
498                        return Err(eyre!("Generation interrupted by user"));
499                    }
500                    generation_progress::graph_finished(
501                        GenerationProcessKind::Amplitude,
502                        &integrand_name,
503                        &graph_name,
504                        &stats,
505                        None,
506                    );
507
508                    Ok(NamedGraphGenerationReport {
509                        integrand_name: integrand_name.clone(),
510                        graph_name,
511                        stats,
512                    })
513                })
514                .collect::<Result<Vec<_>>>()
515        })?;
516
517        drop(preprocess_span_enter);
518        drop(preprocess_span);
519
520        // Groups containing non-default variants need the variant evaluator lane throughout,
521        // including members whose directives reproduce their defaults.
522        for group in &self.graph_group_structure {
523            if group.into_iter().any(|id| {
524                self.graphs[id]
525                    .derived_data
526                    .resolved_threshold_counterterms
527                    .as_ref()
528                    .is_some_and(|resolved| !resolved.legacy_equivalent)
529            }) {
530                for id in group {
531                    if let Some(resolved) =
532                        &mut self.graphs[id].derived_data.resolved_threshold_counterterms
533                    {
534                        resolved.legacy_equivalent = false;
535                    }
536                }
537            }
538        }
539        self.generate_grouped_derived_data()?;
540
541        Ok(preprocess_reports)
542    }

550    pub fn build_integrand(
551        &mut self,
552        model: &Model,
553        process_name: &str,
554        global_settings: &GlobalSettings,
555        runtime_default: LockedRuntimeSettings,
556        thread_pool: &ThreadPool,
557    ) -> Result<Vec<NamedGraphGenerationReport>> {
558        let started = std::time::Instant::now();
559        crate::debug_tags!(#generation, #profile, #graph, #summary;
560            stage = "amplitude_build_integrand_start",
561            integrand = %self.name,
562            graph_count = self.graphs.len(),
563            "Generation timing milestone"
564        );
565        if crate::is_interrupted() {
566            return Err(eyre!("Generation interrupted by user"));
567        }
568        let integrand_name = self.name.clone();
569        generation_progress::begin_phase(
570            GenerationProgressPhase::GraphGeneration,
571            GenerationProcessKind::Amplitude,
572            process_name,
573            &integrand_name,
574            self.graphs.len(),
575            None,
576        );
577        let mut graph_reports = Vec::new();
578        let terms: Vec<_> = thread_pool.install(|| {
579            self.graphs
580                .par_iter_mut()
581                .enumerate()
582                .map(|(graph_id, graph)| {
583                    if crate::is_interrupted() {
584                        return Err(eyre!("Generation interrupted by user"));
585                    }
586                    let graph_started = std::time::Instant::now();
587                    let group_id = graph.graph.group_id.unwrap(); // should always be set
588                    let esurface_map = &self.group_derived_data[group_id].esurface_map;
589                    let group_pos = self.graph_group_structure[group_id]
590                        .find_position(graph_id)
591                        .unwrap();
592
593                    crate::debug_tags!(#generation, #profile, #graph, #summary;
594                        stage = "amplitude_generate_term_for_graph_start",
595                        integrand = %integrand_name,
596                        graph = %graph.graph.name,
597                        graph_id,
598                        group_id = %group_id.0,
599                        "Generation timing milestone"
600                    );
601                    generation_progress::graph_started(
602                        GenerationProcessKind::Amplitude,
603                        &integrand_name,
604                        &graph.graph.name,
605                        None,
606                    );
607                    let _progress_context_guard =
608                        generation_progress::enter_progress_context(graph.graph.name.clone());
609                    let (term, mut stats) = graph.generate_term_for_graph(
610                        model,
611                        group_pos,
612                        esurface_map.clone(),
613                        global_settings,
614                    )?;
615                    if crate::is_interrupted() {
616                        return Err(eyre!("Generation interrupted by user"));
617                    }
618                    stats.evaluator_count = term.generic_evaluator_count();
619                    stats.total_time += graph_started.elapsed();
620                    crate::debug_tags!(#generation, #profile, #graph, #summary;
621                        stage = "amplitude_generate_term_for_graph_done",
622                        integrand = %integrand_name,
623                        graph = %graph.graph.name,
624                        graph_id,
625                        group_id = %group_id.0,
626                        elapsed_ms = graph_started.elapsed().as_secs_f64() * 1000.0,
627                        "Generation timing milestone"
628                    );
629                    generation_progress::graph_finished(
630                        GenerationProcessKind::Amplitude,
631                        &integrand_name,
632                        &graph.graph.name,
633                        &stats,
634                        None,
635                    );
636                    Ok((
637                        term,
638                        NamedGraphGenerationReport {
639                            integrand_name: integrand_name.clone(),
640                            graph_name: graph.graph.name.clone(),
641                            stats,
642                        },
643                    ))
644                })
645                .collect::<Result<Vec<_>>>()
646        })?;
647        if crate::is_interrupted() {
648            return Err(eyre!("Generation interrupted by user"));
649        }
650        for (_, report) in &terms {
651            graph_reports.push(report.clone());
652        }
653        let mut terms = terms.into_iter().map(|(term, _)| term).collect::<Vec<_>>();
654
655        for group in self.graph_group_structure.iter() {
656            let master_graph_id = group.master();
657            let mc_of_master = self.graphs[master_graph_id]
658                .derived_data
659                .multi_channeling_setup
660                .as_ref()
661                .unwrap();
662            let master_graph = &self.graphs[master_graph_id].graph;
663            let master_external_signature = master_graph.get_external_signature();
664            let master_external_pdgs = master_graph
665                .get_external_partcles()
666                .into_iter()
667                .map(|particle| particle.pdg_code)
668                .collect_vec();
669
670            for graph_id in group.into_iter() {
671                terms[graph_id].multi_channeling_setup = mc_of_master.clone();
672                terms[graph_id].master_external_signature = master_external_signature.clone();
673                terms[graph_id].master_external_pdgs = master_external_pdgs.clone();
674            }
675        }
676
677        let backend_started = std::time::Instant::now();
678        let graph_count = terms.len();
679        crate::debug_tags!(#generation, #profile, #compile, #graph, #summary;
680            stage = "amplitude_prepare_runtime_backends_start",
681            integrand = %self.name,
682            graph_count,
683            elapsed_ms = started.elapsed().as_secs_f64() * 1000.0,
684            "Generation timing milestone"
685        );
686        generation_progress::backend_started(
687            GenerationProcessKind::Amplitude,
688            &self.name,

788        // for each group we must collect all inequivalent esurfaces.
789
790        let group_derived_data = self
791            .graph_group_structure
792            .iter()
793            .map(|group| {
794                let mut group_esurface_structure =
795                    BTreeMap::<Atom, TiVec<GraphGroupPosition, Option<RaisedEsurfaceId>>>::default(
796                    );
797
798                for (graph_group_position, graph_id) in group.iter_enumerated() {
799                    let amplitude_graph = &self.graphs[graph_id];
800                    let lmb_reps = amplitude_graph.graph.integrand_replacement(
801                        &amplitude_graph.graph.full_filter(),
802                        &amplitude_graph.graph.loop_momentum_basis,
803                        &[W_.x___],
804                    );
805
806                    let esurfaces = &amplitude_graph.graph.surface_cache.esurface_cache;
807
808                    // Cached surfaces without a surviving pole are not threshold candidates.
809                    for (raised_esurface_id, raised_group) in amplitude_graph
810                        .derived_data
811                        .raised_data
812                        .raised_groups
813                        .iter_enumerated()
814                        .filter(|(_, raised_group)| raised_group.max_occurence > 0)
815                    {
816                        let esurface = &esurfaces[raised_group.esurface_ids[0]];
817                        let esurface_atom = esurface.lmb_atom(&amplitude_graph.graph, &lmb_reps);
818
819                        group_esurface_structure
820                            .entry(esurface_atom)
821                            .or_insert(ti_vec![None; group.len()])[graph_group_position] =
822                            Some(raised_esurface_id);
823                    }
824                }
825
826                let (surface_atoms, esurface_map) = group_esurface_structure.into_iter().unzip();
827
828                GroupDerivedData {
829                    esurface_map,
830                    esurface_atoms: surface_atoms,
831                }
832            })
833            .collect::<TiVec<GroupId, _>>();
834
835        let _: () = self.group_derived_data = group_derived_data;
836        Ok(())

1050        &mut self,
1051        model: &Model,
1052        settings: &GenerationSettings,
1053        locked_runtime_settings: &LockedRuntimeSettings,
1054    ) -> Result<GraphGenerationStats> {
1055        let _progress_guard = generation_progress::enter_detailed_progress_span("preprocessing");
1056        let preprocess_started = std::time::Instant::now();
1057        let vk = crate::utils::vakint()?;
1058
1059        self.generate_cff(settings)?;
1060
1061        // UV orchestration can extend the graph surface cache, while raised IDs
1062        // belong to the CFF expression generated above.
1063        let raised_data = settings.threshold_subtraction.enable_thresholds.then(|| {
1064            self.graph.determine_raised_esurfaces_from_expression(
1065                &self
1066                    .derived_data
1067                    .cff_expression
1068                    .as_ref()
1069                    .expect("cff_expression should have been created")

1072        });
1073
1074        self.build_integrands(settings, vk)?;
1075
1076        if self.graph.is_group_master {
1077            self.build_tropical_sampler(settings)?;
1078        }
1079
1080        self.build_lmbs();
1081
1082        if self.graph.is_group_master {
1083            self.build_multi_channeling_channels(settings.override_lmb_heuristics);
1084        }
1085
1086        if let Some(mut raised_data) = raised_data {
1087            let max_order = raised_data
1088                .raised_groups
1089                .iter()
1090                .map(|raised_group| raised_group.max_occurence)
1091                .max()
1092                .unwrap_or(0);

1094                self.graph.param_builder.initialize_duals(max_order);
1095            }
1096            raised_data.pass_two_evaluator = Some(
1097                (1..=max_order)

1099                        threshold_counterterm_helper(
1100                            order as u8,

1109                &raised_data,
1110                settings,
1111                vk,
1112                locked_runtime_settings,
1113                model,
1114            )?;
1115            self.derived_data.threshold_counterterms = build.legacy_counterterms;
1116            self.derived_data.threshold_counterterm_variants = build.variants;
1117            self.derived_data.resolved_threshold_counterterms = Some(build.resolved);
1118            self.derived_data.raised_esurface_ids = build.raised_esurface_ids;
1119            self.derived_data.raised_data = raised_data;
1120        } else {
1121            self.derived_data.resolved_threshold_counterterms = None;
1122            self.derived_data.threshold_counterterms.clear();
1123            self.derived_data.threshold_counterterm_variants.clear();
1124            self.derived_data.raised_esurface_ids.clear();
1125            self.derived_data.raised_data.raised_groups.clear();
1126            self.derived_data.raised_data.pass_two_evaluator = None;
1127        }
1128
1129        Ok(GraphGenerationStats {
1130            total_time: preprocess_started.elapsed(),
1131            ..GraphGenerationStats::default()
1132        })
1133    }

1554    fn resolve_amplitude_threshold_variant_subspace(
1555        &self,
1556        variant: &ThresholdCountertermVariant,
1557        legacy_subspace: &SubspaceData,
1558        context: &str,
1559    ) -> Result<SubspaceData> {
1560        let (topology, all_lmbs) = self.derived_data.threshold_topology.as_ref().map_or_else(
1561            || {
1562                (
1563                    &self.graph,
1564                    self.derived_data
1565                        .lmbs
1566                        .as_ref()
1567                        .expect("amplitude threshold resolution requires loop-momentum bases"),
1568                )
1569            },
1570            |(graph, lmbs)| (graph, lmbs),
1571        );
1572        let containing_subgraph = topology.no_dummy();
1573        let build_in_parent = |parent_lmb_index| match &variant.subspace {
1574            Some(requested_basis_edges) => SubspaceData::new_from_parent_basis_edges(
1575                requested_basis_edges,
1576                &containing_subgraph,
1577                parent_lmb_index,
1578                topology,
1579                all_lmbs,
1580            ),
1581            None => SubspaceData::new_with_user_selected_lmb(
1582                containing_subgraph.clone(),
1583                parent_lmb_index,
1584                topology,
1585                all_lmbs,
1586            ),
1587        };
1588
1589        if let Some(requested_parent_lmb) = &variant.parent_lmb {
1590            let matching_lmbs = all_lmbs
1591                .iter_enumerated()
1592                .filter_map(|(lmb_index, lmb)| {
1593                    lmb.loop_edges
1594                        .iter()
1595                        .eq(requested_parent_lmb.iter())
1596                        .then_some(lmb_index)
1597                })
1598                .collect_vec();
1599            if matching_lmbs.is_empty() {
1600                return Err(eyre!(
1601                    "{context} requests parent_lmb {:?}, which is not among graph '{}' generated LMBs",
1602                    requested_parent_lmb,
1603                    self.graph.name,
1604                ));
1605            }
1606            let mut built = matching_lmbs
1607                .iter()
1608                .map(|&lmb_index| build_in_parent(lmb_index).map(|subspace| (lmb_index, subspace)))
1609                .collect::<Result<Vec<_>>>()
1610                .with_context(|| {
1611                    format!("{context} is incompatible with its requested parent_lmb")
1612                })?;
1613            let (_, selected) = built.remove(0);
1614            if built
1615                .iter()
1616                .all(|(_, candidate)| selected.has_equivalent_embedding(candidate, all_lmbs))
1617            {
1618                return Ok(selected);
1619            }
1620            return Err(eyre!(
1621                "{context} parent_lmb {:?} has genuinely different generated embeddings",
1622                requested_parent_lmb,
1623            ));
1624        }
1625
1626        if variant.subspace.is_none() {
1627            return Ok(legacy_subspace.clone());
1628        }
1629
1630        let preferred_parent = legacy_subspace.parent_lmb_index();
1631        if let Ok(subspace) = build_in_parent(preferred_parent) {
1632            return Ok(subspace);
1633        }
1634
1635        let mut compatible = Vec::new();
1636        let mut rejections = Vec::new();
1637        for (lmb_index, lmb) in all_lmbs.iter_enumerated() {
1638            if lmb_index == preferred_parent {
1639                continue;
1640            }
1641            match build_in_parent(lmb_index) {
1642                Ok(subspace) => compatible.push((lmb_index, subspace)),
1643                Err(error) => rejections.push(format!("parent {:?}: {error:#}", lmb.loop_edges,)),
1644            }
1645        }
1646
1647        match compatible.len() {
1648            1 => Ok(compatible.pop().unwrap().1),
1649            0 => Err(eyre!(
1650                "{context} cannot resolve subspace {:?} in any generated parent LMB. Rejections:\n{}",
1651                variant.subspace,
1652                rejections.join("\n"),
1653            )),
1654            _ => {
1655                let (_, selected) = compatible.remove(0);
1656                if compatible
1657                    .iter()
1658                    .all(|(_, candidate)| selected.has_equivalent_embedding(candidate, all_lmbs))
1659                {
1660                    Ok(selected)
1661                } else {
1662                    Err(eyre!(
1663                        "{context} subspace {:?} has multiple genuinely different non-preferred parent LMB embeddings {:?}; specify parent_lmb",
1664                        variant.subspace,
1665                        compatible
1666                            .iter()
1667                            .map(|(lmb_index, _)| &all_lmbs[*lmb_index].loop_edges)
1668                            .collect_vec(),
1669                    ))
1670                }
1671            }
1672        }
1673    }

2579fn threshold_counterterm_helper_atoms(
2580    order: u8,
2581    loop_number: usize,
2582) -> (SingleThresholdPieces<Atom>, Atom) {
2583    let loop_3 = loop_number as i64 * 3;
2584
2585    let laurent_coeff_indices = (1..=order).map(|i| -(i as i8));
2586
2587    let mut laurent_coeffs = laurent_coeff_indices.map(|laurent_coeff_index| {
2588        build_derivative_structure_atom(order, laurent_coeff_index)
2589            .replace(GS.rescale_star)
2590            .with(GS.radius_star_left)
2591    });
2592
2593    let i = Atom::i();
2594
2595    let radius = Atom::var(GS.radius_left);
2596    let radius_star = Atom::var(GS.radius_star_left);
2597    let uv_damp_plus = Atom::var(GS.uv_damp_plus_left);
2598    let uv_damp_minus = Atom::var(GS.uv_damp_minus_left);
2599    let hfunction = Atom::var(GS.hfunction_left_th);
2600
2601    let delta_r_plus = &radius - &radius_star;
2602    let delta_r_minus = -&radius - &radius_star;
2603
2604    let jacobian_ratio = (Atom::one() / &radius).pow(loop_3 - 1);
2605
2606    let local_prefactor =
2607        &jacobian_ratio * (uv_damp_plus / &delta_r_plus + uv_damp_minus / &delta_r_minus);
2608
2609    let integrated_prefactor = -i * Atom::var(GS.pi) * &jacobian_ratio * hfunction;
2610
2611    let leading_laurent_coeff = laurent_coeffs.next().unwrap();
2612    let mut raised_local = Atom::Zero;
2613
2614    for pow in 2..=order {
2615        raised_local += laurent_coeffs.next().unwrap()
2616            * &jacobian_ratio
2617            * (Atom::one() / delta_r_plus.pow(pow as i64)
2618                + Atom::one() / delta_r_minus.pow(pow as i64));
2619    }
2620
2621    let local = local_prefactor.clone() * &leading_laurent_coeff + &raised_local;
2622    let integrated = integrated_prefactor.clone() * &leading_laurent_coeff;
2623    // Keep the no-directive helper structurally identical to the historical single-output lane.
2624    let legacy = (local_prefactor + integrated_prefactor) * leading_laurent_coeff + raised_local;
2625
2626    debug!(
2627        "Threshold counterterm helper atoms for order {} and loop number {}: local={}, integrated={}, legacy={}",
2628        order, loop_number, local, integrated, legacy,
2629    );
2630    (SingleThresholdPieces { local, integrated }, legacy)
2631}

2633enum ThresholdCountertermHelperOutputs {
2634    Legacy,
2635    Pieces,
2636    LegacyAndPieces,
2637}

2639fn build_threshold_counterterm_helper(
2640    order: u8,
2641    loop_number: usize,
2642    evaluator_settings: &EvaluatorSettings,
2643    outputs: ThresholdCountertermHelperOutputs,
2644) -> GenericEvaluator {
2645    let (pieces, legacy) = threshold_counterterm_helper_atoms(order, loop_number);
2646    let atoms = match outputs {
2647        ThresholdCountertermHelperOutputs::Legacy => vec![legacy],
2648        ThresholdCountertermHelperOutputs::Pieces => vec![pieces.local, pieces.integrated],
2649        ThresholdCountertermHelperOutputs::LegacyAndPieces => {
2650            vec![legacy, pieces.local, pieces.integrated]
2651        }
2652    };
2653    let fn_map = FunctionMap::default();
2654
2655    let mut params = params_for_derivative_order(order)
2656        .into_iter()
2657        .map(|param| param.replace(GS.rescale_star).with(GS.radius_star_left))
2658        .collect_vec();
2659
2660    let radius = Atom::var(GS.radius_left);
2661    let radius_star = Atom::var(GS.radius_star_left);
2662    let uv_damp_plus = Atom::var(GS.uv_damp_plus_left);
2663    let uv_damp_minus = Atom::var(GS.uv_damp_minus_left);
2664    let hfunction = Atom::var(GS.hfunction_left_th);
2665
2666    params.push(radius);
2667    params.push(radius_star);
2668    params.push(uv_damp_plus);
2669    params.push(uv_damp_minus);
2670    params.push(hfunction);
2671
2672    GenericEvaluator::new_from_raw_params(
2673        atoms,
2674        &params,
2675        &fn_map,
2676        vec![],
2677        evaluator_settings.optimization_settings(),
2678        None,
2679        evaluator_settings,
2680    )
2681    .unwrap()
2682    .into_eager_only()
2683}
