12/// Scalar definitions shared by multiplier expressions before binding their cut-local inputs.
13pub(super) struct ThresholdMultiplierFunctions {
14    entries: Vec<FnMapEntry>,
15}
16
17impl ThresholdMultiplierFunctions {
18    pub(super) fn parse(
19        definitions: &BTreeMap<String, String>,
20        parse_atom: impl Fn(&str) -> Result<Atom>,
21    ) -> Result<Self> {
22        let mut entries = Vec::<FnMapEntry>::new();
23        let default = (
24            "dE(m_a,m_b,Ea,Eb,ax,ay,az,bx,by,bz)",
25            "((m_a-m_b)*(m_a+m_b)+(ax-bx)*(ax+bx)+(ay-by)*(ay+by)+(az-bz)*(az+bz))/(Ea+Eb)",
26        );
27        for (definition_index, (signature, body)) in definitions
28            .iter()
29            .map(|(signature, body)| (signature.as_str(), body.as_str()))
30            .chain(std::iter::once(default))
31            .enumerate()
32        {
33            let header = parse_atom(signature)
34                .with_context(|| format!("invalid threshold function signature `{signature}`"))?;
35            let (name, arguments) = match header.as_view() {
36                AtomView::Var(variable) => (variable.get_symbol(), Vec::new()),
37                AtomView::Fun(function) => (
38                    function.get_symbol(),
39                    function
40                        .iter()
41                        .map(|argument| match argument {
42                            AtomView::Var(variable) => Ok(variable.get_symbol()),
43                            _ => Err(eyre!(
44                                "threshold function `{signature}` requires distinct scalar argument names; tags and static edge/frame formals are not supported"
45                            )),
46                        })
47                        .collect::<Result<Vec<_>>>()?,
48                ),
49                _ => {
50                    return Err(eyre!(
51                        "threshold function signature `{signature}` must be a name or scalar function call"
52                    ));
53                }
54            };
55            if Self::reserved(name) {
56                return Err(eyre!("reserved threshold function name `{name}`"));
57            }
58            if entries
59                .iter()
60                .any(|entry| entry.lhs.as_fun_view().unwrap().get_symbol() == name)
61            {
62                if definition_index == definitions.len() {
63                    // An explicit user definition replaces the default, without depending on
64                    // Symbolica's first-registration-wins behavior.
65                    continue;
66                }
67                return Err(eyre!(
68                    "duplicate threshold function head `{name}` in `{signature}`"
69                ));
70            }
71            for (index, argument) in arguments.iter().enumerate() {
72                if arguments[..index].contains(argument) || Self::reserved(*argument) {
73                    return Err(eyre!(
74                        "duplicate or reserved argument `{argument}` in threshold function `{signature}`"
75                    ));
76                }
77            }
78
79            // Private formal symbols keep a caller's argument from shadowing a captured
80            // graph input in another function, in both Symbolica translation modes.
81            let formals = arguments
82                .iter()
83                .enumerate()
84                .map(|(index, _)| {
85                    symbol!(&format!(
86                        "gammalooprs::threshold_multiplier_arg::{}::arg_{index}",
87                        name.get_name()
88                    ))
89                })
90                .collect::<Vec<_>>();
91            let body = parse_atom(body)
92                .with_context(|| format!("invalid body of threshold function `{signature}`"))?;
93            let body = body.replace_map(|view, _, output| {
94                if let AtomView::Var(variable) = view
95                    && let Some(index) = arguments
96                        .iter()
97                        .position(|argument| *argument == variable.get_symbol())
98                {
99                    **output = Atom::var(formals[index]);
100                }
101            });
102            entries.push(FnMapEntry {
103                lhs: FunctionBuilder::new(name)
104                    .add_args(formals.iter().map(|formal| Atom::var(*formal)))
105                    .finish(),
106                rhs: body,
107                args: formals.into_iter().map(Indeterminate::from).collect(),
108                tags: Vec::new(),
109            });
110        }
111        Ok(Self { entries })
112    }

114    pub(super) fn bind(
115        &self,
116        source: Atom,
117        normalize: impl Fn(&Atom) -> Result<Atom>,
118    ) -> Result<(Atom, FunctionMap, Vec<FnMapEntry>)> {
119        let mut entries = Vec::new();
120        let source = self.bind_expression(
121            &source,
122            &normalize,
123            &mut vec![0; self.entries.len()],
124            &mut entries,
125        )?;
126        let mut functions = FunctionMap::new();
127        for entry in &entries {
128            functions
129                .add_function(
130                    entry.lhs.as_fun_view().unwrap().get_symbol(),
131                    entry.args.clone(),
132                    entry.rhs.clone(),
133                )
134                .map_err(|error| eyre!("failed to bind threshold function: {error}"))?;
135        }
136        Ok((source, functions, entries))
137    }
138
139    fn bind_expression(
140        &self,
141        source: &Atom,
142        normalize: &impl Fn(&Atom) -> Result<Atom>,
143        states: &mut [u8],
144        entries: &mut Vec<FnMapEntry>,
145    ) -> Result<Atom> {
146        let mut dependencies = BTreeSet::new();
147        let mut error = None;
148        let source = source.replace_map(|view, _, output| {
149            let (name, arity) = match view {
150                AtomView::Var(variable) => (variable.get_symbol(), 0),
151                AtomView::Fun(function) => (function.get_symbol(), function.get_nargs()),
152                _ => return,
153            };
154            let Some((index, entry)) = self
155                .entries
156                .iter()
157                .enumerate()
158                .find(|(_, entry)| entry.lhs.as_fun_view().unwrap().get_symbol() == name)
159            else {
160                return;
161            };
162            if arity != entry.args.len() {
163                error = Some(eyre!(
164                    "threshold function `{name}` expects {} arguments, got {arity}",
165                    entry.args.len()
166                ));
167                return;
168            }
169            dependencies.insert(index);
170            if matches!(view, AtomView::Var(_)) {
171                **output = FunctionBuilder::new(name).finish();
172            }
173        });
174        if let Some(error) = error {
175            return Err(error);
176        }
177        for index in dependencies {
178            let entry = &self.entries[index];
179            match states[index] {
180                1 => {
181                    return Err(eyre!(
182                        "cyclic threshold function dependency at `{}`",
183                        entry.lhs
184                    ));
185                }
186                2 => continue,
187                _ => {}
188            }
189            states[index] = 1;
190            let rhs = self.bind_expression(&entry.rhs, normalize, states, entries)?;
191            entries.push(FnMapEntry {
192                rhs,
193                ..entry.clone()
194            });
195            states[index] = 2;
196        }
197        normalize(&source)
198    }
199
200    fn reserved(symbol: Symbol) -> bool {
201        let name = symbol.get_name().rsplit("::").next().unwrap();
202        symbol.is_builtin()
203            || name.starts_with("threshold_multiplier_")
204            || matches!(
205                name,
206                "E" | "Q3" | "Q" | "P" | "K" | "eta" | "eset" | "cind" | "effective" | "star"
207            )
208    }

216    #[test]
217    fn nested_aliases_and_scalar_functions_capture_inputs_without_shadowing() {
218        crate::initialisation::test_initialise().unwrap();
219        let functions = ThresholdMultiplierFunctions::parse(
220            &BTreeMap::from([
221                ("captured".into(), "x+1".into()),
222                ("nested(x)".into(), "x*captured".into()),
223                ("outer(y)".into(), "nested(y+1)+captured()".into()),
224            ]),
225            |source| try_parse!(source).map_err(|error| eyre!("{error}")),
226        )
227        .unwrap();
228        let (source, map, entries) = functions
229            .bind(parse!("outer(2)+captured"), |atom| Ok(atom.clone()))
230            .unwrap();
231        assert_eq!(entries.len(), 3);
232        assert!(matches!(entries[0].lhs.as_view(), AtomView::Fun(_)));
233        for direct in [false, true] {
234            let mut evaluator = source
235                .evaluator(&[parse!("x")])
236                .function_map(map.clone())
237                .direct_translation(direct)
238                .build()
239                .unwrap()
240                .map_coeff(&|coefficient| coefficient.re.to_f64());
241            assert_eq!(evaluator.evaluate_single(&[5.0]), 30.0);
242            assert_eq!(evaluator.evaluate_single(&[7.0]), 40.0);
243        }
244    }

246    #[test]
247    fn only_reachable_functions_bind_cut_geometry() {
248        crate::initialisation::test_initialise().unwrap();
249        let functions = ThresholdMultiplierFunctions::parse(
250            &BTreeMap::from([
251                ("local".into(), "2".into()),
252                ("foreign".into(), "eta(star,eset(7,8))".into()),
253                ("unreachable_cycle".into(), "unreachable_cycle()".into()),
254            ]),
255            |source| try_parse!(source).map_err(|error| eyre!("{error}")),
256        )
257        .unwrap();
258        let normalize = |atom: &Atom| {
259            if atom.contains_symbol(symbol!("eta")) {
260                Err(eyre!("foreign eta is absent from this cut"))
261            } else {
262                Ok(atom.clone())
263            }
264        };
265        let (_, _, entries) = functions.bind(parse!("local"), normalize).unwrap();
266        assert_eq!(entries.len(), 1);
267        assert!(functions.bind(parse!("foreign"), normalize).is_err());
268        assert!(
269            functions
270                .bind(parse!("unreachable_cycle"), normalize)
271                .is_err()
272        );
273    }

282            BTreeMap::from([("f(1)".into(), "1".into())]),
283            BTreeMap::from([("E(a)".into(), "a".into())]),

285            BTreeMap::from([("f(star)".into(), "star".into())]),
286        ] {
287            assert!(
288                ThresholdMultiplierFunctions::parse(&definitions, |source| {
289                    try_parse!(source).map_err(|error| eyre!("{error}"))

293        }
294        let functions = ThresholdMultiplierFunctions::parse(

323    #[test]
324    fn default_energy_difference_and_explicit_override_are_composable() {
325        crate::initialisation::test_initialise().unwrap();
326        for definitions in [
327            BTreeMap::new(),
328            BTreeMap::from([("dE(a,b)".into(), "a-b".into())]),
329        ] {
330            let functions = ThresholdMultiplierFunctions::parse(&definitions, |source| {
331                try_parse!(source).map_err(|error| eyre!("{error}"))
332            })
333            .unwrap();
334            let source = if definitions.is_empty() {
335                // The first energy has mass 3 and momentum (4,0,0); the second is
336                // at rest with mass/energy 2. This also checks unequal masses.
337                parse!("dE(3,2,5,2,4,0,0,0,0,0)")
338            } else {
339                parse!("dE(5,2)")
340            };
341            let (source, map, entries) = functions.bind(source, |atom| Ok(atom.clone())).unwrap();
342            assert_eq!(entries.len(), 1);
343            let mut evaluator = source
344                .evaluator(&[] as &[Atom])
345                .function_map(map)
346                .build()
347                .unwrap()
348                .map_coeff(&|coefficient| coefficient.re.to_f64());
349            assert_eq!(evaluator.evaluate_single(&[]), 3.0);
350        }
