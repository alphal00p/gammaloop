1//! Localization of zero-temperature distribution derivatives on independent cyclic chains.
2//!
3//! Each factor `N^(r)(E)` becomes `delta^(r-1)(E - sigma*mu)`. The CFF coefficient
4//! already owns the cyclic-chain sign and factorial. This module supplies the
5//! distribution action and spatial measure, and evaluates compiled smooth
6//! coefficients in the amplitude's existing momentum and parameter layout.
7//!
8//! In a basis containing one edge from each chain, insert a normalized positive
9//! scale integral and dilate that edge momentum, `q = t*k`. With shell offset
10//! `u = E - sigma*mu`, the inverse map is analytic:
11//! `t(u) = sqrt((sigma*mu + u)^2 - m^2) / |k|`. Its complete weight is
12//! `h(t) * t^3 / g'(t) = h(t) * E * (E^2 - m^2) / |k|^4`.
13//! Evaluating this weight and the coefficient on energy-offset jets implements
14//! the same change-of-variable derivative as a raised-cut residue, including
15//! every derivative of the profile, Jacobian and coefficient.
16

43/// A product of Fermi distributions whose cyclic-chain momenta are independent.
44///
45/// The first loop coordinates of `lmb()` follow `factors()` in order. External
46/// shifts are retained in the basis routing, so the selected coordinates are
47/// the actual edge momenta about which the Fermi spheres are centred.
48#[derive(Clone, Debug, Encode, Decode)]
49pub(crate) struct FermiSurfaceProduct {
50    factors: Vec<ThermalDistributionFactor>,
51    lmb: LoopMomentumBasis,
52    derivative_orders: Vec<usize>,
53    shape: Vec<Vec<usize>>,
54}
55
56impl FermiSurfaceProduct {
57    /// Localize a smooth coefficient while preserving the spatial integration dimension.
58    ///
59    /// `loop_momenta` is in `lmb()`. Masses and oriented chemical potentials
60    /// (`sigma*mu`, not their absolute values) follow `factors()` in order;
61    /// only the squared mass enters the shell. `profile(t)` must integrate to
62    /// one over `t > 0`.
63    /// Both callbacks must return jets with the supplied shape. The coefficient
64    /// includes all routed numerator/denominator dependence; the localizer owns
65    /// the scale profiles, spatial dilation factors and delta Jacobians.
66    ///
67    /// A future conditional Cutkosky localization can be evaluated inside the
68    /// coefficient callback. Its root and Jacobian must retain their dependence
69    /// on all shell-offset jets before the Fermi derivatives are extracted.
70    /// No delta or discontinuous step function may be numerically differentiated
71    /// inside this smooth callback. Expand such derivatives symbolically first,
72    /// and validate any generated intersection together with its host surface
73    /// using `FermiSurfaceProduct::new`.
74    pub(crate) fn localize<T: FloatLike>(
75        &self,
76        loop_momenta: &LoopMomenta<F<T>>,
77        masses: &[F<T>],
78        oriented_chemical_potentials: &[F<T>],
79        mut profile: impl FnMut(&HyperDual<F<T>>) -> Result<HyperDual<F<T>>>,
80        coefficient: impl FnOnce(&LoopMomenta<HyperDual<F<T>>>) -> Result<HyperDual<Complex<F<T>>>>,
81    ) -> Result<Complex<F<T>>> {
82        if loop_momenta.len() != self.lmb.loop_edges.len()
83            || masses.len() != self.factors.len()
84            || oriented_chemical_potentials.len() != self.factors.len()
85        {
86            return Err(eyre!(
87                "Fermi localization requires {} loop momenta and {} masses/chemical potentials",
88                self.lmb.loop_edges.len(),
89                self.factors.len()
90            ));
91        }
92        let representative = &oriented_chemical_potentials[0];
93        for (index, (mass, nu)) in masses.iter().zip(oriented_chemical_potentials).enumerate() {
94            if !mass.is_finite() || !nu.is_finite() {
95                return Err(eyre!(
96                    "Fermi surface on edge {} requires a finite mass and finite oriented chemical potential",
97                    self.factors[index].edge_id
98                ));
99            }
100        }
101        let masses = masses.iter().map(|mass| mass.abs()).collect::<Vec<_>>();
102        if loop_momenta.iter().any(|momentum| {
103            [&momentum.px, &momentum.py, &momentum.pz]
104                .into_iter()
105                .any(|value| !value.is_finite())
106        }) {
107            return Err(SamplingEvaluationError::Unrepresentable {
108                operation: "Fermi localization",
109                detail: "Fermi localization requires finite loop momenta".to_owned(),
110            }
111            .into());
112        }
113        // Empty support is a physical zero, independent of the smooth coefficient.
114        // A zero chemical potential leaves the occupation constant on E > 0,
115        // also for a massless fermion, exactly as for a particle without one.
116        if masses
117            .iter()
118            .zip(oriented_chemical_potentials)
119            .any(|(mass, nu)| nu < mass || nu.is_zero())
120        {
121            return Ok(Complex::new_re(representative.zero()));
122        }
123        let mut template = HyperDual::from_values(
124            self.shape.clone(),
125            vec![representative.zero(); self.shape.len()],
126        );
127        template.values[0] = representative.one();
128        let mut momenta = LoopMomenta::from_iter(
129            loop_momenta
130                .iter()
131                .map(|momentum| momentum.map_ref(&|value| new_constant(&template, value))),
132        );
133        let mut measures = Vec::with_capacity(self.factors.len());
134
135        for (axis, ((factor, mass), nu)) in self
136            .factors
137            .iter()
138            .zip(&masses)
139            .zip(oriented_chemical_potentials)
140            .enumerate()
141        {
142            // Check the selected sector at its actual evaluation precision, after
143            // the empty-product return above. Binary64 warm-up can round a shell
144            // resolved by Quad/Arb to an apparent onset.
145            if nu == mass {
146                return Err(eyre!(
147                    "Degenerate Fermi surface on edge {} at sigma*mu = mass; the regular-shell localization does not define the onset",
148                    factor.edge_id
149                ));
150            }
151            let momentum = &loop_momenta[axis.into()];
152            let norm_squared = momentum.norm_squared();
153            if norm_squared <= representative.zero() || !norm_squared.is_finite() {
154                return Err(SamplingEvaluationError::Unrepresentable {
155                    operation: "Fermi localization",
156                    detail: format!(
157                        "Fermi surface on edge {} requires a finite nonzero radial sampling momentum",
158                        factor.edge_id
159                    ),
160                }
161                .into());
162            }
163            let mut energy = new_constant(&template, nu);
164            if self.derivative_orders[axis] > 0 {
165                let mut unit = vec![0; self.factors.len()];
166                unit[axis] = 1;
167                let index = self
168                    .shape
169                    .iter()
170                    .position(|orders| *orders == unit)
171                    .expect("mixed derivative shape contains each active first derivative");
172                energy.values[index] = representative.one();
173            }
174            // Take square roots before multiplying, preserving representable
175            // radii even when their squares underflow. The massless radius is
176            // exactly E; using it directly also avoids cancelling higher jets.
177            let radius = if *mass == mass.zero() {
178                energy.clone()
179            } else {
180                let mass = new_constant(&template, mass);
181                (energy.clone() - &mass).sqrt() * (energy.clone() + &mass).sqrt()
182            };
183            let scale = radius / new_constant(&template, &norm_squared.sqrt());
184            if scale.values[0] <= representative.zero()
185                || scale.values.iter().any(|value| !value.is_finite())
186            {
187                return Err(SamplingEvaluationError::Unrepresentable {
188                    operation: "Fermi localization",
189                    detail: format!("Nonfinite Fermi root jet on edge {}", factor.edge_id),
190                }
191                .into());
192            }
193            let density = profile(&scale)?;
194            if density.get_shape() != template.get_shape()
195                || density.values.len() != self.shape.len()
196            {
197                return Err(eyre!(
198                    "Fermi scale profile returned an incompatible derivative shape"
199                ));
200            }
201            momenta[axis.into()] =
202                momentum.map_ref(&|value| new_constant(&template, value) * &scale);
203            measures.push((density, energy, scale, norm_squared));
204        }
205
206        let value = coefficient(&momenta)?;
207        if value.get_shape() != template.get_shape() || value.values.len() != self.shape.len() {
208            return Err(eyre!(
209                "Fermi coefficient returned an incompatible derivative shape"
210            ));
211        }
212        // E*t*t/|k|^2 equals t^3/g'(t) without forming |k|^4. Keep its factors
213        // separate until the coefficient is available: either multiplying the
214        // small profile first or the large geometry first can lose a finite
215        // fully weighted result. Logs below only choose the arithmetic order.
216        let mut factors = Vec::with_capacity(5 * measures.len());
217        for (density, energy, scale, norm_squared) in measures {
218            for (factor, divide) in [
219                (density, false),
220                (energy, false),
221                (scale.clone(), false),
222                (scale, false),
223                (new_constant(&template, &norm_squared), true),
224            ] {
225                if factor.values.iter().any(|value| !value.is_finite()) {
226                    return Err(SamplingEvaluationError::Unrepresentable {
227                        operation: "Fermi localization",
228                        detail: "Nonfinite Fermi measure jet".to_owned(),
229                    }
230                    .into());
231                }
232                let magnitude = factor
233                    .values
234                    .iter()
235                    .map(|value| value.abs())
236                    .max_by(|left, right| left.partial_cmp(right).unwrap())
237                    .unwrap();
238                let score = if divide {
239                    -magnitude.log()
240                } else {
241                    magnitude.log()
242                };
243                factors.push((factor, divide, score));
244            }
245        }
246        let index = self
247            .shape
248            .iter()
249            .position(|orders| orders == &self.derivative_orders)
250            .expect("mixed derivative shape contains the requested derivative");
251        let mut components = [representative.zero(), representative.zero()];
252        for (component, imaginary) in components.iter_mut().zip([false, true]) {
253            let mut weighted = HyperDual::from_values(
254                self.shape.clone(),
255                value
256                    .values
257                    .iter()
258                    .map(|value| {
259                        if imaginary {
260                            value.im.clone()
261                        } else {
262                            value.re.clone()
263                        }
264                    })
265                    .collect(),
266            );
267            if weighted.values.iter().any(|value| !value.is_finite()) {
268                return Err(SamplingEvaluationError::Unrepresentable {
269                    operation: "Fermi localization",
270                    detail: "Nonfinite Fermi coefficient jet".to_owned(),
271                }
272                .into());
273            }
274            // Balance each observable component independently. A large imaginary
275            // coefficient must not determine the scaling of a small real one.
276            let mut remaining = (0..factors.len()).collect::<Vec<_>>();
277            remaining
278                .sort_by(|left, right| factors[*left].2.partial_cmp(&factors[*right].2).unwrap());
279            while !remaining.is_empty() {
280                let magnitude = weighted
281                    .values
282                    .iter()
283                    .map(|value| value.abs())
284                    .max_by(|left, right| left.partial_cmp(right).unwrap())
285                    .unwrap();
286                let next = if magnitude >= representative.one() {
287                    remaining.remove(0)
288                } else {
289                    remaining.pop().unwrap()
290                };
291                let (factor, divide, _) = &factors[next];
292                if *divide {
293                    for value in &mut weighted.values {
294                        *value /= &factor.values[0];
295                    }
296                } else {
297                    weighted *= factor;
298                }
299                if weighted.values.iter().any(|value| !value.is_finite()) {
300                    return Err(SamplingEvaluationError::Unrepresentable {
301                        operation: "Fermi localization",
302                        detail: "Nonfinite localized Fermi contribution".to_owned(),
303                    }
304                    .into());
305                }
306            }
307            *component = weighted.values[index].clone();
308            // Convert Taylor coefficients to ordinary distribution derivatives,
309            // avoiding an independently overflowing integer or float factorial.
310            for order in &self.derivative_orders {
311                for factor in 1..=*order {
312                    *component *= -representative.from_usize(factor);
313                }
314            }
315        }
316        let [real, imaginary] = components;
317        let result = Complex::new(real, imaginary);
318        if !result.re.is_finite() || !result.im.is_finite() {
319            return Err(SamplingEvaluationError::Unrepresentable {
320                operation: "Fermi localization",
321                detail: "Nonfinite localized Fermi contribution".to_owned(),
322            }
323            .into());
324        }
325        Ok(result)
326    }
327}
