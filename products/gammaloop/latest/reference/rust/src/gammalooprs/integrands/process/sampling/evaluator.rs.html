1//! Eager and first-derivative evaluators used by compiled sampling maps.
2//!
3//! Sampling maps are assembled from symbolic expressions during warmup.  This
4//! wrapper keeps the expression compiler and its hyper-dual variant together so
5//! a map and its Jacobian always use the same Symbolica expressions.  It is a
6//! deliberately small layer around the process evaluator: graph-specific map
7//! construction remains in the process sampler.
8

27/// Values and first partial derivatives of one evaluated expression.
28#[derive(Clone, Debug, PartialEq)]
29pub struct SamplingDualValue<T: FloatLike = f64> {
30    /// The value of the expression at the supplied parameters.
31    pub value: Complex<F<T>>,
32    /// First derivatives in the compiled parameter-column order supplied to
33    /// [`SamplingExpressionEvaluator::new`].
34    pub derivatives: Vec<Complex<F<T>>>,
35}
36
37/// Real-valued output and first-derivative matrix of a sampling expression.
38///
39/// Sampling maps are real maps even though the shared Symbolica evaluator uses
40/// its complex domain.  This type performs that boundary conversion once and
41/// computes the signed determinant of a square output/input Jacobian.  The
42/// map density must use [`Self::absolute_determinant`], since a coordinate
43/// chart may reverse orientation.
44#[derive(Clone, Debug, PartialEq)]
45pub struct SamplingJacobianEvaluation<T: FloatLike = f64> {
46    /// Expression values in the order supplied to the evaluator.
47    pub values: Vec<T>,
48    /// Rows are outputs; columns follow the requested active-parameter order
49    /// (the compiled column order when no subset was supplied).
50    pub derivatives: Vec<Vec<T>>,
51    /// Signed determinant of `derivatives`.
52    pub determinant: T,
53}
54
55impl<T: FloatLike> SamplingJacobianEvaluation<T> {
56    /// Absolute determinant required by a push-forward volume density.
57    pub fn absolute_determinant(&self) -> T {
58        F(self.determinant.clone()).abs().0
59    }
60}

62/// A warmup-compiled evaluator for one or more sampling-map expressions.
63///
64/// The eager program is always retained. If derivative columns are requested,
65/// Symbolica's hyper-dual evaluator is built from the very same expression
66/// tree, yielding exact first partials suitable for a forward-map Jacobian.
67/// Cloning copies the already compiled programs and their mutable buffers;
68/// it does not run expression optimization or compilation again.
69#[derive(Clone)]
70pub struct SamplingExpressionEvaluator {
71    evaluator: GenericEvaluator,
72    derivative_evaluator: Option<GenericEvaluator>,
73    parameter_count: usize,
74    output_count: usize,
75    derivative_parameters: Vec<usize>,
76}
77
78impl SamplingExpressionEvaluator {
79    /// Build an eager evaluator from expressions and their ordered parameters.
80    ///
81    /// Parameters are ordinary Symbolica atoms (usually variables).  The
82    /// expression list must be non-empty; each expression contributes one
83    /// output in the returned vector. Only `derivative_parameters` are dualized;
84    /// an empty slice builds a scalar-only evaluator. Scalar reconstruction and
85    /// active derivatives compile from the same expressions during warmup.
86    pub fn new(
87        expressions: impl IntoIterator<Item = Atom>,
88        parameters: impl IntoIterator<Item = Atom>,
89        derivative_parameters: &[usize],
90    ) -> Result<Self> {
91        let expressions = expressions.into_iter().collect::<Vec<_>>();
92        let parameters = parameters.into_iter().collect::<Vec<_>>();
93        if expressions.is_empty() {
94            return Err(eyre!("sampling evaluator requires at least one expression"));
95        }
96        if parameters.is_empty() {
97            return Err(eyre!("sampling evaluator requires at least one parameter"));
98        }
99        for (column, &parameter) in derivative_parameters.iter().enumerate() {
100            if parameter >= parameters.len() {
101                return Err(eyre!(
102                    "sampling derivative parameter {parameter} is out of range"
103                ));
104            }
105            if derivative_parameters[..column].contains(&parameter) {
106                return Err(eyre!("sampling derivative parameters must be unique"));
107            }
108        }
109        let compile = |dual_config| -> Result<GenericEvaluator> {
110            let mut evaluator = GenericEvaluator::new_from_raw_params(
111                expressions.clone(),
112                &parameters,
113                &FunctionMap::new(),
114                Vec::new(),
115                OptimizationSettings::default(),
116                dual_config,
117                &EvaluatorSettings::default(),
118            )?;
119            // Map rational coefficients once, before any source draw. Physical
120            // evaluators retain an empty cache and their existing three lanes.
121            let program = evaluator
122                .rational
123                .as_ref()
124                .expect("new sampling evaluators retain their rational program")
125                .clone()
126                .map_coeff(&|r| Complex::new(F::from(&r.re), F::from(&r.im)));
127            evaluator.sampling_fixed256.set(program);
128            Ok(evaluator)
129        };
130        let evaluator = compile(None)?;
131        let derivative_evaluator = if derivative_parameters.is_empty() {
132            None
133        } else {
134            let zero_components = (0..parameters.len())
135                .flat_map(|parameter| {
136                    derivative_parameters
137                        .iter()
138                        .enumerate()
139                        .filter(move |&(_, &active)| active != parameter)
140                        .map(move |(derivative, _)| (parameter, derivative + 1))
141                })
142                .collect();
143            Some(compile(Some((
144                first_derivative_shape(derivative_parameters.len()),
145                zero_components,
146            )))?)
147        };
148        Ok(Self {
149            evaluator,
150            derivative_evaluator,
151            parameter_count: parameters.len(),
152            output_count: expressions.len(),
153            derivative_parameters: derivative_parameters.to_vec(),
154        })
155    }
156
157    /// Compile a declared-positive proxy once for all supported precisions.
158    ///
159    /// Non-real and nonpositive constants are rejected before compilation.
160    /// This is not a positivity certificate for arbitrary expressions: native
161    /// score evaluation must still check every result, and the graph compiler
162    /// must separately establish coverage of the intended singular features.
163    pub(crate) fn new_positive_proxy(
164        expression: Atom,
165        parameters: impl IntoIterator<Item = Atom>,
166    ) -> Result<Self> {
167        if let AtomView::Num(number) = expression.as_view() {
168            let coefficient = number.get_coeff_view().to_owned();
169            if !coefficient.is_real() || coefficient.is_zero() || coefficient.is_negative() {
170                return Err(eyre!(
171                    "sampling Symbolica score constant must be real and strictly positive: {expression}"
172                ));
173            }
174        }
175        Self::new([expression], parameters, &[])
176    }
177
178    /// Compile the auxiliary-scale proposal once, including its native fit and
179    /// first derivatives. The two sign inputs select algebraically equivalent
180    /// stable tails: every evaluated exponential has a nonpositive argument.
181    /// Inputs are log(t), log(R), log(beta), focused sign and broad sign;
182    /// outputs are fitted log-scale, shape, CDF, survival and log(raw radius).
183    /// Only log(t) is active in the radial Jacobian; root, scale and stable-tail
184    /// signs remain runtime inputs without unused dual derivative components.
185    pub(crate) fn new_lu_h_profile(
186        profile: &SamplingRadialProfile,
187        h: &HFunctionSettings,
188    ) -> Result<Self> {
189        if !h.sigma.is_finite() || h.sigma <= 0.0 {
190            return Err(eyre!(
191                "LU-h sampling requires a positive finite h_function.sigma"
192            ));
193        }
194        if !profile.broad_fraction.is_finite()
195            || !(0.0..=1.0).contains(&profile.broad_fraction)
196            || [profile.scale, profile.shape]
197                .into_iter()
198                .flatten()
199                .any(|value| !value.is_finite() || value <= 0.0)
200        {
201            return Err(eyre!(
202                "LU-h sampling requires broad_fraction in [0,1] and positive finite scale/shape overrides"
203            ));
204        }
205        let power = h.power.unwrap_or(0);
206        let (log_scale, shape) = match h.function {
207            HFunction::PolyExponential | HFunction::PolyLeftRightExponential => {
208                if ![0, 1, 3, 4, 6, 7, 9, 10, 12, 13, 15, 16].contains(&power) {
209                    return Err(eyre!(
210                        "unsupported LU h-function power {power}; supported powers are 0,1,3,4,6,7,9,10,12,13,15,16"
211                    ));
212                }
213                let a = if matches!(h.function, HFunction::PolyExponential) {
214                    2
215                } else {
216                    1
217                };
218                let b = format!("((1-{power})/(2*{a}))");
219                // asinh(b) written with positive arguments, evaluated in the
220                // requested native precision rather than fitted in binary64.
221                (
222                    format!("log({})+log({b}+sqrt(1+({b})^2))/{a}", h.sigma),
223                    format!("2*{a}*(1+({b})^2)^(1/4)"),
224                )
225            }
226            HFunction::Exponential => (format!("log({}/2)", h.sigma), "1".to_owned()),
227            HFunction::ExponentialCT => {
228                return Err(eyre!(
229                    "exponential_ct is a local threshold localization, not a normalized auxiliary LU h-function"
230                ));
231            }
232        };
233        let log_scale = profile
234            .scale
235            .map_or(log_scale, |scale| format!("log({scale})"));
236        let shape = profile.shape.map_or(shape, |shape| shape.to_string());
237        let parameters = [
238            "lu_profile_y",
239            "lu_profile_log_root",
240            "lu_profile_log_beta",
241            "lu_profile_focus_sign",
242            "lu_profile_broad_sign",
243        ];
244        let x = format!("exp(lu_profile_focus_sign*({shape})*(lu_profile_y-({log_scale})))");
245        let b = "exp(lu_profile_broad_sign*(lu_profile_y-lu_profile_log_root+lu_profile_log_beta))";
246        let focused_cdf =
247            format!("((1+lu_profile_focus_sign)*({x})+1-lu_profile_focus_sign)/(2*(1+({x})))");
248        let focused_survival =
249            format!("(1+lu_profile_focus_sign+(1-lu_profile_focus_sign)*({x}))/(2*(1+({x})))");
250        let broad_cdf =
251            format!("((1+lu_profile_broad_sign)*({b})+1-lu_profile_broad_sign)/(2*(1+({b})))");
252        let broad_survival =
253            format!("(1+lu_profile_broad_sign+(1-lu_profile_broad_sign)*({b}))/(2*(1+({b})))");
254        let epsilon = profile.broad_fraction;
255        let expressions = [
256            log_scale,
257            shape,
258            format!("(1-{epsilon})*({focused_cdf})+{epsilon}*({broad_cdf})"),
259            format!("(1-{epsilon})*({focused_survival})+{epsilon}*({broad_survival})"),
260            "lu_profile_log_root-lu_profile_y".to_owned(),
261        ];
262        Self::new(
263            expressions
264                .iter()
265                .map(|expression| try_parse!(expression))
266                .collect::<std::result::Result<Vec<_>, _>>()
267                .map_err(|error| eyre!(error))?,
268            parameters
269                .iter()
270                .map(|parameter| try_parse!(*parameter))
271                .collect::<std::result::Result<Vec<_>, _>>()
272                .map_err(|error| eyre!(error))?,
273            &[0],
274        )
275    }

289    /// Evaluate all outputs eagerly at a real parameter point.
290    pub fn evaluate<T: FloatLike>(&mut self, parameters: &[T]) -> Result<Vec<Complex<F<T>>>> {
291        self.validate_parameters(parameters)?;
292        let input = parameters
293            .iter()
294            .cloned()
295            .map(|value| Complex::new_re(F(value)))
296            .collect::<Vec<_>>();
297        let values = <T as GenericEvaluatorFloat>::get_evaluator(&mut self.evaluator)(&input);
298        values
299            .into_iter()
300            .map(|value| match value {
301                DualOrNot::NonDual(value) => Ok(value),
302                DualOrNot::Dual(_) => Err(eyre!("scalar sampling evaluator returned dual outputs")),
303            })
304            .collect()
305    }

307    /// Evaluate all outputs and their compiled first partial derivatives.
308    pub fn evaluate_with_derivatives<T: FloatLike>(
309        &mut self,
310        parameters: &[T],
311    ) -> Result<Vec<SamplingDualValue<T>>> {
312        self.validate_parameters(parameters)?;
313        let input = self.evaluator_parameters(parameters);
314        let evaluator = self
315            .derivative_evaluator
316            .as_mut()
317            .ok_or_else(|| eyre!("sampling evaluator was built without derivative support"))?;
318        let values = <T as GenericEvaluatorFloat>::get_evaluator(evaluator)(&input);
319        values
320            .into_iter()
321            .map(|value| match value {
322                DualOrNot::Dual(value) => {
323                    if value.values.len() != self.derivative_parameters.len() + 1 {
324                        return Err(eyre!(
325                            "sampling evaluator returned {} dual components, expected {}",
326                            value.values.len(),
327                            self.derivative_parameters.len() + 1
328                        ));
329                    }
330                    Ok(SamplingDualValue {
331                        value: value.values[0].clone(),
332                        derivatives: value.values[1..].to_vec(),
333                    })
334                }
335                DualOrNot::NonDual(_) => Err(eyre!(
336                    "sampling evaluator did not return hyper-dual outputs"
337                )),
338            })
339            .collect()
340    }

342    /// Evaluate a real map and its exact Symbolica dual Jacobian with respect
343    /// to the ordered active parameters. `None` selects every compiled column.
344    ///
345    /// This is the common boundary used by map kernels: the expression tree
346    /// is compiled once during warmup, while coordinates are supplied at
347    /// runtime.  A non-real output is rejected rather than silently projected
348    /// onto its real part, since doing so would invalidate the advertised
349    /// volume Jacobian. Parameters outside the selected columns are held fixed;
350    /// their values can still change the map and its active Jacobian. This uses
351    /// the same compiled active dual program with statically zero identity-seed
352    /// components to supply the requested columns. Prepared-only singular
353    /// derivatives do not contaminate active columns. Singular intermediates
354    /// that depend on active parameters must still have finite derivatives;
355    /// this is not an algebraic extension through their removable singularities.
356    pub fn evaluate_with_real_jacobian<T: FloatLike>(
357        &mut self,
358        parameters: &[T],
359        active_parameters: Option<&[usize]>,
360    ) -> Result<SamplingJacobianEvaluation<T>> {
361        let active_parameters =
362            active_parameters.map_or_else(|| self.derivative_parameters.clone(), <[usize]>::to_vec);
363        if self.output_count != active_parameters.len() {
364            return Err(eyre!(
365                "sampling Jacobian requires a square map ({} outputs, {} active parameters)",
366                self.output_count,
367                active_parameters.len()
368            ));
369        }
370        for (column, &parameter) in active_parameters.iter().enumerate() {
371            if parameter >= self.parameter_count {
372                return Err(eyre!(
373                    "sampling Jacobian parameter index {parameter} is out of range for {} parameters",
374                    self.parameter_count
375                ));
376            }
377            if active_parameters[..column].contains(&parameter) {
378                return Err(eyre!(
379                    "sampling Jacobian active parameters must be unique; index {parameter} is repeated"
380                ));
381            }
382        }
383        let columns = active_parameters
384            .iter()
385            .map(|parameter| {
386                self.derivative_parameters
387                    .iter()
388                    .position(|compiled| compiled == parameter)
389                    .ok_or_else(|| {
390                        eyre!("sampling derivative parameter {parameter} was not compiled")
391                    })
392            })
393            .collect::<Result<Vec<_>>>()?;
394        let dual_values = self.evaluate_with_derivatives(parameters)?;
395        let mut values = Vec::with_capacity(dual_values.len());
396        let mut derivatives = Vec::with_capacity(dual_values.len());
397        for (output_index, value) in dual_values.into_iter().enumerate() {
398            values.push(Self::real_component(
399                value.value,
400                "value",
401                output_index,
402                None,
403            )?);
404            let row = columns
405                .iter()
406                .zip(&active_parameters)
407                .map(|(&column, &parameter_index)| {
408                    Self::real_component(
409                        value.derivatives[column].clone(),
410                        "derivative",
411                        output_index,
412                        Some(parameter_index),
413                    )
414                })
415                .collect::<Result<Vec<_>>>()?;
416            derivatives.push(row);
417        }
418        let determinant = signed_determinant(&derivatives)?;
419        Ok(SamplingJacobianEvaluation {
420            values,
421            derivatives,
422            determinant,
423        })
424    }

426    fn validate_parameters<T: FloatLike>(&self, parameters: &[T]) -> Result<()> {
427        if parameters.len() != self.parameter_count {
428            return Err(eyre!(
429                "sampling evaluator received {} parameters, expected {}",
430                parameters.len(),
431                self.parameter_count
432            ));
433        }
434        if parameters.iter().any(|value| !value.is_finite()) {
435            return Err(eyre!("sampling evaluator parameters must be finite"));
436        }
437        Ok(())
438    }

440    /// Symbolica vectorizes a dual expression by storing one hyper-dual block
441    /// for every scalar parameter. Keep this flattening local to the shared
442    /// evaluator wrapper so eager values and dual Jacobians cannot disagree on
443    /// the input layout.
444    fn evaluator_parameters<T: FloatLike>(&self, parameters: &[T]) -> Vec<Complex<F<T>>> {
445        let dual_width = self.derivative_parameters.len() + 1;
446        let mut input = Vec::with_capacity(self.parameter_count * dual_width);
447        for (parameter_index, value) in parameters.iter().cloned().enumerate() {
448            let value = F(value);
449            input.push(Complex::new_re(value.clone()));
450            for &derivative_parameter in &self.derivative_parameters {
451                input.push(Complex::new_re(
452                    if parameter_index == derivative_parameter {
453                        value.one()
454                    } else {
455                        value.zero()
456                    },
457                ));
458            }
459        }
460        debug_assert_eq!(input.len(), self.parameter_count * dual_width);
461        input
462    }

464    pub(crate) fn real_component<T: FloatLike>(
465        value: Complex<F<T>>,
466        role: &str,
467        output_index: usize,
468        parameter_index: Option<usize>,
469    ) -> Result<T> {
470        // Expressions in the map language are required to be real. Permit only
471        // roundoff-level imaginary residue from complex constants, and retain the
472        // diagnostic location when a genuinely complex map is supplied. Scale
473        // relative to the native output, including values smaller than f64 can hold.
474        let tolerance = value.re.epsilon() * value.re.from_usize(512) * value.re.abs();
475        if !value.re.0.is_finite() || !value.im.0.is_finite() {
476            return Err(SamplingEvaluationError::Unrepresentable {
477                operation: "eager sampling expression",
478                detail: format!(
479                    "sampling {role} {output_index}{} is non-finite: {value:?}",
480                    parameter_index
481                        .map(|index| format!("/{index}"))
482                        .unwrap_or_default()
483                ),
484            }
485            .into());
486        }
487        if value.im.abs() > tolerance {
488            return Err(eyre!(
489                "sampling {role} {output_index}{} has non-negligible imaginary part {}",
490                parameter_index
491                    .map(|index| format!("/{index}"))
492                    .unwrap_or_default(),
493                value.im
494            ));
495        }
496        Ok(value.re.0)
497    }
498}

500fn signed_determinant<T: FloatLike>(matrix: &[Vec<T>]) -> Result<T> {
501    let dimension = matrix.len();
502    if dimension == 0 || matrix.iter().any(|row| row.len() != dimension) {
503        return Err(eyre!(
504            "sampling Jacobian determinant requires a non-empty square matrix"
505        ));
506    }
507    let mut work = matrix
508        .iter()
509        .map(|row| row.iter().cloned().map(F).collect::<Vec<_>>())
510        .collect::<Vec<_>>();
511    if work.iter().flatten().any(|value| !value.0.is_finite()) {
512        return Err(SamplingEvaluationError::Unrepresentable {
513            operation: "sampling Jacobian determinant",
514            detail: "matrix contains non-finite entries".into(),
515        }
516        .into());
517    }
518    let mut determinant = work[0][0].one();
519    for pivot in 0..dimension {
520        if (pivot..dimension).any(|row| !work[row][pivot].0.is_finite()) {
521            return Err(SamplingEvaluationError::Unrepresentable {
522                operation: "sampling Jacobian determinant",
523                detail: "elimination produced a non-finite pivot".into(),
524            }
525            .into());
526        }
527        let (pivot_row, pivot_value) = (pivot..dimension)
528            .map(|row| (row, work[row][pivot].abs()))
529            .max_by(|(_, left), (_, right)| left.partial_cmp(right).expect("finite pivots"))
530            .expect("non-empty pivot range");
531        if pivot_value == pivot_value.zero() {
532            return Ok(pivot_value.0);
533        }
534        if pivot_row != pivot {
535            work.swap(pivot_row, pivot);
536            determinant = -determinant;
537        }
538        let diagonal = work[pivot][pivot].clone();
539        determinant *= &diagonal;
540        if !determinant.0.is_finite() || determinant == determinant.zero() {
541            return Err(SamplingEvaluationError::Unrepresentable {
542                operation: "sampling Jacobian determinant",
543                detail: "product of nonzero pivots overflowed or underflowed".into(),
544            }
545            .into());
546        }
547        let (completed, pending) = work.split_at_mut(pivot + 1);
548        let pivot_values = &completed[pivot];
549        for row in pending {
550            let factor = &row[pivot] / &diagonal;
551            for (entry, pivot_entry) in row.iter_mut().zip(pivot_values).skip(pivot + 1) {
552                *entry -= &factor * pivot_entry;
553            }
554        }
555    }
556    Ok(determinant.0)
557}

559fn first_derivative_shape(parameter_count: usize) -> Vec<Vec<usize>> {
560    let mut shape = Vec::with_capacity(parameter_count + 1);
561    shape.push(vec![0; parameter_count]);
562    for derivative in 0..parameter_count {
563        let mut order = vec![0; parameter_count];
564        order[derivative] = 1;
565        shape.push(order);
566    }
567    shape
568}

897        test_initialise().unwrap();
898        let parameters = ["u", "m", "v", "s", "w"]
899            .map(|name| Atom::var(symbol!(format!("sampling_reduced::{name}"))));

905                    .err()
906                    .expect("invalid derivative columns must fail before compilation");
907            assert!(error.to_string().contains(message));
908        }
909        let mut full = SamplingExpressionEvaluator::new(
910            expressions.clone(),
911            parameters.clone(),
912            &[0, 1, 2, 3, 4],
913        )
914        .unwrap();
915        let mut evaluator =
916            SamplingExpressionEvaluator::new(expressions, parameters, &[0, 2, 4]).unwrap();
917        assert!(evaluator.evaluator.dual_shape.is_none());
918        assert_eq!(
919            evaluator
920                .derivative_evaluator
921                .as_ref()
922                .unwrap()
923                .dual_shape
924                .as_ref()
925                .unwrap()
926                .len(),
927            4
928        );
929
930        fn check<T: FloatLike>(
931            evaluator: &mut SamplingExpressionEvaluator,
932            full: &mut SamplingExpressionEvaluator,
933        ) {
934            let one = F::<T>::default().one();
935            let u = &one / one.from_usize(4);
936            let v = &one / one.from_usize(2);
937            let w = &one * one.from_usize(3) / one.from_usize(4);
938            let s = one.from_usize(2) + &one / one.from_usize(10).powi(25);
939            let point = [
940                u.0.clone(),
941                one.zero().0,
942                v.0.clone(),
943                s.0.clone(),
944                w.0.clone(),
945            ];
946            // sqrt(m) at m=0 has a finite value and no active tangent. The
947            // scalar path never executes its singular unused m derivative.
948            let scalar = evaluator.evaluate(&point).unwrap();
949            let jacobian = evaluator.evaluate_with_real_jacobian(&point, None).unwrap();
950            let previous = full
951                .evaluate_with_real_jacobian(&point, Some(&[0, 2, 4]))
952                .unwrap();
953            assert_eq!(jacobian, previous);
954            for (scalar, previous) in scalar.iter().zip(&previous.values) {
955                assert_eq!(&scalar.re.0, previous);
956            }
957            let tolerance = one.epsilon() * one.from_usize(2048);
958            for ((scalar, value), expected) in
959                scalar
960                    .iter()
961                    .zip(&jacobian.values)
962                    .zip([&s * &u, &v + u.square(), &w + &u * &v])
963            {
964                assert_eq!(scalar.im, one.zero());
965                assert!((&scalar.re - &expected).abs() < tolerance);
966                assert!((F(value.clone()) - expected).abs() < tolerance);
967            }
968            assert!((F(jacobian.determinant) - &s).abs() < tolerance);
969            let reordered = evaluator
970                .evaluate_with_real_jacobian(&point, Some(&[4, 2, 0]))
971                .unwrap();
972            assert!((F(reordered.determinant) + &s).abs() < tolerance);
973            let error = evaluator
974                .evaluate_with_real_jacobian(&point, Some(&[0, 1, 4]))
975                .unwrap_err();
976            assert!(error.to_string().contains("was not compiled"));
977        }
978        check::<f64>(&mut evaluator, &mut full);
979        check::<crate::utils::QuadFloat>(&mut evaluator, &mut full);
980        check::<crate::utils::ArbPrec>(&mut evaluator, &mut full);
981
982        // Independent Cartesian differences use only the scalar program and
983        // vary the three compiled input columns, keeping m=0 and s fixed.
984        let point = [0.25, 0.0, 0.5, 2.0, 0.75];
985        let jacobian = evaluator.evaluate_with_real_jacobian(&point, None).unwrap();
986        let step = 1e-5;
987        for (column, parameter) in [0, 2, 4].into_iter().enumerate() {
988            let mut plus = point;
989            let mut minus = point;
990            plus[parameter] += step;
991            minus[parameter] -= step;
992            let plus = evaluator.evaluate(&plus).unwrap();
993            let minus = evaluator.evaluate(&minus).unwrap();
994            for (row, (plus, minus)) in plus.iter().zip(minus).enumerate() {
995                let difference = (plus.re.0 - minus.re.0) / (2.0 * step);
996                assert!((difference - jacobian.derivatives[row][column]).abs() < 1e-9);
997            }
998        }
999    }
