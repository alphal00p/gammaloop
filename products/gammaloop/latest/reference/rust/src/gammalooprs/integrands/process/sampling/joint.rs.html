83// Each endpoint of the conservative box range is itself enclosed. This keeps
84// interval dependency/overestimation distinct from MPFR rounding uncertainty:
85// only a certified sign of the mathematical lower endpoint changes the disk.
86// This arithmetic is deliberately private to the nine rational predicates.
87#[derive(Clone, Debug)]
88struct JointRange {
89    lower: [Float; 2],
90    upper: [Float; 2],
91}
92
93impl JointRange {
94    const PRECISION: u32 = 2048;
95    const PRECISIONS: [u32; 5] = [128, 256, 512, 1024, Self::PRECISION];
96
97    fn native<T: FloatLike>(value: &T, precision: u32) -> Self {
98        let (lo, hi) = value.mpfr_enclosure(precision);
99        Self {
100            lower: [lo.clone(), hi.clone()],
101            upper: [lo, hi],
102        }
103    }
104
105    fn integer(value: i32, precision: u32) -> Self {
106        let value = Float::with_val(precision, value);
107        Self {
108            lower: [value.clone(), value.clone()],
109            upper: [value.clone(), value],
110        }
111    }
112
113    fn bounds(lower: Self, upper: Self) -> Self {
114        Self {
115            lower: lower.lower,
116            upper: upper.upper,
117        }
118    }
119
120    fn add_bound(a: &[Float; 2], b: &[Float; 2]) -> [Float; 2] {
121        [
122            Float::with_val_round(a[0].prec(), &a[0] + &b[0], Round::Down).0,
123            Float::with_val_round(a[0].prec(), &a[1] + &b[1], Round::Up).0,
124        ]
125    }
126
127    fn mul_bound(a: &[Float; 2], b: &[Float; 2]) -> [Float; 2] {
128        let products = a.iter().flat_map(|x| b.iter().map(move |y| (x, y)));
129        let mut lower = Float::with_val(a[0].prec(), Special::Infinity);
130        let mut upper = -lower.clone();
131        for (x, y) in products {
132            lower = lower.min(&Float::with_val_round(a[0].prec(), x * y, Round::Down).0);
133            upper = upper.max(&Float::with_val_round(a[0].prec(), x * y, Round::Up).0);
134        }
135        [lower, upper]
136    }
137
138    fn add(&self, rhs: &Self) -> Self {
139        Self {
140            lower: Self::add_bound(&self.lower, &rhs.lower),
141            upper: Self::add_bound(&self.upper, &rhs.upper),
142        }
143    }
144
145    fn neg(&self) -> Self {
146        Self {
147            lower: [-self.upper[1].clone(), -self.upper[0].clone()],
148            upper: [-self.lower[1].clone(), -self.lower[0].clone()],
149        }
150    }
151
152    fn sub(&self, rhs: &Self) -> Self {
153        self.add(&rhs.neg())
154    }
155
156    fn mul(&self, rhs: &Self) -> Self {
157        let products = [&self.lower, &self.upper].into_iter().flat_map(|a| {
158            [&rhs.lower, &rhs.upper]
159                .into_iter()
160                .map(move |b| Self::mul_bound(a, b))
161        });
162        let infinity = Float::with_val(self.lower[0].prec(), Special::Infinity);
163        let mut lower = [infinity.clone(), infinity.clone()];
164        let mut upper = [-infinity.clone(), -infinity];
165        for product in products {
166            for index in 0..2 {
167                lower[index] = lower[index].clone().min(&product[index]);
168                upper[index] = upper[index].clone().max(&product[index]);
169            }
170        }
171        Self { lower, upper }
172    }
173
174    fn square(&self) -> Self {
175        // The dependency-aware lower square avoids artificially negative
176        // lower bounds when a normal coordinate crosses zero.
177        let zero = Float::with_val(self.lower[0].prec(), 0);
178        let a = Self::mul_bound(&self.lower, &self.lower);
179        let b = Self::mul_bound(&self.upper, &self.upper);
180        let upper = [a[0].clone().max(&b[0]), a[1].clone().max(&b[1])];
181        let lower = if self.lower[1] <= 0 && self.upper[0] >= 0 {
182            [zero.clone(), zero]
183        } else if self.lower[0] > 0 || self.upper[1] < 0 {
184            [a[0].clone().min(&b[0]), a[1].clone().min(&b[1])]
185        } else {
186            // Only arithmetic uncertainty remains about crossing zero.
187            [zero, a[1].clone().min(&b[1])]
188        };
189        Self { lower, upper }
190    }
191
192    fn reciprocal_positive(&self) -> Result<Self> {
193        if self.lower[0] <= 0 {
194            return Err(SamplingEvaluationError::UncertainGeometry {
195                detail: "joint certificate denominator is not certified positive".into(),
196            }
197            .into());
198        }
199        let reciprocal = |bound: &[Float; 2]| {
200            [
201                Float::with_val_round(self.lower[0].prec(), 1 / &bound[1], Round::Down).0,
202                Float::with_val_round(self.lower[0].prec(), 1 / &bound[0], Round::Up).0,
203            ]
204        };
205        Ok(Self {
206            lower: reciprocal(&self.upper),
207            upper: reciprocal(&self.lower),
208        })
209    }
210
211    /// Sign of the mathematical conservative lower endpoint. A negative
212    /// endpoint declines a box; it does not prove physical absence.
213    fn lower_positive(&self) -> Result<bool> {
214        if self.lower[0] > 0 {
215            return Ok(true);
216        }
217        if self.lower[1] <= 0 {
218            return Ok(false);
219        }
220        Err(SamplingEvaluationError::UncertainGeometry {
221            detail: "joint certificate lower-endpoint sign is unresolved by directed arithmetic"
222                .into(),
223        }
224        .into())
225    }
226
227    fn sqrt_point(&self) -> Result<Self> {
228        if self.lower[0] < 0 {
229            return Err(SamplingEvaluationError::UncertainGeometry {
230                detail: "joint support energy square is not certified nonnegative".into(),
231            }
232            .into());
233        }
234        let mut lo = self.lower[0].clone();
235        let mut hi = self.upper[1].clone();
236        lo.sqrt_round(Round::Down);
237        hi.sqrt_round(Round::Up);
238        Ok(Self {
239            lower: [lo.clone(), hi.clone()],
240            upper: [lo, hi],
241        })
242    }
243}

245impl<T: FloatLike> SharedEnergyJointGeometry<T> {
246    fn validate(&self) -> Result<()> {
247        if self
248            .shifts
249            .iter()
250            .flatten()
251            .chain(&self.masses)
252            .chain(&self.energy_sums)
253            .any(|x| !x.is_finite())
254        {
255            return Err(SamplingEvaluationError::Unrepresentable {
256                operation: "joint geometry",
257                detail: "non-finite prepared parameter".into(),
258            }
259            .into());
260        }
261        if self.masses.iter().any(|m| m < &m.zero()) {
262            return Err(eyre!("joint energy masses must be nonnegative"));
263        }
264        Ok(())
265    }
266
267    /// Prepare radius-independent directed ranges for one native geometry.
268    /// Only this call's radius trials share them; each new conditional point
269    /// prepares its own ranges. Keep the original arithmetic association and
270    /// lower-endpoint predicate order when evaluating each candidate disk.
271    fn prepare_certificate(
272        &self,
273        scale: &T,
274        precision: u32,
275    ) -> Result<impl Fn(&T) -> Result<bool> + use<T>> {
276        let a = self.shifts[0]
277            .each_ref()
278            .map(|value| JointRange::native(value, precision));
279        let b = self.shifts[1]
280            .each_ref()
281            .map(|value| JointRange::native(value, precision));
282        let m = self
283            .masses
284            .each_ref()
285            .map(|value| JointRange::native(value, precision));
286        let dot = |a: &[JointRange; 3], b: &[JointRange; 3]| {
287            a.iter()
288                .zip(b)
289                .fold(JointRange::integer(0, precision), |sum, (a, b)| {
290                    sum.add(&a.mul(b))
291                })
292        };
293        let aa = dot(&a, &a);
294        let ab = dot(&a, &b);
295        let bb = dot(&b, &b);
296        let gram = aa.mul(&bb).sub(&ab.square());
297        // A nonpositive Gram declines every candidate before checking any
298        // denominator. The caller still performs all native radius halvings
299        // and their underflow checks before recording the ordinary decision.
300        let fixed = if gram.lower_positive()? {
301            let inverse_scale = JointRange::native(scale, precision).reciprocal_positive()?;
302            let energy_sums = self
303                .energy_sums
304                .each_ref()
305                .map(|value| JointRange::native(value, precision));
306            let inverse_two = JointRange::integer(2, precision).reciprocal_positive()?;
307            let mass_squares = m.each_ref().map(JointRange::square);
308            let inverse_gram = gram.reciprocal_positive()?;
309            Some((
310                energy_sums,
311                mass_squares,
312                [inverse_scale, inverse_two, inverse_gram],
313                JointRange::integer(1, precision),
314            ))
315        } else {
316            None
317        };
318        Ok(move |radius: &T| {
319            let Some((energy_sums, mass_squares, [inverse_scale, inverse_two, inverse_gram], one)) =
320                &fixed
321            else {
322                return Ok(false);
323            };
324            let r = JointRange::native(radius, precision);
325            let z = r.mul(inverse_scale);
326            let s = energy_sums[0].add(&JointRange::bounds(r.neg(), r));
327            let t = energy_sums[1].add(&JointRange::bounds(z.neg(), z));
328            let da = s
329                .square()
330                .add(&mass_squares[0])
331                .sub(&mass_squares[1])
332                .sub(&aa)
333                .mul(inverse_two);
334            let db = t
335                .square()
336                .add(&mass_squares[0])
337                .sub(&mass_squares[2])
338                .sub(&bb)
339                .mul(inverse_two);
340            let d = [
341                bb.mul(&da).sub(&ab.mul(&db)).mul(inverse_gram),
342                aa.mul(&db).sub(&ab.mul(&da)).mul(inverse_gram),
343            ];
344            let e = [
345                ab.mul(&t).sub(&bb.mul(&s)).mul(inverse_gram),
346                ab.mul(&s).sub(&aa.mul(&t)).mul(inverse_gram),
347            ];
348            let plane_dot = |x: &[JointRange; 2], y: &[JointRange; 2]| {
349                aa.mul(&x[0])
350                    .mul(&y[0])
351                    .add(&ab.mul(&x[0].mul(&y[1]).add(&x[1].mul(&y[0]))))
352                    .add(&bb.mul(&x[1]).mul(&y[1]))
353            };
354            let k = plane_dot(&e, &e).sub(one);
355            let b = plane_dot(&d, &e);
356            let c = mass_squares[0].add(&plane_dot(&d, &d));
357            let discriminant = b.square().sub(&k.mul(&c));
358            let limits = [
359                b.neg().sub(&k.mul(&m[0])),
360                k.mul(&s.sub(&m[1])).add(&b),
361                k.mul(&t.sub(&m[2])).add(&b),
362            ];
363            for predicate in [k, discriminant.clone()].into_iter().chain(
364                limits
365                    .into_iter()
366                    .flat_map(|l| [l.clone(), l.square().sub(&discriminant)]),
367            ) {
368                if !predicate.lower_positive()? {
369                    return Ok(false);
370                }
371            }
372            Ok(true)
373        })
374    }

376    fn residuals(&self, point: &[T]) -> ([F<T>; 2], [F<T>; 3]) {
377        let zero = F(self.masses[0].zero());
378        let energies = std::array::from_fn(|index| {
379            let square = point.iter().enumerate().fold(
380                F(self.masses[index].clone()).square(),
381                |sum, (component, x)| {
382                    let shifted = F(x.clone())
383                        + if index == 0 {
384                            zero.clone()
385                        } else {
386                            F(self.shifts[index - 1][component].clone())
387                        };
388                    sum + shifted.square()
389                },
390            );
391            square.sqrt()
392        });
393        (
394            [
395                &energies[0] + &energies[1] - F(self.energy_sums[0].clone()),
396                &energies[0] + &energies[2] - F(self.energy_sums[1].clone()),
397            ],
398            energies,
399        )
400    }
401
402    fn support_relation(&self, point: &[T], radius: &T, scale: &T) -> Result<std::cmp::Ordering> {
403        // Refine only arithmetic at the identical point and disk. An unresolved
404        // boundary cannot decline support or change the native source law.
405        let evaluate = |precision| -> Result<Option<std::cmp::Ordering>> {
406            let energies = (0..3)
407                .map(|index| {
408                    let square = point.iter().enumerate().fold(
409                        JointRange::native(&self.masses[index], precision).square(),
410                        |sum, (component, x)| {
411                            let x = JointRange::native(x, precision);
412                            let shifted = if index == 0 {
413                                x
414                            } else {
415                                x.add(&JointRange::native(
416                                    &self.shifts[index - 1][component],
417                                    precision,
418                                ))
419                            };
420                            sum.add(&shifted.square())
421                        },
422                    );
423                    square.sqrt_point()
424                })
425                .collect::<Result<Vec<_>>>()?;
426            let h = energies[0]
427                .add(&energies[1])
428                .sub(&JointRange::native(&self.energy_sums[0], precision));
429            let z = energies[0]
430                .add(&energies[2])
431                .sub(&JointRange::native(&self.energy_sums[1], precision));
432            let difference = h
433                .square()
434                .add(&z.mul(&JointRange::native(scale, precision)).square())
435                .sub(&JointRange::native(radius, precision).square());
436            if difference.lower[0] > 0 {
437                return Ok(Some(std::cmp::Ordering::Greater));
438            }
439            if difference.upper[1] < 0 {
440                return Ok(Some(std::cmp::Ordering::Less));
441            }
442            Ok(None)
443        };
444        for precision in JointRange::PRECISIONS {
445            match evaluate(precision) {
446                Ok(Some(relation)) => return Ok(relation),
447                Ok(None) => {}
448                Err(error)
449                    if precision != JointRange::PRECISION
450                        && matches!(
451                            error.downcast_ref::<SamplingEvaluationError>(),
452                            Some(SamplingEvaluationError::UncertainGeometry { .. })
453                        ) => {}
454                Err(error) => return Err(error),
455            }
456        }
457        Err(SamplingEvaluationError::UncertainGeometry {
458            detail: "joint inverse lies on an uncertified normal-disk boundary".into(),
459        }
460        .into())
461    }
462}

464impl<T: FloatLike> SharedEnergyJointMap<T> {
465    pub fn new(
466        geometry: SharedEnergyJointGeometryEvaluator<T>,
467        context_dimension: usize,
468        max_radius: T,
469        normal_scale: T,
470        beta: f64,
471        program: SamplingExpressionEvaluator,
472    ) -> Result<Self> {
473        if [&max_radius, &normal_scale]
474            .iter()
475            .any(|x| !x.is_finite() || **x <= x.zero())
476        {
477            return Err(eyre!(
478                "joint normal radius and scale must be positive and finite"
479            ));
480        }
481        if program.parameter_count() != 17
482            || program.output_count() != 3
483            || program.derivative_parameters() != [0, 1, 2]
484        {
485            return Err(eyre!(
486                "joint chart requires its three-output, seventeen-parameter eager program with active columns [0, 1, 2]"
487            ));
488        }
489        let fallback = SurfaceRadialMap::absent(3, vec![max_radius.zero(); 3], beta, 1.0)?;
490        Ok(Self {
491            geometry,
492            context_dimension,
493            max_radius,
494            normal_scale,
495            fallback,
496            program: Mutex::new(program),
497        })
498    }
499
500    /// Compile the neutral map once; native bindings and worker clones reuse
501    /// these programs. Prepared inputs occupy inactive derivative columns.
502    pub fn compile_program() -> Result<SamplingExpressionEvaluator> {
503        Self::compile_program_with_derivatives(&[0, 1, 2])
504    }
505
506    fn compile_program_with_derivatives(
507        derivative_parameters: &[usize],
508    ) -> Result<SamplingExpressionEvaluator> {
509        let names = [
510            "jr", "jt", "jp", "rho", "alpha", "tau", "ax", "ay", "az", "bx", "by", "bz", "m0",
511            "m1", "m2", "c1", "c2",
512        ];
513        let p = names
514            .iter()
515            .map(|name| try_parse!(*name).map_err(|e| eyre!(e)))
516            .collect::<Result<Vec<_>>>()?;
517        let a = &p[6..9];
518        let b = &p[9..12];
519        let dot = |a: &[Atom], b: &[Atom]| {
520            a.iter()
521                .zip(b)
522                .fold(Atom::num(0), |sum, (a, b)| sum + a * b)
523        };
524        let anorm = dot(a, a).sqrt();
525        let ea = a.iter().map(|x| x / &anorm).collect::<Vec<_>>();
526        let along = dot(b, &ea);
527        let bp = b
528            .iter()
529            .zip(&ea)
530            .map(|(b, ea)| b - &along * ea)
531            .collect::<Vec<_>>();
532        let bnorm = dot(&bp, &bp).sqrt();
533        let eb = bp.iter().map(|x| x / &bnorm).collect::<Vec<_>>();
534        let normal = [
535            &ea[1] * &eb[2] - &ea[2] * &eb[1],
536            &ea[2] * &eb[0] - &ea[0] * &eb[2],
537            &ea[0] * &eb[1] - &ea[1] * &eb[0],
538        ];
539        let r = &p[3] * &p[0];
540        let theta = &p[5] * &p[1];
541        let phi = &p[5] * &p[2];
542        let s = &p[15] + &r * theta.cos();
543        let t = &p[16] + &r * theta.sin() / &p[4];
544        let d1 = (&s * &s + &p[12] * &p[12] - &p[13] * &p[13] - &anorm * &anorm)
545            / (Atom::num(2) * &anorm);
546        let e1 = -&s / &anorm;
547        let d2 = ((&t * &t + &p[12] * &p[12] - &p[14] * &p[14] - dot(b, b)) / Atom::num(2)
548            - &along * &d1)
549            / &bnorm;
550        let e2 = (-&t - &along * &e1) / &bnorm;
551        let k = &e1 * &e1 + &e2 * &e2 - Atom::num(1);
552        let big_b = &d1 * &e1 + &d2 * &e2;
553        let c = &p[12] * &p[12] + &d1 * &d1 + &d2 * &d2;
554        let delta = (&big_b * &big_b - &k * c).sqrt() / &k;
555        let u = -&big_b / &k + &delta * phi.cos();
556        let w = k.sqrt() * delta * phi.sin();
557        let outputs =
558            (0..3).map(|i| (&d1 + &e1 * &u) * &ea[i] + (&d2 + &e2 * &u) * &eb[i] + &w * &normal[i]);
559        SamplingExpressionEvaluator::new(outputs, p.clone(), derivative_parameters)
560    }

562    fn prepare(
563        &self,
564        context: &mut SamplingMapContext<'_, T>,
565    ) -> Result<(SharedEnergyJointGeometry<T>, Option<T>)> {
566        if context.previous.len() != self.context_dimension {
567            return Err(eyre!(
568                "joint chart context has dimension {}, expected {}",
569                context.previous.len(),
570                self.context_dimension
571            ));
572        }
573        if context.previous.iter().any(|x| !x.is_finite()) {
574            return Err(SamplingEvaluationError::Unrepresentable {
575                operation: "joint context",
576                detail: "non-finite previous coordinates".into(),
577            }
578            .into());
579        }
580        let geometry = (self.geometry)(context.previous)?;
581        geometry.validate()?;
582        let mut radius = F(self.max_radius.clone());
583        let two = radius.from_usize(2);
584        let mut compact = None;
585        {
586            let mut precisions = JointRange::PRECISIONS.into_iter().peekable();
587            let mut certificate = geometry.prepare_certificate(
588                &self.normal_scale,
589                *precisions
590                    .peek()
591                    .expect("joint precision ladder is nonempty"),
592            );
593            // A deterministic complement-only policy, not a claim of absence.
594            // Refine arithmetic ambiguity at this same native radius before
595            // halving; exhausted geometric boxes use the normalized ordinary map.
596            // Fixed terms live only in this preparation. Once refinement is
597            // needed, later radius trials reuse that more precise certificate.
598            for dyadic_exponent in 0..32 {
599                let admitted = loop {
600                    let precision = *precisions.peek().expect("joint precision ceiling retained");
601                    match certificate {
602                        Ok(prepared) => {
603                            let result = prepared(&radius.0);
604                            certificate = Ok(prepared);
605                            match result {
606                                Err(error)
607                                    if precision != JointRange::PRECISION
608                                        && matches!(
609                                            error.downcast_ref::<SamplingEvaluationError>(),
610                                            Some(SamplingEvaluationError::UncertainGeometry { .. })
611                                        ) => {}
612                                result => break result?,
613                            }
614                        }
615                        Err(error)
616                            if precision != JointRange::PRECISION
617                                && matches!(
618                                    error.downcast_ref::<SamplingEvaluationError>(),
619                                    Some(SamplingEvaluationError::UncertainGeometry { .. })
620                                ) => {}
621                        Err(error) => return Err(error),
622                    }
623                    precisions.next();
624                    certificate = geometry.prepare_certificate(
625                        &self.normal_scale,
626                        *precisions.peek().expect("joint precision ceiling retained"),
627                    );
628                };
629                if admitted {
630                    compact = Some((dyadic_exponent, radius.0));
631                    break;
632                }
633                radius /= &two;
634                if !radius.is_finite() || radius <= radius.zero() {
635                    return Err(SamplingEvaluationError::Unrepresentable {
636                        operation: "joint disk",
637                        detail: "dyadic radius underflow".into(),
638                    }
639                    .into());
640                }
641            }
642        }
643        let decision = compact.as_ref().map_or(
644            SamplingProposalDecision::Ordinary,
645            |(dyadic_exponent, _)| SamplingProposalDecision::Compact {
646                dyadic_exponent: *dyadic_exponent,
647            },
648        );
649        // Revalidate the complete native choice, not merely whether a stored
650        // smaller disk would also be valid. Different valid radii are different laws.
651        context.validate_decision(decision).wrap_err_with(|| format!(
652            "joint native decision {decision:?}, precision {:?}, max radius {:?}, normal scale {:?}, represented geometry {geometry:?}",
653            T::sampling_precision(), self.max_radius, self.normal_scale,
654        ))?;
655        Ok((geometry, compact.map(|(_, radius)| radius)))
656    }
657
658    fn parameters(
659        &self,
660        geometry: &SharedEnergyJointGeometry<T>,
661        radius: &T,
662        coordinates: &[T],
663    ) -> Vec<T> {
664        coordinates
665            .iter()
666            .cloned()
667            .chain([radius.clone(), self.normal_scale.clone(), radius.TAU()])
668            .chain(geometry.shifts.iter().flatten().cloned())
669            .chain(geometry.masses.iter().cloned())
670            .chain(geometry.energy_sums.iter().cloned())
671            .collect()
672    }

675// Native inverse geometry. The production forward determinant comes from the
676// Symbolica map above; the inverse density uses the original supplied point's
677// energies and residual radius, rather than a reconstructed forward point.
678struct JointCircle<T: FloatLike> {
679    axes: [[F<T>; 3]; 3],
680    k: F<T>,
681    center: F<T>,
682    delta: F<T>,
683    cross_norm: F<T>,
684}
685
686impl<T: FloatLike> JointCircle<T> {
687    fn new(geometry: &SharedEnergyJointGeometry<T>, residuals: &[F<T>; 2]) -> Result<Self> {
688        let a = geometry.shifts[0].each_ref().map(|x| F(x.clone()));
689        let b = geometry.shifts[1].each_ref().map(|x| F(x.clone()));
690        let zero = a[0].zero();
691        let dot = |a: &[F<T>; 3], b: &[F<T>; 3]| {
692            a.iter()
693                .zip(b)
694                .fold(zero.clone(), |sum, (a, b)| sum + a * b)
695        };
696        let anorm = dot(&a, &a).sqrt();
697        let ea = a.each_ref().map(|x| x / &anorm);
698        let along = dot(&b, &ea);
699        let bp = std::array::from_fn(|i| &b[i] - &along * &ea[i]);
700        let bnorm = dot(&bp, &bp).sqrt();
701        let eb = bp.each_ref().map(|x| x / &bnorm);
702        let n = [
703            &ea[1] * &eb[2] - &ea[2] * &eb[1],
704            &ea[2] * &eb[0] - &ea[0] * &eb[2],
705            &ea[0] * &eb[1] - &ea[1] * &eb[0],
706        ];
707        let s = F(geometry.energy_sums[0].clone()) + &residuals[0];
708        let t = F(geometry.energy_sums[1].clone()) + &residuals[1];
709        let m = geometry.masses.each_ref().map(|x| F(x.clone()).square());
710        let two = s.from_usize(2);
711        let d1 = (s.square() + &m[0] - &m[1] - anorm.square()) / (&two * &anorm);
712        let e1 = -&s / &anorm;
713        let d2 = ((t.square() + &m[0] - &m[2] - dot(&b, &b)) / &two - &along * &d1) / &bnorm;
714        let e2 = (-&t - &along * &e1) / &bnorm;
715        let k = e1.square() + e2.square() - s.one();
716        let big_b = &d1 * &e1 + &d2 * &e2;
717        let discriminant = big_b.square() - &k * (&m[0] + d1.square() + d2.square());
718        let center = -big_b / &k;
719        let delta = discriminant.sqrt() / &k;
720        let cross_norm = anorm * bnorm;
721        if [&k, &center, &delta, &cross_norm]
722            .into_iter()
723            .any(|x| !x.is_finite())
724            || k <= zero
725            || delta <= zero
726            || cross_norm <= zero
727        {
728            return Err(SamplingEvaluationError::Unrepresentable {
729                operation: "joint circle",
730                detail: "certified circle is not representable in the active native precision"
731                    .into(),
732            }
733            .into());
734        }
735        Ok(Self {
736            axes: [ea, eb, n],
737            k,
738            center,
739            delta,
740            cross_norm,
741        })
742    }
743}

745impl<T: FloatLike> SamplingMapComponent<T> for SharedEnergyJointMap<T> {
746    fn dimensions(&self) -> usize {
747        3
748    }
749    fn output_dimensions(&self) -> usize {
750        3
751    }
752    fn name(&self) -> &'static str {
753        "shared_energy_joint"
754    }
755    fn contract(&self) -> SamplingMapContract {
756        SamplingMapContract {
757            support: SamplingSupport::Restricted,
758            requires_context: self.context_dimension > 0,
759            requires_proposal_policy: true,
760            jacobian: SamplingJacobian::ExactForward,
761        }
762    }
763
764    fn forward(
765        &self,
766        coordinates: &[T],
767        context: &mut SamplingMapContext<'_, T>,
768    ) -> Result<SamplingMapEvaluation<T>> {
769        if coordinates.len() != 3
770            || coordinates.iter().enumerate().any(|(i, x)| {
771                !x.is_finite() || x < &x.zero() || x >= &x.one() || (i == 0 && x == &x.zero())
772            })
773        {
774            return Err(eyre!(
775                "joint chart requires radial coordinate in (0,1) and half-open angular coordinates in [0,1)"
776            ));
777        }
778        let (geometry, radius) = self.prepare(context)?;
779        let Some(radius) = radius else {
780            // The full-circle seams are regular in the focused chart. The
781            // ordinary polar fallback has a different, measure-zero angular
782            // boundary there; preserve that law rather than remapping it.
783            if coordinates[1..].iter().any(|x| x == &x.zero()) {
784                return Err(SamplingEvaluationError::Unrepresentable {
785                    operation: "joint ordinary fallback angular boundary",
786                    detail: "exact angular zero lies on the ordinary chart boundary".into(),
787                }
788                .into());
789            }
790            let mut output = SamplingMapComponent::forward(
791                &self.fallback,
792                coordinates,
793                &mut context.reborrow(&[], None),
794            )?;
795            output
796                .diagnostics
797                .push("joint: normalized ordinary fallback; disk policy declined".into());
798            return Ok(output);
799        };
800        let output = self
801            .program
802            .lock()
803            .map_err(|_| eyre!("joint evaluator poisoned"))?
804            .evaluate_with_real_jacobian(
805                &self.parameters(&geometry, &radius, coordinates),
806                Some(&[0, 1, 2]),
807            )?;
808        let jacobian = F(output.absolute_determinant());
809        let (residuals, _) = geometry.residuals(&output.values);
810        let r = F(radius.clone()) * F(coordinates[0].clone());
811        let angle = F(radius.TAU()) * F(coordinates[1].clone());
812        let residual = (&residuals[0] - &r * angle.cos())
813            .abs()
814            .max((residuals[1].clone() - r * angle.sin() / F(self.normal_scale.clone())).abs());
815        SamplingMapEvaluation {
816            coordinates: coordinates.to_vec(),
817            point: output.values,
818            inverse_jacobian: jacobian.inv().0,
819            jacobian: jacobian.0,
820            residual: residual.0,
821            support: SamplingSupport::Restricted,
822            diagnostics: vec![format!("joint: certified normal disk rho={radius}")],
823        }
824        .validate("joint forward")
825    }
826
827    fn inverse(
828        &self,
829        point: &[T],
830        context: &mut SamplingMapContext<'_, T>,
831    ) -> Result<Option<SamplingMapEvaluation<T>>> {
832        self.inverse_evaluation(point, context, true)
833    }
834
835    fn inverse_density(
836        &self,
837        point: &[T],
838        context: &mut SamplingMapContext<'_, T>,
839    ) -> Result<Option<T>> {
840        self.inverse_evaluation(point, context, false)
841            .map(|evaluation| evaluation.map(|evaluation| evaluation.inverse_jacobian))
842    }
843}

845impl<T: FloatLike> SharedEnergyJointMap<T> {
846    fn inverse_evaluation(
847        &self,
848        point: &[T],
849        context: &mut SamplingMapContext<'_, T>,
850        reconstruct: bool,
851    ) -> Result<Option<SamplingMapEvaluation<T>>> {
852        if point.len() != 3 {
853            return Err(eyre!("joint chart point must have three components"));
854        }
855        if point.iter().any(|x| !x.is_finite()) {
856            return Err(SamplingEvaluationError::Unrepresentable {
857                operation: "joint inverse point",
858                detail: "non-finite supplied point".into(),
859            }
860            .into());
861        }
862        let (geometry, radius) = self.prepare(context)?;
863        let Some(radius) = radius else {
864            let mut output = SamplingMapComponent::inverse(
865                &self.fallback,
866                point,
867                &mut context.reborrow(&[], None),
868            )?;
869            if let Some(output) = &mut output {
870                output
871                    .diagnostics
872                    .push("joint: normalized ordinary fallback; disk policy declined".into());
873            }
874            return Ok(output);
875        };
876        if geometry.support_relation(point, &radius, &self.normal_scale)?
877            == std::cmp::Ordering::Greater
878        {
879            return Ok(None);
880        }
881        let (residuals, energies) = geometry.residuals(point);
882        let scaled_z = F(self.normal_scale.clone()) * &residuals[1];
883        let r = (residuals[0].square() + scaled_z.square()).sqrt();
884        if !r.is_finite() || r <= r.zero() {
885            return Err(SamplingEvaluationError::Unrepresentable {
886                operation: "joint inverse normal radius",
887                detail: "R=0 is singular; a rounded zero requires native retry".into(),
888            }
889            .into());
890        }
891        let circle = JointCircle::new(&geometry, &residuals)?;
892        let w = point
893            .iter()
894            .zip(&circle.axes[2])
895            .fold(r.zero(), |sum, (x, n)| sum + F(x.clone()) * n);
896        let tau = r.TAU();
897        let wrap = |angle: F<T>| {
898            if angle < angle.zero() {
899                angle + &tau
900            } else {
901                angle
902            }
903        };
904        let theta = wrap(scaled_z.atan2(&residuals[0]));
905        let phi = wrap(
906            (w / (circle.k.sqrt() * &circle.delta))
907                .atan2(&((&energies[0] - &circle.center) / &circle.delta)),
908        );
909        let coordinates = vec![
910            (r.clone() / F(radius.clone())).0,
911            (theta / &tau).0,
912            (phi / &tau).0,
913        ];
914        if coordinates[0] <= radius.zero()
915            || coordinates
916                .iter()
917                .any(|x| !x.is_finite() || x < &x.zero() || x >= &x.one())
918        {
919            return Err(SamplingEvaluationError::Unrepresentable {
920                operation: "joint inverse coordinates",
921                detail: "derived radial or half-open angular coordinate rounded outside its chart"
922                    .into(),
923            }
924            .into());
925        }
926        let jacobian =
927            tau.square() * F(radius.clone()) * &r * &energies[0] * &energies[1] * &energies[2]
928                / (F(self.normal_scale.clone()) * &circle.cross_norm * circle.k.sqrt());
929        // A density-only query discards this private diagnostic field; full
930        // inverses always fill it from the independent forward reconstruction.
931        let mut residual = r.zero();
932        if reconstruct {
933            // Reconstruction is diagnostic only: the density above is evaluated
934            // at the original supplied point, including its original normal radius.
935            let reconstructed = self
936                .program
937                .lock()
938                .map_err(|_| eyre!("joint evaluator poisoned"))?
939                .evaluate(&self.parameters(&geometry, &radius, &coordinates))?;
940            for (actual, expected) in reconstructed.iter().zip(point) {
941                if !actual.re.is_finite() || actual.im != actual.im.zero() {
942                    return Err(SamplingEvaluationError::Unrepresentable {
943                        operation: "joint inverse reconstruction",
944                        detail: "recovered cube does not produce finite real Cartesian coordinates"
945                            .into(),
946                    }
947                    .into());
948                }
949                residual = residual.max((&actual.re - F(expected.clone())).abs());
950            }
951        }
952        let evaluation = SamplingMapEvaluation {
953            coordinates,
954            point: point.to_vec(),
955            inverse_jacobian: jacobian.inv().0,
956            jacobian: jacobian.0,
957            residual: residual.0,
958            support: SamplingSupport::Restricted,
959            diagnostics: vec![format!(
960                "joint: certified normal disk rho={radius}; density at supplied point"
961            )],
962        }
963        .validate("joint inverse")?;
964        Ok(Some(evaluation))
965    }
966}

1205            assert!((output.jacobian * inverse.inverse_jacobian - 1.0).abs() < 2e-9);
1206            for (a, b) in cube.iter().zip(&inverse.coordinates) {
1207                let difference = (a - b).abs();
1208                assert!(difference.min((1.0 - difference).abs()) < 2e-9);
1209            }
1210            // Three-dimensional Cartesian FD includes both square-root circle
1211            // branches and their finite joining points. Angular input is periodic.
1212            let step = 1e-5;
1213            let mut matrix = [[0.0; 3]; 3];
1214            for column in 0..3 {
1215                let mut plus = cube;
1216                let mut minus = cube;
1217                plus[column] += step;
1218                minus[column] -= step;
1219                if column == 2 {
1220                    plus[column] = plus[column].rem_euclid(1.0);
1221                    minus[column] = minus[column].rem_euclid(1.0);
1222                }
1223                let plus = map
1224                    .forward(&plus, &mut SamplingMapContext::detached(&[]))
1225                    .unwrap();
1226                let minus = map
1227                    .forward(&minus, &mut SamplingMapContext::detached(&[]))
1228                    .unwrap();
1229                for (row, values) in matrix.iter_mut().enumerate() {
1230                    values[column] = (plus.point[row] - minus.point[row]) / (2.0 * step);
1231                }
1232            }
1233            let determinant = matrix[0][0]
1234                * (matrix[1][1] * matrix[2][2] - matrix[1][2] * matrix[2][1])
1235                - matrix[0][1] * (matrix[1][0] * matrix[2][2] - matrix[1][2] * matrix[2][0])
1236                + matrix[0][2] * (matrix[1][0] * matrix[2][1] - matrix[1][1] * matrix[2][0]);
1237            assert!(
1238                (determinant.abs() / output.jacobian - 1.0).abs() < 3e-6,
1239                "phi={phi}: FD={determinant}, eager={}",
1240                output.jacobian
1241            );
1242        }
1243        assert!(
1244            map.inverse(
1245                &[100.0, 100.0, 100.0],
1246                &mut SamplingMapContext::detached(&[])
1247            )
1248            .unwrap()
1249            .is_none()
1250        );
1251    }
