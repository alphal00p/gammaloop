1//! Native classification for graph-aware sampling maps.
2//!
3//! Graph, host cut, side and complete parent identity belong to the immutable
4//! resolved block binding. Each conditional map prepares its actual native t*
5//! and fixed physical coordinates together from the declared raw prerequisites.
6//! Each foreign inverse uses its supplied raw point, reusing a native preparation
7//! only for an identical plan and source. No partially sampled data are labelled
8//! a complete MomentumSample. Production exactly embeds one completed source
9//! draw in canonical Arb storage; physical precision attempts materialize
10//! directly from that immutable authority, never promoting a previous physical
11//! retry or pairing its point with stale cut data.

56/// Native source authority. Physical attempts materialize directly from its
57/// canonical Arb instance; rounded inverse prerequisites cannot replace it.
58#[derive(Clone, Debug)]
59pub(crate) struct PreparedLUHost<T: FloatLike> {
60    pub(crate) plan: Arc<SamplingLUHostPlan>,
61    pub(crate) source: SamplingProposalKey,
62    pub(crate) prior: Vec<T>,
63    pub(crate) ray: EsurfaceRay<T>,
64    pub(crate) solution: NewtonIterationResult<T>,
65}

67impl<T: FloatLike> PreparedLUHost<T> {
68    /// Freeze the completed source authority without reconstructing its ray or
69    /// revisiting any root, prerequisite or discrete source decision.
70    pub(crate) fn to_arb_exact(&self) -> Result<PreparedLUHost<ArbPrec>> {
71        Ok(PreparedLUHost {
72            plan: Arc::clone(&self.plan),
73            source: self.source.clone(),
74            prior: self
75                .prior
76                .iter()
77                .map(|value| F(value.clone()).to_arb_exact().map(|value| value.0))
78                .collect::<Result<_>>()?,
79            ray: self.ray.to_arb_exact()?,
80            solution: NewtonIterationResult {
81                solution: self.solution.solution.to_arb_exact()?,
82                derivative_at_solution: self.solution.derivative_at_solution.to_arb_exact()?,
83                error_of_function: self.solution.error_of_function.to_arb_exact()?,
84                num_iterations_used: self.solution.num_iterations_used,
85            },
86        })
87    }
88}

90impl PreparedLUHost<ArbPrec> {
91    /// Convert the canonical authority, retaining its identities and root
92    /// history. This is not a new solve or a native map-preparation observation.
93    pub(crate) fn materialize<T: FloatLike>(&self, tolerance: f64) -> Result<PreparedLUHost<T>> {
94        let positive = |value: &F<ArbPrec>, label: &str| -> Result<F<T>> {
95            let native = F::<T>::from_arb(&value.0)?;
96            if value <= &value.zero() || native <= native.zero() {
97                return Err(SamplingEvaluationError::UncertainGeometry {
98                    detail: format!("LU host materialization requires positive {label}"),
99                }
100                .into());
101            }
102            // Compare to the original represented source, including separated
103            // native limbs. The existing physical adoption subsequently checks
104            // the completed ray, residual, slope and inverse-volume budget.
105            native
106                .verify_arb_materialization(&value.0, tolerance)
107                .wrap_err_with(|| format!("LU host {label} for {:?}", self.source))?;
108            Ok(native)
109        };
110        Ok(PreparedLUHost {
111            plan: Arc::clone(&self.plan),
112            source: self.source.clone(),
113            prior: self
114                .prior
115                .iter()
116                .map(|value| F::<T>::from_arb(value).map(|value| value.0))
117                .collect::<Result<_>>()?,
118            ray: self.ray.materialize()?,
119            solution: NewtonIterationResult {
120                solution: positive(&self.solution.solution, "root")?,
121                derivative_at_solution: positive(
122                    &self.solution.derivative_at_solution,
123                    "root derivative",
124                )?,
125                error_of_function: F::<T>::from_arb(&self.solution.error_of_function.0)?,
126                num_iterations_used: self.solution.num_iterations_used,
127            },
128        })
129    }
130}

165/// Borrowed view of a channel's current numerical prerequisites and the sole
166/// original-draw policy owner. Detached access is explicit for represented-
167/// geometry fixtures; production uses collecting or sealed metadata records.
168#[derive(Debug)]
169pub struct SamplingMapContext<'a, T: FloatLike = f64> {
170    pub previous: &'a [T],
171    pub(crate) policies: Option<&'a mut SamplingProposalPolicies>,
172    pub(crate) key: SamplingProposalKey,
173    pub(crate) prepared_lu_hosts: Option<&'a mut Vec<PreparedLUHost<T>>>,
174    pub(crate) radial_root_diagnostics: Option<&'a mut RadialRootDiagnostics>,
175    pub(crate) selecting_lu_hosts: bool,
176}
177
178impl<'a, T: FloatLike> SamplingMapContext<'a, T> {
179    pub fn detached(previous: &'a [T]) -> Self {
180        Self {
181            previous,
182            policies: None,
183            key: SamplingProposalKey {
184                graph_id: 0,
185                generating_channel: SamplingChannelId(0),
186                target_channel: SamplingChannelId(0),
187                block_path: Vec::new(),
188            },
189            prepared_lu_hosts: None,
190            radial_root_diagnostics: None,
191            selecting_lu_hosts: false,
192        }
193    }
194
195    /// Wrappers keep lineage while replacing prerequisites. A real compiled
196    /// child appends its immutable index, identically in forward and inverse.
197    pub(crate) fn reborrow<'b>(
198        &'b mut self,
199        previous: &'b [T],
200        child: Option<usize>,
201    ) -> SamplingMapContext<'b, T> {
202        let mut key = self.key.clone();
203        key.block_path.extend(child);
204        SamplingMapContext {
205            previous,
206            policies: self.policies.as_deref_mut(),
207            key,
208            prepared_lu_hosts: self.prepared_lu_hosts.as_deref_mut(),
209            radial_root_diagnostics: self.radial_root_diagnostics.as_deref_mut(),
210            selecting_lu_hosts: self.selecting_lu_hosts,
211        }
212    }

214    /// Reuse only exact represented prerequisites. During the initial selected
215    /// operation one plan has one source; inverse work may prepare a different
216    /// source but cannot extend the authoritative prefix frozen by the bridge.
217    pub(crate) fn prepare_lu_host(
218        &mut self,
219        plan: &Arc<SamplingLUHostPlan>,
220        prior: Vec<T>,
221        prepare: impl FnOnce(
222            &mut RadialRootDiagnostics,
223            &RadialRootIdentity,
224        ) -> Result<(EsurfaceRay<T>, NewtonIterationResult<T>)>,
225    ) -> Result<&PreparedLUHost<T>> {
226        if prior.len() != 3 * plan.required_prior_lmb.len() {
227            return Err(eyre!(
228                "LU host plan {plan:?} needs {} canonical prior components, received {}",
229                3 * plan.required_prior_lmb.len(),
230                prior.len()
231            ));
232        }
233        if prior.iter().any(|value| !value.is_finite()) {
234            return Err(SamplingEvaluationError::Unrepresentable {
235                operation: "LU host prerequisites",
236                detail: format!("nonfinite native prior for {plan:?}"),
237            }
238            .into());
239        }
240        let hosts = self
241            .prepared_lu_hosts
242            .as_deref_mut()
243            .ok_or_else(|| eyre!("LU host preparation requires a runtime row context"))?;
244        if self.selecting_lu_hosts
245            && hosts
246                .iter()
247                .any(|host| host.plan == *plan && host.prior != prior)
248        {
249            return Err(SamplingEvaluationError::UncertainGeometry {
250                detail: format!(
251                    "selected LU host plan {plan:?} received inconsistent native prerequisites"
252                ),
253            }
254            .into());
255        }
256        if let Some(index) = hosts
257            .iter()
258            .position(|host| host.plan == *plan && host.prior == prior)
259        {
260            return Ok(&hosts[index]);
261        }
262        let identity = RadialRootIdentity::new(format!(
263            "sampling LU host {plan:?}, source {:?}, {} {}",
264            self.key,
265            if self
266                .policies
267                .as_deref()
268                .is_some_and(SamplingProposalPolicies::is_collecting)
269            {
270                "canonical"
271            } else {
272                "native"
273            },
274            if self.selecting_lu_hosts {
275                "selected"
276            } else {
277                "inverse"
278            },
279        ));
280        let diagnostics = self
281            .radial_root_diagnostics
282            .as_deref_mut()
283            .ok_or_else(|| eyre!("LU host preparation requires the runtime root diagnostics"))?;
284        let (ray, solution) = prepare(diagnostics, &identity)?;
285        hosts.push(PreparedLUHost {
286            plan: Arc::clone(plan),
287            source: self.key.clone(),
288            prior,
289            ray,
290            solution,
291        });
292        Ok(hosts.last().expect("prepared native host was inserted"))
293    }

295    /// Compare the complete native policy, not merely validity of an older
296    /// disk. Two valid radii can induce different densities for the same draw.
297    pub(crate) fn validate_decision(&mut self, decision: SamplingProposalDecision) -> Result<()> {
298        let Some(policies) = self.policies.as_deref_mut() else {
299            return Ok(());
300        };
301        let canonical = policies.records.get(&self.key).copied();
302        if canonical == Some(decision) {
303            return Ok(());
304        }
305        if canonical.is_none() && policies.collecting {
306            policies.records.insert(self.key.clone(), decision);
307            return Ok(());
308        }
309        Err(SamplingEvaluationError::UncertainGeometry {
310            detail: format!(
311                "proposal policy mismatch for {:?}: canonical Arb(1000-bit) decision {canonical:?}, native {:?} decision {decision:?}; phase {}",
312                self.key,
313                T::sampling_precision(),
314                if policies.collecting { "collecting" } else { "sealed" }
315            ),
316        }.into())
317    }
318}

327/// Pointwise classification of an energy surface for prepared complement or cut data.
328///
329/// `Absent` is a normal branch for cross-section kinematics.  Callers should
330/// use the full-support fallback map in that branch rather than dropping the
331/// channel.  `Pinched` remains explicit because it has different diagnostics
332/// and may require a dedicated local map in a later implementation.
333#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
334pub enum PreparedSurfaceStatus<T: FloatLike = f64> {
335    Existing { normalized_margin: Option<T> },
336    Pinched { normalized_margin: T },
337    Absent { reason: String },
338}
339
340impl<T: FloatLike> PreparedSurfaceStatus<T> {
341    pub fn existing(normalized_margin: Option<T>) -> Result<Self> {
342        if let Some(margin) = &normalized_margin {
343            validate_finite(margin, "normalized surface margin")?;
344        }
345        Ok(Self::Existing { normalized_margin })
346    }
347
348    pub fn pinched(normalized_margin: T) -> Result<Self> {
349        validate_finite(&normalized_margin, "normalized surface margin")?;
350        Ok(Self::Pinched { normalized_margin })
351    }
352
353    pub fn absent(reason: impl Into<String>) -> Result<Self> {
354        let reason = reason.into();
355        if reason.trim().is_empty() {
356            return Err(eyre!(
357                "an absent sampling surface needs a diagnostic reason"
358            ));
359        }
360        Ok(Self::Absent { reason })
361    }
362
363    /// A regular threshold radius is direction dependent and belongs to the
364    /// radial map evaluation, not to this complement-only classification.
365    pub fn is_existing(&self) -> bool {
366        matches!(self, Self::Existing { .. })
367    }
368
369    pub fn uses_full_support_fallback(&self) -> bool {
370        matches!(self, Self::Absent { .. } | Self::Pinched { .. })
371    }
372}
373
374fn validate_finite<T: FloatLike>(value: &T, label: &str) -> Result<()> {
375    if !value.is_finite() {
376        return Err(eyre!("{label} must be finite, got {value}"));
377    }
378    Ok(())
379}
