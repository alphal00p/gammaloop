15impl FermiSurfaceProduct {
16    /// Choose a graph basis whose first loop momenta are the selected fermion
17    /// momenta. Independent internal routes extend to a unimodular graph basis;
18    /// its complete edge signatures retain the original external shifts.
19    pub(crate) fn new(graph: &Graph, factors: &[ThermalDistributionFactor]) -> Result<Self> {
20        ensure!(
21            !factors.is_empty(),
22            "Fermi-surface product must not be empty"
23        );
24        let parent = &graph.loop_momentum_basis;
25        let mut edges = BTreeSet::new();
26        let mut rows = Vec::with_capacity(factors.len());
27        for factor in factors {
28            ensure!(
29                factor.derivative_order > 0,
30                "Fermi-surface edge {} requires a positive distribution derivative order",
31                factor.edge_id
32            );
33            ensure!(
34                matches!(factor.sign, -1..=1),
35                "Fermi-surface edge {} has invalid thermal sign {}",
36                factor.edge_id,
37                factor.sign
38            );
39            ensure!(
40                edges.insert(factor.edge_id),
41                "Fermi-surface product repeats edge {}",
42                factor.edge_id
43            );
44            let (pair, _, edge) = graph
45                .iter_edges()
46                .find(|(_, edge_id, _)| *edge_id == factor.edge_id)
47                .ok_or_else(|| eyre!("Unknown Fermi-surface edge {}", factor.edge_id))?;
48            ensure!(
49                pair.is_paired() && edge.data.is_fermion(),
50                "Fermi-surface edge {} must be an internal fermion edge",
51                factor.edge_id
52            );
53            ensure!(
54                edge.data.chemical_potential_atom().is_some(),
55                "Fermi-surface edge {} has no chemical potential",
56                factor.edge_id
57            );
58            let signature = parent.edge_signatures.get(factor.edge_id).ok_or_else(|| {
59                eyre!("Missing routing for Fermi-surface edge {}", factor.edge_id)
60            })?;
61            ensure!(
62                signature.internal.len() == parent.loop_edges.len()
63                    && signature.external.len() == parent.ext_edges.len(),
64                "Incomplete routing for Fermi-surface edge {}",
65                factor.edge_id
66            );
67            rows.push(
68                signature
69                    .internal
70                    .iter()
71                    .map(|sign| *sign * 1_i64)
72                    .collect(),
73            );
74        }
75        // Fixed external shifts cannot make dependent integration directions
76        // independent. Check before asking the spanning-forest builder to omit
77        // these edges, because a dependent choice can disconnect its guide.
78        ensure!(
79            rank_i64(&rows) == factors.len(),
80            "Fermi-surface product requires independent loop-momentum routes; selected edges {:?} are dependent",
81            edges
82        );
83        let selected_edges = factors
84            .iter()
85            .map(|factor| factor.edge_id)
86            .collect::<Vec<_>>();
87        // Complete the active directions with independent fermion routes that
88        // have chemical potentials and can carry occupation steps,
89        // before choosing arbitrary spanning-tree chords. Otherwise an
90        // untouched occupation step can acquire an artificial dependence on
91        // a localized radius (e.g. the other fermion of a dotted sunset).
92        let mut basis_edges = selected_edges.clone();
93        for (pair, edge_id, edge) in graph.iter_edges() {
94            if basis_edges.len() == parent.loop_edges.len() {
95                break;
96            }
97            if !pair.is_paired()
98                || !edge.data.is_fermion()
99                || edge.data.chemical_potential_atom().is_none()
100            {
101                continue;
102            }
103            rows.push(
104                parent.edge_signatures[edge_id]
105                    .internal
106                    .iter()
107                    .map(|sign| *sign * 1_i64)
108                    .collect(),
109            );
110            if rank_i64(&rows) == rows.len() {
111                basis_edges.push(edge_id);
112            } else {
113                rows.pop();
114            }
115        }
116        let mut lmb = graph
117            .lmb_with_loop_edges(basis_edges.as_slice())
118            .wrap_err("Could not construct an independent Fermi-surface basis")?;
119        graph.canonicalize_lmb_external_order(&mut lmb);
120        ensure!(
121            lmb.loop_edges.len() == parent.loop_edges.len() && lmb.ext_edges == parent.ext_edges,
122            "Fermi-surface basis does not preserve the graph's loop and external momentum spaces"
123        );
124        for (slot, edge) in selected_edges.iter().enumerate() {
125            let position = lmb
126                .loop_edges
127                .iter()
128                .position(|candidate| candidate == edge)
129                .ok_or_else(|| {
130                    eyre!("Fermi-surface edge {edge} is absent from the adapted basis")
131                })?;
132            lmb.swap_loops(LoopIndex(slot), LoopIndex(position));
133        }
134        for (slot, edge) in selected_edges.iter().enumerate() {
135            let signature = &lmb.edge_signatures[*edge];
136            ensure!(
137                signature.internal.iter().enumerate().all(|(index, sign)| {
138                    *sign
139                        == if index == slot {
140                            SignOrZero::Plus
141                        } else {
142                            SignOrZero::Zero
143                        }
144                }) && signature
145                    .external
146                    .iter()
147                    .all(|sign| *sign == SignOrZero::Zero),
148                "Fermi-surface edge {edge} is not a defining momentum in its adapted basis"
149            );
150        }
151        let derivative_orders = factors
152            .iter()
153            .map(|factor| factor.derivative_order - 1)
154            .collect::<Vec<_>>();
155        let shape = mixed_derivative_shape(&derivative_orders);
156        Ok(Self {
157            factors: factors.to_vec(),
158            lmb,
159            derivative_orders,
160            shape,
161        })
162    }
163
164    pub(crate) fn lmb(&self) -> &LoopMomentumBasis {
165        &self.lmb
166    }

168    pub(crate) fn factors(&self) -> &[ThermalDistributionFactor] {
169        &self.factors
170    }

201    #[test]
202    fn independent_fermi_basis_preserves_affine_routes() -> Result<()> {
203        let graph = test_graph()?;
204        let factors = [
205            ThermalDistributionFactor {
206                edge_id: EdgeIndex(3),
207                sign: -1,
208                derivative_order: 2,
209            },
210            ThermalDistributionFactor {
211                edge_id: EdgeIndex(1),
212                sign: 1,
213                derivative_order: 3,
214            },
215        ];
216        let product = FermiSurfaceProduct::new(&graph, &factors)?;
217        assert_eq!(product.factors(), factors);
218        assert_eq!(product.lmb().loop_edges.raw, [EdgeIndex(3), EdgeIndex(1)]);
219        assert_eq!(product.derivative_orders, [1, 2]);
220        assert_eq!(product.shape.len(), 6);
221        let parent = &graph.loop_momentum_basis;
222        assert!(factors.iter().any(|factor| {
223            parent.edge_signatures[factor.edge_id]
224                .external
225                .iter()
226                .any(|sign| *sign != SignOrZero::Zero)
227        }));
228
229        let loops = [
230            ThreeMomentum::new(F(1.0), F(2.0), F(3.0)),
231            ThreeMomentum::new(F(4.0), F(5.0), F(6.0)),
232        ];
233        let externals = vec![ThreeMomentum::new(F(2.0), F(-3.0), F(4.0)); parent.ext_edges.len()];
234        let adapted = product
235            .lmb()
236            .loop_edges
237            .iter()
238            .map(|edge| parent.edge_signatures[*edge].compute_momentum_untyped(&loops, &externals))
239            .collect::<Vec<_>>();
240        for (edge, signature) in &parent.edge_signatures {
241            assert_eq!(
242                signature.compute_momentum_untyped(&loops, &externals),
243                product.lmb().edge_signatures[edge].compute_momentum_untyped(&adapted, &externals),
244                "affine route for edge {edge}"
245            );
246        }
247
248        let partial = FermiSurfaceProduct::new(&graph, &factors[..1])?;
249        assert_eq!(partial.lmb().loop_edges.len(), 2);
250        assert_eq!(partial.lmb().loop_edges[LoopIndex(0)], EdgeIndex(3));
251        Ok(())
252    }

293    #[test]
294    fn fermi_basis_prioritizes_occupation_steps_over_unstepped_fermions() -> Result<()> {
295        use three_dimensional_reps::ThermalDistributionCall;
296
297        use super::super::FermiSurfaceSector;
298        use crate::model::{ParameterName, UFOSymbol};
299
300        test_initialise()?;
301        let mut graph: Graph = dot!(digraph mixed_density_three_loop {
302            node [num=1]
303            edge [num=1]
304            D -> E [id=0 particle="e-" lmb_id=1]
305            E -> D [id=1 particle="e-"]
306            A -> C [id=2 particle="d" lmb_id=0]
307            C -> B [id=3 particle="d"]
308            B -> A [id=4 particle="d"]
309            A -> D [id=5 particle="a" lmb_id=2]
310            E -> B [id=6 particle="a"]
311        })?;
312        // Both unstepped lepton edges can fill the remaining loop slots, but
313        // doing so makes the quark occupation depend on the localized radius.
314        let unstepped_basis =
315            graph.lmb_with_loop_edges([EdgeIndex(2), EdgeIndex(0), EdgeIndex(1)].as_slice())?;
316        let active_slot = unstepped_basis
317            .loop_edges
318            .iter()
319            .position(|edge| *edge == EdgeIndex(2))
320            .unwrap();
321        assert_ne!(
322            unstepped_basis.edge_signatures[EdgeIndex(4)].internal[LoopIndex(active_slot)],
323            SignOrZero::Zero
324        );
325
326        let step = ThermalDistributionCall::atom(4, 0, 0, 1, 1);
327        let expression = ThermalDistributionCall::atom(2, 1, 0, 1, 1) * &step;
328        for chemical_potential in [None, Some(ParameterName(UFOSymbol::zero()))] {
329            for edge in [EdgeIndex(0), EdgeIndex(1)] {
330                let mut particle = graph[edge].particle().unwrap();
331                std::sync::Arc::make_mut(&mut particle.0).chemical_potential = chemical_potential;
332                graph.underlying[edge].particle = particle.into();
333                assert!(graph[edge].chemical_potential_atom().is_none());
334            }
335            let (bulk, sectors) = FermiSurfaceSector::extract(&graph, &expression)?;
336            assert!(bulk.is_zero());
337            assert_eq!(sectors.len(), 1);
338            assert_eq!(sectors[0].coefficient, step);
339            let lmb = sectors[0].product.lmb();
340            assert_eq!(lmb.loop_edges[LoopIndex(0)], EdgeIndex(2));
341            assert!(lmb.loop_edges.contains(&EdgeIndex(4)));
342            assert_eq!(
343                lmb.edge_signatures[EdgeIndex(4)].internal[LoopIndex(0)],
344                SignOrZero::Zero,
345                "the quark occupation must remain independent of the localized radius"
346            );
347        }
348        Ok(())
349    }

353        let graph = test_graph()?;
354        let parent = &graph.loop_momentum_basis;
