1use eyre::{Result, eyre};

3/// Physical inputs and their archived constants use parameter-major jets with
4/// the same derivative shape for every evaluator in a stack.
5/// Cross-section archives may omit representative inputs and require caller inputs.
6pub(crate) fn validate_standalone_input_layout<'a>(
7    parameter_count: usize,
8    input_count: Option<usize>,
9    mult_offset: usize,
10    starts: [usize; 2],
11    dual_shapes: impl IntoIterator<Item = Option<&'a [Vec<usize>]>>,
12) -> Result<()> {
13    let mut shapes = dual_shapes.into_iter();
14    let shape = shapes
15        .next()
16        .ok_or_else(|| eyre!("Standalone stack has no evaluator"))?;
17    let width = shape.map_or(1, <[Vec<usize>]>::len);
18    if width == 0
19        || mult_offset != width
20        || input_count.is_some_and(|count| parameter_count.checked_mul(width) != Some(count))
21        // Empty parameter groups can start immediately after the last input.
22        || starts.into_iter().any(|start| start > parameter_count)
23        || shapes.any(|other| other != shape)
24    {
25        return Err(eyre!(
26            "Standalone stack has inconsistent physical input or dual layout"
27        ));
28    }
29    Ok(())
30}
