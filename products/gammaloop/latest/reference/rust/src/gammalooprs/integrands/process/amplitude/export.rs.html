59fn atom_to_bytes(atom: &Atom) -> Result<Vec<u8>> {
60    let mut out = Vec::new();
61    atom.as_view()
62        .write(&mut out)
63        .with_context(|| "Failed to write atom to standalone binary stream")?;
64    Ok(out)
65}

67fn export_generic_evaluator<T: ExportAtomTo>(
68    evaluator: &GenericEvaluator,
69) -> Result<StandaloneGenericEvaluatorArchive<T>> {
70    let exprs = evaluator
71        .exprs
72        .as_ref()
73        .ok_or_else(|| {
74            eyre!("Standalone export requires stored evaluator atoms (`store_atom=true`)")
75        })?
76        .iter()
77        .map(|a| T::export_atom_to(a))
78        .collect::<Result<Vec<_>>>()?;
79
80    let additional_fn_map_entries = evaluator
81        .fn_map_entries
82        .iter()
83        .map(|e| e.archive())
84        .collect::<Result<_>>()?;
85
86    Ok(StandaloneGenericEvaluatorArchive {
87        exprs,
88        additional_fn_map_entries,
89        dual_shape: evaluator.dual_shape.clone(),
90    })
91}

93fn export_evaluator_stack<T: ExportAtomTo>(
94    evaluator_stack: &crate::integrands::process::evaluators::EvaluatorStack,
95    start: usize,
96    residue_map_id_start: usize,
97    mult_offset: usize,
98    representative_input: Vec<StandaloneComplexInput>,
99) -> Result<StandaloneEvaluatorStackArchive<T>> {
100    Ok(StandaloneEvaluatorStackArchive {
101        explicit_orientation_sum_only: evaluator_stack.explicit_orientation_sum_only,
102        production_orientation_ids: evaluator_stack
103            .production_orientation_ids()
104            .iter()
105            .map(|id| id.0)
106            .collect(),
107        start,
108        residue_map_id_start,
109        mult_offset,
110        representative_input,
111        single_parametric: export_generic_evaluator(&evaluator_stack.single_parametric)?,
112        iterative: evaluator_stack
113            .iterative
114            .as_ref()
115            .map(|(eval, _)| export_generic_evaluator(eval))
116            .transpose()?,
117        summed_function_map: evaluator_stack
118            .summed_function_map
119            .as_ref()
120            .map(export_generic_evaluator)
121            .transpose()?,
122        summed: evaluator_stack
123            .summed
124            .as_ref()
125            .map(export_generic_evaluator)
126            .transpose()?,
127    })
128}

138fn export_evaluator_map<T: ExportAtomTo>(
139    evaluator_stacks: &BTreeMap<
140        CutCFFIndex,
141        crate::integrands::process::evaluators::EvaluatorStack,
142    >,
143    start: usize,
144    residue_map_id_start: usize,
145    mult_offset: usize,
146    representative_input: &[StandaloneComplexInput],
147) -> Result<Vec<StandaloneIndexedEvaluatorStackArchive<T>>> {
148    evaluator_stacks
149        .iter()
150        .map(|(cut_cff_index, evaluator_stack)| {
151            Ok(StandaloneIndexedEvaluatorStackArchive {
152                cut_cff_index: export_cut_cff_index(cut_cff_index),
153                evaluator_stack: export_evaluator_stack(
154                    evaluator_stack,
155                    start,
156                    residue_map_id_start,
157                    mult_offset,
158                    representative_input.to_vec(),
159                )?,
160            })
161        })
162        .collect()

219impl AmplitudeIntegrand {
220    fn create_archive<T: ExportAtomTo, S>(
221        &self,
222        symbolica_state: S,
223        numeric_target: StandaloneNumericTarget,
224    ) -> Result<StandaloneEvaluatorArchive<S, T>> {
225        let sample_inputs = match numeric_target {
226            StandaloneNumericTarget::Double => self.representative_input_for::<f64>()?,
227            StandaloneNumericTarget::Quad => self.representative_input_for::<f128>()?,
228            StandaloneNumericTarget::Arb => self.representative_input_for::<ArbPrec>()?,
229        };
230
231        let graph_terms = self
232            .data
233            .graph_terms
234            .iter()
235            .zip(sample_inputs)
236            .map(
237                |(
238                    term,
239                    (
240                        representative_input,
241                        multiplicative_offset,
242                        orientation_start,
243                        residue_map_id_start,
244                    ),
245                )| {
246                    let param_builder_params = (&term.param_builder.pairs)
247                        .into_iter()
248                        .flat_map(|pair| pair.params.iter())
249                        .map(|a| T::export_atom_to(a))
250                        .collect::<Result<Vec<_>>>()?;
251
252                    let orientations: Vec<Vec<_>> = term
253                        .orientations
254                        .iter()
255                        .map(|a| {
256                            a.iter()
257                                .map(|(_, o)| match o {
258                                    Orientation::Default => 1i8,
259                                    Orientation::Reversed => -1i8,
260                                    Orientation::Undirected => 0,
261                                })
262                                .collect()
263                        })
264                        .collect();
265
266                    let fn_map_entries = term
267                        .param_builder
268                        .reps
269                        .iter()
270                        .map(|entry| entry.archive())
271                        .collect::<Result<Vec<_>>>()?;
272
273                    let original_integrand = export_evaluator_stack(
274                        &term.original_integrand,
275                        orientation_start,
276                        residue_map_id_start,
277                        multiplicative_offset,
278                        representative_input.clone(),
279                    )?;
280
281                    let export_counterterm = |counterterm: &crate::subtraction::amplitude_counterterm::AmplitudeCountertermEvaluator| {
282                        export_evaluator_map(
283                            &counterterm.evaluator_stacks,
284                            orientation_start,
285                            residue_map_id_start,
286                            multiplicative_offset,
287                            &representative_input,
288                        )
289                    };
290                    let threshold_counterterms = if term.threshold_counterterm.legacy_equivalent {
291                        term.threshold_counterterm
292                            .evaluators
293                            .iter()
294                            .map(export_counterterm)
295                            .collect::<Result<Vec<_>>>()?
296                    } else {
297                        term.threshold_counterterm
298                            .variant_evaluators
299                            .iter()
300                            .map(export_counterterm)
301                            .collect::<Result<Vec<_>>>()?
302                    };
303                    let threshold_variants = term
304                        .threshold_counterterm
305                        .variant_metadata
306                        .iter_enumerated()
307                        .map(|(variant_id, variant)| {
308                            let multiplier = variant.multiplier.as_ref();
309                            StandaloneAmplitudeThresholdVariant {
310                                variant_id: variant_id.0,
311                                name: variant.name.clone(),
312                                raised_esurface_id: term
313                                    .threshold_counterterm
314                                    .variant_raised_esurfaces[variant_id]
315                                    .0,
316                                generated: term.threshold_counterterm.variant_generated_mask
317                                    [variant_id],
318                                active: term.threshold_counterterm.variant_active_mask[variant_id],
319                                requested_subspace: variant
320                                    .requested_subspace
321                                    .as_ref()
322                                    .map(|edges| edges.iter().map(|edge| edge.0).collect()),
323                                requested_parent_lmb: variant
324                                    .requested_parent_lmb
325                                    .as_ref()
326                                    .map(|edges| edges.iter().map(|edge| edge.0).collect()),
327                                resolved_parent_lmb: variant
328                                    .resolved_parent_lmb
329                                    .iter()
330                                    .map(|edge| edge.0)
331                                    .collect(),
332                                resolved_subspace: variant
333                                    .subspace
334                                    .iter_basis_edges(&term.threshold_counterterm.lmbs)
335                                    .map(|edge| edge.0)
336                                    .collect(),
337                                subspace_loop_count: variant.subspace_loop_count,
338                                multiplier_expression: multiplier
339                                    .map(|multiplier| multiplier.expression.clone()),
340                                multiplier_symmetrize: multiplier
341                                    .is_some_and(|multiplier| multiplier.symmetrize),
342                                multiplier_opaque_derivatives: multiplier
343                                    .is_none_or(|multiplier| multiplier.opaque_derivatives),
344                                threshold_edge_sets: variant
345                                    .associations
346                                    .iter()
347                                    .map(|association| {
348                                        association
349                                            .threshold_edges
350                                            .iter()
351                                            .map(|edge| edge.0)
352                                            .collect()
353                                    })
354                                    .collect(),
355                                explicit_associations: variant
356                                    .associations
357                                    .iter()
358                                    .map(|association| {
359                                        association.origin == ThresholdCountertermOrigin::Explicit
360                                    })
361                                    .collect(),
362                            }
363                        })
364                        .collect();
365                    let threshold_multipliers = term
366                        .threshold_counterterm
367                        .threshold_multipliers
368                        .as_ref()
369                        .map(export_threshold_multiplier_collection)
370                        .transpose()?;
371
372                    Ok(StandaloneGraphTermArchive {
373                        graph_name: term.graph.name.clone(),
374                        orientations,
375                        param_builder_params,
376                        fn_map_entries,
377                        original_integrand,
378                        threshold_counterterms,
379                        threshold_counterterms_are_variants: !term
380                            .threshold_counterterm
381                            .legacy_equivalent,
382                        threshold_variants,
383                        threshold_multipliers,
384                        metadata_registry: term.threshold_counterterm.metadata_registry.clone(),
385                    })
386                },
387            )
388            .collect::<Result<Vec<_>>>()?;
389        Ok(StandaloneEvaluatorArchive {
390            version: STANDALONE_EVALUATORS_VERSION,
391            numeric_target,
392            symbolica_state,
393            graph_terms,
394        })
395    }

397    #[allow(clippy::type_complexity)]
398    fn representative_input_for<T>(
399        &self,
400    ) -> Result<Vec<(Vec<StandaloneComplexInput>, usize, usize, usize)>>
401    where
402        T: StandaloneRepresentativeInputPrecision,
403    {
404        let hel = self.settings.kinematics.externals.get_helicities();
405        let mut rng = rand::rng();
406
407        let mut base_mom_samples = Vec::new();
408        {
409            let dependent_momenta_constructor = self.get_dependent_momenta_constructor();
410            for t in self.data.graph_terms.iter() {
411                let loop_moms = LoopMomenta::from_iter(
412                    (0..t.graph.loop_momentum_basis.loop_edges.len()).map(|_| ThreeMomentum {
413                        px: F(rng.random_range(-1.0..1.0)),
414                        py: F(rng.random_range(-1.0..1.0)),
415                        pz: F(rng.random_range(-1.0..1.0)),
416                    }),
417                );
418                let momentum_sample = MomentumSample::new(
419                    loop_moms,
420                    0,
421                    &self.settings.kinematics.externals,
422                    0,
423                    F(1.0),
424                    dependent_momenta_constructor,
425                    None,
426                )?;
427                base_mom_samples.push(momentum_sample);
428            }
429        }
430        let mut inputs = Vec::new();
431
432        for (t, base_momentum_sample) in self.data.graph_terms.iter().zip(&base_mom_samples) {
433            let mut p_build = t.param_builder.clone();
434            let momentum_sample = T::sample_from_base(base_momentum_sample);
435            let input = T::get_parameters(
436                &mut p_build,
437                (false, false),
438                &t.graph,
439                &momentum_sample,
440                hel,
441                &self.settings.additional_params(),
442                None,
443                None,
444                None,
445            );
446
447            inputs.push((
448                serialize_representative_input(input.as_slice()),
449                input.multiplicative_offset,
450                input.orientations_start,
451                input.residue_map_id_start,
452            ));
453        }
454
455        Ok(inputs)
456    }

460    pub(crate) fn export_standalone(
461        &self,
462        path: impl AsRef<Path>,
463        settings: &StandaloneExportSettings,
464    ) -> Result<()> {
465        let mut symbolica_state = Vec::new();
466        State::export(&mut symbolica_state)
467            .with_context(|| "Failed to export Symbolica state for standalone evaluators")?;
468
469        let mut standalone_path = path.as_ref().join(STANDALONE_DATA_FILE);
470
471        match settings.format {
472            StandaloneDataFormat::Binary => {
473                let standalone: StandaloneEvaluatorArchive<Vec<u8>, Vec<u8>> =
474                    self.create_archive(symbolica_state, settings.precision)?;
475                let binary = bincode::encode_to_vec(&standalone, bincode::config::standard())?;
476                standalone_path.add_extension("bin");
477                fs::write(&standalone_path, binary).with_context(|| {
478                    format!(
479                        "Failed writing standalone evaluator binary to {}",
480                        standalone_path.display()
481                    )
482                })?;
483            }
484            StandaloneDataFormat::Json => {
485                let standalone: StandaloneEvaluatorArchive<(), String> =
486                    self.create_archive((), settings.precision)?;
487                let json = serde_json::to_vec_pretty(&standalone)?;
488                standalone_path.add_extension("json");
489                fs::write(&standalone_path, json).with_context(|| {
490                    format!(
491                        "Failed writing standalone evaluator JSON to {}",
492                        standalone_path.display()
493                    )
494                })?
495            }
496        }
497
498        match settings.mode {
499            StandaloneExportMode::Python => {
500                return Err(eyre!("Python Export mode not implemented"));
501            }
502            StandaloneExportMode::Rust => {
503                let script = standalone_rust_script();
504                let script_path = path.as_ref().join(STANDALONE_RUST_SCRIPT_FILE);
505                fs::write(&script_path, script).with_context(|| {
506                    format!(
507                        "Failed writing standalone rust-script loader to {}",
508                        script_path.display()
509                    )
510                })?;
511                make_script_executable(&script_path)?;
512            }
513        }
514
515        Ok(())
516    }
517}
