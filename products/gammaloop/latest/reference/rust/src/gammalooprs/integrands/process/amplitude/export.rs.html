58fn atom_to_bytes(atom: &Atom) -> Result<Vec<u8>> {
59    let mut out = Vec::new();
60    atom.as_view()
61        .write(&mut out)
62        .with_context(|| "Failed to write atom to standalone binary stream")?;
63    Ok(out)
64}

66fn export_generic_evaluator<T: ExportAtomTo>(
67    evaluator: &GenericEvaluator,
68) -> Result<StandaloneGenericEvaluatorArchive<T>> {
69    let exprs = evaluator
70        .exprs
71        .as_ref()
72        .ok_or_else(|| {
73            eyre!("Standalone export requires stored evaluator atoms (`store_atom=true`)")
74        })?
75        .iter()
76        .map(|a| T::export_atom_to(a))
77        .collect::<Result<Vec<_>>>()?;
78
79    let additional_fn_map_entries = evaluator
80        .fn_map_entries
81        .iter()
82        .map(|e| e.archive())
83        .collect::<Result<_>>()?;
84
85    Ok(StandaloneGenericEvaluatorArchive {
86        exprs,
87        additional_fn_map_entries,
88        dual_shape: evaluator.dual_shape.clone(),
89    })
90}

92fn export_evaluator_stack<T: ExportAtomTo>(
93    evaluator_stack: &crate::integrands::process::evaluators::EvaluatorStack,
94    start: usize,
95    mult_offset: usize,
96    representative_input: Vec<StandaloneComplexInput>,
97) -> Result<StandaloneEvaluatorStackArchive<T>> {
98    Ok(StandaloneEvaluatorStackArchive {
99        start,
100        mult_offset,
101        representative_input,
102        single_parametric: export_generic_evaluator(&evaluator_stack.single_parametric)?,
103        iterative: evaluator_stack
104            .iterative
105            .as_ref()
106            .map(|(eval, _)| export_generic_evaluator(eval))
107            .transpose()?,
108        summed_function_map: evaluator_stack
109            .summed_function_map
110            .as_ref()
111            .map(export_generic_evaluator)
112            .transpose()?,
113        summed: evaluator_stack
114            .summed
115            .as_ref()
116            .map(export_generic_evaluator)
117            .transpose()?,
118    })
119}

129fn export_evaluator_map<T: ExportAtomTo>(
130    evaluator_stacks: &BTreeMap<
131        CutCFFIndex,
132        crate::integrands::process::evaluators::EvaluatorStack,
133    >,
134    start: usize,
135    mult_offset: usize,
136    representative_input: &[StandaloneComplexInput],
137) -> Result<Vec<StandaloneIndexedEvaluatorStackArchive<T>>> {
138    evaluator_stacks
139        .iter()
140        .map(|(cut_cff_index, evaluator_stack)| {
141            Ok(StandaloneIndexedEvaluatorStackArchive {
142                cut_cff_index: export_cut_cff_index(cut_cff_index),
143                evaluator_stack: export_evaluator_stack(
144                    evaluator_stack,
145                    start,
146                    mult_offset,
147                    representative_input.to_vec(),
148                )?,
149            })
150        })
151        .collect()
152}

158pub trait ExportAtomTo: Sized {
159    fn export_atom_to(atom: &Atom) -> Result<Self>;
160}
161
162impl ExportAtomTo for Vec<u8> {
163    fn export_atom_to(atom: &Atom) -> Result<Self> {
164        atom_to_bytes(atom)
165    }
166}

168impl ExportAtomTo for String {
169    fn export_atom_to(atom: &Atom) -> Result<Self> {
170        Ok(atom.printer(PrintOptions::file()).to_string())
171    }

208impl AmplitudeIntegrand {
209    fn create_archive<T: ExportAtomTo, S>(
210        &self,
211        symbolica_state: S,
212        numeric_target: StandaloneNumericTarget,
213    ) -> Result<StandaloneEvaluatorArchive<S, T>> {
214        let sample_inputs = match numeric_target {
215            StandaloneNumericTarget::Double => self.representative_input_for::<f64>()?,
216            StandaloneNumericTarget::Quad => self.representative_input_for::<f128>()?,
217            StandaloneNumericTarget::Arb => self.representative_input_for::<ArbPrec>()?,
218        };
219
220        let graph_terms = self
221            .data
222            .graph_terms
223            .iter()
224            .zip(sample_inputs)
225            .map(
226                |(term, (representative_input, multiplicative_offset, orientation_start))| {
227                    let param_builder_params = (&term.param_builder.pairs)
228                        .into_iter()
229                        .flat_map(|pair| pair.params.iter())
230                        .map(|a| T::export_atom_to(a))
231                        .collect::<Result<Vec<_>>>()?;
232
233                    let orientations: Vec<Vec<_>> = term
234                        .orientations
235                        .iter()
236                        .map(|a| {
237                            a.iter()
238                                .map(|(_, o)| match o {
239                                    Orientation::Default => 1i8,
240                                    Orientation::Reversed => -1i8,
241                                    Orientation::Undirected => 0,
242                                })
243                                .collect()
244                        })
245                        .collect();
246
247                    let fn_map_entries = term
248                        .param_builder
249                        .reps
250                        .iter()
251                        .map(|entry| entry.archive())
252                        .collect::<Result<Vec<_>>>()?;
253
254                    let original_integrand = StandaloneEvaluatorStackArchive {
255                        start: orientation_start,
256                        mult_offset: multiplicative_offset,
257                        representative_input: representative_input.clone(),
258                        single_parametric: export_generic_evaluator(
259                            &term.original_integrand.single_parametric,
260                        )?,
261                        iterative: term
262                            .original_integrand
263                            .iterative
264                            .as_ref()
265                            .map(|(eval, _)| export_generic_evaluator(eval))
266                            .transpose()?,
267                        summed_function_map: term
268                            .original_integrand
269                            .summed_function_map
270                            .as_ref()
271                            .map(export_generic_evaluator)
272                            .transpose()?,
273                        summed: term
274                            .original_integrand
275                            .summed
276                            .as_ref()
277                            .map(export_generic_evaluator)
278                            .transpose()?,
279                    };
280
281                    let threshold_counterterms = term
282                        .threshold_counterterm
283                        .evaluators
284                        .iter()
285                        .map(|counterterm| {
286                            export_evaluator_map(
287                                &counterterm.evaluator_stacks,
288                                orientation_start,
289                                multiplicative_offset,
290                                &representative_input,
291                            )
292                        })
293                        .collect::<Result<Vec<_>>>()?;
294
295                    Ok(StandaloneGraphTermArchive {
296                        graph_name: term.graph.name.clone(),
297                        orientations,
298                        param_builder_params,
299                        fn_map_entries,
300                        original_integrand,
301                        threshold_counterterms,
302                    })
303                },
304            )
305            .collect::<Result<Vec<_>>>()?;
306        Ok(StandaloneEvaluatorArchive {
307            version: STANDALONE_EVALUATORS_VERSION,
308            numeric_target,
309            symbolica_state,
310            graph_terms,
311        })
312    }

314    #[allow(clippy::type_complexity)]
315    fn representative_input_for<T>(
316        &self,
317    ) -> Result<Vec<(Vec<StandaloneComplexInput>, usize, usize)>>
318    where
319        T: StandaloneRepresentativeInputPrecision,
320    {
321        let hel = self.settings.kinematics.externals.get_helicities();
322        let mut rng = rand::rng();
323
324        let mut base_mom_samples = Vec::new();
325        {
326            let dependent_momenta_constructor = self.get_dependent_momenta_constructor();
327            for t in self.data.graph_terms.iter() {
328                let loop_moms = LoopMomenta::from_iter(
329                    (0..t.graph.loop_momentum_basis.loop_edges.len()).map(|_| ThreeMomentum {
330                        px: F(rng.random_range(-1.0..1.0)),
331                        py: F(rng.random_range(-1.0..1.0)),
332                        pz: F(rng.random_range(-1.0..1.0)),
333                    }),
334                );
335                let momentum_sample = MomentumSample::new(
336                    loop_moms,
337                    0,
338                    &self.settings.kinematics.externals,
339                    0,
340                    F(1.0),
341                    dependent_momenta_constructor,
342                    None,
343                )?;
344                base_mom_samples.push(momentum_sample);
345            }
346        }
347        let mut inputs = Vec::new();
348
349        for (t, base_momentum_sample) in self.data.graph_terms.iter().zip(&base_mom_samples) {
350            let mut p_build = t.param_builder.clone();
351            let momentum_sample = T::sample_from_base(base_momentum_sample);
352            let input = T::get_parameters(
353                &mut p_build,
354                (false, false),
355                &t.graph,
356                &momentum_sample,
357                hel,
358                &self.settings.additional_params(),
359                None,
360                None,
361                None,
362            );
363
364            inputs.push((
365                serialize_representative_input(input.as_slice()),
366                input.multiplicative_offset,
367                input.orientations_start,
368            ));
369        }
370
371        Ok(inputs)
372    }

376    pub(crate) fn export_standalone(
377        &self,
378        path: impl AsRef<Path>,
379        settings: &StandaloneExportSettings,
380    ) -> Result<()> {
381        let mut symbolica_state = Vec::new();
382        State::export(&mut symbolica_state)
383            .with_context(|| "Failed to export Symbolica state for standalone evaluators")?;
384
385        let mut standalone_path = path.as_ref().join(STANDALONE_DATA_FILE);
386
387        match settings.format {
388            StandaloneDataFormat::Binary => {
389                let standalone: StandaloneEvaluatorArchive<Vec<u8>, Vec<u8>> =
390                    self.create_archive(symbolica_state, settings.precision)?;
391                let binary = bincode::encode_to_vec(&standalone, bincode::config::standard())?;
392                standalone_path.add_extension("bin");
393                fs::write(&standalone_path, binary).with_context(|| {
394                    format!(
395                        "Failed writing standalone evaluator binary to {}",
396                        standalone_path.display()
397                    )
398                })?;
399            }
400            StandaloneDataFormat::Json => {
401                let standalone: StandaloneEvaluatorArchive<(), String> =
402                    self.create_archive((), settings.precision)?;
403                let json = serde_json::to_vec_pretty(&standalone)?;
404                standalone_path.add_extension("json");
405                fs::write(&standalone_path, json).with_context(|| {
406                    format!(
407                        "Failed writing standalone evaluator JSON to {}",
408                        standalone_path.display()
409                    )
410                })?
411            }
412        }
413
414        match settings.mode {
415            StandaloneExportMode::Python => {
416                return Err(eyre!("Python Export mode not implemented"));
417            }
418            StandaloneExportMode::Rust => {
419                let script = standalone_rust_script();
420                let script_path = path.as_ref().join(STANDALONE_RUST_SCRIPT_FILE);
421                fs::write(&script_path, script).with_context(|| {
422                    format!(
423                        "Failed writing standalone rust-script loader to {}",
424                        script_path.display()
425                    )
426                })?;
427                make_script_executable(&script_path)?;
428            }
429        }
430
431        Ok(())
432    }
