141fn parse_fn_map_entries<A: ImportWithMap>(
142    entries: &[SerializedFnMapEntry<A>],
143    state_map: &StateMap,
144) -> Result<Vec<ParsedFnMapEntry>> {
145    entries
146        .iter()
147        .map(|(lhs, rhs, tags, args)| {
148            let lhs_atom = lhs.import_with_map(state_map)?;
149            let rhs_atom = rhs.import_with_map(state_map)?;
150            let tags = tags
151                .iter()
152                .map(|tag| tag.import_with_map(state_map))
153                .collect::<Result<Vec<_>>>()?;
154            let args = args
155                .iter()
156                .map(|arg| {
157                    let arg = arg.import_with_map(state_map)?;
158                    arg.clone().try_into().map_err(|_| {
159                        eyre!("Expected indeterminate in function argument, got {arg}")
160                    })
161                })
162                .collect::<Result<Vec<_>>>()?;
163
164            Ok((lhs_atom, rhs_atom, tags, args))
165        })
166        .collect()
167}

169fn apply_fn_map_entries(
170    parsed_entries: Vec<ParsedFnMapEntry>,
171) -> Result<(Vec<Replacement>, Vec<Replacement>, FunctionMap)> {
172    let mut all_replacements = Vec::new();
173    let mut fn_map = FunctionMap::new();
174    let mut replacements = Vec::new();
175    fn_map
176        .add_aliases([(parse_lit!(gammalooprs::x), Atom::Zero)])
177        .map_err(|e| eyre!(e))?;
178
179    // Graph and evaluator archives can share definitions. Register exact
180    // duplicates once; different bodies still trigger Symbolica's conflict check.
181    let mut seen = HashSet::new();
182    for (lhs, rhs, tags, args) in parsed_entries
183        .into_iter()
184        .filter(|entry| seen.insert(entry.clone()))
185    {
186        if let AtomView::Var(_) = lhs.as_view() {
187            if let Ok(value) = Complex::<Rational>::try_from(rhs.as_view()) {
188                fn_map
189                    .add_aliases([(lhs.clone(), Atom::num(value))])
190                    .map_err(|e| eyre!(e))?;
191                all_replacements.push(Replacement::new(lhs.to_pattern(), rhs.clone()));
192            } else {
193                replacements.push(Replacement::new(lhs.to_pattern(), rhs.clone()));
194            }
195        } else if let AtomView::Fun(f) = lhs.as_view() {
196            if tags.is_empty() {
197                let wildcards = args
198                    .iter()
199                    .enumerate()
200                    .map(|(i, arg)| {
201                        let atom: Atom = arg.clone().into();
202                        Replacement::new(
203                            atom.to_pattern(),
204                            Atom::var(symbolica::symbol!(format!("x{i}_"))),
205                        )
206                        .allow_new_wildcards_on_rhs(true)
207                    })
208                    .collect::<Vec<_>>();
209
210                fn_map
211                    .add_function(f.get_symbol(), args, rhs.clone())
212                    .map_err(|e| eyre!(e))?;
213
214                all_replacements.push(Replacement::new(
215                    lhs.replace_multiple(&wildcards).to_pattern(),
216                    rhs.replace_multiple(&wildcards),
217                ));
218            } else {
219                fn_map
220                    .add_tagged_function(f.get_symbol(), tags, args, rhs.clone())
221                    .map_err(|e| eyre!(e))?;
222                all_replacements.push(Replacement::new(lhs.to_pattern(), rhs.clone()));
223            }
224        } else {
225            all_replacements.push(Replacement::new(lhs.to_pattern(), rhs.clone()));
226            replacements.push(Replacement::new(lhs.to_pattern(), rhs.clone()));
227        }
228    }
229
230    Ok((replacements, all_replacements, fn_map))
231}

233fn build_evaluator<A: ImportWithMap>(
234    payload: StandaloneGenericEvaluatorArchive<A>,
235    graph_params: &[Atom],
236    mut fn_map_entries: Vec<ParsedFnMapEntry>,
237    state_map: &StateMap,
238    iterate: bool,
239) -> Result<LoadedGenericEvaluator> {
240    let optimization_settings = OptimizationSettings::new()
241        .horner_iterations(10)
242        .cores(10)
243        .abort_check(None);
244    let cpe_iterations = None;
245    let exprs = payload
246        .exprs
247        .iter()
248        .map(|expr| expr.import_with_map(state_map))
249        .collect::<Result<Vec<_>>>()?;
250    let parameter_override = payload
251        .parameter_override
252        .as_ref()
253        .map(|params| {
254            params
255                .iter()
256                .map(|param| param.import_with_map(state_map))
257                .collect::<Result<Vec<_>>>()
258        })
259        .transpose()?;
260    let params = parameter_override.as_deref().unwrap_or(graph_params);
261
262    let additional_reps = parse_fn_map_entries(&payload.additional_fn_map_entries, state_map)?;
263    fn_map_entries.extend(additional_reps);
264
265    let result = vec![Complex::new(0.0, 0.0); exprs.len()];
266    let (replacements, all_replacements, fn_map) = apply_fn_map_entries(fn_map_entries)?;
267
268    if iterate {
269        let mut tree: Option<RationalExpressionTree> = None;
270
271        for expr in &exprs {
272            let eval = expr
273                .replace_multiple(&replacements)
274                .evaluator(params)
275                .function_map(fn_map.clone())
276                .optimization_settings(optimization_settings.clone())
277                .build()
278                .map_err(|e| eyre!("{e} for {expr}:{}", expr.replace_multiple(&replacements)))?;
279
280            tree = Some(if let Some((atoms, mut existing)) = tree {
281                existing.merge(eval, cpe_iterations).map_err(|e| eyre!(e))?;
282                (atoms, existing)
283            } else {
284                (exprs.clone(), eval)
285            });
286        }
287
288        tree.map(|(atoms, eval)| {
289            (
290                atoms,
291                all_replacements,
292                eval.map_coeff(&|r| Complex {
293                    re: r.re.to_f64(),
294                    im: r.im.to_f64(),
295                }),
296                result,
297            )
298        })
299        .ok_or_else(|| eyre!("No expressions in evaluator payload"))
300    } else {
301        let replaced_exprs = exprs
302            .iter()
303            .map(|expr| expr.replace_multiple(&replacements))
304            .collect::<Vec<_>>();
305
306        Atom::evaluator_multiple(&replaced_exprs, params)
307            .function_map(fn_map)
308            .optimization_settings(optimization_settings)
309            .build()
310            .map(|eval| {
311                (
312                    exprs,
313                    all_replacements,
314                    eval.map_coeff(&|r| Complex {
315                        re: r.re.to_f64(),
316                        im: r.im.to_f64(),
317                    }),
318                    result,
319                )
320            })
321            .map_err(|e| eyre!("{e}"))
322    }
323}

325fn build_stack<A: ImportWithMap>(
326    stack: StandaloneEvaluatorStackArchive<A>,
327    params: &[Atom],
328    fn_map_entries: &[SerializedFnMapEntry<A>],
329    state_map: &StateMap,
330    graph_name: &str,
331    label: &str,
332) -> Result<LoadedStandaloneEvaluatorStack> {
333    let parsed_fn_map_entries = parse_fn_map_entries(fn_map_entries, state_map)?;
334    let timed_build = |payload: StandaloneGenericEvaluatorArchive<A>,
335                       iterate: bool,
336                       component: &str|
337     -> Result<LoadedGenericEvaluator> {
338        let started = Instant::now();
339        let evaluator = build_evaluator(
340            payload,
341            params,
342            parsed_fn_map_entries.clone(),
343            state_map,
344            iterate,
345        )?;
346        println!(
347            "[timing] build_evaluator {graph_name}::{label}::{component} took {:?}",
348            started.elapsed()
349        );
350        Ok(evaluator)
351    };
352
353    Ok(LoadedStandaloneEvaluatorStack {
354        explicit_orientation_sum_only: stack.explicit_orientation_sum_only,
355        production_orientation_ids: stack.production_orientation_ids,
356        representative_input: stack.representative_input,
357        orientation_start: stack.start,
358        residue_map_id_start: stack.residue_map_id_start,
359        mult_offset: stack.mult_offset,
360        single_parametric: timed_build(stack.single_parametric, false, "single_parametric")?,
361        iterative: stack
362            .iterative
363            .map(|payload| timed_build(payload, true, "iterative"))
364            .transpose()?,
365        summed_function_map: stack
366            .summed_function_map
367            .map(|payload| timed_build(payload, false, "summed_function_map"))
368            .transpose()?,
369        summed: stack
370            .summed
371            .map(|payload| timed_build(payload, false, "summed"))
372            .transpose()?,
373    })
374}

422impl StandaloneCrossSectionArchive<(), String> {
423    pub fn load(self) -> Result<LoadedStandaloneCrossSection> {
424        if self.version != STANDALONE_EVALUATORS_VERSION {
425            return Err(eyre!(
426                "Unsupported version {} (expected {})",
427                self.version,
428                STANDALONE_EVALUATORS_VERSION
429            ));
430        }
431
432        let mut symbolica_state = Vec::new();
433        State::export(&mut symbolica_state)
434            .with_context(|| "Failed to export Symbolica state for standalone cross section")?;
435        let mut state_cursor = Cursor::new(&symbolica_state);
436        let state_map = State::import(&mut state_cursor, None)?;
437
438        self.load_impl(&state_map)
439    }
440}

458impl<S, A: ImportWithMap + Clone> StandaloneCrossSectionArchive<S, A> {
459    pub fn load_impl(self, state_map: &StateMap) -> Result<LoadedStandaloneCrossSection> {
460        let mut graph_terms = Vec::new();
461
462        for graph in self.graph_terms {
463            let params = graph
464                .param_builder_params
465                .iter()
466                .map(|param| param.import_with_map(state_map))
467                .collect::<Result<Vec<_>>>()?;
468
469            let build_indexed_stack_collection = |payloads: Vec<
470                StandaloneIndexedEvaluatorStackArchive<A>,
471            >,
472                                                  label: &str|
473             -> Result<
474                BTreeMap<StandaloneCutCFFIndex, LoadedStandaloneEvaluatorStack>,
475            > {
476                payloads
477                    .into_iter()
478                    .enumerate()
479                    .map(|(slot, payload)| {
480                        Ok((
481                            payload.cut_cff_index,
482                            build_stack(
483                                payload.evaluator_stack,
484                                &params,
485                                &graph.fn_map_entries,
486                                state_map,
487                                &graph.graph_name,
488                                &format!("{label}.slot[{slot}]"),
489                            )?,
490                        ))
491                    })
492                    .collect::<Result<BTreeMap<_, _>>>()
493            };
494
495            let build_indexed_generic_evaluator_collection = |payloads: Vec<
496                StandaloneIndexedGenericEvaluatorArchive<A>,
497            >,
498                                                              label: &str|
499             -> Result<
500                BTreeMap<StandaloneCutCFFIndex, LoadedGenericEvaluator>,
501            > {
502                let parsed_fn_map_entries = parse_fn_map_entries(&graph.fn_map_entries, state_map)?;
503                payloads
504                    .into_iter()
505                    .enumerate()
506                    .map(|(slot, payload)| {
507                        let started = Instant::now();
508                        let evaluator = build_evaluator(
509                            payload.evaluator,
510                            &params,
511                            parsed_fn_map_entries.clone(),
512                            state_map,
513                            false,
514                        )?;
515                        println!(
516                            "[timing] build_evaluator {}::{}.slot[{}] took {:?}",
517                            graph.graph_name,
518                            label,
519                            slot,
520                            started.elapsed()
521                        );
522                        Ok((payload.cut_cff_index, evaluator))
523                    })
524                    .collect::<Result<BTreeMap<_, _>>>()
525            };
526
527            let cut_group_integrands = graph
528                .cut_group_integrands
529                .into_iter()
530                .enumerate()
531                .map(|(cut_group_id, derivative_stacks)| {
532                    build_indexed_stack_collection(
533                        derivative_stacks,
534                        &format!("cut_group[{cut_group_id}]"),
535                    )
536                })
537                .collect::<Result<Vec<_>>>()?;
538
539            let counterterms = graph
540                .counterterms
541                .into_iter()
542                .enumerate()
543                .map(|(cut_id, counterterm)| {
544                    let build_stack_collection =
545                        |payloads: Vec<StandaloneIndexedEvaluatorStackArchive<A>>,
546                         label: &str|
547                         -> Result<BTreeMap<StandaloneCutCFFIndex, LoadedStandaloneEvaluatorStack>> {
548                            build_indexed_stack_collection(
549                                payloads,
550                                &format!("counterterm[{cut_id}]::{label}"),
551                            )
552                        };
553
554                    let build_stack_iterated =
555                        |payloads: StandaloneIteratedCollectionArchive<_>,
556                         label: &str,
557                         num_left_thresholds: usize,
558                         num_right_thresholds: usize|
559                         -> Result<
560                            LoadedStandaloneIteratedCollection<
561                                BTreeMap<StandaloneCutCFFIndex, LoadedStandaloneEvaluatorStack>,
562                            >,
563                        > {
564                            let expected_len = num_left_thresholds
565                                .checked_mul(num_right_thresholds)
566                                .ok_or_else(|| eyre!("{label} dimensions overflow usize"))?;
567                            if payloads.num_right_thresholds != num_right_thresholds
568                                || payloads.data.len() != expected_len
569                            {
570                                return Err(eyre!(
571                                    "counterterm[{cut_id}]::{label} has dimensions inconsistent with its left/right threshold collections: data_len={}, stored_right_count={}, expected={}x{}",
572                                    payloads.data.len(),
573                                    payloads.num_right_thresholds,
574                                    num_left_thresholds,
575                                    num_right_thresholds,
576                                ));
577                            }
578                            let data = payloads
579                                .data
580                                .into_iter()
581                                .enumerate()
582                                .map(|(i, payload)| {
583                                    build_stack_collection(payload, &format!("{label}[{i}]"))
584                                })
585                                .collect::<Result<Vec<_>>>()?;
586
587                            Ok(LoadedStandaloneIteratedCollection {
588                                data,
589                                num_right_thresholds: payloads.num_right_thresholds,
590                            })
591                        };
592
593                    let build_helper_iterated =
594                        |payloads: StandaloneIteratedCollectionArchive<_>,
595                         label: &str,
596                         num_left_thresholds: usize,
597                         num_right_thresholds: usize|
598                         -> Result<
599                            LoadedStandaloneIteratedCollection<
600                                BTreeMap<StandaloneCutCFFIndex, LoadedGenericEvaluator>,
601                            >,
602                        > {
603                            let expected_len = num_left_thresholds
604                                .checked_mul(num_right_thresholds)
605                                .ok_or_else(|| eyre!("{label} dimensions overflow usize"))?;
606                            if payloads.num_right_thresholds != num_right_thresholds
607                                || payloads.data.len() != expected_len
608                            {
609                                return Err(eyre!(
610                                    "counterterm[{cut_id}]::{label} has dimensions inconsistent with its left/right threshold collections: data_len={}, stored_right_count={}, expected={}x{}",
611                                    payloads.data.len(),
612                                    payloads.num_right_thresholds,
613                                    num_left_thresholds,
614                                    num_right_thresholds,
615                                ));
616                            }
617                            let data = payloads
618                                .data
619                                .into_iter()
620                                .enumerate()
621                                .map(|(i, payload)| {
622                                    build_indexed_generic_evaluator_collection(
623                                        payload,
624                                        &format!("counterterm[{cut_id}]::{label}[{i}]"),
625                                    )
626                                })
627                                .collect::<Result<Vec<_>>>()?;
628
629                            Ok(LoadedStandaloneIteratedCollection {
630                                data,
631                                num_right_thresholds: payloads.num_right_thresholds,
632                            })
633                        };
634
635                    let pass_two_evaluator = counterterm
636                        .pass_two_evaluator
637                        .into_iter()
638                        .enumerate()
639                        .map(|(order, payload)| {
640                            let started = Instant::now();
641                            let evaluator = build_evaluator(
642                                payload,
643                                &params,
644                                parse_fn_map_entries(&graph.fn_map_entries, state_map)?,
645                                state_map,
646                                false,
647                            )?;
648                            println!(
649                                "[timing] build_evaluator {}::counterterm[{cut_id}]::pass_two_evaluator[{order}] took {:?}",
650                                graph.graph_name,
651                                started.elapsed()
652                            );
653                            Ok(evaluator)
654                        })
655                        .collect::<Result<Vec<_>>>()?;
656
657                    let left_thresholds_evaluator = counterterm
658                        .left_thresholds_evaluator
659                        .into_iter()
660                        .enumerate()
661                        .map(|(i, payload)| {
662                            build_stack_collection(
663                                payload,
664                                &format!("left_thresholds_evaluator[{i}]"),
665                            )
666                        })
667                        .collect::<Result<Vec<_>>>()?;
668                    let right_thresholds_evaluator = counterterm
669                        .right_thresholds_evaluator
670                        .into_iter()
671                        .enumerate()
672                        .map(|(i, payload)| {
673                            build_stack_collection(
674                                payload,
675                                &format!("right_thresholds_evaluator[{i}]"),
676                            )
677                        })
678                        .collect::<Result<Vec<_>>>()?;
679                    let left_threshold_helpers = counterterm
680                        .left_threshold_helpers
681                        .into_iter()
682                        .enumerate()
683                        .map(|(i, payload)| {
684                            build_indexed_generic_evaluator_collection(
685                                payload,
686                                &format!(
687                                    "counterterm[{cut_id}]::left_threshold_helpers[{i}]"
688                                ),
689                            )
690                        })
691                        .collect::<Result<Vec<_>>>()?;
692                    let right_threshold_helpers = counterterm
693                        .right_threshold_helpers
694                        .into_iter()
695                        .enumerate()
696                        .map(|(i, payload)| {
697                            build_indexed_generic_evaluator_collection(
698                                payload,
699                                &format!(
700                                    "counterterm[{cut_id}]::right_threshold_helpers[{i}]"
701                                ),
702                            )
703                        })
704                        .collect::<Result<Vec<_>>>()?;
705                    let iterated_evaluator = build_stack_iterated(
706                        counterterm.iterated_evaluator,
707                        "iterated_evaluator",
708                        left_thresholds_evaluator.len(),
709                        right_thresholds_evaluator.len(),
710                    )?;
711                    let iterated_helpers = build_helper_iterated(
712                        counterterm.iterated_helpers,
713                        "iterated_helpers",
714                        left_threshold_helpers.len(),
715                        right_threshold_helpers.len(),
716                    )?;
717
718                    Ok(LoadedStandaloneCounterterm {
719                        left_thresholds_evaluator,
720                        right_thresholds_evaluator,
721                        iterated_evaluator,
722                        left_threshold_helpers,
723                        right_threshold_helpers,
724                        iterated_helpers,
725                        pass_two_evaluator,
726                    })
727                })
728                .collect::<Result<Vec<_>>>()?;
729
730            graph_terms.push(LoadedStandaloneCrossSectionGraphTerm {
731                graph_name: graph.graph_name,
732                orientations: graph.orientations,
733                param_builder_params: params,
734                cut_group_integrands,
735                counterterms,
736            });
737        }
738
739        Ok(LoadedStandaloneCrossSection { graph_terms })
740    }
741}

743fn load_bin(path: impl AsRef<Path>) -> Result<LoadedStandaloneCrossSection> {
744    let binary =
745        fs::read(&path).with_context(|| format!("Cannot read {}", path.as_ref().display()))?;
746    let (archive, _): (StandaloneCrossSectionArchive, _) =
747        bincode::decode_from_slice(&binary, bincode::config::standard())?;
748    archive.load()
749}
750
751fn load_json(path: impl AsRef<Path>) -> Result<LoadedStandaloneCrossSection> {
752    let binary =
753        fs::read(&path).with_context(|| format!("Cannot read {}", path.as_ref().display()))?;
754    let archive: StandaloneCrossSectionArchive<(), String> = serde_json::from_slice(&binary)?;
755    archive.load()
756}
757
758fn default_input_path() -> PathBuf {
759    let binary = PathBuf::from("standalone_cross_section.bin");
760    if binary.exists() {
761        binary
762    } else {
763        PathBuf::from("standalone_cross_section.json")
764    }
765}
766
767fn main() -> Result<()> {
768    let input = std::env::args()
769        .nth(1)
770        .map(PathBuf::from)
771        .unwrap_or_else(default_input_path);
772
773    let Some(ext) = input.extension() else {
774        return Err(eyre!("No extension, expected .bin or .json"));
775    };
776
777    let loaded = match ext.to_string_lossy().as_ref() {
778        "bin" => load_bin(&input)?,
779        "json" => load_json(&input)?,
780        _ => {
781            return Err(eyre!(
782                "Unsupported file extension {}, expected .bin or .json",
783                ext.to_string_lossy()
784            ));
785        }
786    };
787
788    println!("Loaded {} graph terms", loaded.graph_terms.len());
789    for graph in &loaded.graph_terms {
790        println!(
791            "graph={} orientations={} cut_groups={} counterterms={}",
792            graph.graph_name,
793            graph.orientations.len(),
794            graph.cut_group_integrands.len(),
795            graph.counterterms.len()
796        );
797        for (cut_group_id, derivative_stacks) in graph.cut_group_integrands.iter().enumerate() {
798            println!(
799                "  cut_group[{cut_group_id}] derivative_evaluators={}",
800                derivative_stacks.len()
801            );
802        }
803        for (cut_id, counterterm) in graph.counterterms.iter().enumerate() {
804            println!(
805                "  counterterm[{cut_id}] left={} right={} iterated={} left_helpers={} right_helpers={} iterated_helpers={} pass_two_exprs={}",
806                counterterm.left_thresholds_evaluator.len(),
807                counterterm.right_thresholds_evaluator.len(),
808                counterterm.iterated_evaluator.data.len(),
809                counterterm.left_threshold_helpers.len(),
810                counterterm.right_threshold_helpers.len(),
811                counterterm.iterated_helpers.data.len(),
812                counterterm.pass_two_evaluator.len()
813            );
814        }
815    }
816
817    Ok(())
818}
