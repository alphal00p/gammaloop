28/// Persistent ownership carried by an EMR momentum through a local 4D UV
29/// Taylor expansion. Wire value two is deliberately reserved for the
30/// transient denominator marker used while the Taylor operator is acting.
31#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
32#[repr(i64)]
33pub(crate) enum UvMomentumProvenanceRole {
34    /// A pre-existing numerator factor fixed to its original owner during raw
35    /// Taylor recursion. Completed UV projection may certify a denominator
36    /// class and transfer this factor to that class's occurrence pool.
37    TaylorFixed = 0,
38    /// A hard numerator factor created by differentiating a denominator. Raw
39    /// provenance permits copies of the same owner; completed UV projection
40    /// may certify equivalent signed channels before class-wide dispatch.
41    DenominatorDerived = 1,
42    /// A physical source momentum reconstructed from its stored hard lift,
43    /// independently of occurrence-local contact samples.
44    PhysicalSourceFixed = 3,
45    /// A soft momentum created by a denominator Taylor derivative. Its exact
46    /// off-shell routing may be chosen when the outer numerator is complete;
47    /// it is not an occurrence of the differentiated hard denominator.
48    DenominatorDerivedSoft = 4,
49}
50
51/// An algebraic denominator channel in one completed UV projection view.
52/// This namespace is distinct from physical source edges and CFF occurrences.
53#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
54pub(crate) struct UvDenominatorClassId(pub usize);

56impl From<bool> for UvMomentumProvenanceRole {
57    fn from(denominator_derived: bool) -> Self {
58        if denominator_derived {
59            Self::DenominatorDerived
60        } else {
61            Self::TaylorFixed
62        }
63    }
64}

279    pub fn collect_orientation_if<'a>(&self, arg: impl Into<AtomOrView<'a>>) -> Atom {
280        let arg = arg.into();
281        if !arg.as_view().contains_symbol(self.theta) && !arg.as_view().contains_symbol(Symbol::IF)
282        {
283            return arg.into_owned();
284        }
285        arg.replace(self.sign_theta(W_.a_))
286            .with(Symbol::IF.call(Atom::var(W_.a_) + 1))
287            // A generalized residue-map delta is represented as
288            // IF(current_id-key, 0, 1). Move the selected branch body inside
289            // that conditional so an inactive key is discarded before any of
290            // its selector-local inverses are evaluated.
291            .replace(Symbol::IF.call_args([Atom::var(W_.a_), Atom::Zero, Atom::one()]) * W_.b___)
292            .with(Symbol::IF.call_args([Atom::var(W_.a_), Atom::Zero, Atom::var(W_.b___)]))
293            .replace(Symbol::IF.call(W_.a_) * Symbol::IF.call(W_.b_))
294            .repeat()
295            .with(Symbol::IF.call(W_.a_ * W_.b_))
296            .replace(Symbol::IF.call(W_.a_) * W_.b___)
297            .with(Symbol::IF.call_args([Atom::var(W_.a_), Atom::var(W_.b___), Atom::Zero]))
298            .replace(Symbol::IF.call_args([Atom::var(W_.a_), Atom::Zero]))
299            .with(Symbol::IF.call_args([Atom::var(W_.a_), Atom::one(), Atom::Zero]))
300            .replace(Symbol::IF.call_args([Atom::var(W_.a_)]))
301            .with(Symbol::IF.call_args([Atom::var(W_.a_), Atom::one(), Atom::Zero]))
302            // Tensor contractions are complete here. Combine scalar contributions
303            // selected by the same key while keeping their complete bodies lazy.
304            .replace(
305                Symbol::IF.call_args([Atom::var(W_.a_), Atom::Zero, Atom::var(W_.b_)])
306                    + Symbol::IF.call_args([Atom::var(W_.a_), Atom::Zero, Atom::var(W_.c_)]),
307            )
308            .repeat()
309            .with(Symbol::IF.call_args([Atom::var(W_.a_), Atom::Zero, Atom::var(W_.b_) + W_.c_]))
310    }

438macro_rules! spenso_print_scripted_indexed {
439    ($a:ident, $opt:ident, $symbol:expr) => {
440        spenso_print_scripted_indexed!($a, $opt, $symbol, $symbol)
441    };
442    ($a:ident, $opt:ident, $symbol:expr, $typst_symbol:expr) => {{
443        match $opt.custom_print_mode.get("spenso") {
444            Some(PrintUserData::Integer(i)) => {
445                let SpensoPrintSettings {
446                    parens,
447                    symbol_scripts,
448                    commas,
449                    with_dim,
450                    ..
451                } = SpensoPrintSettings::from(*i as usize);
452
453                let AtomView::Fun(f) = $a else {
454                    return None;
455                };
456
457                let mut argiter = f.iter();
458                let id = argiter.next()?;
459                let Ok(i) = usize::try_from(id) else {
460                    return None;
461                };
462
463                let is_typst = $opt.typst_mode().is_some();
464                let mut out = if is_typst {
465                    $typst_symbol.to_string()
466                } else {
467                    $symbol.to_string()
468                };
469                if is_typst {
470                    out.push('_');
471                    out.push_str(&i.to_string());
472                } else {
473                    out.push_str(&to_subscript(i as isize));
474                }
475                if $opt.color_builtin_symbols && !is_typst {
476                    out = nu_ansi_term::Color::Magenta.paint(out).to_string();
477                }
478
479                let mut printed_args = false;
480                for arg in argiter {
481                    let hidden_representation = matches!(
482                        arg,
483                        AtomView::Fun(a)
484                            if a.get_symbol().has_tag(&SPENSO_TAG.representation)
485                                && a.get_nargs() == 1
486                                && !with_dim
487                    );
488                    if hidden_representation {
489                        continue;
490                    }
491
492                    if printed_args {
493                        out.push(if commas { ',' } else { ' ' });
494                    } else {
495                        if symbol_scripts {
496                            out.push('^');
497                        }
498                        if parens {
499                            out.push('(');
500                        }
501                        printed_args = true;
502                    }
503
504                    arg.format(&mut out, $opt, PrintState::new()).unwrap();
505                }
506                if printed_args && parens {
507                    out.push(')');
508                }
509                Some(out)
510            }
511            _ => None,
512        }
513    }};
514}

542                let mut first = true;
543                for arg in args {

545                        first = false;
546                        out.push('(');
547                    } else {
548                        out.push(',');
549                    }
550                    arg.format(&mut out, $opt, PrintState::new()).unwrap();

553                    out.push(')');
554                }

1426    pub(crate) fn localizing_integrand(&self, lmb: &LoopMomentumBasis) -> Atom {
1427        // Normalize each factor with int d^3k / (|k|^2 + rls^2)^2 = pi^2 / rls,
1428        // so the localizing integrand itself integrates to one.
1429        let pi_atom = (Symbol::PI).to_atom();
1430        let normalization_term_integral = (pi_atom.pow(2)) / GS.renormalization_localization_scale;
1431
1432        let mut res = Atom::one();
1433
1434        for l in lmb.loop_edges.iter() {
1435            // Orientation ownership is applied by `Localizer`: orientation-local 3D uses one
1436            // deterministic valid host, while selector-free 3D and local 4D explicitly sum their
1437            // source orientations. It is therefore not an extra factor in this normalized radial
1438            // kernel.
1439            res /= normalization_term_integral.as_view();
1440
1441            let spatial_norm_sq = function!(self.emr_mom, l.0, GS.cind(1)).pow(2)
1442                + function!(self.emr_mom, l.0, GS.cind(2)).pow(2)
1443                + function!(self.emr_mom, l.0, GS.cind(3)).pow(2);
1444
1445            // Per-orientation CFF localizer of the normalized cubic tadpole.
1446            let denominator = spatial_norm_sq
1447                + GS.renormalization_localization_scale * GS.renormalization_localization_scale;
1448            res /= denominator.as_view() * denominator.as_view();
1449        }
1450
1451        if res.is_one() {
1452            res
1453        } else {
1454            function!(self.localizing_integrand, res)
1455        }
1456    }
