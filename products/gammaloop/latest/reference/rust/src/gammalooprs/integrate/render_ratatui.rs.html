194        let Some(cell) = self.row.slot_cell(slot_index) else {
195            return 0.0;
196        };
197        let statistics = cell.statistics(view);

207                .map(|value| value.raw.0.abs().0)
208                .unwrap_or_default(),
209            ContributionSortMode::Error => statistics
210                .value
211                .as_ref()
212                .map(|value| value.raw.1.abs().0)
213                .unwrap_or_default(),
214        }

476    fn push_history_point(&mut self, update: &StatusUpdate) {
477        let signed_real_slot_values =
478            self.collect_history_slot_values(update, ComponentKind::Real, IntegralView::Signed);
479        let signed_imag_slot_values =
480            self.collect_history_slot_values(update, ComponentKind::Imag, IntegralView::Signed);
481        let absolute_real_slot_values =
482            self.collect_history_slot_values(update, ComponentKind::Real, IntegralView::Absolute);
483        let absolute_imag_slot_values =
484            self.collect_history_slot_values(update, ComponentKind::Imag, IntegralView::Absolute);
485        if signed_real_slot_values.iter().all(Option::is_none)
486            && signed_imag_slot_values.iter().all(Option::is_none)
487            && absolute_real_slot_values.iter().all(Option::is_none)
488            && absolute_imag_slot_values.iter().all(Option::is_none)
489        {
490            return;
491        }
492
493        let point = HistoryPoint {
494            iteration: update.meta.iteration,
495            samples: update.meta.total_points,
496            signed_real_slot_values,
497            signed_imag_slot_values,
498            absolute_real_slot_values,
499            absolute_imag_slot_values,
500            completed_iteration: !matches!(update.kind(), super::IntegrationStatusKind::Live),
501        };
502
503        if self
504            .history
505            .last()
506            .is_some_and(|previous| previous.samples == point.samples)
507        {
508            let preserve_completed_point = self.history.last().is_some_and(|previous| {
509                previous.completed_iteration
510                    && !point.completed_iteration
511                    && update.is_initial_live_status()
512            });
513            if !preserve_completed_point {
514                let _ = self.history.pop();
515            }
516        }
517        if self
518            .history
519            .last()
520            .is_some_and(|previous| previous.samples > point.samples)
521        {
522            self.history.clear();
523        }
524        self.history.push(point);
525        while self.history.len() > MAX_HISTORY_POINTS {
526            self.compact_history_preserving_span();
527        }
528    }

530    fn compact_history_preserving_span(&mut self) {
531        if self.history.len() <= MAX_HISTORY_POINTS {
532            return;
533        }
534
535        let retain_recent = MAX_HISTORY_POINTS / 2;
536        let split_index = self.history.len().saturating_sub(retain_recent);
537        if split_index == 0 {
538            return;
539        }
540
541        let older = &self.history[..split_index];
542        let recent = &self.history[split_index..];
543        let max_older_points = MAX_HISTORY_POINTS.saturating_sub(recent.len()).max(2);
544        let step = older.len().div_ceil(max_older_points);
545
546        let mut compacted = Vec::with_capacity(MAX_HISTORY_POINTS);
547        for (index, point) in older.iter().enumerate() {
548            if index == 0 || index + 1 == older.len() || index % step == 0 {
549                compacted.push(point.clone());
550            }
551        }
552        compacted.extend(recent.iter().cloned());
553
554        if compacted.len() > MAX_HISTORY_POINTS {
555            let keep_from_older = compacted.len().saturating_sub(recent.len());
556            let extra = compacted.len() - MAX_HISTORY_POINTS;
557            if keep_from_older > extra + 1 {
558                compacted.drain(1..=extra);
559            } else {
560                compacted.truncate(MAX_HISTORY_POINTS);
561            }
562        }
563
564        self.history = compacted;
565    }

567    fn collect_history_slot_values(
568        &self,
569        update: &StatusUpdate,
570        component: ComponentKind,
571        view: IntegralView,
572    ) -> Vec<Option<(f64, f64)>> {

581                row.slot_cell(slot_index)
582                    .and_then(|cell| cell.statistics(view).value.as_ref())
583                    .map(|value| (value.raw.0.0, value.raw.1.0.abs()))
584            })

588    fn visible_history(&self) -> Vec<&HistoryPoint> {
589        match self.chart_history_window {
590            ChartHistoryWindow::Full => self.history.iter().collect(),
591            ChartHistoryWindow::RecentIterations(iterations) => {
592                if iterations == 0 || self.history.is_empty() {
593                    return self.history.iter().collect();
594                }
595                let latest_iteration = self
596                    .history
597                    .last()
598                    .map(|point| point.iteration)
599                    .unwrap_or_default();
600                let start_iteration = latest_iteration
601                    .saturating_add(1)
602                    .saturating_sub(iterations);
603                self.history
604                    .iter()
605                    .filter(|point| point.iteration >= start_iteration)
606                    .collect()
607            }
608        }
609    }

2232                    }
2233                    _ => Ordering::Equal,
2234                },
2235                ContributionSortMode::Integral | ContributionSortMode::Error => lhs_key
2236                    .partial_cmp(&rhs_key)
2237                    .unwrap_or(Ordering::Equal)
2238                    .then(lhs.component().cmp(&rhs.component())),
2239            };

2241                ordering.reverse()

2598fn mix_bar_line(segments: &[StatisticsMixSegment], width: usize) -> Line<'static> {
2599    let bar_width = width.max(8);
2600    let mut spans = Vec::new();
2601    let mut used = 0usize;
2602    for (index, segment) in segments.iter().enumerate() {
2603        let color = color_from_styled_text(&segment.label, Color::DarkGray);
2604        let width = if index + 1 == segments.len() {
2605            bar_width.saturating_sub(used)
2606        } else {
2607            let segment_width = ((segment.percentage / 100.0) * bar_width as f64).round() as usize;
2608            used += segment_width;
2609            segment_width
2610        };
2611        spans.push(Span::styled("█".repeat(width), Style::default().fg(color)));
2612    }
2613    Line::from(spans)
2614}

2656fn evenly_spaced_values(min: f64, max: f64, count: usize) -> Vec<f64> {
2657    if count <= 1 || !min.is_finite() || !max.is_finite() || (max - min).abs() < f64::EPSILON {
2658        return vec![min];
2659    }
2660
2661    let step = (max - min) / (count.saturating_sub(1) as f64);
2662    (0..count).map(|index| min + step * index as f64).collect()
2663}
