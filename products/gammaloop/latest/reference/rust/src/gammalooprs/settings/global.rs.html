432fn deserialize_orientation_pattern_atom<'de, D>(
433    deserializer: D,
434) -> std::result::Result<Option<StringSerializedAtom>, D::Error>
435where
436    D: Deserializer<'de>,
437{
438    let raw = Option::<String>::deserialize(deserializer)?;
439    raw.map(|value| OrientationPattern::parse_user_pattern(&value))
440        .transpose()
441        .map(|parsed| parsed.map(StringSerializedAtom))
442        .map_err(serde::de::Error::custom)
443}

445impl From<Atom> for OrientationPattern {
446    fn from(value: Atom) -> Self {
447        OrientationPattern {
448            pat: Some(StringSerializedAtom(value)),
449        }
450    }
451}

453impl OrientationPattern {
454    fn is_orientation_delta_atom(atom: &Atom) -> bool {
455        matches!(
456            atom.as_view(),
457            AtomView::Fun(function)
458                if function.get_symbol().get_stripped_name() == "orientation_delta"
459        )
460    }
461
462    fn split_top_level_args(input: &str) -> EyreResult<Vec<String>> {
463        let mut args = Vec::new();
464        let mut start = 0usize;
465        let mut depth = 0usize;
466
467        for (index, ch) in input.char_indices() {
468            match ch {
469                '(' | '[' | '{' => depth += 1,
470                ')' | ']' | '}' => {
471                    if depth == 0 {
472                        return Err(eyre!(
473                            "Unbalanced delimiter in orientation pattern: {input}"
474                        ));
475                    }
476                    depth -= 1;
477                }
478                ',' if depth == 0 => {
479                    let arg = input[start..index].trim();
480                    if arg.is_empty() {
481                        return Err(eyre!("Empty orientation-pattern entry in pattern: {input}"));
482                    }
483                    args.push(arg.to_string());
484                    start = index + ch.len_utf8();
485                }
486                _ => {}
487            }
488        }
489
490        if depth != 0 {
491            return Err(eyre!(
492                "Unbalanced delimiter in orientation pattern: {input}"
493            ));
494        }
495
496        let tail = input[start..].trim();
497        if tail.is_empty() {
498            if args.is_empty() {
499                return Err(eyre!("Orientation pattern cannot be empty"));
500            }
501            return Err(eyre!("Empty orientation-pattern entry in pattern: {input}"));
502        }
503        args.push(tail.to_string());
504
505        Ok(args)
506    }
507
508    fn normalize_user_pattern(pattern: &str) -> EyreResult<String> {
509        let trimmed = pattern.trim();
510        if trimmed.is_empty() {
511            return Err(eyre!("Orientation pattern cannot be empty"));
512        }
513
514        let args = if let Some(rest) = trimmed.strip_prefix("orientation_delta") {
515            let rest = rest.trim();
516            if !(rest.starts_with('(') && rest.ends_with(')')) {
517                return Err(eyre!(
518                    "orientation_delta patterns must use parentheses, got: {pattern}"
519                ));
520            }
521            Self::split_top_level_args(&rest[1..rest.len() - 1])?
522        } else if trimmed.starts_with('(') && trimmed.ends_with(')') {
523            Self::split_top_level_args(&trimmed[1..trimmed.len() - 1])?
524        } else {
525            Self::split_top_level_args(trimmed)?
526        };
527
528        let normalized_args = args
529            .into_iter()
530            .map(|arg| match arg.as_str() {
531                "+" | "+1" => "1".to_string(),
532                "-" | "-1" => "-1".to_string(),
533                _ => arg,
534            })
535            .collect::<Vec<_>>()
536            .join(",");
537
538        Ok(format!("orientation_delta({normalized_args})"))
539    }
540
541    pub fn parse_user_pattern(pattern: &str) -> EyreResult<Atom> {
542        let trimmed = pattern.trim();
543        if trimmed.is_empty() {
544            return Err(eyre!("Orientation pattern cannot be empty"));
545        }
546
547        if let Ok(parsed) = try_parse!(trimmed)
548            && Self::is_orientation_delta_atom(&parsed)
549        {
550            return Ok(parsed);
551        }
552
553        let normalized = Self::normalize_user_pattern(pattern)?;
554        try_parse!(normalized.as_str())
555            .map_err(|error| eyre!("Symbolica parsing error for orientation pattern: {error}"))
556    }
557
558    pub fn from_user_pattern(pattern: &str) -> EyreResult<Self> {
559        Ok(Self {
560            pat: Some(StringSerializedAtom(Self::parse_user_pattern(pattern)?)),
561        })
562    }
563
564    pub fn from_orientation<O: GraphOrientation>(orientation: &O) -> Self {
565        orientation.orientation_delta().into()
566    }
567
568    pub fn select_pattern(&self, atom: impl AtomCore) -> Option<Atom> {
569        Some(
570            atom.replace(self.pat.as_ref()?.as_view().to_pattern())
571                .with(function!(GS.selected, &self.pat.as_ref()?.0))
572                .replace(function!(GS.orientation_delta, W_.a___))
573                .level_range((0, Some(0)))
574                .with(Atom::Zero),
575        )
576    }
577
578    pub fn filter<O: GraphOrientation>(&self, orientation: &O) -> bool {
579        if let Some(pat) = &self.pat {
580            let a = orientation.orientation_delta();
581
582            // println!("{a}vs{}", pat.0);
583
584            // println!("{a}");
585            a.pattern_match(&pat.to_pattern(), None, None)
586                .next()
587                .is_some()
588        } else {
589            true
590        }
591    }
592
593    pub fn alt_filter<O: GraphOrientation>(&self, orientation: &O) -> bool {
594        self.filter(orientation)
595    }
