69pub enum UVLimitSelection {
70    /// One representative for every expected divergent cycle union: the generation LMB when
71    /// suitable, otherwise the first suitable basis in the deterministically sorted full list.
72    #[default]
73    OnlyDivergent,
74    /// Every nonempty loop subset in every generated physical LMB.
75    All,
76}
77
78impl UVLimitSelection {
79    fn cycle_union(
80        graph: &Graph,
81        lmb: &LoopMomentumBasis,
82        subset: &SubSet<LoopIndex>,
83    ) -> SuBitGraph {
84        let mut subgraph: SuBitGraph = graph.empty_subgraph();
85        for (pair, edge_id, _) in graph.underlying.iter_edges() {
86            if matches!(pair, HedgePair::Paired { .. })
87                && subset.included_iter().any(|loop_index| {
88                    lmb.edge_signatures[edge_id].internal[loop_index] != SignOrZero::Zero
89                })
90            {
91                subgraph.add(pair);
92            }
93        }
94        subgraph
95    }
96
97    fn all_lmb_limits(
98        graph: &Graph,
99        lmb: &LoopMomentumBasis,
100        profile_domain: &SuBitGraph,
101        compatible_cuts: Option<&[SuBitGraph]>,
102    ) -> Vec<(SubSet<LoopIndex>, i32)> {
103        let mut loops = PowersetIterator::<LoopIndex>::new(lmb.loop_edges.len() as u8);
104        loops.next();
105        loops
106            .filter_map(|subset| {
107                let cycle = Self::cycle_union(graph, lmb, &subset);
108                let is_in_profile_domain = cycle.subtract(profile_domain).is_empty();
109                let is_cut_compatible = compatible_cuts
110                    .is_none_or(|cuts| cuts.iter().any(|cut| !cycle.intersects(cut)));
111                (is_in_profile_domain && is_cut_compatible)
112                    .then(|| (subset, graph.compute_dod(&cycle)))
113            })
114            .collect()
115    }
116
117    fn lmb_limits(
118        self,
119        graph: &Graph,
120        all_lmbs: &TiVec<LmbIndex, LoopMomentumBasis>,
121        profile_domain: &SuBitGraph,
122        compatible_cuts: Option<&[SuBitGraph]>,
123    ) -> Result<ProfileLmbLimits> {
124        if matches!(self, Self::All) {
125            return Ok(all_lmbs
126                .iter()
127                .cloned()
128                .map(|lmb| {
129                    let limits = Self::all_lmb_limits(graph, &lmb, profile_domain, compatible_cuts);
130                    (lmb, limits)
131                })
132                .collect());
133        }
134
135        let mut sorted_lmbs = all_lmbs.iter().collect_vec();
136        sorted_lmbs.sort_by_key(|lmb| {
137            (
138                lmb.loop_edges.iter().map(|edge| edge.0).collect_vec(),
139                lmb.ext_edges.iter().map(|edge| edge.0).collect_vec(),
140                lmb.tree.string_label(),
141            )
142        });
143        let mut seen_lmbs = HashSet::new();
144        let candidate_lmbs = std::iter::once(&graph.loop_momentum_basis)
145            .chain(sorted_lmbs)
146            .filter(|lmb| seen_lmbs.insert(*lmb))
147            .collect_vec();
148        let profile_domains = compatible_cuts
149            .map(|cuts| {
150                cuts.iter()
151                    .map(|cut| profile_domain.subtract(cut))
152                    .collect_vec()
153            })
154            .unwrap_or_else(|| vec![profile_domain.clone()]);
155        let mut targets = profile_domains
156            .into_iter()
157            .filter(|domain| !domain.is_empty())
158            .flat_map(|domain| graph.all_cycle_unions(&domain))
159            .filter_map(|cycle| {
160                let cycle = cycle.filter;
161                (!cycle.is_empty()).then(|| {
162                    let dod = graph.compute_dod(&cycle);
163                    (cycle, dod)
164                })
165            })
166            .filter(|(_, dod)| *dod >= 0)
167            .collect_vec();
168        targets.sort_by_key(|(cycle, _)| cycle.string_label());
169        targets.dedup_by(|(left, _), (right, _)| left == right);
170        // Most graphs are fully represented by the generation LMB. Build support maps lazily so
171        // stored spanning-tree bases are touched only when an earlier basis cannot represent a
172        // divergent cycle union.
173        let mut candidate_supports = vec![None; candidate_lmbs.len()];
174
175        let mut selected: ProfileLmbLimits = Vec::new();
176        for (target, dod) in targets {
177            let mut representation = None;
178            for (candidate_index, &lmb) in candidate_lmbs.iter().enumerate() {
179                let supports = candidate_supports[candidate_index].get_or_insert_with(|| {
180                    (0..lmb.loop_edges.len())
181                        .map(|loop_index| {
182                            let loop_index = LoopIndex(loop_index);
183                            let mut support: SuBitGraph = graph.empty_subgraph();
184                            for (pair, edge_id, _) in graph.underlying.iter_edges() {
185                                if matches!(pair, HedgePair::Paired { .. })
186                                    && lmb.edge_signatures[edge_id].internal[loop_index]
187                                        != SignOrZero::Zero
188                                {
189                                    support.add(pair);
190                                }
191                            }
192                            support
193                        })
194                        .collect_vec()
195                });
196                let mut subset = SubSet::empty(lmb.loop_edges.len());
197                let mut represented_cycle: SuBitGraph = graph.empty_subgraph();
198                for (loop_index, support) in supports.iter().enumerate() {
199                    let loop_index = LoopIndex(loop_index);
200                    if !support.is_empty() && support.subtract(&target).is_empty() {
201                        subset.add(loop_index);
202                        represented_cycle.union_with(support);
203                    }
204                }
205
206                if represented_cycle == target {
207                    representation = Some((lmb, subset));
208                    break;
209                }
210            }
211
212            let Some((lmb, subset)) = representation else {
213                return Err(eyre!(
214                    "Divergent UV cycle {} (DOD {dod}) in graph '{}' is not represented by the generation LMB or any basis in the complete generated LMB list",
215                    target.string_label(),
216                    graph.name,
217                ));
218            };
219            if let Some((_, limits)) = selected
220                .iter_mut()
221                .find(|(selected_lmb, _)| selected_lmb == lmb)
222            {
223                limits.push((subset, dod));
224            } else {
225                selected.push((lmb.clone(), vec![(subset, dod)]));
226            }
227        }
228        Ok(selected)
229    }

269pub struct UVProfileFixedRay {
270    directions: Vec<[f64; 3]>,
271    norms: Vec<f64>,
272}
273
274impl UVProfileFixedRay {
275    pub fn from_flat_components(directions: &[f64], norms: &[f64]) -> Result<Self> {
276        if directions.is_empty() {
277            return Err(eyre!(
278                "Fixed UV ray directions cannot be empty when the option is used."
279            ));
280        }
281        if !directions.len().is_multiple_of(3) {
282            return Err(eyre!(
283                "Fixed UV ray directions must contain a multiple of 3 components, got {}.",
284                directions.len()
285            ));
286        }
287        if norms.is_empty() {
288            return Err(eyre!(
289                "Fixed UV ray norms cannot be empty when directions are supplied."
290            ));
291        }
292
293        let directions = directions
294            .as_chunks::<3>()
295            .0
296            .iter()
297            .map(|direction| {
298                if direction.iter().any(|component| !component.is_finite()) {
299                    return Err(eyre!("Fixed UV ray directions must be finite."));
300                }
301                let scale = direction
302                    .iter()
303                    .map(|component| component.abs())
304                    .fold(0.0, f64::max);
305                if scale == 0.0 {
306                    return Err(eyre!(
307                        "Fixed UV ray directions cannot contain a zero vector."
308                    ));
309                }
310                // Normalize after scaling to preserve both subnormal and very large rays.
311                let direction = direction.map(|component| component / scale);
312                let norm = direction[0].hypot(direction[1]).hypot(direction[2]);
313                Ok(direction.map(|component| component / norm))
314            })
315            .collect::<Result<Vec<_>>>()?;
316
317        if norms.iter().any(|norm| !norm.is_finite() || *norm <= 0.0) {
318            return Err(eyre!(
319                "Fixed UV ray norms must be finite and strictly positive."
320            ));
321        }
322
323        Ok(Self {
324            directions,
325            norms: norms.to_vec(),
326        })
327    }
328
329    fn sample(&self, loop_count: usize) -> Result<LoopMomentumSample> {
330        if self.directions.len() != 1 && self.directions.len() != loop_count {
331            return Err(eyre!(
332                "Fixed UV ray directions contain {} ray(s), but this LMB has {} loop variable(s). Use either one direction or one direction per loop variable.",
333                self.directions.len(),
334                loop_count
335            ));
336        }
337        if self.norms.len() != 1 && self.norms.len() != loop_count {
338            return Err(eyre!(
339                "Fixed UV ray norms contain {} value(s), but this LMB has {} loop variable(s). Use either one norm or one norm per loop variable.",
340                self.norms.len(),
341                loop_count
342            ));
343        }
344
345        (0..loop_count)
346            .map(|i| {
347                let direction = self.directions[if self.directions.len() == 1 { 0 } else { i }];
348                let norm = self.norms[if self.norms.len() == 1 { 0 } else { i }];
349                Ok(ThreeMomentum {
350                    px: F(norm * direction[0]),
351                    py: F(norm * direction[1]),
352                    pz: F(norm * direction[2]),
353                })
354            })
355            .collect()
356    }
357}

1714    let log_start = start;
1715    let log_stop = stop;

1717
1718    (0..num)
1719        .map(|i| {

1721            base.powf(exponent)
1722        })
1723        .collect()
1724}

2722fn inspect_failure_reason(
2723    analysis: Option<&InspectAnalysis>,
2724    fit_status: InspectFitStatus,
2725    max_dod: f64,
2726    allow_vanishing_missing_fits: bool,
2727) -> Option<&'static str> {
2728    if allow_vanishing_missing_fits && fit_status.vanishes_without_power_law(analysis, max_dod) {
2729        return None;
2730    }
2731    match analysis {
2732        None => Some("missing_fit"),
2733        Some(analysis) if !analysis.result.is_asymptotic() => Some("unstable_fit"),
2734        Some(analysis) if analysis.result.slope > max_dod || analysis.result.slope.is_nan() => {
2735            Some("dod_exceeds_threshold")
2736        }
2737        Some(_) => None,
2738    }
2739}

2741/// Slope, r², DOD and status cells of one sampled ray.
2742fn inspect_cells(
2743    analysis: Option<&InspectAnalysis>,
2744    fit_status: InspectFitStatus,
2745    max_dod: f64,
2746    allow_vanishing_missing_fits: bool,
2747) -> [String; 4] {
2748    let failure =
2749        inspect_failure_reason(analysis, fit_status, max_dod, allow_vanishing_missing_fits);
2750    let decay_slope = fit_status
2751        .decay_slope
2752        .filter(|_| fit_status.vanishes_without_power_law(analysis, max_dod));
2753    let status = inspect_label(analysis, failure, decay_slope);
2754    if let Some(bound) = decay_slope.filter(|_| failure.is_none()) {
2755        let slope = if bound.is_finite() {
2756            format!("≤ {bound:.6}").green().to_string()
2757        } else {
2758            "-".to_string()
2759        };
2760        return [slope, "-".to_string(), "-".to_string(), status];
2761    }
2762    match analysis {
2763        Some(analysis) => {
2764            let r2_text = format!("{:.3}", analysis.result.r_squared);
2765            let r2_text = if analysis.result.is_asymptotic() {
2766                r2_text.green()
2767            } else {
2768                r2_text.red()
2769            };
2770            let dod_text = analysis.estimated_dod.to_string();
2771            let dod_text = if failure.is_none() {
2772                dod_text.green()
2773            } else {
2774                dod_text.red()
2775            };
2776            [
2777                format!("{:.6}", analysis.result.slope),
2778                r2_text.to_string(),
2779                dod_text.to_string(),
2780                status,
2781            ]
2782        }
2783        None => ["-".to_string(), "-".to_string(), "-".to_string(), status],
2784    }
2785}

3600fn log_log_slope(inspect: &[InspectResult], scales: &[f64]) -> Option<FitResult> {
3601    let mut valid_samples = Vec::new();
3602    let mut points = vec![];
3603    let mut valid_scales = vec![];
3604
3605    for (x, s) in inspect.iter().zip(scales) {
3606        // A failed sample must trigger the Arb retry, and remain a failed ray
3607        // if reevaluation cannot recover it. Finite zero samples are distinct.
3608        let norm = x.magnitude.as_ref()?;
3609        if norm.is_nan() || norm.is_infinite() {
3610            return None;
3611        }
3612        if norm <= &norm.zero() {
3613            debug!(scale = s, "zero UV profile sample");
3614            continue;
3615        }
3616        let y = norm.log10().into_ff64().0;
3617        let x = s.log10();
3618        if !y.is_finite() || !x.is_finite() {
3619            continue;
3620        }
3621        points.push(norm.clone());
3622        valid_scales.push(*s);
3623        valid_samples.push((x, y));
3624    }
3625
3626    if valid_samples.len() < 2 {
3627        return None;
3628    }
3629
3630    let fit = |samples: &[(f64, f64)]| {
3631        let n = samples.len() as f64;
3632        let (sum_x, sum_y, sum_xy, sum_x2) = samples.iter().fold(
3633            (0.0, 0.0, 0.0, 0.0),
3634            |(sum_x, sum_y, sum_xy, sum_x2), (x, y)| {
3635                (sum_x + x, sum_y + y, sum_xy + x * y, sum_x2 + x * x)
3636            },
3637        );
3638        let denominator = n * sum_x2 - sum_x * sum_x;
3639        if denominator.abs() < 1e-15 {
3640            return None;
3641        }
3642        let slope = (n * sum_xy - sum_x * sum_y) / denominator;
3643        let intercept = (sum_y - slope * sum_x) / n;
3644        let y_mean = sum_y / n;
3645        let (ss_tot, ss_res) = samples.iter().fold((0.0, 0.0), |(ss_tot, ss_res), (x, y)| {
3646            let residual = y - (intercept + slope * x);
3647            (ss_tot + (y - y_mean).powi(2), ss_res + residual.powi(2))
3648        });
3649        let r_squared = if ss_tot > 1e-15 {
3650            1.0 - ss_res / ss_tot
3651        } else {
3652            0.0
3653        };
3654        Some((slope, intercept, r_squared))
3655    };
3656
3657    let full_range_fit = fit(&valid_samples)?;
3658    // UV power counting is asymptotic. Retain the longest high-scale suffix which is already
3659    // linear, dropping only a pre-asymptotic prefix.
3660    let minimum_suffix_len = asymptotic_suffix_len(valid_samples.len());
3661    let (fit_start, (slope, intercept, r_squared)) = (0..=valid_samples.len() - minimum_suffix_len)
3662        .find_map(|start| {
3663            let candidate = fit(&valid_samples[start..])?;
3664            (candidate.2 >= UV_PROFILE_ASYMPTOTIC_MIN_R_SQUARED).then_some((start, candidate))
3665        })
3666        .unwrap_or((0, full_range_fit));
3667    if slope > UV_PROFILE_MAX_DOD || r_squared < UV_PROFILE_ASYMPTOTIC_MIN_R_SQUARED {
3668        debug!(
3669            slope,
3670            intercept,
3671            r_squared,
3672            full_range_r_squared = full_range_fit.2,
3673            fit_start,
3674            ?valid_samples,
3675            "unstable UV profile log-log fit"
3676        );
3677    }
3678
3679    Some(FitResult {
3680        points,
3681        scales: valid_scales,
3682        fit_start,
3683        slope,
3684        intercept,
3685        r_squared,
3686        full_range_slope: full_range_fit.0,
3687        full_range_intercept: full_range_fit.1,
3688        full_range_r_squared: full_range_fit.2,
3689    })
3690}

3722#[derive(Debug, Clone)]
3723pub struct InspectResult {
3724    /// Fully weighted magnitude; None is an invalid sample, not a zero tail.
3725    magnitude: Option<F<ArbPrec>>,
3726}
3727
3728impl InspectResult {
3729    fn from_evaluation<T: FloatLike>(
3730        result: GenericEvaluationResult<T>,
3731        prefactor: &F<ArbPrec>,
3732        compatible_event_cut_ids: Option<&[CutId]>,
3733    ) -> Self {
3734        // The evaluator already sums orientations in its native precision. Only
3735        // cut projection and the profile measure remain before taking a magnitude.
3736        let magnitude = (|| {
3737            if result.evaluation_metadata.is_nan {
3738                return None;
3739            }
3740            let value = if let Some(cuts) = compatible_event_cut_ids {
3741                result
3742                    .event_groups
3743                    .iter()
3744                    .flat_map(|group| group.iter())
3745                    .filter(|event| cuts.contains(&CutId(event.cut_info.cut_id)))
3746                    .try_fold(Complex::new_re(prefactor.zero()), |sum, event| {
3747                        Some(
3748                            sum + Complex::new(
3749                                event.weight.re.to_arb_exact().ok()?,
3750                                event.weight.im.to_arb_exact().ok()?,
3751                            ),
3752                        )
3753                    })?
3754            } else {
3755                Complex::new(
3756                    result.integrand_result.re.to_arb_exact().ok()?,
3757                    result.integrand_result.im.to_arb_exact().ok()?,
3758                ) * result.integrator_weight.to_arb_exact().ok()?
3759            };
3760            let weighted = value * prefactor;
3761            let magnitude = weighted.re.hypot(&weighted.im);
3762            (!magnitude.is_nan() && !magnitude.is_infinite()).then_some(magnitude)
3763        })();
3764        // Nonfinite values and native values which cannot be embedded exactly
3765        // request reevaluation in Arb. Never round before a cut cancellation or
3766        // abort the entire report for an invalid profile sample.
3767        Self { magnitude }
3768    }
3769}

3771/// Sampled evidence of an accelerating decay or a zero tail, without an asymptotic power-law fit.
3772#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize)]
3773struct InspectFitStatus {
3774    /// Smallest initial slope of a high-scale log-log majorant whose decay accelerates or ends
3775    /// at zero; `-∞` for a window ending at zero with at most one nonzero sample.
3776    /// `None` if no window supplies that evidence or any sample is invalid.
3777    #[serde(
3778        default,
3779        serialize_with = "InspectFitStatus::serialize_decay_slope",
3780        deserialize_with = "InspectFitStatus::deserialize_decay_slope"
3781    )]
3782    decay_slope: Option<f64>,
3783}
3784
3785impl InspectFitStatus {
3786    // JSON has no infinity literal; keep exact-zero evidence distinct from missing data.
3787    fn serialize_decay_slope<S: serde::Serializer>(
3788        slope: &Option<f64>,
3789        serializer: S,
3790    ) -> Result<S::Ok, S::Error> {
3791        match slope {
3792            None => serializer.serialize_none(),
3793            Some(slope) if slope.is_finite() => serializer.serialize_some(slope),
3794            Some(slope) if *slope == f64::NEG_INFINITY => serializer.serialize_some("-inf"),
3795            Some(_) => Err(serde::ser::Error::custom("expected a finite slope or -inf")),
3796        }
3797    }
3798
3799    fn deserialize_decay_slope<'de, D: serde::Deserializer<'de>>(
3800        deserializer: D,
3801    ) -> Result<Option<f64>, D::Error> {
3802        #[derive(Deserialize)]
3803        #[serde(untagged)]
3804        enum Slope {
3805            Number(f64),
3806            Marker(String),
3807        }
3808
3809        match Option::<Slope>::deserialize(deserializer)? {
3810            None => Ok(None),
3811            Some(Slope::Number(slope)) if slope.is_finite() => Ok(Some(slope)),
3812            Some(Slope::Marker(marker)) if marker == "-inf" => Ok(Some(f64::NEG_INFINITY)),
3813            Some(_) => Err(serde::de::Error::custom("expected a finite slope or -inf")),
3814        }
3815    }
3816
3817    fn from_results(inspect: &[InspectResult], scales: &[f64]) -> Self {
3818        let mut samples = Vec::with_capacity(inspect.len());
3819        for (result, scale) in inspect.iter().zip(scales) {
3820            let Some(norm) = result.magnitude.as_ref() else {
3821                return Self::default();
3822            };
3823            if norm.is_nan() || norm.is_infinite() {
3824                // An invalid sample cannot certify an inactive, vanishing LU ray.
3825                return Self::default();
3826            }
3827            let log_magnitude = (norm > &norm.zero()).then(|| norm.log10().into_ff64().0);
3828            samples.push((scale.log10(), log_magnitude));
3829        }
3830        // As for the power-law fit, a pre-asymptotic prefix of at most half the ray is ignored.
3831        let window_len = asymptotic_suffix_len(samples.len());
3832        Self {
3833            decay_slope: (0..=samples.len() - window_len)
3834                .filter_map(|start| Self::window_decay_slope(&samples[start..]))
3835                .min_by(f64::total_cmp),
3836        }
3837    }
3838
3839    /// Initial, hence shallowest, slope of the least concave majorant of the window's nonzero
3840    /// `(log λ, log |f|)` samples, provided the samples follow it to their end. The majorant
3841    /// bridges isolated dips, such as an exact cancellation resolved by Arb, but its final segment
3842    /// must be the final sampled step: a tail which flattens or rises leaves it. With at least two
3843    /// nonzero samples, exact zeros may only trail them and then replace that requirement.
3844    /// A sparse window with at most one nonzero sample instead requires an exact zero endpoint.
3845    /// Nonzero windows must also show two successive doublings of their decay rate at separated
3846    /// checkpoints. This dimensionless heuristic rejects noisy power laws and isolated final drops;
3847    /// it does not prove asymptotic behavior beyond the sampled range.
3848    fn window_decay_slope(window: &[(f64, Option<f64>)]) -> Option<f64> {
3849        let nonzero = window
3850            .iter()
3851            .filter_map(|&(log_scale, log_magnitude)| Some((log_scale, log_magnitude?)))
3852            .collect_vec();
3853        let [earlier @ .., penultimate, last] = nonzero.as_slice() else {
3854            // A sparse, inactive ray must end at zero; a late revival is unresolved.
3855            return window
3856                .last()
3857                .is_some_and(|(_, log_magnitude)| log_magnitude.is_none())
3858                .then_some(f64::NEG_INFINITY);
3859        };
3860        let trailing_zeros = window
3861            .iter()
3862            .rev()
3863            .take_while(|(_, log_magnitude)| log_magnitude.is_none())
3864            .count();
3865        if nonzero.len() + trailing_zeros < window.len() {
3866            return None;
3867        }
3868        let slope = |(x0, y0): (f64, f64), (x1, y1): (f64, f64)| (y1 - y0) / (x1 - x0);
3869        let final_slope = slope(*penultimate, *last);
3870        if trailing_zeros == 0
3871            && !earlier
3872                .iter()
3873                .all(|point| final_slope <= slope(*point, *penultimate))
3874        {
3875            return None;
3876        }
3877        let initial_slope = |start| {
3878            nonzero[start + 1..]
3879                .iter()
3880                .map(|point| slope(nonzero[start], *point))
3881                .max_by(f64::total_cmp)
3882        };
3883        let initial = initial_slope(0)?;
3884        if trailing_zeros == 0 {
3885            if nonzero.len() < 4 {
3886                return None;
3887            }
3888            let middle = initial_slope((nonzero.len() - 1) / 3)?;
3889            let late = initial_slope(2 * (nonzero.len() - 1) / 3)?;
3890            if !(initial < 0.0 && middle <= 2.0 * initial && late <= 2.0 * middle) {
3891                return None;
3892            }
3893        }
3894        Some(initial)
3895    }
3896
3897    /// Accept sampled zero tails or accelerating decay only when no asymptotic power law describes
3898    /// them and the initial decay bound meets `max_dod`. A shallow bound alone is insufficient.
3899    fn vanishes_without_power_law(self, analysis: Option<&InspectAnalysis>, max_dod: f64) -> bool {
3900        !analysis.is_some_and(|analysis| analysis.result.is_asymptotic())
3901            && self.decay_slope.is_some_and(|slope| slope <= max_dod)
3902    }
3903}
