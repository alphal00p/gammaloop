91impl UVProfileFixedRay {
92    pub fn from_flat_components(directions: &[f64], norms: &[f64]) -> Result<Self> {
93        if directions.is_empty() {
94            return Err(eyre!(
95                "Fixed UV ray directions cannot be empty when the option is used."
96            ));
97        }
98        if !directions.len().is_multiple_of(3) {
99            return Err(eyre!(
100                "Fixed UV ray directions must contain a multiple of 3 components, got {}.",
101                directions.len()
102            ));
103        }
104        if norms.is_empty() {
105            return Err(eyre!(
106                "Fixed UV ray norms cannot be empty when directions are supplied."
107            ));
108        }
109
110        let directions = directions
111            .chunks_exact(3)
112            .map(|direction| {
113                let norm = (direction[0] * direction[0]
114                    + direction[1] * direction[1]
115                    + direction[2] * direction[2])
116                    .sqrt();
117                if norm == 0.0 {
118                    return Err(eyre!(
119                        "Fixed UV ray directions cannot contain a zero vector."
120                    ));
121                }
122                Ok([
123                    direction[0] / norm,
124                    direction[1] / norm,
125                    direction[2] / norm,
126                ])
127            })
128            .collect::<Result<Vec<_>>>()?;
129
130        if norms.iter().any(|norm| !norm.is_finite() || *norm <= 0.0) {
131            return Err(eyre!(
132                "Fixed UV ray norms must be finite and strictly positive."
133            ));
134        }
135
136        Ok(Self {
137            directions,
138            norms: norms.to_vec(),
139        })
140    }
141
142    fn sample(&self, loop_count: usize) -> Result<LoopMomentumSample> {
143        if self.directions.len() != 1 && self.directions.len() != loop_count {
144            return Err(eyre!(
145                "Fixed UV ray directions contain {} ray(s), but this LMB has {} loop variable(s). Use either one direction or one direction per loop variable.",
146                self.directions.len(),
147                loop_count
148            ));
149        }
150        if self.norms.len() != 1 && self.norms.len() != loop_count {
151            return Err(eyre!(
152                "Fixed UV ray norms contain {} value(s), but this LMB has {} loop variable(s). Use either one norm or one norm per loop variable.",
153                self.norms.len(),
154                loop_count
155            ));
156        }
157
158        (0..loop_count)
159            .map(|i| {
160                let direction = self.directions[if self.directions.len() == 1 { 0 } else { i }];
161                let norm = self.norms[if self.norms.len() == 1 { 0 } else { i }];
162                Ok(ThreeMomentum {
163                    px: F(norm * direction[0]),
164                    py: F(norm * direction[1]),
165                    pz: F(norm * direction[2]),
166                })
167            })
168            .collect()
169    }
170}

197pub fn logspace(start: f64, stop: f64, num: usize, base: f64) -> Vec<f64> {
198    let log_start = start;
199    let log_stop = stop;
200    let step = (log_stop - log_start) / (num - 1) as f64;
201
202    (0..num)
203        .map(|i| {
204            let exponent = log_start + step * i as f64;
205            base.powf(exponent)
206        })
207        .collect()
208}

210fn lmb_seed(base_seed: u64, graph_id: usize, lmb_index: usize) -> u64 {
211    base_seed
212        .wrapping_add((graph_id as u64).wrapping_mul(0x9E3779B97F4A7C15))
213        .wrapping_add(lmb_index as u64)
214}

279        let runner = UVProfileRunner {
280            integrand: &integrand,
281            scales: &scales,
282            externals: &externals,
283            model,
284            settings: &settings,
285            profile_settings,
286            base_seed,
287        };
288
289        let per_graph = self
290            .graphs
291            .par_iter()
292            .enumerate()
293            .map(|(i, g)| {
294                let res = runner.sample_graph(i, g)?;
295                profile_span.pb_inc(1);

298            .collect::<Result<Vec<_>>>()?;
299
300        drop(_profile_span_enter);
301        drop(profile_span);
302
303        let integrand = Arc::try_unwrap(integrand)
304            .map_err(|_| color_eyre::eyre::eyre!("integrand still shared"))?
305            .into_inner()
306            .expect("integrand mutex poisoned");
307        self.integrand = Some(integrand);
308
309        Ok(UVProfile {
310            per_graph,
311            scales,
312            allow_vanishing_missing_fits: false,
313        })
314        // results.push((inspect_res, analytic_res));
315    }

408impl UVProfile {
409    pub fn analyse(&self) -> UVProfileAnalysis {
410        let graphs = self
411            .per_graph
412            .iter()
413            .enumerate()
414            .map(|(graph_index, graph)| {
415                let lmbs: Vec<UVProfileLmbAnalysis> = graph
416                    .per_lmb
417                    .iter()
418                    .enumerate()
419                    .map(|(lmb_index, lmb)| {
420                        let lmb_label = lmb_label(&lmb.lmb);
421                        let subsets: Vec<UVProfileSubsetAnalysis> = lmb
422                            .per_subsets
423                            .iter()
424                            .enumerate()
425                            .map(|(subset_index, (subset, subset_result))| {
426                                let mut not_included: SubSet<LoopIndex> =
427                                    SubSet::full(subset.size());
428                                not_included.subtract_with(subset);
429                                let free: Vec<EdgeIndex> = subset
430                                    .included_iter()
431                                    .map(|loop_index| lmb.lmb.loop_edges[loop_index])
432                                    .collect();
433
434                                let fixed: Vec<EdgeIndex> = not_included
435                                    .included_iter()
436                                    .map(|loop_index| lmb.lmb.loop_edges[loop_index])
437                                    .collect();
438                                let analysis = subset_result.analyse(&self.scales);
439                                let per_orientation_inspect_entries =
440                                    analysis.per_orientation_inspect_entries();
441                                let analytic_entries =
442                                    analysis.analytic.as_ref().and_then(|analytic| {
443                                        let entries = analytic
444                                            .per_orientation
445                                            .iter()
446                                            .map(|(orientation, orientation_analysis)| {
447                                                let (orientation_edges, orientation_signs) =
448                                                    orientation_signs(orientation);
449                                                UVProfileAnalyticEntry {
450                                                    graph_index,
451                                                    lmb_index,
452                                                    subset_index,
453                                                    fixed: fixed.clone(),
454                                                    free: free.clone(),
455                                                    orientation_edges,
456                                                    orientation_signs,
457                                                    is_constant: orientation_analysis.is_constant,
458                                                    leading_coef: orientation_analysis
459                                                        .leading_coef
460                                                        .to_string(),
461                                                }
462                                            })
463                                            .collect::<Vec<_>>();
464                                        if entries.is_empty() {
465                                            None
466                                        } else {
467                                            Some(entries)
468                                        }
469                                    });
470                                UVProfileSubsetAnalysis {
471                                    subset_index,
472                                    fixed,
473                                    free,
474                                    initial_dod: subset_result.initial_dod,
475                                    analysis,
476                                    per_orientation_inspect_entries,
477                                    analytic_entries,
478                                }
479                            })
480                            .collect();
481
482                        UVProfileLmbAnalysis {
483                            lmb_index,
484                            lmb_label,
485                            subsets,
486                        }
487                    })
488                    .collect();
489
490                UVProfileGraphAnalysis { graph_index, lmbs }
491            })
492            .collect();
493
494        UVProfileAnalysis {
495            scales: self.scales.clone(),
496            graphs,
497            allow_vanishing_missing_fits: self.allow_vanishing_missing_fits,
498        }
499    }

669    pub fn tables_per_graph(&self, max_dod: f64) -> Vec<Table> {
670        self.graphs
671            .iter()
672            .map(|graph| {
673                let rows = graph
674                    .lmbs
675                    .iter()
676                    .flat_map(|lmb| {
677                        lmb.subsets.iter().map(|subset| {
678                            let (slope, r_squared, estimated_dod) =
679                                match &subset.analysis.inspect_level {
680                                    Some(analysis) => {
681                                        let r2_text = format!("{:.3}", analysis.result.r_squared);
682                                        let r2_text = if analysis.result.r_squared >= 0.99 {
683                                            r2_text.green()
684                                        } else {
685                                            r2_text.red()
686                                        }
687                                        .to_string();
688
689                                        let dod = analysis.estimated_dod;
690                                        let dod_text = dod.to_string();
691                                        let dod_text = if (dod as f64) <= max_dod {
692                                            dod_text.green()
693                                        } else {
694                                            dod_text.red()
695                                        }
696                                        .to_string();
697
698                                        (format!("{:.6}", analysis.result.slope), r2_text, dod_text)
699                                    }
700                                    None => ("-".to_string(), "-".to_string(), "-".to_string()),
701                                };
702
703                            UVProfileSubsetRow {
704                                fixed: format!(
705                                    "{{{}}}",
706                                    subset.fixed.iter().map(ToString::to_string).join(",")
707                                ),
708                                free: format!(

735    pub fn analytic_tables_per_graph(&self) -> Vec<Option<Table>> {
736        self.graphs
737            .iter()
738            .map(|graph| {
739                let mut groups: Vec<Vec<&UVProfileAnalyticEntry>> = Vec::new();
740                let mut orientation_headers: Option<Vec<String>> = None;
741
742                for lmb in &graph.lmbs {
743                    for subset in &lmb.subsets {
744                        if let Some(entries) = &subset.analytic_entries {
745                            if orientation_headers.is_none() {
746                                orientation_headers =
747                                    entries.first().map(|entry| entry.orientation_edges.clone());
748                            }
749                            groups.push(entries.iter().collect());
750                        }
751                    }
752                }
753
754                if groups.is_empty() {
755                    return None;
756                }
757
758                let mut builder = Builder::new();
759                let mut header = vec!["fixed".to_string(), "→ ∞".to_string()];
760                header.extend(orientation_headers.unwrap_or_default());
761                header.extend(["const".to_string(), "leading coef".to_string()]);
762                builder.push_record(header);
763
764                let mut span_ops: Vec<(usize, usize, usize)> = Vec::new();
765                let mut row_index = 1;
766
767                for group in groups {
768                    let span_len = group.len();
769                    let start_row = row_index;
770
771                    for (i, entry) in group.into_iter().enumerate() {
772                        let fixed = if i == 0 {
773                            format!(
774                                "{{{}}}",
775                                entry.fixed.iter().map(ToString::to_string).join(",")
776                            )
777                        } else {
778                            String::new()
779                        };
780                        let free = if i == 0 {
781                            format!(
782                                "{{{}}}",
783                                entry.free.iter().map(ToString::to_string).join(",")
784                            )
785                        } else {
786                            String::new()
787                        };
788                        let mut row = vec![fixed, free];
789                        row.extend(entry.orientation_signs.iter().cloned());
790                        row.extend([entry.is_constant.to_string(), "".to_string()]);
791                        builder.push_record(row);
792                        row_index += 1;
793                    }
794
795                    if span_len > 1 {
796                        span_ops.push((start_row, 0, span_len));
797                        span_ops.push((start_row, 1, span_len));
798                    }
799                }
800
801                let mut table = builder.build();
802                for (row, col, span_len) in span_ops {
803                    table.with(Modify::new((row, col)).with(Span::row(span_len as isize)));
804                }
805                table.with(Style::rounded());
806                Some(table)
807            })
808            .collect()
809    }

903    pub fn pass_fail(&self, max_dod: f64) -> UVProfilePassFail {
904        let mut failures = Vec::new();
905        let mut total = 0;
906
907        for graph in &self.graphs {
908            for lmb in &graph.lmbs {
909                for subset in &lmb.subsets {
910                    let per_orientation_reason_count = subset
911                        .analysis
912                        .per_orientation_inspect
913                        .as_ref()
914                        .map(|entries| {
915                            entries
916                                .iter()
917                                .filter(|entry| {
918                                    inspect_failure_reason(
919                                        entry.analysis.as_ref(),
920                                        entry.inspect_fit_status,
921                                        max_dod,
922                                        self.allow_vanishing_missing_fits,
923                                    )
924                                    .is_some()
925                                })
926                                .count()
927                        })
928                        .unwrap_or(0);
929                    let all_orientations_pass = subset.analysis.per_orientation_inspect.is_some()
930                        && per_orientation_reason_count == 0;
931
932                    total += 1;
933                    let reason = inspect_failure_reason(
934                        subset.analysis.inspect_level.as_ref(),
935                        subset.analysis.inspect_fit_status,
936                        max_dod,
937                        self.allow_vanishing_missing_fits,
938                    );
939
940                    if let Some(reason) = reason.filter(|_| !all_orientations_pass) {
941                        failures.push(UVProfileFailure {
942                            graph_index: graph.graph_index,
943                            lmb_index: lmb.lmb_index,
944                            fixed: subset.fixed.clone(),
945                            free: subset.free.clone(),
946                            orientation_label: None,
947                            reason: reason.to_string(),
948                        });
949                    }
950
951                    for entry in subset.analysis.per_orientation_inspect.iter().flatten() {
952                        total += 1;
953                        let reason = inspect_failure_reason(
954                            entry.analysis.as_ref(),
955                            entry.inspect_fit_status,
956                            max_dod,
957                            self.allow_vanishing_missing_fits,
958                        );
959
960                        if let Some(reason) = reason {
961                            failures.push(UVProfileFailure {
962                                graph_index: graph.graph_index,
963                                lmb_index: lmb.lmb_index,
964                                fixed: subset.fixed.clone(),
965                                free: subset.free.clone(),
966                                orientation_label: Some(entry.orientation_label.clone()),
967                                reason: reason.to_string(),
968                            });
969                        }
970                    }
971                }
972            }
973        }
974
975        UVProfilePassFail {
976            max_dod,
977            total,
978            failed: failures.len(),
979            failures,
980        }
981    }

984fn inspect_failure_reason(
985    analysis: Option<&InspectAnalysis>,
986    fit_status: InspectFitStatus,
987    max_dod: f64,
988    allow_vanishing_missing_fits: bool,
989) -> Option<&'static str> {
990    match analysis {
991        None if allow_vanishing_missing_fits && fit_status.missing_fit_is_vanishing() => None,
992        None => Some("missing_fit"),
993        Some(analysis) if analysis.result.slope > max_dod || analysis.result.slope.is_nan() => {
994            Some("dod_exceeds_threshold")
995        }
996        _ => None,
997    }

1188pub struct LMBResult {
1189    pub(crate) lmb: LoopMomentumBasis,
1190    pub(crate) per_subsets: BTreeMap<SubSet<LoopIndex>, SubSetResult>,
1191}
1192
1193impl<'a> UVProfileRunner<'a> {
1194    fn sample_lmb(
1195        &self,
1196        graph_id: usize,
1197        graph: &Graph,
1198        lmb_index: usize,
1199        lmb: &LoopMomentumBasis,
1200        orientation_labels: Option<&[String]>,
1201    ) -> Result<LMBResult> {
1202        let sample: LoopMomentumSample =
1203            if let Some(fixed_uv_ray) = &self.profile_settings.fixed_uv_ray {
1204                fixed_uv_ray.sample(lmb.loop_edges.len())?
1205            } else {
1206                let mut rng = MonteCarloRng::new(lmb_seed(self.base_seed, graph_id, lmb_index), 0);
1207                lmb.loop_edges
1208                    .iter()
1209                    .map(|_| ThreeMomentum {
1210                        px: F(rng.random_range(
1211                            -self.settings.kinematics.e_cm..self.settings.kinematics.e_cm,
1212                        )),
1213                        py: F(rng.random_range(
1214                            -self.settings.kinematics.e_cm..self.settings.kinematics.e_cm,
1215                        )),
1216                        pz: F(rng.random_range(
1217                            -self.settings.kinematics.e_cm..self.settings.kinematics.e_cm,
1218                        )),
1219                    })
1220                    .collect()
1221            };
1222
1223        let mut loops = PowersetIterator::<LoopIndex>::new(lmb.loop_edges.len() as u8);
1224        loops.next();
1225
1226        let subsets: Vec<_> = loops.collect();
1227        let subset_span = info_span!(
1228            "Profiling subsets",
1229            indicatif.pb_show = true,
1230            graph_id = graph_id,
1231            lmb_index = lmb_index
1232        );
1233        subset_span.pb_set_style(
1234            &ProgressStyle::with_template("{wide_bar} {pos}/{len} {msg}")
1235                .expect("invalid progress bar template"),
1236        );
1237        subset_span.pb_set_length(subsets.len() as u64);
1238        subset_span.pb_set_message("Profiling subsets");
1239        let _subset_span_enter = subset_span.enter();
1240
1241        let per_subsets_vec: Vec<(SubSet<LoopIndex>, SubSetResult)> = subsets
1242            .into_par_iter()
1243            .map_init(
1244                || {
1245                    self.integrand
1246                        .lock()
1247                        .expect("integrand mutex poisoned")
1248                        .clone()
1249                },
1250                |integrand, ls| {
1251                    let res = self.sample_subset(
1252                        integrand,
1253                        SubsetSampleInput {
1254                            graph_id,
1255                            graph,
1256                            subset: &ls,
1257                            lmb,
1258                            sample: &sample,
1259                            orientation_labels,
1260                        },
1261                    )?;
1262                    subset_span.pb_inc(1);
1263                    Ok((ls, res))
1264                },
1265            )
1266            .collect::<Result<Vec<_>>>()?;
1267        let per_subsets = per_subsets_vec.into_iter().collect();
1268
1269        drop(_subset_span_enter);
1270        drop(subset_span);
1271
1272        Ok(LMBResult {
1273            lmb: lmb.clone(),
1274            per_subsets,
1275        })
1276    }

1315    graph_id: usize,
1316    graph: &'a Graph,
1317    subset: &'a SubSet<LoopIndex>,
1318    lmb: &'a LoopMomentumBasis,
1319    sample: &'a LoopMomentumSample,
1320    orientation_labels: Option<&'a [String]>,
1321}

1323impl<'a> UVProfileRunner<'a> {
1324    fn sample_subset(
1325        &self,
1326        integrand: &mut ProcessIntegrand,
1327        input: SubsetSampleInput<'_>,
1328    ) -> Result<SubSetResult> {
1329        let mut subgraph: SuBitGraph = input.graph.empty_subgraph();
1330        for l in input.subset.included_iter() {
1331            let eid = input.lmb.loop_edges[l];
1332            let cut = input.graph[&eid].1.any_hedge();
1333            let root_node = input.graph.node_id(cut);
1334
1335            let tree = SimpleTraversalTree::depth_first_traverse(
1336                input.graph,
1337                &input.lmb.tree,
1338                &root_node,
1339                None,
1340            )
1341            .unwrap();
1342            subgraph.union_with(
1343                &tree
1344                    .get_cycle(cut, input.graph.underlying.as_ref())
1345                    .unwrap()
1346                    .filter,
1347            );
1348        }
1349
1350        let initial_dod = input.graph.compute_dod(&subgraph);
1351        let inspect = if let Some(orientation_labels) = input.orientation_labels {
1352            let per_orientation = orientation_labels
1353                .iter()
1354                .enumerate()
1355                .map(|(orientation_id, label)| {
1356                    let inspect = self.sample_subset_orientation(
1357                        integrand,
1358                        SubsetOrientationInput {
1359                            graph_id: input.graph_id,
1360                            subset: input.subset,
1361                            lmb: input.lmb,
1362                            sample: input.sample,
1363                            orientation: Some(orientation_id),
1364                        },
1365                    )?;
1366                    Ok(OrientationInspectSamples {
1367                        label: label.clone(),
1368                        inspect: inspect.inspect,
1369                        used_arb_prec_retry: inspect.used_arb_prec_retry,
1370                    })
1371                })
1372                .collect::<Result<Vec<_>>>()?;
1373            let summed = sum_orientation_inspect_samples(&per_orientation);
1374            InspectSamples {
1375                summed,
1376                summed_used_arb_prec_retry: per_orientation
1377                    .iter()
1378                    .any(|orientation| orientation.used_arb_prec_retry),
1379                per_orientation,
1380            }
1381        } else {
1382            let inspect = self.sample_subset_orientation(
1383                integrand,
1384                SubsetOrientationInput {
1385                    graph_id: input.graph_id,
1386                    subset: input.subset,
1387                    lmb: input.lmb,
1388                    sample: input.sample,
1389                    orientation: None,
1390                },
1391            )?;
1392            InspectSamples {
1393                summed: inspect.inspect,
1394                summed_used_arb_prec_retry: inspect.used_arb_prec_retry,
1395                per_orientation: Vec::new(),
1396            }
1397        };
1398        let analytic = None;
1399
1400        Ok(SubSetResult {
1401            inspect,
1402            initial_dod,
1403            analytic,
1404        })
1405    }

1407    fn sample_subset_orientation(
1408        &self,
1409        integrand: &mut ProcessIntegrand,
1410        input: SubsetOrientationInput<'_>,
1411    ) -> Result<InspectRun> {
1412        let inspect = self.sample_subset_orientation_with_precision(
1413            integrand,
1414            input,
1415            self.profile_settings.use_f128,
1416        )?;
1417        if !self.profile_settings.use_f128
1418            && inspect_results_need_arbprec_retry(&inspect, self.scales)
1419        {
1420            return Ok(InspectRun {
1421                inspect: self.sample_subset_orientation_with_precision(integrand, input, true)?,
1422                used_arb_prec_retry: true,
1423            });
1424        }
1425        Ok(InspectRun {
1426            inspect,
1427            used_arb_prec_retry: false,
1428        })
1429    }

1431    fn sample_subset_orientation_with_precision(
1432        &self,
1433        integrand: &mut ProcessIntegrand,
1434        input: SubsetOrientationInput<'_>,
1435        use_arb_prec: bool,
1436    ) -> Result<Vec<InspectResult>> {
1437        let n_included = input.subset.n_included() as i32;
1438        self.scales
1439            .iter()
1440            .map(|s| {
1441                let prefactor = s.powi(3 * n_included);
1442                let mut scaled_sample = input.sample.clone();
1443                for l in input.subset.included_iter() {
1444                    scaled_sample[l] = scaled_sample[l].map_ref(&|a| a * F(*s));
1445                }
1446                let loop_momenta = input
1447                    .lmb
1448                    .loop_edges
1449                    .iter()
1450                    .map(|edge| {
1451                        input.lmb.edge_signatures[*edge]
1452                            .compute_momentum(&scaled_sample, self.externals)
1453                    })
1454                    .collect::<Vec<_>>();
1455
1456                let inspect_res_eval = evaluate_momentum_space_point(
1457                    integrand,
1458                    self.model,
1459                    loop_momenta,
1460                    input.graph_id,
1461                    input.orientation,
1462                    use_arb_prec,
1463                )?;
1464
1465                Ok(InspectResult {
1466                    result: inspect_res_eval,
1467                    prefactor,
1468                })
1469            })
1470            .collect()
1471    }
1472}

1580fn log_log_slope(inspect: &[InspectResult], scales: &[f64]) -> Option<FitResult> {
1581    let mut valid_samples = Vec::new();
1582    let mut sum_x = 0.0;
1583    let mut sum_y = 0.0;
1584    let mut sum_xy = 0.0;
1585    let mut sum_x2 = 0.0;
1586    let mut points = vec![];
1587    let mut points_detail = vec![];
1588
1589    for (x, s) in inspect.iter().zip(scales) {
1590        let norm = x.magnitude();
1591        if norm <= 0.0 {
1592            debug!("{s}:\t{}", x.result.evaluation_metadata);
1593            continue;
1594        }
1595        if !norm.is_finite() {
1596            continue;
1597        }
1598        points_detail.push(x.result.clone());
1599        points.push(norm);
1600        let y = (norm).log10();
1601        let x = s.log10();
1602        if !y.is_finite() || !x.is_finite() {
1603            continue;
1604        }
1605        valid_samples.push((x, y));
1606        sum_x += x;
1607        sum_y += y;
1608        sum_xy += x * y;
1609        sum_x2 += x * x;
1610    }
1611
1612    if valid_samples.len() < 2 {
1613        return None;
1614    }
1615
1616    let n = valid_samples.len() as f64;
1617    let denominator = n * sum_x2 - sum_x * sum_x;
1618    if denominator.abs() < 1e-15 {
1619        return None;
1620    }
1621
1622    let slope = (n * sum_xy - sum_x * sum_y) / denominator;
1623    let intercept = (sum_y - slope * sum_x) / n;
1624
1625    let y_mean = sum_y / n;
1626    let mut ss_tot = 0.0;
1627    let mut ss_res = 0.0;
1628    for (x, y) in &valid_samples {
1629        let y_pred = intercept + slope * x;
1630        ss_tot += (y - y_mean).powi(2);
1631        ss_res += (y - y_pred).powi(2);
1632    }
1633
1634    let r_squared = if ss_tot > 1e-15 {
1635        1.0 - ss_res / ss_tot
1636    } else {
1637        0.0
1638    };
1639
1640    Some(FitResult {
1641        points,
1642        points_detail,
1643        slope,
1644        intercept,
1645        r_squared,
1646    })
1647}

1717struct InspectFitStatus {
1718    finite_samples: usize,
1719    positive_finite_samples: usize,
1720}
1721
1722impl InspectFitStatus {
1723    fn from_results(inspect: &[InspectResult]) -> Self {
1724        inspect.iter().fold(Self::default(), |mut status, result| {
1725            let norm = result.magnitude();
1726            if norm.is_finite() {
1727                status.finite_samples += 1;
1728                if norm > 0.0 {
1729                    status.positive_finite_samples += 1;
1730                }
1731            }
1732            status
1733        })
1734    }
1735
1736    fn missing_fit_is_vanishing(self) -> bool {
1737        self.finite_samples > self.positive_finite_samples && self.positive_finite_samples < 2
1738    }
1739}
