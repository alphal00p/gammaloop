79impl IntegratedCts {
80    pub(crate) fn factorized_product<'a>(
81        factors: impl IntoIterator<Item = &'a Self>,
82        depth: usize,
83    ) -> Result<Self> {
84        let mut factors = factors.into_iter();
85        let first = factors
86            .next()
87            .ok_or_else(|| eyre!("a factorized integrated counterterm cannot be empty"))?;
88        let mut pole = truncate(&first.expansion, false);
89        let mut finite_counterterm = -truncate(&first.expansion, true);
90        let mut scale_power = first.scale_power;
91
92        for factor in factors {
93            pole *= truncate(&factor.expansion, false);
94            finite_counterterm *= -truncate(&factor.expansion, true);
95            scale_power += factor.scale_power;
96        }
97
98        Ok(Self {
99            // The public projections add the finite-counterterm sign, so store
100            // its negative while keeping the pole product unchanged.
101            expansion: series(&(pole - finite_counterterm), depth)?,
102            scale_power,
103        })
104    }

106    fn projected_atom(&self, finite: bool) -> Atom {
107        // The enclosing forest Taylor operation rescales the vacuum mass in
108        // this coefficient along with its active momenta. The bookkeeping
109        // factor restores only consumed loop measures; physical projections
110        // and the next integration set it to one.
111        truncate(&self.expansion, finite)
112            * Atom::var(GS.integrated_loop_scale).pow(self.scale_power)
113    }

116        self.projected_atom(false)

119    pub(crate) fn finite_counterterm_atom(&self) -> Atom {
120        -self.projected_atom(true)
121    }

140fn truncate(series: &Series<AtomField>, finite: bool) -> Atom {
141    let mut truncated = Atom::Zero;
142
143    for (power, p) in series.terms() {
144        if (power >= 0) == finite {
145            truncated += p * Atom::var(GS.dim_epsilon).pow(power);
146        }
147    }
148
149    truncated
150}

161fn simplify(integrand: &Atom) -> Result<Atom> {
162    if integrand.contains_symbol(AGS.sigma) {
163        return Err(eyre!(
164            "Sigma tensors are not supported by d-dimensional analytic UV numerator algebra"
165        ));
166    }
167    // Only the analytically integrated UV subgraph reaches this function.
168    // Expose spin/vector slots before closing traces; the representation-specific
169    // collectors below retain factorized scalar coefficients.
170    let integrand =
171        if integrand.contains_symbol(AGS.gamma) || integrand.contains_symbol(SPENSO_TAG.trace) {
172            // Scalar products already bind their contracted slots. Materializing
173            // a powered dot before symbolic multiplication reuses its dummy index
174            // and turns (p.k)^2 into p^2 k^2. Protect completed scalar spectators,
175            // while keeping spin/vector contractions with open slots visible.
176            // The symbolic network also checks hidden shorthand
177            // slots without requiring concrete tensor dimensions or execution.
178            let mut used = integrand.get_all_symbols(true);
179            let mut aliases = BTreeMap::<Atom, Atom>::new();
180            let mut serial = 0usize;
181            let protected = integrand.replace_map(|arg, _context, out| {
182                // The complete scalar expression in a production denominator is
183                // metadata for propagator conversion, not a tensor-network factor.
184                // Protect it alongside scalar spectators before closing spin traces.
185                if matches!(arg, AtomView::Fun(fun) if fun.get_symbol() == GS.den)
186                    || (matches!(arg, AtomView::Fun(fun)
187                    if fun.get_symbol() == SPENSO_TAG.dot || fun.get_symbol() == ETS.metric)
188                        && !arg.contains_symbol(AGS.gamma)
189                        && !arg.contains_symbol(SPENSO_TAG.trace)
190                        && arg
191                            .parse_to_symbolic_net::<Aind>(&ParseSettings {
192                                shorthand_parsing: ShorthandParsing::expand_all(),
193                                ..Default::default()
194                            })
195                            .is_ok_and(|network| network.graph.dangling_indices().is_empty()))
196                {
197                    let alias = aliases.entry(arg.to_owned()).or_insert_with(|| {
198                        loop {
199                            let symbol = symbol!(&format!("gammaloop::uv_scalar_product_{serial}"));
200                            serial += 1;
201                            if used.insert(symbol) {
202                                break Atom::var(symbol);
203                            }
204                        }
205                    });
206                    **out = alias.clone();
207                }
208            });
209            let mut protected = AliasedAtom::from(protected);
210            for (original, alias) in aliases {
211                protected.register_alias(alias, original);
212            }
213            let prepared = protected
214                .get_root()
215                .parse_to_symbolic_net::<Aind>(&ParseSettings {
216                    shorthand_parsing: ShorthandParsing::Expand {
217                        schoonschip: SchoonschipExpansionMode::full(),
218                        trace: false,
219                        chain: true,
220                    },
221                    ..Default::default()
222                })
223                .map_err(|error| eyre!("invalid analytic UV spin tensor notation: {error}"))?
224                .simple_execute::<()>()?;
225            protected.map_root(|_| prepared).into_inner()
226        } else {
227            integrand.clone()
228        };
229    let collected = integrand
230        .collect_rep(Minkowski {}.into())
231        .simplify_metrics()
232        .collect_rep((Bispinor {}).into())
233        .collect_gamma_chains();
234    debug_tags!(#uv,#integrated,#collect;log.expr = collected, "After gamma chain collection");
235
236    let schoonschip = collected
237        .schoonschip_with_settings(&SchoonschipSettings {
238            simplify_chain_like_functions: true,
239            schoonschip_rank1_tensors: true,
240            ..Default::default()
241        })
242        .normalize_chains();
243    debug_tags!(#uv, #integrated, #profile, #trace, #start, #collect;
244        log.expr = schoonschip,
245        "After gamma schoonschip"
246    );
247    // Keep each Taylor term's propagator powers. Vakint receives the individual
248    // denominator topologies: collecting factors across the sum clears denominators
249    // and manufactures higher-rank numerator factors before reduction.
250    // Color is a spectator of the spin algebra; collecting all chains and traces
251    // here distributes its shared coefficient across the kinematic terms.
252    let collected = schoonschip.simplify_metrics().collect_gamma_chains();
253    debug_tags!(#uv, #integrated, #profile, #trace, #start, #collect;
254        log.expr = collected,
255        "After gamma collection"
256    );
257
258    let simplified = collected
259        .simplify_gamma_with(GammaSimplifySettings::canonical())
260        .collect_rep(Minkowski {}.into())
261        .expand_num();
262    debug_tags!(#uv, #integrated, #vakint, #profile, #trace, #start, #gamma;
263        log.expr = simplified,
264        "After gamma simplification"
265    );
266    let schoonschipped = simplified.schoonschip_net::<Aind>()?;
267    debug_tags!(#uv, #integrated, #vakint, #profile, #trace,#schoonschip, #start;
268        log.expr = schoonschipped,
269        "After Schoonschip net"
270    );
271    let dotted = schoonschipped.to_dots().normalize_dots();
272    debug_tags!(#uv, #integrated, #vakint, #profile, #trace, #dots;
273        log.expr = dotted,
274        "After dots"
275    );
276
277    if dotted
278        .replace(function!(
279            SPENSO_TAG.trace,
280            Bispinor {}.to_symbolic([W_.d_]),
281            W_.x___
282        ))
283        .matches()
284    {
285        return Err(eyre!(
286            "an unresolved Dirac trace remains after d-dimensional UV numerator algebra"
287        ));
288    }
289    if dotted
290        .replace(function!(
291            SPENSO_TAG.trace,
292            Minkowski {}.to_symbolic([W_.d_]),
293            W_.x___
294        ))
295        .matches()
296    {
297        return Err(eyre!(
298            "an unresolved Lorentz trace remains after d-dimensional UV numerator algebra"
299        ));
300    }
301    Ok(dotted)
302}

320    pub(crate) fn run<S: super::ForestNodeLike, M: super::ForestNodeLike>(
321        &self,
322        integrand: &Local4dCts,
323        ctx: &UVCtx<'_>,
324        current: &S,
325        given: &S,
326        marker_current: &M,
327        marker_given: &M,
328    ) -> Result<IntegratedCts> {
329        let graph = ctx.graph;
330
331        let n_loops = graph.n_loops(current.subgraph());
332
333        let scheme = current.renormalization_scheme();
334        match scheme {
335            ApproximationType::MUV | ApproximationType::PolePart | ApproximationType::IR => {
336                // Integrate the complete signed operator in each retained frame.
337                // Linearity permits this sector sum; no soft branch is discarded
338                // before the nested forest has supplied its completed coefficient.
339                let mut integrated = Atom::Zero;
340                for sector in integrand.active_sectors() {
341                    let physical = sector
342                        .atom
343                        .replace(GS.integrated_loop_scale)
344                        .with(Atom::one());
345                    let simplified = simplify(&physical)?;
346                    if !simplified.is_zero() {
347                        integrated += self
348                            .integrate(&simplified, sector, ctx, current, given)
349                            .wrap_err_with(|| {
350                                format!(
351                                    "while integrating {scheme} counterterm {} with retained component frames {:?}",
352                                    current.subgraph().string_label(),
353                                    sector.active_components,
354                                )
355                            })?;
356                    }
357                }
358                let marker = UvMarker::new(ctx.settings);
359                let integrated = marker.apply(
360                    UvOperation::Integrate,
361                    marker_current.subgraph(),
362                    marker_given.subgraph(),
363                    &integrated,
364                );
365                let expansion_depth =
366                    usize::try_from(self.vakint_settings.number_of_terms_in_epsilon_expansion)
367                        .wrap_err("Vakint epsilon expansion depth must be nonnegative")?;
368                let expanded = series(&integrated, expansion_depth.max(n_loops + 1))?.map_coeff(
369                    |coefficient| {
370                        marker.apply(
371                            UvOperation::Series,
372                            marker_current.subgraph(),
373                            marker_given.subgraph(),
374                            coefficient,
375                        )
376                    },
377                );
378                let expansion = expanded.map_coeff(|coefficient| {
379                    marker.apply(
380                        UvOperation::Truncate,
381                        marker_current.subgraph(),
382                        marker_given.subgraph(),
383                        coefficient,
384                    )
385                });
386
387                // Retain the consumed loop measures for subsequent UV rescalings. Keep this
388                // marker independent of mUV so enclosing limits still rescale the vacuum mass.
389                Ok(IntegratedCts {
390                    expansion,
391                    scale_power: 4 * n_loops as i64,
392                })
393            }
394            ApproximationType::VaccuumLimit => Err(eyre!("Not yet implemented VaccuumLimit")),
395            ApproximationType::OS => Err(eyre!("Not yet implemented OS")),
396            ApproximationType::Unsubtracted => {
397                panic!("should have been kept out of the wood");
398            }
399        }
400    }

402    #[debug_instrument(
403        current = %current.log_display(),
404        given = %given.log_display(),
405        reduced,
406    )]
407    fn integrate<S: ForestNodeLike>(
408        &self,
409        integrand: &Atom,
410        sector: &FourDSector,
411        ctx: &UVCtx<'_>,
412        current: &S,
413        given: &S,
414    ) -> Result<Atom> {
415        let graph = ctx.graph;
416        let reduced = current.reduced_subgraph(given);
417        let settings = ctx.settings;
418        let reduced_label = reduced.string_label();
419        tracing::Span::current().record("reduced", reduced_label.as_str());
420        debug_tags!(#uv, #integrated, #vakint, #trace, #input;
421            log.integrand = integrand,
422            "Integrating and truncating"
423        );
424        // Match the vacuum result to the unintegrated CFF loop measure through
425        // Vakint's configured per-loop normalization hook. Forest-subtraction
426        // signs are folded into the expression at the integrated CT composition
427        // sites.
428        let integrand_vakint = to_vakint_integrand(
429            integrand,
430            graph,
431            current.subgraph(),
432            given.subgraph(),
433            &sector.active_components,
434            &settings.vakint,
435            false,
436        )?;
437
438        let propagator = function!(
439            vakint::symbols::S.prop,
440            W_.a_,
441            W_.b_,
442            W_.mom_,
443            W_.mass_,
444            W_.e_
445        )
446        .to_pattern();
447        let retained_loop_count = sector
448            .active_components
449            .iter()
450            .flat_map(|(_, _, lmb)| lmb.loop_edges.iter())
451            .collect::<std::collections::HashSet<_>>()
452            .len();
453        let mut res = Atom::Zero;
454        for source_term in integrand_vakint.0 {
455            // A product of independent one-loop denominators can carry a
456            // coupled numerator. Integrate one factor at a time, treating all
457            // spectator loops as fixed vectors for its angular projection.
458            let mut loop_factors = std::collections::BTreeMap::<i64, Atom>::new();
459            let mut masses = std::collections::HashSet::new();
460            let mut independent = true;
461            for matched in source_term.integral.pattern_match(&propagator, None, None) {
462                let momentum = &matched[&W_.mom_];
463                let loop_id = [1, -1].into_iter().find_map(|sign| {
464                    let momentum = momentum * Atom::num(sign);
465                    let AtomView::Fun(vector) = momentum.as_view() else {
466                        return None;
467                    };
468                    (vector.get_symbol() == vakint::symbols::S.k && vector.get_nargs() == 1)
469                        .then(|| i64::try_from(vector.get(0)).ok())
470                        .flatten()
471                });
472                let Some(loop_id) = loop_id else {
473                    independent = false;
474                    break;
475                };
476                masses.insert(matched[&W_.mass_].clone());
477                *loop_factors.entry(loop_id).or_insert_with(Atom::one) *= propagator
478                    .replace_wildcards(&matched)
479                    .map_err(|error| eyre!(error))?;
480            }
481            let factorized = independent
482                && retained_loop_count > 1
483                && loop_factors.len() == retained_loop_count
484                && masses.len() > 1;
485            let loop_ids = loop_factors.keys().copied().collect::<Vec<_>>();
486            let stages = if factorized {
487                loop_factors
488                    .into_iter()
489                    .map(|(loop_id, propagators)| {
490                        (
491                            function!(vakint::symbols::S.topo, propagators),
492                            loop_ids
493                                .iter()
494                                .copied()
495                                .filter(|id| *id != loop_id)
496                                .collect::<Vec<_>>(),
497                        )
498                    })
499                    .collect::<Vec<_>>()
500            } else {
501                vec![(source_term.integral.clone(), Vec::new())]
502            };
503            let mut coefficient = source_term.numerator.clone();
504            if factorized {
505                // Preserve even custom normalization hooks that are not a
506                // product of identical per-loop factors. The converter's
507                // additional normalization already belongs to the full measure.
508                let normalization = self
509                    .vakint_settings
510                    .get_integral_normalization_factor_atom()?;
511                coefficient *= normalization
512                    .replace(vakint::symbols::S.n_loops.to_pattern())
513                    .with(Atom::num(retained_loop_count))
514                    / normalization
515                        .replace(vakint::symbols::S.n_loops.to_pattern())
516                        .with(Atom::one())
517                        .pow(retained_loop_count);
518            }
519            // Keep the same conservative epsilon target at every stage. Later
520            // vacuum poles can consume positive orders, while Vakint reserves
521            // coefficient and normalization poles independently.
522            for (integral, spectators) in stages {
523                let stage_loop_count = if factorized { 1 } else { retained_loop_count };
524                let external = function!(vakint::symbols::S.p, W_.i_, W_.x___).to_pattern();
525                let mut used_ids = coefficient
526                    .pattern_match(&external, None, None)
527                    .map(|matched| i64::try_from(matched[&W_.i_].as_view()))
528                    .collect::<Result<std::collections::HashSet<_>, _>>()
529                    .map_err(|error| eyre!(error))?;
530                let mut next_id = 0i64;
531                let mut restore_spectators = Vec::new();
532                for spectator in spectators {
533                    while used_ids.contains(&next_id) {
534                        next_id = next_id.checked_add(1).ok_or_else(|| {
535                            eyre!("no free external momentum ID for vacuum spectator")
536                        })?;
537                    }
538                    used_ids.insert(next_id);
539                    let source = function!(vakint::symbols::S.k, spectator, W_.x___);
540                    let target = function!(vakint::symbols::S.p, next_id, W_.x___);
541                    coefficient = coefficient
542                        .replace(source.to_pattern())
543                        .with(target.to_pattern());
544                    restore_spectators.push(Replacement::new(target, source));
545                }
546                let mut stage_term = source_term.clone();
547                stage_term.integral = integral;
548                stage_term.numerator = coefficient;
549                let mut integrand_vakint = VakintExpression(vec![stage_term]);
550                let mut same_momentum = std::collections::BTreeMap::<Atom, Vec<_>>::new();
551                for matched in integrand_vakint.0[0]
552                    .integral
553                    .pattern_match(&propagator, None, None)
554                {
555                    let momentum = &matched[&W_.mom_];
556                    same_momentum
557                        .entry(momentum.clone().min(-momentum))
558                        .or_default()
559                        .push((
560                            usize::try_from(matched[&W_.a_].as_view())
561                                .map_err(|error| eyre!(error))?,
562                            matched[&W_.mass_].clone(),
563                            matched[&W_.e_].clone(),
564                        ));
565                }
566                for (momentum, propagators) in same_momentum {
567                    let (_, first_mass, _) = &propagators[0];
568                    if propagators.iter().all(|(_, mass, _)| mass == first_mass)
569                        || momentum.contains_symbol(vakint::symbols::S.p)
570                        || propagators.iter().any(|(_, mass, power)| {
571                            mass.contains_symbol(W_.x_)
572                                || !i64::try_from(power.as_view()).is_ok_and(|power| power > 0)
573                        })
574                    {
575                        continue;
576                    }
577
578                    // A mixed-mass one-loop vacuum is a sum of single-mass
579                    // tadpoles: 1/(D_a D_b) = (1/D_a - 1/D_b)/(m_a²-m_b²).
580                    // The same identity applies to any equal-momentum group
581                    // with other vacuum propagators held fixed. Partial-fraction
582                    // only the denominator, including raised powers; the graph
583                    // numerator remains an unchanged factor.
584                    let loop_square = Atom::var(W_.x_);
585                    let denominator = propagators
586                        .iter()
587                        .fold(Atom::one(), |product, (_, mass, power)| {
588                            product * (&loop_square - mass).pow(-power)
589                        });
590                    let fractions = denominator
591                        .try_to_rational_polynomial::<_, _, u16>(&Q, &Z, [W_.x_])?
592                        .apart_factored_denominators(0);
593                    for term in std::mem::take(&mut integrand_vakint.0) {
594                        let powers = term
595                            .integral
596                            .pattern_match(&propagator, None, None)
597                            .map(|matched| {
598                                Ok((
599                                    usize::try_from(matched[&W_.a_].as_view())
600                                        .map_err(|error| eyre!(error))?,
601                                    matched[&W_.e_].clone(),
602                                ))
603                            })
604                            .collect::<Result<std::collections::BTreeMap<_, _>>>()?;
605                        let graph = vakint::graph::Graph::new_from_atom(
606                            term.integral.as_view(),
607                            *powers.keys().next_back().unwrap(),
608                        )?;
609                        for (coefficient, denominator, power) in &fractions {
610                            let denominator = denominator.to_expression();
611                            let constant = denominator.replace(W_.x_).with(Atom::Zero);
612                            let leading = denominator.replace(W_.x_).with(Atom::one()) - &constant;
613                            let (survivor, _, _) = propagators.iter().find(|(_, mass, _)| {
614                                (&denominator - &leading * (&loop_square - mass)).expand().is_zero()
615                            }).ok_or_else(|| eyre!("vacuum partial fraction has an unexpected denominator {denominator}"))?;
616                            let coefficient = coefficient.to_expression() / leading.pow(*power);
617                            if coefficient.contains_symbol(W_.x_) {
618                                return Err(eyre!(
619                                    "vacuum partial fraction retained loop dependence in its coefficient"
620                                ));
621                            }
622                            let mut contracted = graph.clone();
623                            for (id, _, _) in &propagators {
624                                if id != survivor {
625                                    contracted.contract_edges(&[*id].into_iter().collect());
626                                }
627                            }
628                            // Removing a serial line contracts its endpoints.
629                            // Dropping it without that contraction would silently
630                            // lower a sunset's graph rank although both loop
631                            // generators still occur in its denominators.
632                            if contracted.to_symbolica_graph(false).num_loops() != stage_loop_count
633                            {
634                                return Err(eyre!(
635                                    "vacuum partial fraction changed the retained {stage_loop_count}-loop integration domain for component {reduced_label}"
636                                ));
637                            }
638                            let mut fraction = term.clone();
639                            fraction.numerator *= coefficient;
640                            fraction.integral = function!(
641                                vakint::symbols::S.topo,
642                                contracted
643                                    .edges
644                                    .values()
645                                    .fold(Atom::one(), |product, edge| {
646                                        product
647                                            * function!(
648                                                vakint::symbols::S.prop,
649                                                edge.id,
650                                                function!(
651                                                    vakint::symbols::S.edge,
652                                                    edge.left_node_id,
653                                                    edge.right_node_id
654                                                ),
655                                                &edge.momentum,
656                                                &edge.mass,
657                                                if edge.id == *survivor {
658                                                    Atom::num(*power)
659                                                } else {
660                                                    powers[&edge.id].clone()
661                                                }
662                                            )
663                                    })
664                            );
665                            integrand_vakint.0.push(fraction);
666                        }
667                    }
668                }
669                // Only a vacuum topology whose actual masses all vanish is scaleless.
670                // Certify polynomial loop dependence as well: an external scale hidden
671                // inside a nonpolynomial numerator must not be erased with the soft jet.
672                integrand_vakint.0.retain(|term| {
673                    let propagators = term
674                        .integral
675                        .pattern_match(&propagator, None, None)
676                        .map(|matched| {
677                            matched[&W_.mass_].is_zero()
678                                && matched[&W_.mom_].contains_symbol(vakint::symbols::S.k)
679                        })
680                        .collect::<Vec<_>>();
681                    if propagators.is_empty()
682                        || propagators.iter().any(|massless_loop| !massless_loop)
683                        || term.integral.contains_symbol(vakint::symbols::S.p)
684                    {
685                        return true;
686                    }
687                    let mut pending = vec![term.numerator.as_view()];
688                    let mut scaleless = true;
689                    while let Some(view) = pending.pop() {
690                        if !view.contains_symbol(vakint::symbols::S.k) {
691                            continue;
692                        }
693                        match view {
694                            AtomView::Add(sum) => pending.extend(sum.iter()),
695                            AtomView::Mul(product) => pending.extend(product.iter()),
696                            AtomView::Pow(power) => {
697                                let (base, exponent) = power.get_base_exp();
698                                if i64::try_from(exponent).is_ok_and(|power| power >= 0) {
699                                    pending.push(base);
700                                } else {
701                                    scaleless = false;
702                                    break;
703                                }
704                            }
705                            AtomView::Fun(function)
706                                if function.get_symbol() == vakint::symbols::S.k
707                                    && function.get_nargs() > 0
708                                    && usize::try_from(function.get(0)).is_ok() => {}
709                            AtomView::Fun(function)
710                                if function.get_symbol() == vakint::symbols::S.dot
711                                    && function.get_nargs() == 2
712                                    && function.iter().all(|argument| {
713                                        matches!(argument, AtomView::Fun(vector)
714                                            if [vakint::symbols::S.k, vakint::symbols::S.p]
715                                                .contains(&vector.get_symbol())
716                                                && vector.get_nargs() == 1
717                                                && usize::try_from(vector.get(0)).is_ok())
718                                    }) => {}
719                            _ => {
720                                scaleless = false;
721                                break;
722                            }
723                        }
724                    }
725                    if scaleless {
726                        debug_tags!(#uv, #integrated, #vakint, #trace;
727                            stage = "certified_scaleless_vacuum",
728                            log.integral = term.integral,
729                            log.numerator = term.numerator,
730                            "Dropping a massless vacuum term with polynomial loop numerator"
731                        );
732                    }
733                    !scaleless
734                });
735                if integrand_vakint.0.is_empty() {
736                    coefficient = Atom::Zero;
737                    break;
738                }
739
740                for (term_index, t) in integrand_vakint.0.iter().enumerate() {
741                    debug_tags!(#uv,#integrated,#vakint,#trace,#to_vakint;
742                        term_index = %term_index,
743                        log.integral = t.integral,
744                        log.numerator = t.numerator,
745                        "Vakint term as input"
746                    );
747                }
748                debug_tags!(#uv,#integrated,#vakint;settings = ?&self.vakint_settings,"Vakint args");
749
750                // let mut res = vakint
751                //     .0
752                //     .evaluate(&vakint.1, integrand_vakint.as_view())
753                //     .unwrap();
754
755                // The analytic backends identify a vacuum mass through a
756                // symbol or its square. Alias other exact nonzero mass values
757                // only in propagator mass slots, then restore the values below.
758                let mut mass_aliases = std::collections::BTreeMap::new();
759                let mut used_symbols = integrand_vakint
760                    .0
761                    .iter()
762                    .flat_map(|term| [&term.integral, &term.numerator])
763                    .flat_map(|atom| atom.get_all_symbols(true))
764                    .collect::<std::collections::HashSet<_>>();
765                let mut mass_serial = 0;
766                for term in &mut integrand_vakint.0 {
767                    let mut replacements = Vec::new();
768                    for matched in term.integral.pattern_match(&propagator, None, None) {
769                        let mass = &matched[&W_.mass_];
770                        if mass.contains_symbol(vakint::symbols::S.k)
771                            || mass.contains_symbol(vakint::symbols::S.p)
772                        {
773                            return Err(eyre!(
774                                "a vacuum propagator mass must be independent of momentum"
775                            ));
776                        }
777                        if mass.is_zero()
778                            || matches!(mass.as_view(), AtomView::Var(_))
779                            || matches!(mass.as_view(), AtomView::Pow(power)
780                                if matches!(power.get_base_exp().0, AtomView::Var(_))
781                                    && power.get_base_exp().1 == Atom::num(2).as_view())
782                        {
783                            continue;
784                        }
785                        let alias = mass_aliases.entry(mass.clone()).or_insert_with(|| {
786                            loop {
787                                let symbol = vakint_symbol!(format!(
788                                    "integrated_mass_squared_{mass_serial}"
789                                ));
790                                mass_serial += 1;
791                                if used_symbols.insert(symbol) {
792                                    break Atom::var(symbol);
793                                }
794                            }
795                        });
796                        replacements.push(Replacement::new(
797                            propagator
798                                .replace_wildcards(&matched)
799                                .map_err(|error| eyre!(error))?,
800                            function!(
801                                vakint::symbols::S.prop,
802                                &matched[&W_.a_],
803                                &matched[&W_.b_],
804                                &matched[&W_.mom_],
805                                &*alias,
806                                &matched[&W_.e_]
807                            ),
808                        ));
809                    }
810                    term.integral = term.integral.replace_multiple(&replacements);
811                }
812                integrand_vakint.canonicalize(
813                    self.vakint_settings,
814                    &self.vakint.topologies,
815                    false,
816                )?;
817                for (term_index, t) in integrand_vakint.0.iter().enumerate() {
818                    debug_tags!(#uv,#integrated,#vakint,#trace,#canonicalize;
819                        term_index = %term_index,
820                        log.integral = t.integral,
821                        log.numerator = t.numerator,
822                        "Vakint term after canonicalization"
823                    );
824                }
825                integrand_vakint.tensor_reduce(self.vakint, self.vakint_settings)?;
826                for (term_index, t) in integrand_vakint.0.iter().enumerate() {
827                    debug_tags!(#uv,#integrated,#vakint,#trace,#tensor_reduce;
828                        term_index = %term_index,
829                        log.integral = t.integral,
830                        log.numerator = t.numerator,
831                        "Vakint term after tensor reduction"
832                    );
833                }
834                for (term_index, term) in integrand_vakint.0.iter_mut().enumerate() {
835                    term.numerator =
836                        self.simplify_projected_numerator(&term.numerator, current.topo_order())?;
837                    debug_tags!(#uv, #integrated, #vakint, #trace;
838                        stage = "projected_numerator_after_d_dimensional_algebra",
839                        term_index = %term_index,
840                        log.numerator = term.numerator,
841                        "Completed projected numerator algebra before Laurent expansion"
842                    );
843                }
844                integrand_vakint.evaluate_integral(self.vakint, self.vakint_settings)?;
845                for (term_index, t) in integrand_vakint.0.iter().enumerate() {
846                    debug_tags!(#uv,#integrated,#vakint,#trace,#evaluate;
847                        term_index = %term_index,
848                        log.integral = t.integral,
849                        log.numerator = t.numerator,
850                        "Vakint term after evaluation"
851                    );
852                }
853
854                coefficient = Atom::from(integrand_vakint).replace_multiple(
855                    mass_aliases
856                        .into_iter()
857                        .map(|(mass, alias)| Replacement::new(alias, mass))
858                        .collect::<Vec<_>>(),
859                );
860                if coefficient.contains_symbol(vakint::symbols::S.k) {
861                    return Err(eyre!(
862                        "integrated vacuum factor retained an active loop momentum"
863                    ));
864                }
865                coefficient = coefficient.replace_multiple(&restore_spectators);
866            }
867            res += coefficient;
868        }
869
870        debug_tags!(#uv,#integrated,#vakint,#trace,#raw;
871            log.res = res,
872            "Raw post vakint "
873        );
874
875        let mut res = Self::restore_numerator(res, current.topo_order());
876
877        res = Self::dimensionally_regularized(&res.simplify_metrics().metric_shorthand_to_dot());
878
879        debug_tags!(#uv, #integrated, #vakint, #inspect, #trace, #replace;
880            log.res = res,
881            "Replaced post vakint "
882        );
883
884        // This strips as many dummies as possible after undoing chains and traces,
885        // so that terms can merge later on.
886        let bispinor_rep = Bispinor {}.into();
887        let after_chainify = res.chainify(bispinor_rep);
888        debug_tags!(#uv, #integrated, #vakint, #profile, #trace, #chainify;
889            log.expr = after_chainify,
890            "Integrated UV chain cleanup after chainify"
891        );
892
893        let after_collect_chains = after_chainify.collect_chains(bispinor_rep);
894        debug_tags!(#uv, #integrated, #vakint, #profile, #trace, #collect;
895            log.expr = after_collect_chains,
896            "Integrated UV chain cleanup after collect_chains"
897        );
898
899        res = after_collect_chains.undo_single_length();
900        debug_tags!(#uv, #integrated, #vakint, #profile, #trace, #undo_single_length;
901            log.expr = res,
902            "Integrated UV chain cleanup after undo_single_length"
903        );
904
905        // println!("\nIntegrated CT:\n{}\n", res);
906        // Normalize the fully integrated UV subgraph before Laurent projection
907        // and reinsertion: identical scalar masters can arrive as 2*(A+B) or
908        // 2*A+2*B. No cograph numerator is present at this analytic boundary.
909        Ok(res.expand())
910    }

912    fn simplify_projected_numerator(
913        &self,
914        numerator: &Atom,
915        topology_order: usize,
916    ) -> Result<Atom> {
917        // Tensor projection can identify Lorentz slots belonging to different
918        // gamma matrices. Complete their d-dimensional algebra before Vakint
919        // expands scalar coefficients in epsilon; otherwise an evanescent
920        // contraction can multiply a pole only after its finite term was lost.
921        let numerator = Vakint::convert_to_dot_notation(self.vakint_settings, numerator.as_view())?;
922        let numerator = Self::restore_numerator(numerator, topology_order);
923        let numerator = simplify(&numerator)?;
924        Self::ensure_resolved_lorentz_contractions(&numerator)?;
925        let numerator = Self::dimensionally_regularized(&numerator)
926            .undo_schoonschip::<Aind>()?
927            .undo_chain::<Aind>()?
928            .undo_trace::<Aind>()?
929            .metric_shorthand_to_dot();
930        Ok(Self::to_vakint_numerator(&numerator))
931    }

933    fn ensure_resolved_lorentz_contractions(numerator: &Atom) -> Result<()> {
934        let minkowski = Minkowski {}.to_symbolic([GS.dim]).get_symbol().unwrap();
935        if !numerator.contains_symbol(minkowski) {
936            return Ok(());
937        }
938        // Inspect each expanded analytic branch using the tensor parser's
939        // incidence rules. Expose compact inter-chain contractions first;
940        // keeping only non-momentum Lorentz tensors allows scalar products and
941        // slashed external momenta while detecting unresolved tensor operators.
942        let explicit = numerator
943            .parse_to_symbolic_net::<Aind>(&ParseSettings {
944                shorthand_parsing: ShorthandParsing::Expand {
945                    schoonschip: SchoonschipExpansionMode::full(),
946                    trace: false,
947                    chain: true,
948                },
949                ..Default::default()
950            })
951            .map_err(|error| eyre!("invalid analytic UV Lorentz tensor notation: {error}"))?
952            .simple_execute::<()>()?
953            .expand();
954        let terms = if let AtomView::Add(sum) = explicit.as_view() {
955            sum.iter().collect::<Vec<_>>()
956        } else {
957            vec![explicit.as_view()]
958        };
959        let settings = ParseSettings {
960            shorthand_parsing: ShorthandParsing::Opaque {
961                inference: StructureInferenceMode::Fast,
962            },
963            ..Default::default()
964        };
965        for term in terms {
966            let mut unresolved_inner_product = false;
967            let tensors = term.replace_map(|view, _, out| {
968                if !matches!(view, AtomView::Num(_)) && !view.contains_symbol(minkowski) {
969                    // Remove scalar subtrees atomically: replacing the d in
970                    // 1/(d-1) alone would manufacture a singular coefficient.
971                    // Numerical exponents of tensor powers keep their value.
972                    **out = Atom::one();
973                    return;
974                }
975                match view {
976                    AtomView::Fun(f) => {
977                        if [SPENSO_TAG.dot, ETS.metric].contains(&f.get_symbol())
978                            && view.contains_symbol(minkowski)
979                            && !f
980                                .iter()
981                                .all(|arg| Slot::<Minkowski, Aind>::try_from(arg).is_ok())
982                        {
983                            // Full shorthand parsing must expose every compact
984                            // Lorentz contraction before scalar coefficients can
985                            // be discarded by this diagnostic.
986                            unresolved_inner_product = true;
987                        }
988                        let momentum = [
989                            GS.emr_mom,
990                            GS.loop_mom,
991                            vakint::symbols::S.p,
992                            vakint::symbols::S.k,
993                        ]
994                        .contains(&f.get_symbol());
995                        **out = if !momentum
996                            && f.iter()
997                                .any(|arg| Slot::<Minkowski, Aind>::try_from(arg).is_ok())
998                        {
999                            view.to_owned()
1000                        } else {
1001                            Atom::one()
1002                        };
1003                    }
1004                    AtomView::Var(_) => **out = Atom::one(),
1005                    _ => {}
1006                }
1007            });
1008            if unresolved_inner_product {
1009                return Err(eyre!(
1010                    "an unresolved compact Lorentz inner product remains after d-dimensional UV numerator algebra"
1011                ));
1012            }
1013            let network = tensors
1014                .parse_to_symbolic_net::<Aind>(&settings)
1015                .map_err(|error| {
1016                    eyre!("invalid residual d-dimensional tensor structure: {error}")
1017                })?;
1018            let external = network.graph.dangling_indices();
1019            for (pair, _, edge) in network.graph.graph.iter_edges() {
1020                if pair.is_paired()
1021                    && let NetworkEdge::Slot(slot) = edge.data
1022                    && slot.rep_name().symbol() == minkowski
1023                    && !external.contains(slot)
1024                {
1025                    return Err(eyre!(
1026                        "an unresolved internal Lorentz tensor contraction remains after d-dimensional UV numerator algebra; a d-dimensional tensor-operator reduction is required before analytic integration"
1027                    ));
1028                }
1029            }
1030        }
1031        Ok(())
1032    }

1034    fn dimensionally_regularized(numerator: &Atom) -> Atom {
1035        let minkowski = Minkowski {}.to_symbolic([GS.dim]).get_symbol().unwrap();
1036        let dimension = Atom::var(GS.dim);
1037        let regulated = Atom::num(4) - Atom::num(2) * Atom::var(GS.dim_epsilon);
1038        numerator.replace_map(|view, _, out| {
1039            if view.get_symbol() == Some(minkowski) {
1040                // A retained open Lorentz basis keeps its representation. All
1041                // scalar occurrences, including inside function coefficients,
1042                // must instead participate in the Laurent expansion.
1043                **out = view.to_owned();
1044            } else if view == dimension.as_view() {
1045                **out = regulated.clone();
1046            }
1047        })
1048    }

1050    fn restore_numerator(mut res: Atom, topology_order: usize) -> Atom {
1051        // Projector indices may close two opaque tensor leaves. Restore their
1052        // complete Minkowski representation before exposing the tensor bodies.
1053        let mink = Minkowski {}.new_rep(GS.dim);
1054        let mut projected_indices = std::collections::BTreeMap::new();
1055        res = res.replace_map(|view, _, out| {
1056            if matches!(view, AtomView::Fun(index) if index.get_symbol() == vakint::symbols::S.tensor_index) {
1057                let replacement = projected_indices.entry(view.to_owned()).or_insert_with(|| loop {
1058                    let candidate = mink.to_symbolic([Aind::new_dummy().to_atom()]);
1059                    if res.pattern_match(&candidate.to_pattern(), None, None).next().is_none() {
1060                        break candidate;
1061                    }
1062                });
1063                **out = replacement.clone();
1064            }
1065        });
1066
1067        res = res
1068            // Opaque tensor slots are needed only while Vakint validates and
1069            // projects the Lorentz domain. Restore the caller's tensor leaves.
1070            .replace(function!(vakint::symbols::S.tensor, W_.x_, W_.x___))
1071            .with(Atom::var(W_.x_))
1072            .replace(parse_lit!(vakint::cl2))
1073            .with(parse_lit!(cl2))
1074            .replace(parse_lit!(vakint::sqrt3))
1075            .with(parse_lit!(sqrt(3)));
1076
1077        let vk_metric = vakint_symbol!("g");
1078
1079        // apply metric
1080        res = res
1081            .replace(vakint::symbols::S.p.call_args([W_.i_, W_.j_]))
1082            .when(W_.j_.filter(|r| r.is_integer().is_true()))
1083            .with(
1084                vakint::symbols::S.p.call_args([
1085                    Atom::var(W_.i_),
1086                    mink.to_symbolic([GS
1087                        .uvaind
1088                        .call_args([Atom::num(topology_order), Atom::var(W_.j_)])]),
1089                ]),
1090            )
1091            .replace(
1092                vakint::symbols::S
1093                    .p
1094                    .call_args([Atom::var(W_.i_), vakint::symbols::S.dot_dummy_ind(W_.j_)]),
1095            )
1096            .when(W_.j_.filter(|r| r.is_integer().is_true()))
1097            .with(
1098                vakint::symbols::S.p.call_args([
1099                    Atom::var(W_.i_),
1100                    mink.to_symbolic([GS
1101                        .uvaind
1102                        .call_args([Atom::num(topology_order), Atom::var(W_.j_)])]),
1103                ]),
1104            )
1105            .replace(vakint::symbols::S.p.call_args([W_.x__]))
1106            .with(GS.emr_mom.call_args([W_.x__]));
1107        res = res
1108            .replace(function!(vk_metric, W_.x_, W_.y_) * function!(GS.emr_mom, W_.x___, W_.x_))
1109            .with(function!(GS.emr_mom, W_.x___, W_.y_))
1110            .replace(function!(
1111                vk_metric,
1112                vakint::symbols::S.dot_dummy_ind(W_.x_),
1113                W_.y_
1114            ))
1115            .when(W_.x_.filter(|r| r.is_integer().is_true()))
1116            .with(function!(
1117                vk_metric,
1118                mink.to_symbolic([GS
1119                    .uvaind
1120                    .call_args([Atom::num(topology_order), Atom::var(W_.x_)])]),
1121                W_.y_
1122            ))
1123            .replace(function!(
1124                vk_metric,
1125                W_.x_,
1126                vakint::symbols::S.dot_dummy_ind(W_.y_)
1127            ))
1128            .when(W_.y_.filter(|r| r.is_integer().is_true()))
1129            .with(function!(
1130                vk_metric,
1131                mink.to_symbolic([GS
1132                    .uvaind
1133                    .call_args([Atom::num(topology_order), Atom::var(W_.y_)])]),
1134                W_.x_
1135            ))
1136            .replace(function!(vk_metric, W_.x_, W_.y_))
1137            .with(function!(ETS.metric, W_.x_, W_.y_));
1138
1139        res = res.replace(vakint::symbols::S.cmplx_i).with(Atom::i());
1140
1141        res
1142    }

1144    fn to_vakint_numerator(numerator: &Atom) -> Atom {
1145        let numerator = numerator
1146            .replace(function!(GS.loop_mom, W_.x___))
1147            .with(function!(vakint::symbols::S.k, W_.x___))
1148            .replace(function!(GS.emr_mom, W_.x___))
1149            .with(function!(vakint::symbols::S.p, W_.x___))
1150            .replace(function!(
1151                SPENSO_TAG.dot,
1152                function!(W_.a_, W_.a___, Minkowski {}.new_rep(GS.dim).to_symbolic([])),
1153                function!(W_.b_, W_.b___, Minkowski {}.new_rep(GS.dim).to_symbolic([]))
1154            ))
1155            .with(vakint::symbols::S.dot(function!(W_.a_, W_.a___), function!(W_.b_, W_.b___)))
1156            .replace(function!(
1157                ETS.metric,
1158                Minkowski {}.to_symbolic([W_.a__]),
1159                Minkowski {}.to_symbolic([W_.b__])
1160            ))
1161            .with(function!(
1162                vakint::symbols::S.metric,
1163                Minkowski {}.to_symbolic([W_.a__]),
1164                Minkowski {}.to_symbolic([W_.b__])
1165            ));
1166        // Preserve the Lorentz slots of arbitrary spin tensors without
1167        // teaching Vakint Spenso's representation syntax or tensor algebra.
1168        // Direct slots use the same parser as the tensor network. The original
1169        // leaf stays opaque, including its spin and color indices.
1170        numerator.replace_map(|view, _, out| {
1171            let AtomView::Fun(tensor) = view else { return };
1172            if [
1173                vakint::symbols::S.k,
1174                vakint::symbols::S.p,
1175                vakint::symbols::S.g,
1176                vakint::symbols::S.dot,
1177                vakint::symbols::S.tensor,
1178            ]
1179            .contains(&tensor.get_symbol())
1180            {
1181                **out = view.to_owned();
1182                return;
1183            }
1184            let slots = tensor
1185                .iter()
1186                .filter(|argument| Slot::<Minkowski, Aind>::try_from(*argument).is_ok())
1187                .collect::<Vec<_>>();
1188            if !slots.is_empty() {
1189                let mut wrapped = FunctionBuilder::new(vakint::symbols::S.tensor).add_arg(view);
1190                for slot in slots {
1191                    wrapped = wrapped.add_arg(slot);
1192                }
1193                **out = wrapped.finish();
1194            }
1195        })
1196    }

1756            reduced = %reduced_label,

1798    replacements: Vec<Replacement>,
1799    free_variables: Vec<Atom>,
1800}

1802impl VakintMomentumSolution {
1803    fn solve(
1804        system: &[Atom],
1805        variables: &[Atom],
1806        add_additional_args: &[Replacement],
1807    ) -> Result<Self> {
1808        if variables.is_empty() {
1809            return Ok(Self {
1810                replacements: vec![],
1811                free_variables: vec![],
1812            });
1813        }
1814        if system.is_empty() {
1815            return Ok(Self {
1816                replacements: vec![],
1817                free_variables: variables.to_vec(),
1818            });
1819        }
1820
1821        let solutions = Atom::solve(system)
1822            .wrt_with_exponent::<u8, _>(variables)
1823            .map_err(|source| eyre!("{source}"))?;
1824        let [solution] = solutions.iter().as_slice() else {
1825            return Err(eyre!(
1826                "expected one Vakint momentum solution, got {} branches",
1827                solutions.len()
1828            ));
1829        };
1830        let solution = variables
1831            .iter()
1832            .map(|variable| {
1833                let polynomial_variable =
1834                    PolyVariable::try_from(variable.clone()).map_err(|source| eyre!("{source}"))?;
1835                if solution.free_variables().contains(&polynomial_variable) {
1836                    Ok(variable.clone())
1837                } else {
1838                    solution.get(&polynomial_variable).cloned().ok_or_else(|| {
1839                        eyre!("Vakint momentum solution has no value for {polynomial_variable}")
1840                    })
1841                }
1842            })
1843            .collect::<Result<Vec<_>>>()?;
1844        Ok(Self::from_solution(
1845            &solution,
1846            variables,
1847            add_additional_args,
1848        ))
1849    }
1850
1851    fn from_solution(
1852        solution: &[Atom],
1853        variables: &[Atom],
1854        add_additional_args: &[Replacement],
1855    ) -> Self {
1856        debug_assert_eq!(solution.len(), variables.len());
1857        let mut replacements = vec![];
1858        let mut free_variables = vec![];
1859
1860        for index in (0..variables.len()).rev() {
1861            let replacement = &solution[index];
1862            let variable = &variables[index];
1863            let replacement = replacement.replace_multiple(&replacements);
1864            if replacement == variable {
1865                free_variables.push(variable.clone());
1866            } else {
1867                replacements.push(Replacement::new(
1868                    variable.replace_multiple(add_additional_args).to_pattern(),
1869                    replacement
1870                        .replace_multiple(add_additional_args)
1871                        .to_pattern(),
1872                ));
1873            }
1874        }
1875
1876        Self {
1877            replacements,
1878            free_variables,
1879        }
1880    }
1881
1882    fn ensure_free_variables_eliminated(
1883        &self,
1884        term_index: usize,
1885        expression_kind: &str,
1886        expression: &Atom,
1887        add_additional_args: &[Replacement],
1888    ) -> Result<()> {
1889        self.ensure_variables_eliminated(
1890            term_index,
1891            expression_kind,
1892            expression,
1893            self.free_variables.iter(),
1894            add_additional_args,
1895        )
1896    }
1897
1898    fn ensure_variables_eliminated<'a>(
1899        &self,
1900        term_index: usize,
1901        expression_kind: &str,
1902        expression: &Atom,
1903        variables: impl IntoIterator<Item = &'a Atom>,
1904        add_additional_args: &[Replacement],
1905    ) -> Result<()> {
1906        for variable in variables {
1907            let bare_pattern = variable.to_pattern();
1908            let indexed_pattern = variable.replace_multiple(add_additional_args).to_pattern();
1909            if expression
1910                .pattern_match(&bare_pattern, None, None)
1911                .next()
1912                .is_some()
1913                || expression
1914                    .pattern_match(&indexed_pattern, None, None)
1915                    .next()
1916                    .is_some()
1917            {
1918                return Err(eyre!(
1919                    "Underdetermined Vakint momentum solve left free variable {} in {} of term {}",
1920                    variable,
1921                    expression_kind,
1922                    term_index
1923                ));
1924            }
1925        }
1926        Ok(())
1927    }
1928
1929    fn rewrite_numerator(&self, numerator: &Atom) -> Atom {
1930        numerator
1931            .replace_multiple(&self.replacements)
1932            .normalize_dots()
1933    }
1934
1935    fn rewrite_integral(&self, integral: &Atom, add_additional_args: &[Replacement]) -> Atom {
1936        let free_variable_zero_replacements =
1937            self.free_variable_zero_replacements(add_additional_args);
1938        Self::normalize_topology_momenta(
1939            integral
1940                .replace_multiple(&self.replacements)
1941                .replace_multiple(&free_variable_zero_replacements),
1942        )
1943    }
1944
1945    fn free_variable_zero_replacements(
1946        &self,
1947        add_additional_args: &[Replacement],
1948    ) -> Vec<Replacement> {
1949        let mut replacements = Vec::with_capacity(2 * self.free_variables.len());
1950        for variable in &self.free_variables {
1951            replacements.push(Replacement::new(variable.to_pattern(), Atom::Zero));
1952            let indexed_variable = variable.replace_multiple(add_additional_args);
1953            if indexed_variable != *variable {
1954                replacements.push(Replacement::new(indexed_variable.to_pattern(), Atom::Zero));
1955            }
1956        }
1957        replacements
1958    }
1959
1960    fn normalize_topology_momenta(expression: Atom) -> Atom {
1961        if !expression
1962            .replace(function!(vakint::symbols::S.topo, W_.x_))
1963            .matches()
1964        {
1965            return expression;
1966        }
1967
1968        expression
1969            .replace(function!(
1970                vakint::symbols::S.prop,
1971                W_.edgeid_,
1972                W_.x_,
1973                W_.mom_,
1974                W_.mass_,
1975                W_.prop_
1976            ))
1977            .with_map(|matches| {
1978                function!(
1979                    vakint::symbols::S.prop,
1980                    matches.get(W_.edgeid_).unwrap().to_atom(),
1981                    matches.get(W_.x_).unwrap().to_atom(),
1982                    matches.get(W_.mom_).unwrap().to_atom().expand(),
1983                    matches.get(W_.mass_).unwrap().to_atom(),
1984                    matches.get(W_.prop_).unwrap().to_atom()
1985                )
1986            })
1987    }
