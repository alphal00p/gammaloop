85impl UVForestNodeTerm {
86    pub fn file_name(&self) -> String {
87        let key = sanitize_file_component(&self.node_key);
88        let residue = residue_suffix(self.residue_index);
89        format!(
90            "node_{:03}_{}_term_{:03}_{}.dot",
91            self.node_index, key, self.term_index, residue
92        )
93    }
94
95    /// Return the atom-free provenance embedded in this computed Hedge DOT.
96    ///
97    /// Parsing remains encapsulated here so callers do not need a direct DOT
98    /// parser dependency. Legacy and structure-only exports return `None`.
99    pub fn forest_provenance(&self) -> Result<Option<serde_json::Value>> {
100        let dot: DotGraph = DotGraph::from_string(&self.dot)
101            .map_err(|error| eyre!("failed to parse UV forest node DOT: {error}"))?;
102        let Some(json) = dot.global_data.statements.get("forest_provenance") else {
103            return Ok(None);
104        };
105        Ok(Some(
106            serde_json::from_str(json).context("while parsing UV forest provenance JSON")?,
107        ))
108    }
109}

121impl Process {
122    pub fn export_uv_forest_graph(
123        &self,
124        integrand_name: &str,
125        graph_id: usize,
126        settings: &UVForestExportSettings,
127    ) -> Result<UVForestExport> {
128        let generation_settings = &self
129            .settings_history
130            .as_ref()
131            .ok_or_else(|| {
132                eyre!(
133                    "Cannot export UV forests for process {} without generation settings history",
134                    self.definition.folder_name
135                )
136            })?
137            .generation;
138        let resolved = self.get_integrand(integrand_name)?;
139        let integrand = resolved.require_generated()?;
140        let source = if settings.computed {
141            let (graph, expression) = match &self.collection {
142                ProcessCollection::Amplitudes(amplitudes) => {
143                    let source = amplitudes[&resolved.canonical_name]
144                        .graphs
145                        .get(graph_id)
146                        .ok_or_else(|| eyre!("Missing source amplitude graph {graph_id}"))?;
147                    (&source.graph, source.derived_data.cff_expression.as_ref())
148                }
149                ProcessCollection::CrossSections(cross_sections) => {
150                    let source = cross_sections[&resolved.canonical_name]
151                        .supergraphs
152                        .get(graph_id)
153                        .ok_or_else(|| eyre!("Missing source cross-section graph {graph_id}"))?;
154                    (
155                        &source.graph,
156                        source.derived_data.global_cff_expression.as_ref(),
157                    )
158                }
159            };
160            // Generation and persistent selection preserve graph order;
161            // reject stale runtime metadata instead of pairing a different
162            // graph with this stored production residue map.
163            if integrand.graph_name_by_id(graph_id) != Some(graph.name.as_str()) {
164                return Err(eyre!(
165                    "Source/runtime graph mismatch for computed UV forest export at id {graph_id}"
166                ));
167            }
168            Some((
169                graph,
170                expression.ok_or_else(|| {
171                    eyre!(
172                        "Graph {} has no stored production CFF for computed UV forest export",
173                        graph.name
174                    )
175                })?,
176            ))
177        } else {
178            None
179        };
180        let cff_options = source
181            .map(|(graph, _)| graph.production_cff_3d_expression_options(generation_settings))
182            .transpose()?;
183        let orientation = source
184            .zip(cff_options.as_ref())
185            .map(|((_, expression), options)| {
186                OrientationProjection::exact_expression(
187                    expression,
188                    options,
189                    &generation_settings.orientation_pattern,
190                    generation_settings.explicit_orientation_sum_only,
191                )
192            });
193        match integrand {
194            ProcessIntegrand::Amplitude(integrand) => {
195                let term = integrand.data.graph_terms.get(graph_id).ok_or_else(|| {
196                    eyre!(
197                        "Graph id {} is out of range for amplitude integrand {}",
198                        graph_id,
199                        integrand.data.name
200                    )
201                })?;
202                let cut_structure = CutStructure::empty(&term.graph);
203                export_graph(
204                    &term.graph,
205                    cut_structure,
206                    orientation,
207                    generation_settings,
208                    settings,
209                    |_, atom| atom,
210                )
211            }
212            ProcessIntegrand::CrossSection(integrand) => {
213                let term = integrand.data.graph_terms.get(graph_id).ok_or_else(|| {
214                    eyre!(
215                        "Graph id {} is out of range for cross-section integrand {}",
216                        graph_id,
217                        integrand.data.name
218                    )
219                })?;
220                let cuts = term
221                    .cut_group_data
222                    .cut_groups
223                    .iter()
224                    .map(|cuts| CutSet {
225                        residue_selector: ResidueSelector {
226                            lu: Some(cuts.lu_cut_selection(&term.graph, &term.cuts)),
227                            left_th_cut: None,
228                            right_th_cut: None,
229                        },
230                        union: cuts
231                            .cuts
232                            .iter()
233                            .map(|cut_id| term.cuts[*cut_id].cut.as_subgraph())
234                            .reduce(|cut_1, cut_2| cut_1.union(&cut_2))
235                            .unwrap_or_else(|| term.graph.empty_subgraph()),
236                        canonicalize_external_shifts: false,
237                    })
238                    .collect();
239                let cut_structure = CutStructure { cuts };
240                // Finalized UV nodes already include the CFF spatial measure.
241                // Reuse the physical cut-group conversion, as production does.
242                let lu_prefactors = term
243                    .cut_group_data
244                    .cut_groups
245                    .iter()
246                    .map(|group| group.lu_prefactor(&term.graph, &term.cuts))
247                    .collect::<Result<Vec<_>>>()?;
248
249                export_graph(
250                    &term.graph,
251                    cut_structure,
252                    orientation,
253                    generation_settings,
254                    settings,
255                    |forest_index, atom| atom * &lu_prefactors[forest_index],
256                )
257            }
258        }
259    }
260}

262fn export_graph(
263    graph: &Graph,
264    cut_structure: CutStructure,
265    orientation: Option<OrientationProjection<'_>>,
266    generation_settings: &GenerationSettings,
267    export_settings: &UVForestExportSettings,
268    mut post_process: impl FnMut(usize, Atom) -> Atom,
269) -> Result<UVForestExport> {
270    if generation_settings.uv.orchestrator == UVOrchestrator::Compare {
271        return Err(eyre!(
272            "UV forest export does not support uv.orchestrator = compare"
273        ));
274    }
275    validate_local_counterterm_schemes(graph, &cut_structure, &generation_settings.uv)?;
276
277    let mut forest_dot = String::new();
278    let mut node_terms = Vec::new();
279    for (forest_index, cut) in cut_structure.cuts.into_iter().enumerate() {
280        let single_cut = CutStructure { cuts: vec![cut] };
281        let forest_name = format!(
282            "uv_{}_forest_{forest_index:03}",
283            sanitize_file_component(&graph.name)
284        );
285        let mut graph = graph.clone();
286        match generation_settings.uv.orchestrator {
287            UVOrchestrator::LegacyDagForest => export_legacy_forest(
288                forest_index,
289                &forest_name,
290                &mut graph,
291                single_cut,
292                orientation,
293                generation_settings,
294                export_settings,
295                &mut |atom| post_process(forest_index, atom),
296                &mut forest_dot,
297                &mut node_terms,
298            )?,
299            UVOrchestrator::HedgePoset => export_hedge_poset_forest(
300                forest_index,
301                &forest_name,
302                &mut graph,
303                single_cut,
304                orientation,
305                generation_settings,
306                export_settings,
307                &mut |atom| post_process(forest_index, atom),
308                &mut forest_dot,
309                &mut node_terms,
310            )?,
311            UVOrchestrator::Compare => unreachable!("compare is rejected before export"),
312        }
313    }
314
315    Ok(UVForestExport {
316        graph_name: graph.name.clone(),
317        forest_dot,
318        node_terms,
319    })
320}

322#[allow(clippy::too_many_arguments)]
323fn export_legacy_forest(
324    forest_index: usize,
325    forest_name: &str,
326    graph: &mut Graph,
327    cut_structure: CutStructure,
328    orientation: Option<OrientationProjection<'_>>,
329    generation_settings: &GenerationSettings,
330    export_settings: &UVForestExportSettings,
331    post_process: &mut impl FnMut(Atom) -> Atom,
332    forest_dot: &mut String,
333    node_terms: &mut Vec<UVForestNodeTerm>,
334) -> Result<()> {
335    let cut_woods = CutWoods::new(cut_structure, graph, &generation_settings.uv)?;
336    let mut cut_forests = cut_woods.unfold(graph);
337    let Some(forest) = cut_forests.forests.first_mut() else {
338        return Err(eyre!("Legacy UV exporter produced no forest"));
339    };
340    forest_dot.push_str(&forest.dot_serialize_for_export(forest_name));
341    forest_dot.push('\n');
342
343    if !export_settings.computed {
344        return Ok(());
345    }
346
347    if matches!(
348        generation_settings.uv.final_integrand,
349        FinalIntegrandDimension::FourD
350    ) {
351        return Err(eyre!(
352            "computed 4D UV forest export is not supported by the legacy DAG forest for graph '{}' (forest {forest_index}); use the HedgePoset orchestrator",
353            graph.name,
354        ));
355    }
356    let orientation = orientation.ok_or_else(|| {
357        eyre!("Computed UV forest export requires its stored production CFF expression")
358    })?;
359    compute_legacy_forest(graph, &mut cut_forests, orientation, generation_settings)?;
360    let forest = cut_forests
361        .forests
362        .first()
363        .expect("legacy forest exists after compute");
364    node_terms.extend(
365        forest
366            .export_node_expressions(graph, forest_index, post_process)?
367            .into_iter()
368            .map(|term| node_expression_to_dot(graph, forest_name, term))
369            .collect::<Result<Vec<_>>>()?,
370    );
371    Ok(())
372}

383        orientation,

388#[allow(clippy::too_many_arguments)]
389fn export_hedge_poset_forest(
390    forest_index: usize,
391    forest_name: &str,
392    graph: &mut Graph,
393    cut_structure: CutStructure,
394    orientation: Option<OrientationProjection<'_>>,
395    generation_settings: &GenerationSettings,
396    export_settings: &UVForestExportSettings,
397    post_process: &mut impl FnMut(Atom) -> Atom,
398    forest_dot: &mut String,
399    node_terms: &mut Vec<UVForestNodeTerm>,
400) -> Result<()> {
401    let wood = HedgePosetWood::new(cut_structure, graph, &generation_settings.uv)?;
402    let mut forests = wood.unfold();
403    forest_dot.push_str(&name_dot_graph(forests.dot_serialize(), forest_name));
404    forest_dot.push('\n');
405
406    if !export_settings.computed {
407        return Ok(());
408    }
409
410    let expressions = match &generation_settings.uv.final_integrand {
411        FinalIntegrandDimension::FourD => {
412            forests.integrate(graph, crate::utils::vakint()?, &generation_settings.uv)?;
413            forests.export_4d_node_expressions(forest_index)?
414        }
415        FinalIntegrandDimension::ThreeD => {
416            // A computed 3D export carries provenance from the direct
417            // orientation-local CFF projections themselves. Do not construct
418            // unrelated 4D atoms merely to annotate this diagnostic output.
419            forests.compute(
420                graph,
421                crate::utils::vakint()?,
422                orientation.ok_or_else(|| {
423                    eyre!("Computed UV forest export requires its stored production CFF expression")
424                })?,
425                &generation_settings.uv,
426            )?;
427            forests.export_node_expressions(forest_index, post_process)?
428        }
429    };
430    node_terms.extend(
431        expressions
432            .into_iter()
433            .map(|term| node_expression_to_dot(graph, forest_name, term))
434            .collect::<Result<Vec<_>>>()?,
435    );
436    Ok(())
437}

439fn node_expression_to_dot(
440    graph: &Graph,
441    forest_name: &str,
442    term: UVForestNodeExpression,
443) -> Result<UVForestNodeTerm> {
444    let graph_name = format!(
445        "{}_node_{:03}_term_{:03}",
446        forest_name, term.node_index, term.term_index
447    );
448    let mut dot_graph = graph
449        .with_global_numerator_only(graph_name, term.numerator.clone())
450        .to_dot_graph_with_settings(&DotExportSettings {
451            split_xs_by_initial_states: true,
452            output_full_numerator: false,
453            ..DotExportSettings::default()
454        });
455    dot_graph
456        .global_data
457        .statements
458        .insert("forest_name".into(), forest_name.to_owned());
459    dot_graph
460        .global_data
461        .statements
462        .insert("forest_index".into(), term.forest_index.to_string());
463    dot_graph
464        .global_data
465        .statements
466        .insert("forest_node_index".into(), term.node_index.to_string());
467    dot_graph
468        .global_data
469        .statements
470        .insert("forest_node_key".into(), term.node_key.clone());
471    dot_graph
472        .global_data
473        .statements
474        .insert("forest_term_index".into(), term.term_index.to_string());
475    dot_graph.global_data.statements.insert(
476        "forest_residue_index".into(),
477        residue_suffix(term.residue_index),
478    );
479    if let Some(provenance) = term.forest_provenance.as_deref() {
480        let provenance = serde_json::to_string(provenance)
481            .context("while serializing UV forest provenance as JSON")?;
482        dot_graph
483            .global_data
484            .statements
485            .insert("forest_provenance".into(), provenance);
486    }
487    let full_num = term
488        .numerator
489        .printer(SpensoPrintSettings::typst_options())
490        .to_string();
491    dot_graph
492        .global_data
493        .statements
494        .insert("full_num".into(), full_num);
495
496    let mut dot = Vec::new();
497    dot_graph
498        .write_io(&mut dot)
499        .context("while serializing UV forest node DOT")?;
500    Ok(UVForestNodeTerm {
501        forest_index: term.forest_index,
502        node_index: term.node_index,
503        node_key: term.node_key,
504        term_index: term.term_index,
505        residue_index: term.residue_index,
506        dot: String::from_utf8(dot).context("DOT graph serialization was not UTF-8")?,
507    })
508}

515pub(crate) fn sanitize_file_component(value: &str) -> String {
516    let mut sanitized = String::new();
517    for c in value.chars() {
518        let next = if c.is_ascii_alphanumeric() || c == '-' {
519            c
520        } else {
521            '_'
522        };
523        if next == '_' {
524            if !sanitized.ends_with('_') {
525                sanitized.push(next);
526            }
527        } else {
528            sanitized.push(next);
529        }
530    }
531    sanitized.truncate(48);
532    sanitized = sanitized.trim_matches('_').to_string();
533    if sanitized.is_empty() {
534        "key".to_string()
535    } else {
536        sanitized
537    }
538}

540fn residue_suffix(index: CutCFFIndex) -> String {
541    let suffix = index.to_string();
542    if suffix.is_empty() {
543        "all_none".to_string()
544    } else {
545        sanitize_file_component(&suffix)
546    }

1009                        source.push_str("left -> right [particle=scalar_0];");

1011                    source.push('}');
1012                    (GenerationType::CrossSection, source)
1013                };
1014                let graphs = Graph::from_string(&source, &model)?;
1015                let definition = ProcessDefinition::from_graph_list(&graphs, kind, &model)?;
1016                let mut process = Process::from_graph_list(
1017                    "export_phase".into(),
1018                    "default".into(),
1019                    graphs,
1020                    kind,
1021                    Some(definition),
1022                    None,
1023                    &model,
1024                )?;
1025                let mut settings = GlobalSettings::default();
1026                settings.generation.uv.orchestrator = orchestrator;
1027                settings.generation.uv.subtract_uv = false;
1028                settings.generation.uv.generate_integrated = false;
1029                settings.generation.threshold_subtraction.enable_thresholds = false;
1030                settings.generation.explicit_orientation_sum_only = true;
1031                settings.generation.evaluator.compile = false;
1032                process.preprocess(&model, &settings, &(&runtime).into(), &pool)?;
1033                process.generate_integrands(&model, &settings, (&runtime).into(), &pool)?;
1034                let (graph, expected) = match &process.collection {
1035                    ProcessCollection::Amplitudes(amplitudes) => {
1036                        let graph = &amplitudes["default"].graphs[0];
1037                        (&graph.graph, graph.derived_data.resolved_integrand()?)
1038                    }
1039                    ProcessCollection::CrossSections(cross_sections) => {
1040                        let graph = &cross_sections["default"].supergraphs[0];
1041                        (
1042                            &graph.graph,
1043                            graph
1044                                .derived_data
1045                                .cut_paramatric_integrand
1046                                .iter()
1047                                .map(|integrand| integrand.integrands.resolved())
1048                                .collect::<color_eyre::Result<Vec<_>>>()?
1049                                .iter()
1050                                .flat_map(|integrands| integrands.iter())
1051                                .map(|(_, atom)| atom.clone())
1052                                .sum::<Atom>(),
1053                        )
1054                    }
1055                };
1056                let exported = process.export_uv_forest_graph(
1057                    "default",
1058                    0,
1059                    &UVForestExportSettings { computed: true },
1060                )?;
1061                let split = graph
1062                    .iter_edges_of(
1063                        &graph
1064                            .full_filter()
1065                            .subtract(&graph.initial_state_cut)
1066                            .subtract(&graph.tree_edges),
1067                    )
1068                    .map(|(_, edge, _)| GS.split_mom_pattern_simple(edge))
1069                    .collect::<Vec<_>>();
1070                let mut actual = Atom::Zero;
1071                for term in &exported.node_terms {
1072                    for parsed in Graph::from_string(&term.dot, &model)? {
1073                        actual += UvMarker::new(&settings.generation.uv).finish(
1074                            &parsed
1075                                .global_prefactor
1076                                .num
1077                                .replace_multiple(&split)
1078                                .replace(GS.den(W_.a_, W_.b_, W_.c_, W_.d_))
1079                                .with(W_.d_),
1080                        );
1081                    }
1082                }

1084                // evaluate those trees directly at exact nonsingular points.
1085                // These signed checks certify the sampled values, not a symbolic
1086                // identity; neither numerator is expanded or polynomialized.
1087                if actual != expected {
1088                    let mut inputs = actual
1089                        .get_all_indeterminates(false)
1090                        .into_iter()
1091                        .chain(expected.get_all_indeterminates(false))
1092                        .map(|input| input.to_owned())
1093                        .collect::<Vec<_>>();
1094                    inputs.sort();
1095                    inputs.dedup();
1096                    for base in [3, 5, 7] {
1097                        let mut actual_value = actual.clone();
1098                        let mut expected_value = expected.clone();
1099                        for (index, input) in inputs.iter().enumerate() {

1103                        }

1108                            actual_value, expected_value,
1109                            "{kind}/{orchestrator}: computed export differs from production at base {base}"
1110                        );
1111                    }
1112                }
1113                if multiplicity != 0 {
1114                    let [coupling, expression] = model.get_coupling("SCALAR_COUPLING").rep_rule();
1115                    let lambda: Atom = model.get_parameter("lam").name.into();
1116                    // The runtime tree-denominator function is the identity;
1117                    // these contact Born graphs have an empty product inside it.
1118                    let mut physical = actual
1119                        .replace(GS.wrap_tree_denoms(Atom::one()))
1120                        .with(Atom::one())
1121                        .replace(coupling)
1122                        .with(expression)
1123                        .replace(lambda)
1124                        .with(Atom::one())
1125                        .replace(GS.rescale_star)
1126                        .with(Atom::one())
1127                        .replace(GS.hfunction_lu_cut)
1128                        .with(Atom::one());
1129                    for edge in 0..graph.n_edges() {
1130                        physical = physical.replace(GS.ose(EdgeIndex(edge))).with(Atom::one());
1131                    }
1132                    // With all cut energies set to one, |A|²=1 and no graph
1133                    // symmetry factor, dPhi_n has residue coefficient
1134                    // 2pi / [(2pi)^(3(n-1)) * 2^n]. The LU derivative is separate.
1135                    let two_pi = Atom::num(2) * Atom::var(GS.pi);
1136                    let born = &two_pi
1137                        / (two_pi.pow(3 * (multiplicity - 1)) * Atom::num(2).pow(multiplicity));
1138                    assert_eq!(
1139                        physical, born,
1140                        "{orchestrator}: scalar Born export has the wrong phase or normalization"
1141                    );
1142                }
1143            }
1144        }
1145        Ok(())
