49/// Atom-free record of one direct operation on an orientation-local CFF.
50///
51/// `active_subgraph` is the accumulated local sector, while
52/// `rescaled_subgraph` is the part acted on by this step. They differ during
53/// disconnected replay. `canonical_route_*` records the component route and
54/// `route_*` records the exact local LMB selected by the same code path that
55/// constructs the Laurent projection.
56#[derive(Clone, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
57pub(crate) struct Local3DProjectionStep {
58    pub topo_order: usize,
59    pub current_component: String,
60    pub given_component: String,
61    pub active_subgraph: String,
62    pub rescaled_subgraph: String,
63    pub scheme: ApproximationType,
64    pub dod: i32,
65    pub rescaling: Local3DLoopRescaling,
66    pub canonical_route_loop_edges: Vec<usize>,
67    pub canonical_route_external_edges: Vec<usize>,
68    pub canonical_route_signatures: Vec<Local3DRouteSignature>,
69    pub route_loop_edges: Vec<usize>,
70    pub route_external_edges: Vec<usize>,
71    pub route_signatures: Vec<Local3DRouteSignature>,
72    pub conceptual_branches: Vec<Local3DSignedBranch>,
73    pub materialization: Local3DMaterialization,
74}

76/// One ordered projection history contributing to a local node. Nested steps
77/// are stored child-to-parent; disconnected and local/integrated replay
78/// sectors remain distinct paths.
79#[derive(Clone, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize)]
80pub(crate) struct Local3DProjectionPath {
81    pub steps: Vec<Local3DProjectionStep>,
82}

84impl Local3DProjectionPath {
85    fn with_step(&self, step: &Local3DProjectionStep) -> Self {
86        let mut path = self.clone();
87        path.steps.push(step.clone());
88        path
89    }

92pub(in crate::uv) fn paths_with_step(
93    paths: &[Local3DProjectionPath],
94    step: &Local3DProjectionStep,
95) -> Vec<Local3DProjectionPath> {
96    if paths.is_empty() {
97        vec![Local3DProjectionPath {
98            steps: vec![step.clone()],
99        }]
100    } else {
101        paths.iter().map(|path| path.with_step(step)).collect()
102    }
103}

105impl Local3DLoopRescaling {
106    #[allow(clippy::too_many_arguments)]
107    pub(in crate::uv) fn projection_step<S: ForestNodeLike>(
108        self,
109        ctx: &UVCtx<'_>,
110        current: &S,
111        given: &S,
112        active_subgraph: &SuBitGraph,
113        rescaled_subgraph: Option<&SuBitGraph>,
114        canonical_lmb: &LoopMomentumBasis,
115        lmb: &LoopMomentumBasis,
116    ) -> Local3DProjectionStep {
117        let rescaled_subgraph = rescaled_subgraph.unwrap_or_else(|| current.subgraph());
118        let route_signatures = |lmb: &LoopMomentumBasis, subgraph: &SuBitGraph| {
119            ctx.graph
120                .paired_edges(subgraph)
121                .into_iter()
122                .map(|edge| {
123                    let signature = &lmb.edge_signatures[edge];
124                    Local3DRouteSignature {
125                        edge: usize::from(edge),
126                        loop_signature: signature.internal.to_string(),
127                        external_signature: signature.external.to_string(),
128                    }
129                })
130                .collect::<Vec<_>>()
131        };
132        let (conceptual_branches, materialization) =
133            if current.renormalization_scheme() == ApproximationType::IR && current.dod() > 0 {
134                (
135                    vec![
136                        Local3DSignedBranch {
137                            branch: Local3DConceptualBranch::U,
138                            coefficient: 1,
139                        },
140                        Local3DSignedBranch {
141                            branch: Local3DConceptualBranch::S,
142                            coefficient: 1,
143                        },
144                        Local3DSignedBranch {
145                            branch: Local3DConceptualBranch::US,
146                            coefficient: -1,
147                        },
148                    ],
149                    Local3DMaterialization::FactorizedSoft,
150                )
151            } else {
152                (
153                    vec![Local3DSignedBranch {
154                        branch: Local3DConceptualBranch::U,
155                        coefficient: 1,
156                    }],
157                    Local3DMaterialization::DirectU,
158                )
159            };
160
161        Local3DProjectionStep {
162            topo_order: current.topo_order(),
163            current_component: current.subgraph().string_label(),
164            given_component: given.subgraph().string_label(),
165            active_subgraph: active_subgraph.string_label(),
166            rescaled_subgraph: rescaled_subgraph.string_label(),
167            scheme: current.renormalization_scheme(),
168            dod: current.dod(),
169            rescaling: self,
170            canonical_route_loop_edges: canonical_lmb
171                .loop_edges
172                .iter()
173                .map(|edge| usize::from(*edge))
174                .collect(),
175            canonical_route_external_edges: canonical_lmb
176                .ext_edges
177                .iter()
178                .map(|edge| usize::from(*edge))
179                .collect(),
180            canonical_route_signatures: route_signatures(canonical_lmb, current.subgraph()),
181            route_loop_edges: lmb
182                .loop_edges
183                .iter()
184                .map(|edge| usize::from(*edge))
185                .collect(),
186            route_external_edges: lmb
187                .ext_edges
188                .iter()
189                .map(|edge| usize::from(*edge))
190                .collect(),
191            route_signatures: route_signatures(lmb, rescaled_subgraph),
192            conceptual_branches,
193            materialization,
194        }
195    }
