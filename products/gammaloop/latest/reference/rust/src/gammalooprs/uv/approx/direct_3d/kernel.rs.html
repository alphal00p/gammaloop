30/// One independently framed connected part of a direct forest sector.
31/// Disconnected replay keeps these frames separate until an enclosing Taylor
32/// operation contains and combines their carriers.
33#[derive(Clone, Debug, PartialEq, Eq, Hash)]
34pub(crate) struct DirectCoordinateFrame {
35    pub(crate) active_subgraph: SuBitGraph,
36    pub(crate) lmb: LoopMomentumBasis,
37}

52/// Choose the loop coordinates in which one direct sector enters its next
53/// Taylor operation. An enclosing operation must retain every loop carrier
54/// already owned by the part of the sector contained in `given`;
55/// otherwise an equivalent basis can turn a previously hard child momentum
56/// into a crown-shifted momentum before the outer series is taken.
57pub(super) fn coordinate_lmb<S: ForestNodeLike>(
58    ctx: &UVCtx<'_>,
59    current: &S,
60    given: &S,
61    prior_active_subgraph: Option<&SuBitGraph>,
62    prior_frames: &[DirectCoordinateFrame],
63    active_subgraph: &SuBitGraph,
64) -> Result<LoopMomentumBasis> {
65    let graph = ctx.graph;
66    let active_subgraph = active_subgraph.intersection(current.subgraph());
67    for frame in prior_frames {
68        if given.subgraph().intersects(&frame.active_subgraph)
69            && !given.subgraph().includes(&frame.active_subgraph)
70        {
71            return Err(eyre!(
72                "a nested direct local-3D prefix partially overlaps a prior coordinate frame"
73            ));
74        }
75    }
76    let retained_region = prior_active_subgraph
77        .map(|active| active.intersection(given.subgraph()))
78        .unwrap_or_else(|| graph.empty_subgraph());
79    let mut retained_loop_edges = prior_frames
80        .iter()
81        .flat_map(|frame| frame.lmb.loop_edges.iter())
82        .filter(|edge| retained_region.includes(&graph[*edge].1))
83        .copied()
84        .collect::<Vec<_>>();
85    retained_loop_edges.sort();
86    retained_loop_edges.dedup();
87    let is_compatible = |candidate: &LoopMomentumBasis| {
88        retained_loop_edges
89            .iter()
90            .all(|edge| candidate.loop_edges.contains(edge))
91            && candidate
92                .loop_edges
93                .iter()
94                .filter(|edge| retained_region.includes(&graph[*edge].1))
95                .count()
96                == retained_loop_edges.len()
97    };
98
99    let quotient_lmb = if given.subgraph().is_empty() {
100        current.lmb().clone()
101    } else {
102        let given = InternalSubGraph::try_new(given.subgraph().clone(), graph.as_ref())
103            .ok_or_else(|| {
104                eyre!(
105                    "nested direct local-3D prefix is not an internal subgraph of {}",
106                    current.subgraph().string_label(),
107                )
108            })?;
109        graph.shrunken_sub_lmb(
110            current.subgraph(),
111            &given,
112            graph.dummy_stripped_external_flows_of(current.subgraph()),
113            None,
114        )?
115    };
116    let expected_active_rank = retained_loop_edges.len() + quotient_lmb.loop_edges.len();
117
118    // A sector can have an already-integrated or disconnected inactive prefix.
119    // Contract that prefix only after fixing the enclosing coordinates, so the
120    // quotient keeps the compatible subset of the prior sector's carriers.
121    let coordinate_lmb = if active_subgraph == *current.subgraph() {
122        if is_compatible(current.lmb()) && current.lmb().loop_edges.len() == expected_active_rank {
123            current.lmb().clone()
124        } else {
125            graph
126                .generate_loop_momentum_bases_of(current.subgraph())
127                .into_iter()
128                .filter(|candidate| candidate.loop_edges != current.lmb().loop_edges)
129                .sorted_by_key(|candidate| candidate.loop_edges.iter().copied().collect_vec())
130                .find(|candidate| {
131                    is_compatible(candidate)
132                        && candidate.loop_edges.len() == expected_active_rank
133                })
134                .ok_or_else(|| {
135                    eyre!(
136                        "no direct local-3D enclosing frame has retained rank {} plus quotient rank {}",
137                        retained_loop_edges.len(),
138                        quotient_lmb.loop_edges.len(),
139                    )
140                })?
141        }
142    } else {
143        let shrunken_filter = current.subgraph().subtract(&active_subgraph);
144        let shrunken =
145            InternalSubGraph::try_new(shrunken_filter, graph.as_ref()).ok_or_else(|| {
146                eyre!(
147                    "inactive direct local-3D prefix is not an internal subgraph of {}",
148                    current.subgraph().string_label(),
149                )
150            })?;
151        let externals = graph.dummy_stripped_external_flows_of(current.subgraph());
152        let mut failures = Vec::new();
153        let mut try_enclosing = |enclosing_lmb: &LoopMomentumBasis| {
154            if !is_compatible(enclosing_lmb) {
155                return None;
156            }
157            let enclosing_loop_edges = enclosing_lmb.loop_edges.clone();
158            match graph.shrunken_sub_lmb(
159                current.subgraph(),
160                &shrunken,
161                externals.clone(),
162                Some(enclosing_lmb),
163            ) {
164                Ok(candidate)
165                    if is_compatible(&candidate)
166                        && candidate.loop_edges.len() == expected_active_rank =>
167                {
168                    Some(candidate)
169                }
170                Ok(candidate) => {
171                    failures.push(format!(
172                        "enclosing {:?} contracted to incompatible {:?}",
173                        enclosing_loop_edges, candidate.loop_edges,
174                    ));
175                    None
176                }
177                Err(error) => {
178                    failures.push(format!(
179                        "enclosing {:?} failed contraction: {error}",
180                        enclosing_loop_edges,
181                    ));
182                    None
183                }
184            }
185        };
186        let selected = try_enclosing(current.lmb()).or_else(|| {
187            graph
188                .generate_loop_momentum_bases_of(current.subgraph())
189                .into_iter()
190                .filter(|candidate| candidate.loop_edges != current.lmb().loop_edges)
191                .sorted_by_key(|candidate| candidate.loop_edges.iter().copied().collect_vec())
192                .find_map(|candidate| try_enclosing(&candidate))
193        });
194        selected.ok_or_else(|| {
195            eyre!(
196                "no contracted direct local-3D frame retains exactly {:?} with total rank {}; attempts: {}",
197                retained_loop_edges,
198                expected_active_rank,
199                failures.join("; "),
200            )
201        })?
202    };
203    if !is_compatible(&coordinate_lmb) {
204        return Err(eyre!(
205            "contracting a direct local-3D prefix lost the prior sector frame {:?}; quotient carriers are {:?}",
206            retained_loop_edges,
207            coordinate_lmb.loop_edges,
208        ));
209    }
210    if coordinate_lmb.loop_edges.len() != expected_active_rank {
211        return Err(eyre!(
212            "direct local-3D sector frame has {} loop carriers, expected retained rank {} plus quotient rank {} = {}",
213            coordinate_lmb.loop_edges.len(),
214            retained_loop_edges.len(),
215            quotient_lmb.loop_edges.len(),
216            expected_active_rank,
217        ));
218    }
219
220    debug_tags!(#generation, #uv, #local, #direct, #trace;
221        stage = "direct_3d_coordinate_lmb",
222        current = %current.log_display(),
223        given = %given.log_display(),
224        active_subgraph = %active_subgraph.string_label(),
225        retained_region = %retained_region.string_label(),
226        retained_loop_edges = ?retained_loop_edges,
227        quotient_loop_edges = ?quotient_lmb.loop_edges,
228        expected_active_rank,
229        loop_edges = ?coordinate_lmb.loop_edges,
230        "Selected one coordinate frame for a direct local-3D sector"
231    );
232    Ok(coordinate_lmb)
233}

235/// Apply one local-3D Taylor operation to a complete residue-map family.
236/// Selectors remain outside the atoms. Newly attached factors share one
237/// coefficient-parametric body and retain each complete key's argument row.
238/// This kernel does not add the current operation's subtraction minus;
239/// forest composition supplies it once, retaining signs already in the branch.
240// Keep the forest operation, coordinate frame, and residue family explicit.
241#[allow(clippy::too_many_arguments)]
242pub(super) fn apply_taylor<S: ForestNodeLike>(
243    ctx: &UVCtx<'_>,
244    orientation: OrientationProjection<'_>,
245    current: &S,
246    given: &S,
247    active_subgraph: Option<SuBitGraph>,
248    lmb: &LoopMomentumBasis,
249    integrands: &DirectResidueBranches,
250) -> Result<DirectResidueBranches> {
251    let active_subgraph = active_subgraph
252        .as_ref()
253        .map(|active| active.intersection(current.subgraph()));
254    let reduced = current.reduced_subgraph(given);
255    let numerator = ctx
256        .graph
257        .numerator(&reduced, given.subgraph())
258        .get_single_atom()
259        .expect("graph numerator should be available")
260        .simplify_color_with(ColorSimplifySettings {
261            simplify_non_color: false,
262            ..Default::default()
263        });
264    let scope = DirectResidueBranches::numerator_scope();
265    let numerator_tag = scope.1.clone();
266    let integrands = integrands.multiply_key_mapped(orientation, ctx.graph, &numerator, scope)?;
267    debug_tags!(#generation, #profile, #uv, #local, #direct, #trace;
268        stage = "direct_3d_taylor_family",
269        current = %current.log_display(),
270        given = %given.log_display(),
271        reduced = %reduced.string_label(),
272        loop_edges = ?lmb.loop_edges,
273        branch_count = integrands.iter_keys().count(),
274        "Prepared a shared numerator before its Taylor kernel"
275    );
276    let started = integrands
277        .map_expressions(|atom| start(ctx, current, atom, active_subgraph.as_ref(), lmb))?;
278    match current.renormalization_scheme() {
279        ApproximationType::MUV | ApproximationType::PolePart => {
280            Direct3dApproximation::t(ctx, current, given, &started, lmb)
281        }
282        ApproximationType::IR => {
283            let t_tilde = t_tilde(
284                ctx,
285                current,
286                given,
287                &integrands,
288                &numerator_tag,
289                active_subgraph.as_ref(),
290                lmb,
291            )?;
292            Direct3dApproximation::t(ctx, current, given, &started, lmb)?
293                .zip_add(&t_tilde)?
294                .zip_add(&-Direct3dApproximation::t(
295                    ctx, current, given, &t_tilde, lmb,
296                )?)
297        }
298        ApproximationType::VaccuumLimit => Err(eyre!("Not yet implemented VaccuumLimit")),
299        ApproximationType::OS => Err(eyre!("Not yet implemented OS")),
300        ApproximationType::Unsubtracted => panic!("should have been kept out of the wood"),
301    }
302}

304// #[debug_instrument(
305//     current = %current.log_display(),
306//     given = %given.log_display(),
307//     reduced,
308// )]
309fn t_tilde<S: ForestNodeLike>(
310    ctx: &UVCtx<'_>,
311    current: &S,
312    given: &S,
313    cff: &DirectResidueBranches,
314    numerator_tag: &Atom,
315    active_subgraph: Option<&SuBitGraph>,
316    lmb: &LoopMomentumBasis,
317) -> Result<DirectResidueBranches> {
318    let graph = ctx.graph;
319    let settings = ctx.settings;
320    let reduced = current.reduced_subgraph(given);
321    let rescaled_subgraph = active_subgraph.unwrap_or_else(|| current.subgraph());
322    let lmb_id = lmb
323        .loop_edges
324        .first()
325        .copied()
326        .unwrap_or_else(|| current.lmb_id());
327
328    // split numerator momenta into OSEs and spatial parts
329    let mut reps = Vec::new();
330    for (p, eid, e) in graph.iter_edges_of(rescaled_subgraph) {
331        if p.is_paired() {
332            let e_mass = e.data.mass_atom();
333            reps.push(GS.split_mom_pattern(eid, lmb_id, e_mass, settings.inner_products));
334        }
335    }
336
337    let numerator_scaled = cff.map_numerators(|entry| {
338        if entry.tags.first() != Some(numerator_tag) {
339            return Ok(entry.rhs.clone());
340        }
341        let mut numerator = entry.rhs.replace_multiple(&reps);
342
343        // rescale the external momenta in the added numerator subgraph
344        for e in &lmb.ext_edges {
345            // println!("Rescale {}", e);
346            numerator = numerator
347                .replace(GS.emr_vec_index(*e, W_.x___))
348                .with(GS.emr_vec_index(*e, W_.x___) * GS.rescale)
349                .replace(GS.emr_mom(*e, W_.x___))
350                .with(GS.emr_mom(*e, W_.x___) * GS.rescale);
351        }
352
353        Ok(numerator)
354    })?;
355    let atomarg = numerator_scaled.map_expressions(|atom| {
356        let mut atomarg = atom.clone();
357
358        // add data for OSE computation and add an explicit sqrt
359        for (p, ei, e) in graph.iter_edges_of(rescaled_subgraph) {
360            let eid = usize::from(ei) as i64;
361            if p.is_paired() {
362                // set energies from inner_t on-shell
363                atomarg = atomarg.replace(function!(GS.energy, eid)).with(GS.ose(ei));
364
365                let e_mass = e.data.mass_atom();
366                atomarg = atomarg.replace(GS.ose(ei)).with(GS.ose_full(
367                    ei,
368                    lmb_id,
369                    e_mass,
370                    None,
371                    settings.inner_products,
372                ));
373            }
374        }
375
376        atomarg = atomarg.replace_multiple(&reps);
377
378        let mom_reps = graph.replacement_impl(
379            |e, loops, externals| {
380                Replacement::new(
381                    GS.emr_vec
382                        .call_args([Atom::num(usize::from(e)), Atom::var(W_.x___)])
383                        .to_pattern(),
384                    (loops
385                        .replace(function!(GS.emr_vec, W_.x_))
386                        .allow_new_wildcards_on_rhs(true)
387                        .with(
388                            FunctionBuilder::new(GS.emr_vec)
389                                .add_arg(W_.x_)
390                                .add_args([W_.x___])
391                                .finish(),
392                        )
393                        + externals * GS.rescale)
394                        .to_pattern(),
395                )
396            },
397            &reduced,
398            lmb,
399            GS.emr_vec,
400            GS.emr_vec,
401            &[],
402            &[W_.x___],
403            HedgePair::is_paired,
404            true,
405        );
406
407        atomarg = atomarg.replace_multiple(&mom_reps);
408        atomarg = atomarg
409            .replace(function!(GS.ose, W_.a___))
410            .with(function!(*OSE_FOR_LOCAL_3D_SERIES, W_.a___));
411
412        Ok(atomarg)
413    })?;
414    let a = atomarg.series_preserving_numerators(
415        GS.rescale, Atom::Zero.as_view(), -1, DirectResidueBranches::numerator_scope().1,
416    ).map_err(|error| {
417        eyre!(
418            "local 3D infrared Taylor series through order minus one failed for graph `{}` at {} given {}, in loop coordinates {:?}: {error}",
419            graph.name,
420            current.subgraph().string_label(),
421            given.subgraph().string_label(),
422            lmb.loop_edges,
423        )
424    })?;
425
426    a.map_expressions(|atom| {
427        let mut a = atom
428            .replace(function!(
429                Symbol::DERIVATIVE,
430                0,
431                1,
432                *OSE_FOR_LOCAL_3D_SERIES,
433                W_.y___
434            ))
435            .with(Atom::num(1))
436            .replace(function!(
437                Symbol::DERIVATIVE,
438                W_.x___,
439                *OSE_FOR_LOCAL_3D_SERIES,
440                W_.y___
441            ))
442            .with(Atom::num(0));
443        a = a
444            .replace(function!(*OSE_FOR_LOCAL_3D_SERIES, W_.a___))
445            .with(function!(GS.ose, W_.a___));
446        a = a.replace(GS.rescale).with(Atom::num(1));
447        Ok(a)
448    })
449}

451fn start<S: ForestNodeLike>(
452    ctx: &UVCtx<'_>,
453    current: &S,
454    cff: &Atom,
455    active_subgraph: Option<&SuBitGraph>,
456    lmb: &LoopMomentumBasis,
457) -> Result<Atom> {
458    let graph = ctx.graph;
459    let settings = ctx.settings;
460    let rescaled_subgraph = active_subgraph.unwrap_or_else(|| current.subgraph());
461    let lmb_id = lmb
462        .loop_edges
463        .first()
464        .copied()
465        .unwrap_or_else(|| current.lmb_id());
466    let mut atomarg = cff.clone();
467    debug_tags!(#generation, #profile, #uv, #local, #trace;
468        stage = "local_3d_start_initial",
469        byte_size = atomarg.as_view().get_byte_size(),
470        file.expr = %atomarg,
471        "Local 3D start expression checkpoint"
472    );
473    // println!("CFF: {}", cff);
474
475    // add data for OSE computation and add an explicit sqrt
476    for (p, ei, e) in graph.iter_edges_of(rescaled_subgraph) {
477        let eid = usize::from(ei) as i64;
478        if p.is_paired() {
479            // set energies from inner_t on-shell
480            atomarg = atomarg.replace(function!(GS.energy, eid)).with(GS.ose(ei));
481
482            let e_mass = e.data.mass_atom();
483            atomarg = atomarg.replace(GS.ose(ei)).with(GS.ose_full(
484                ei,
485                lmb_id,
486                e_mass,
487                None,
488                settings.inner_products,
489            ));
490        }
491    }
492    debug_tags!(#generation, #profile, #uv, #local, #trace;
493        stage = "local_3d_start_after_ose_full",
494        byte_size = atomarg.as_view().get_byte_size(),
495        file.expr = %atomarg,
496        "Local 3D start expression checkpoint"
497    );
498
499    // split numerator momenta into OSEs and spatial parts
500    let mut reps = Vec::new();
501    for (p, eid, e) in graph.iter_edges_of(rescaled_subgraph) {
502        if p.is_paired() {
503            let e_mass = e.data.mass_atom();
504            let rep = GS.split_mom_pattern(eid, lmb_id, e_mass, settings.inner_products);
505            debug_tags!(#uv, #local, #momentum, #trace;
506                stage = "local_3d_start_split_mom_pattern",
507                split_rep = %rep,
508                "Local 3D start momentum split"
509            );
510            reps.push(rep);
511        }
512    }
513    let atomarg = atomarg.replace_multiple(&reps);
514    debug_tags!(#generation, #profile, #uv, #local, #trace;
515        stage = "local_3d_start_output",
516        byte_size = atomarg.as_view().get_byte_size(),
517        file.expr = %atomarg,
518        "Local 3D start expression checkpoint"
519    );
520    Ok(atomarg)
521}

523impl Direct3dApproximation<'_> {
524    // #[debug_instrument(
525    //     current = %current.log_display(),
526    //     given = %given.log_display(),
527    //     reduced,
528    // )]
529    fn t_rescale<S: ForestNodeLike>(
530        ctx: &UVCtx<'_>,
531        current: &S,
532        given: &S,
533        integrand: &Atom,
534        lmb: &LoopMomentumBasis,
535        include_measure: bool,
536    ) -> Result<Atom> {
537        let graph = ctx.graph;
538        let reduced = current.reduced_subgraph(given);
539
540        // only apply replacements for edges in the reduced graph
541        let mut mom_reps = graph.uv_spatial_wrapped_replacement(&reduced, lmb, &[W_.x___]);
542        // Explicit spatial components such as `Q(e, cind(1))` remain in the
543        // ordinary EMR wrapper after residue-key mapping; only tensor-vector
544        // occurrences use `EMRvec`. Route both wrappers through the same LMB
545        // before rescaling so every hard numerator momentum is scaled exactly
546        // once, without expanding or redistributing the numerator.
547        mom_reps.extend(graph.uv_wrapped_replacement(&reduced, lmb, &[W_.x___]));
548        for m in &mom_reps {
549            debug_tags!(#uv,#momentum,#trace;mom_rep=%m,"Mom rep");
550        }
551
552        debug_tags!(#generation, #profile, #uv, #local, #summary;
553            stage = "local_3d_t_input",
554            byte_size = integrand.as_view().get_byte_size(),
555            "Local 3D T size checkpoint"
556        );
557        let mut atomarg = integrand.replace_multiple(&mom_reps);
558        debug_tags!(#generation, #profile, #uv, #local, #summary;
559            stage = "local_3d_t_after_momentum_replacements",
560            byte_size = atomarg.as_view().get_byte_size(),
561            "Local 3D T size checkpoint"
562        );
563
564        // Rescale every loop momentum still active in this sector, including
565        // cycles expanded by earlier local operations. Retaining their carriers
566        // can require a different basis, but every branch of this enclosing
567        // Taylor operation must keep the same soft external routing.
568        for e in &lmb.loop_edges {
569            // println!("Rescale {}", e);
570            for momentum_symbol in [GS.emr_vec, GS.emr_mom] {
571                let momentum = function!(momentum_symbol, usize::from(*e) as i64, W_.x___);
572                let soft = current
573                    .lmb()
574                    .ext_atom(*e, momentum_symbol, &[W_.x___], true);
575                let rescaled = (&momentum - &soft) * GS.rescale + soft;
576                atomarg = atomarg.replace(momentum).with(rescaled);
577            }
578        }
579        debug_tags!(#generation, #profile, #uv, #local, #summary;
580            stage = "local_3d_t_after_loop_rescale",
581            byte_size = atomarg.as_view().get_byte_size(),
582            "Local 3D T size checkpoint"
583        );
584
585        // Match the 4D forest Taylor order within the current component: vacuum
586        // masses in its completed coefficients have the hard loop-momentum weight.
587        // Disjoint coefficients retain their owners for a later enclosing operation.
588        // Scale the complete scalar head, leaving its owner independent of Taylor t.
589        // Normalized localization kernels stay outside the series, and the auxiliary
590        // OSE deformation mass keeps its separate role below.
591        let mut invalid_mass_owner = None;
592        atomarg = atomarg.replace_map(|view, _, output| {
593            let AtomView::Fun(mass) = view else {
594                return;
595            };
596            if mass.get_symbol() != GS.m_uv_vacuum {
597                return;
598            }
599            let owner = match mass.get_nargs() {
600                1 => match mass.get(0) {
601                    AtomView::Var(owner) => owner
602                        .get_symbol()
603                        .get_stripped_name()
604                        .strip_prefix("S_")
605                        .and_then(|label| SuBitGraph::from_base62(label, graph.n_hedges())),
606                    _ => None,
607                },
608                _ => None,
609            };
610            let Some(owner) = owner else {
611                invalid_mass_owner = Some("a direct vacuum mass has no valid component owner");
612                return;
613            };
614            if current.subgraph().includes(&owner) {
615                **output = view.to_owned() * GS.rescale;
616            } else if current.subgraph().intersects(&owner) {
617                invalid_mass_owner = Some(
618                    "a direct vacuum-mass owner partially overlaps the current forest component",
619                );
620            }
621        });
622        if let Some(message) = invalid_mass_owner {
623            return Err(eyre!(message));
624        }
625
626        // (re-)expand OSEs from the subgraph only
627        for eid in &lmb.loop_edges {
628            let eid = eid.0 as i64;
629            // rescale the whole OSE so that the function itself has no poles during the expansion
630            atomarg = atomarg.replace(function!(GS.ose, eid, W_.prop_)).with(
631                function!(
632                    GS.ose,
633                    eid,
634                    (GS.m_uv_expansion * GS.m_uv_expansion * GS.rescale * GS.rescale + W_.prop_
635                        - GS.m_uv_expansion * GS.m_uv_expansion)
636                        / GS.rescale
637                        / GS.rescale
638                ) * GS.rescale
639                    * GS.rescale,
640            )
641        }
642        debug_tags!(#generation, #profile, #uv, #local, #summary;
643            stage = "local_3d_t_after_ose_rescale",
644            byte_size = atomarg.as_view().get_byte_size(),
645            "Local 3D T size checkpoint"
646        );
647
648        // The supplied LMB is the integration-space authority. In particular,
649        // a remainder which is a tree in the original incidence can become a
650        // loop after its frozen UV prefix is contracted.
651        if include_measure {
652            atomarg *= Atom::var(GS.rescale).pow(3 * lmb.loop_edges.len() as i64);
653        }
654        atomarg = atomarg.replace(GS.rescale).with(Atom::num(1) / GS.rescale);
655        debug_tags!(#generation, #profile, #uv, #local, #summary;
656            stage = "local_3d_t_before_series",
657            loop_edges = ?lmb.loop_edges,
658            byte_size = atomarg.as_view().get_byte_size(),
659            "Local 3D T size checkpoint"
660        );
661
662        debug_tags!(#uv, #local, #before_series; log.expr = atomarg, "Before series in t");
663
664        Ok(atomarg)
665    }
666
667    fn t<S: ForestNodeLike>(
668        ctx: &UVCtx<'_>,
669        current: &S,
670        given: &S,
671        integrands: &DirectResidueBranches,
672        lmb: &LoopMomentumBasis,
673    ) -> Result<DirectResidueBranches> {
674        let rescaled = integrands
675            .fallible_map(|_, atom| Self::t_rescale(ctx, current, given, atom, lmb, true))?
676            .map_numerators(|entry| Self::t_rescale(ctx, current, given, &entry.rhs, lmb, false))?;
677        let series = rescaled.series_preserving_numerators(
678            GS.rescale, Atom::Zero.as_view(), 0, DirectResidueBranches::numerator_scope().1,
679        ).map_err(|error| eyre!(
680            "local 3D Taylor series through order zero failed for graph `{}` at {} given {}, in loop coordinates {:?}: {error}",
681            ctx.graph.name, current.subgraph().string_label(),
682            given.subgraph().string_label(), lmb.loop_edges,
683        ))?;
684        debug_tags!(#generation, #profile, #uv, #local, #summary;
685            stage = "local_3d_t_after_series",
686            branch_count = series.iter_keys().count(),
687            "Shared local 3D Taylor coefficients prepared"
688        );
689        series.map_expressions(|atom| {
690            let a = atom.replace(GS.rescale).with(Atom::num(1));
691            debug_tags!(#uv, #local; log.expr = a, "Local 3D approximation");
692            Ok(a)
693        })
694    }
