142    pub(super) fn from_keyed(
143        keyed: impl IntoIterator<Item = (DirectResidueKey, Integrands)>,
144    ) -> Result<Self> {
145        let mut branches: Vec<(DirectResidueKey, Integrands)> = Vec::new();
146        for (key, integrands) in keyed {
147            if let Some((_, existing)) = branches
148                .iter_mut()
149                .find(|(existing_key, _)| *existing_key == key)
150            {
151                *existing = existing.clone().zip_add([integrands])?;
152            } else {
153                branches.push((key, integrands));
154            }
155        }
156        let fallback_zero = branches
157            .iter()
158            .find(|(_, integrands)| integrands.iter().all(|(_, atom)| atom.is_zero()))
159            .cloned();
160        branches.retain(|(_, integrands)| !integrands.iter().all(|(_, atom)| atom.is_zero()));
161        if branches.is_empty() {
162            branches.extend(fallback_zero);
163        }
164        if branches.is_empty() {
165            return Err(eyre!("direct local-3D residue branches cannot be empty"));
166        }
167        Ok(Self(branches))
168    }

174    /// Prepare one numerator with Taylor-inert coefficients of the complete
175    /// affine energy maps. The returned tuple is `(numerator, parameters, rows)`;
176    /// each row retains its complete residue key and ordered numeric arguments.
177    /// OSEs, external energies and the continuous sampling scale M stay visible
178    /// to Taylor, while the integer sampling node is a separate coefficient.
179    #[allow(clippy::type_complexity)]
180    pub(crate) fn prepare_numerator(
181        &self,
182        orientation: OrientationProjection<'_>,
183        graph: &Graph,
184        numerator: &Atom,
185        scope: usize,
186    ) -> Result<(Atom, Vec<Atom>, Vec<(DirectResidueKey, Vec<Atom>)>)> {
187        let orientations = orientation.exact_orientations()?;
188        let maps = self
189            .iter_keys()
190            .map(|(key, _)| {
191                let production = orientations.get(key.selector_host).ok_or_else(|| {
192                    eyre!(
193                        "missing production energy map for orientation {}",
194                        key.selector_host.0
195                    )
196                })?;
197                Ok(key
198                    .source_edge_energy_map()
199                    .unwrap_or(&production.edge_energy_map))
200            })
201            .collect::<Result<Vec<_>>>()?;
202        let edge_count = maps[0].len();
203        if maps.iter().any(|map| map.len() != edge_count) {
204            return Err(eyre!(
205                "a direct numerator family requires equal edge-map support; absent entries are not zero samples"
206            ));
207        }
208
209        let coefficient = symbol!("gammalooprs::uv::numerator_coefficient"; Scalar);
210        let mut captured = false;
211        for atom in std::iter::once(numerator).chain(
212            self.iter_keys()
213                .flat_map(|(_, integrands)| integrands.iter().map(|(_, atom)| atom)),
214        ) {
215            let _ = atom.replace_map(|view, _, _| {
216                if let AtomView::Fun(function) = view
217                    && function.get_symbol() == coefficient
218                    && function.get_nargs() == 2
219                    && usize::try_from(function.get(0)).ok() == Some(scope)
220                {
221                    captured = true;
222                }
223            });
224        }
225        if captured {
226            return Err(eyre!(
227                "direct numerator coefficient scope {scope} is already in use"
228            ));
229        }
230
231        // The support is the union of full affine rows, not their orientation
232        // signs. Distinct occurrences and distinct maps on one host keep their
233        // own coefficients, including explicit zero samples.
234        let mut support = BTreeMap::new();
235        for (row, map) in maps.iter().enumerate() {
236            for (slot, energy) in map.iter().enumerate() {
237                let energy = energy.clone().canonical();
238                let terms = energy
239                    .internal_terms
240                    .iter()
241                    .map(|(edge, value)| (0, usize::from(*edge), value, ose_atom_from_index(*edge)))
242                    .chain(energy.external_terms.iter().map(|(edge, value)| {
243                        (
244                            1,
245                            usize::from(*edge),
246                            value,
247                            external_energy_atom_from_index(*edge),
248                        )
249                    }))
250                    .chain([
251                        (
252                            2,
253                            0,
254                            &energy.uniform_scale_coeff,
255                            Atom::var(GS.numerator_sampling_scale),
256                        ),
257                        (3, 0, &energy.constant, Atom::one()),
258                    ]);
259                for (kind, edge, value, basis) in terms {
260                    let value = Atom::num(value.clone());
261                    if !value.is_zero() {
262                        support
263                            .entry((slot, kind, edge))
264                            .or_insert_with(|| (basis, vec![Atom::Zero; maps.len()]))
265                            .1[row] = value;
266                    }
267                }
268            }
269        }
270        let mut energies = vec![Atom::Zero; edge_count];
271        let mut parameters = Vec::with_capacity(support.len());
272        let mut rows = self
273            .iter_keys()
274            .map(|(key, _)| (key.clone(), Vec::with_capacity(support.len())))
275            .collect::<Vec<_>>();
276        for ((slot, _, _), (basis, values)) in support {
277            let parameter = coefficient.call_args([Atom::num(scope), Atom::num(parameters.len())]);
278            energies[slot] += &parameter * basis;
279            parameters.push(parameter);
280            for ((_, arguments), value) in rows.iter_mut().zip(values) {
281                arguments.push(value);
282            }
283        }
284        let numerator = numerator.replace_multiple(energy_map_replacements_gs(energies, graph));
285        let mut used = vec![false; parameters.len()];
286        let _ = numerator.replace_map(|view, _, _| {
287            if let AtomView::Fun(function) = view
288                && function.get_symbol() == coefficient
289                && function.get_nargs() == 2
290                && usize::try_from(function.get(0)).ok() == Some(scope)
291                && let Ok(index) = usize::try_from(function.get(1))
292                && let Some(used) = used.get_mut(index)
293            {
294                *used = true;
295            }
296        });
297        parameters = parameters
298            .into_iter()
299            .zip(&used)
300            .filter_map(|(parameter, used)| used.then_some(parameter))
301            .collect();
302        for (_, arguments) in &mut rows {
303            *arguments = std::mem::take(arguments)
304                .into_iter()
305                .zip(&used)
306                .filter_map(|(argument, used)| used.then_some(argument))
307                .collect();
308        }
309        Ok((numerator, parameters, rows))
310    }

408    /// Transform each shared definition once, preserving its formal binding.
409    pub(crate) fn map_numerators(
410        &self,
411        mut map: impl FnMut(&FnMapEntry) -> Result<Atom>,
412    ) -> Result<Self> {
413        let mut definitions: BTreeMap<Atom, Arc<FnMapEntry>> = BTreeMap::new();
414        let mut transformed = BTreeMap::new();
415        for (_, integrands) in self.iter_keys() {
416            for entry in integrands.numerators() {
417                if let Some(existing) = definitions.get(&entry.lhs) {
418                    ensure!(
419                        existing == entry,
420                        "conflicting retained numerator {}",
421                        entry.lhs
422                    );
423                    continue;
424                }
425                let rhs = map(entry)?;
426                transformed.insert(
427                    entry.lhs.clone(),
428                    if rhs == entry.rhs {
429                        Arc::clone(entry)
430                    } else {
431                        Arc::new(FnMapEntry {
432                            rhs,
433                            ..entry.as_ref().clone()
434                        })
435                    },
436                );
437                definitions.insert(entry.lhs.clone(), Arc::clone(entry));
438            }
439        }
440        Self::from_keyed(
441            self.iter_keys()
442                .map(|(key, integrands)| {
443                    Ok((
444                        key.clone(),
445                        integrands.clone().with_numerators(
446                            integrands
447                                .numerators()
448                                .iter()
449                                .map(|entry| Arc::clone(&transformed[&entry.lhs])),
450                        )?,
451                    ))
452                })
453                .collect::<Result<Vec<_>>>()?,
454        )
455    }

466    pub(crate) fn multiply_key_mapped(
467        &self,
468        orientation: OrientationProjection<'_>,
469        graph: &Graph,
470        factor: &Atom,
471        scope: (usize, Atom),
472    ) -> Result<Self> {
473        if matches!(factor.as_view(), AtomView::Num(_)) || factor.is_zero() {
474            return Ok(self.map(|atom| atom * factor));
475        }
476        let (rhs, parameters, rows) =
477            self.prepare_numerator(orientation, graph, factor, scope.0)?;
478        let lhs = symbol!("gammalooprs::uv::numerator_family")
479            .call_args(std::iter::once(scope.1.clone()).chain(parameters.iter().cloned()));
480        let entry = Arc::new(FnMapEntry {
481            lhs,
482            rhs,
483            args: parameters
484                .iter()
485                .cloned()
486                .map(Indeterminate::try_from)
487                .collect::<std::result::Result<Vec<_>, _>>()
488                .map_err(|error| eyre!(error))?,
489            tags: vec![scope.1],
490            inlining: symbolica::evaluate::InliningPolicy::Always,
491            is_alias: false,
492        });
493        Self::from_keyed(
494            self.iter_keys()
495                .zip(rows)
496                .map(|((key, integrands), (row_key, arguments))| {
497                    // The complete branch key owns one energy substitution;
498                    // every selected cut order reuses that same mapped factor.
499                    ensure!(*key == row_key, "prepared numerator residue order changed");
500                    let call =
501                        entry
502                            .lhs
503                            .replace_multiple(parameters.iter().zip(arguments).map(
504                                |(parameter, value)| {
505                                    Replacement::new(parameter.to_pattern(), value)
506                                },
507                            ));
508                    Ok((
509                        key.clone(),
510                        integrands.map(|atom| atom * &call).with_numerators(
511                            integrands
512                                .numerators()
513                                .iter()
514                                .cloned()
515                                .chain([Arc::clone(&entry)]),
516                        )?,
517                    ))
518                })
519                .collect::<Result<Vec<_>>>()?,
520        )
521    }

545    /// Offline certificate path: all probes and final coefficients use the
546    /// deterministic finite-jet owner, including roots without shared families.
547    #[cfg(test)]
548    pub(crate) fn series_preserving_numerators_exact(
549        &self,
550        variable: Symbol,
551        center: AtomView<'_>,
552        depth: i64,
553        scope: Atom,
554    ) -> Result<Self> {
555        use crate::numerator::exact_soft_jet::ExactSoftJet;
556
557        ensure!(center.is_zero(), "exact soft jets require a zero center");
558        self.series_preserving_numerators_with(
559            variable,
560            center,
561            (depth, depth),
562            scope,
563            |atom, order| ExactSoftJet::series(atom, variable, order),
564            |atom| {
565                // Match the production scalar boundary: independent tensor
566                // factors never enter coefficient arithmetic.
567                let mut independent = Atom::one();
568                let mut dependent = atom.clone();
569                if let AtomView::Mul(product) = atom.as_view() {
570                    dependent = Atom::one();
571                    for factor in product.iter() {
572                        if factor.contains_symbol(variable) {
573                            dependent *= factor;
574                        } else {
575                            independent *= factor;
576                        }
577                    }
578                }
579                let series = ExactSoftJet::series(&dependent, variable, depth.into())?;
580                Ok((independent * series.to_atom()).collect_compact_factors())
581            },
582        )
583    }

585    fn series_preserving_numerators_with(
586        &self,
587        variable: Symbol,
588        center: AtomView<'_>,
589        (depth, root_probe_depth): (i64, i64),
590        scope: Atom,
591        mut expand: impl FnMut(&Atom, Rational) -> Result<Series<AtomField>>,
592        mut finish: impl FnMut(&Atom) -> Result<Atom>,
593    ) -> Result<Self> {
594        // Each instantiated scalar call owns its Laurent valuation: identical
595        // templates at different arguments can have different leading poles.
596        // Feed one temporary parameter-free family per call into the existing
597        // finite-series owner, retaining its native precision throughout.
598        let source_scalar_templates = self
599            .iter_keys()
600            .flat_map(|(_, roots)| roots.scalar_definitions().iter().map(|entry| &entry.tags))
601            .collect::<BTreeSet<_>>()
602            .len();
603        let mut resolve = Integrands::scalar_resolver(
604            self.iter_keys()
605                .flat_map(|(_, roots)| roots.scalar_definitions().iter().cloned()),
606        )?;
607        let family = symbol!("gammalooprs::uv::numerator_family");
608        let mut scalar_sources = BTreeMap::<Atom, Arc<FnMapEntry>>::new();
609        let mut scalar_families = BTreeSet::new();
610        let sources = self.map_numerators(|entry| resolve(&entry.rhs))?;
611        let sources = Self::from_keyed(
612            sources
613                .iter_keys()
614                .map(|(key, roots)| {
615                    let mut used = BTreeSet::new();
616                    let roots = roots.fallible_map(|atom| {
617                        let mut error = None;
618                        let result = atom.replace_map(|view, _, out| {
619                            if error.is_some() {
620                                return;
621                            }
622                            if let AtomView::Fun(call) = view
623                                && call.get_symbol() == Integrands::scalar_symbol()
624                            {
625                                let original = view.to_owned();
626                                if !scalar_sources.contains_key(&original) {
627                                    let rhs = match resolve(&original) {
628                                        Ok(rhs) => rhs,
629                                        Err(err) => {
630                                            error = Some(err);
631                                            return;
632                                        }
633                                    };
634                                    let tag =
635                                        symbol!("gammalooprs::uv::scalar_series_source").call_args(
636                                            [scope.clone(), Atom::num(scalar_sources.len())],
637                                        );
638                                    let lhs = family.call_args([tag.clone()]);
639                                    scalar_families.insert(lhs.clone());
640                                    scalar_sources.insert(
641                                        original.clone(),
642                                        Arc::new(FnMapEntry {
643                                            lhs,
644                                            rhs,
645                                            args: vec![],
646                                            tags: vec![tag],
647                                            inlining: symbolica::evaluate::InliningPolicy::Always,
648                                            is_alias: false,
649                                        }),
650                                    );
651                                }
652                                used.insert(original.clone());
653                                **out = scalar_sources[&original].lhs.clone();
654                            }
655                        });
656                        error.map_or(Ok(result), Err)
657                    })?;
658                    let definitions = roots
659                        .numerators()
660                        .iter()
661                        .cloned()
662                        .chain(used.iter().map(|call| Arc::clone(&scalar_sources[call])))
663                        .collect::<Vec<_>>();
664                    Ok((
665                        key.clone(),
666                        roots
667                            .with_numerators(definitions)?
668                            .with_scalar_definitions([])?,
669                    ))
670                })
671                .collect::<Result<Vec<_>>>()?,
672        )?;
673        let all_definitions = Integrands::from_iter([]).with_numerators(
674            sources
675                .iter_keys()
676                .flat_map(|(_, integrands)| integrands.numerators().iter().cloned()),
677        )?;
678        // One scope-wide owner prevents definition-tag collisions between cuts
679        // and shares coefficients across all residue rows in this operation.
680        let mut scalar_coefficients = ScalarCoefficients::new(variable, scope.clone());
681        let definitions = all_definitions.numerators();
682        crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
683            stage = "shared_taylor_start", scope = %scope, depth,
684            source_families = definitions.len(), residue_rows = self.0.len(),
685            source_scalar_calls = scalar_families.len(),
686            source_scalar_templates,
687            "Starting shared Taylor construction"
688        );
689        if definitions.is_empty() {
690            let result =
691                sources.fallible_map(|_, atom| Ok(scalar_coefficients.retain(&finish(atom)?)))?;
692            return result
693                .with_scalar_definitions(scalar_coefficients.definitions().iter().cloned());
694        }
695        let delta = Atom::var(variable) - center;
696        // Match the polynomial owners already recognized by EnergyPowerAnalyzer.
697        // Inert projectors keep their original attributes outside this operation.
698        let multilinear = |head: Symbol, arity: usize| {
699            head.is_linear()
700                || (head == GS.dot && arity == 2)
701                || [*CYCLIC, *SYM, *ANTISYM].contains(&head)
702        };
703        let mut linear_heads = BTreeMap::<Symbol, Symbol>::new();
704        let mut probes = Vec::with_capacity(definitions.len());
705        let mut bodies = Vec::with_capacity(definitions.len());
706        for (index, entry) in definitions.iter().enumerate() {
707            let parameters = entry
708                .args
709                .iter()
710                .cloned()
711                .map(Atom::from)
712                .collect::<Vec<_>>();
713            ensure!(
714                !entry.tags.is_empty()
715                    && entry.lhs
716                        == family.call_args(
717                            entry.tags.iter().cloned().chain(parameters.iter().cloned())
718                        )
719                    && !entry.lhs.contains_symbol(variable)
720                    && definitions[..index]
721                        .iter()
722                        .all(|other| other.tags != entry.tags)
723                    && parameters
724                        .iter()
725                        .enumerate()
726                        .all(|(i, parameter)| !parameters[..i].contains(parameter)),
727                "invalid or ambiguous retained numerator binding {}",
728                entry.lhs
729            );
730            // Specialization may remove leading coefficients, but must never
731            // introduce a pole in the Taylor variable through a map parameter.
732            let contains_parameter =
733                |atom: AtomView<'_>| parameters.iter().any(|parameter| atom.contains(parameter));
734            let mut invalid = None;
735            let mut inert_heads = std::collections::BTreeSet::new();
736            entry.rhs.visitor(&mut |atom| {
737                if invalid.is_some()
738                    || parameters
739                        .iter()
740                        .any(|parameter| parameter.as_view() == atom)
741                {
742                    return false;
743                }
744                let regular = match atom {
745                    AtomView::Fun(call) => {
746                        let known = multilinear(call.get_symbol(), call.get_nargs());
747                        if known && !call.get_symbol().is_linear() && atom.contains_symbol(variable)
748                        {
749                            inert_heads.insert(call.get_symbol());
750                        }
751                        !contains_parameter(atom) || known
752                    }
753                    AtomView::Pow(power) if contains_parameter(atom) => {
754                        Rational::try_from(power.get_exp())
755                            .is_ok_and(|power| power.is_integer() && !power.is_negative())
756                    }
757                    _ => true,
758                };
759                if !regular {
760                    invalid = Some(atom.to_owned());
761                }
762                regular
763            });
764            ensure!(
765                invalid.is_none(),
766                "retained numerator must be polynomial in map parameters: {:?}",
767                invalid
768            );
769            for head in inert_heads {
770                if linear_heads.contains_key(&head) {
771                    continue;
772                }
773                let linear = loop {
774                    let id = NUMERATOR_SCOPE.fetch_add(1, Ordering::Relaxed);
775                    let name = format!("gammalooprs::uv::numerator_series_linear_{id}");
776                    if symbolica::state::State::symbol_iter().any(|(symbol, existing)| {
777                        existing == name || symbol.get_aliases().iter().any(|alias| alias == &name)
778                    }) {
779                        continue;
780                    }
781                    let mut attributes = head.get_attributes();
782                    attributes.push(SymbolAttribute::Linear);
783                    break SymbolBuilder::new(symbolica::wrap_symbol!(name.as_str()))
784                        .with_attributes(attributes)
785                        .build()
786                        .map_err(|error| eyre!("{error}"))?;
787                };
788                linear_heads.insert(head, linear);
789            }
790            // Normalize only multilinear argument slots, never distribute the
791            // surrounding numerator product or expand projector permutations.
792            let normalized = entry.rhs.replace_map_bottom_up(|atom, _, out| {
793                if let AtomView::Fun(call) = atom
794                    && atom.contains_symbol(variable)
795                    && let Some(linear) = linear_heads.get(&call.get_symbol())
796                {
797                    **out = linear.call_args(call.iter());
798                }
799            });
800            let normalized = normalized.replace_map_bottom_up(|atom, _, out| {
801                if let AtomView::Fun(call) = atom
802                    && let Some((original, _)) = linear_heads
803                        .iter()
804                        .find(|(_, linear)| **linear == call.get_symbol())
805                {
806                    **out = original.call_args(call.iter());
807                }
808            });
809            let mut hidden_taylor = None;
810            normalized.visitor(&mut |atom| {
811                if let AtomView::Fun(call) = atom
812                    && multilinear(call.get_symbol(), call.get_nargs())
813                    && atom.contains_symbol(variable)
814                {
815                    hidden_taylor = Some(atom.to_owned());
816                    return false;
817                }
818                hidden_taylor.is_none()
819            });
820            ensure!(
821                hidden_taylor.is_none(),
822                "Taylor dependence remains inside a multilinear numerator call: {:?}",
823                hidden_taylor
824            );
825            ensure!(
826                linear_heads
827                    .values()
828                    .all(|head| !normalized.contains_symbol(*head)),
829                "temporary multilinear numerator head escaped normalization"
830            );
831            // Keep independent tensor/vertex factors outside native coefficient
832            // collection, just as series_preserving_factors does for scalar roots.
833            // The entire dependent product still owns native Laurent precision.
834            let mut independent = Atom::one();
835            let mut dependent = normalized.clone();
836            if let AtomView::Mul(product) = normalized.as_view() {
837                dependent = Atom::one();
838                for factor in product.iter() {
839                    if factor.contains_symbol(variable) {
840                        dependent *= factor;
841                    } else {
842                        independent *= factor;
843                    }
844                }
845            } else if !normalized.contains_symbol(variable) {
846                independent = normalized.clone();
847                dependent = Atom::one();
848            }
849            crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
850                stage = "shared_taylor_body_probe_start", scope = %scope, index,
851                source_bytes = entry.rhs.as_view().get_byte_size(),
852                normalized_bytes = normalized.as_view().get_byte_size(),
853                independent_bytes = independent.as_view().get_byte_size(),
854                dependent_bytes = dependent.as_view().get_byte_size(),
855                "Probing a shared numerator's Laurent order"
856            );
857            let started = std::time::Instant::now();
858            let probe = expand(&dependent, Rational::from(0))?;
859            crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
860                stage = "shared_taylor_body_probe_done", scope = %scope, index,
861                elapsed_ms = started.elapsed().as_secs_f64() * 1000.0,
862                absolute_order = %probe.absolute_order(), relative_order = %probe.relative_order(),
863                coefficient_count = probe.terms().count(),
864                coefficient_bytes = probe.terms().map(|(_, c)| c.as_view().get_byte_size()).sum::<usize>(),
865                max_coefficient_bytes = probe.terms().map(|(_, c)| c.as_view().get_byte_size()).max().unwrap_or(0),
866                "Shared numerator Laurent probe completed"
867            );
868            let bound = if probe.is_zero() {
869                probe.absolute_order()
870            } else {
871                probe.absolute_order() - probe.relative_order()
872            };
873            probes.push(
874                FnMapEntry {
875                    rhs: if normalized.is_zero() {
876                        Atom::Zero
877                    } else {
878                        &entry.lhs * delta.pow(Atom::num(bound.clone()))
879                    },
880                    ..entry.as_ref().clone()
881                }
882                .replacement(),
883            );
884            bodies.push((independent, dependent, bound, probe));
885        }
886
887        let mut extra = Rational::from(0);
888        let mut root_probes = BTreeMap::new();
889        for (branch_index, (_, integrands)) in sources.iter_keys().enumerate() {
890            for (cut, atom) in integrands.iter() {
891                let mut invalid = None;
892                atom.visitor(&mut |view| {
893                    if invalid.is_some() {
894                        return false;
895                    }
896                    let regular = match view {
897                        AtomView::Fun(call) if call.get_symbol() == family => {
898                            let entry = integrands.numerators().iter().find(|entry| {
899                                call.get_nargs() == entry.tags.len() + entry.args.len()
900                                    && entry
901                                        .tags
902                                        .iter()
903                                        .enumerate()
904                                        .all(|(i, tag)| call.get(i) == tag.as_view())
905                            });
906                            let valid = entry.is_some_and(|entry| {
907                                (entry.tags.len()..call.get_nargs())
908                                    .all(|i| Rational::try_from(call.get(i)).is_ok())
909                                    && !view.contains_symbol(variable)
910                            });
911                            if !valid {
912                                invalid = Some(view.to_owned());
913                            }
914                            return false;
915                        }
916                        AtomView::Fun(_) => !view.contains_symbol(family),
917                        AtomView::Pow(power) if view.contains_symbol(family) => {
918                            Rational::try_from(power.get_exp())
919                                .is_ok_and(|power| power.is_integer() && !power.is_negative())
920                        }
921                        _ => true,
922                    };
923                    if !regular {
924                        invalid = Some(view.to_owned());
925                    }
926                    regular
927                });
928                ensure!(
929                    invalid.is_none(),
930                    "invalid retained numerator occurrence (expected numeric arguments and polynomial use): {:?}",
931                    invalid
932                );
933                let proxy = atom.replace_multiple(&probes);
934                crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
935                    stage = "shared_taylor_root_probe_start", scope = %scope, branch_index, ?cut,
936                    input_bytes = atom.as_view().get_byte_size(),
937                    proxy_bytes = proxy.as_view().get_byte_size(), proxy_terms = proxy.nterms(),
938                    "Probing a residue root's Laurent order"
939                );
940                crate::debug_tags!(#generation, #profile, #uv, #numerator, #trace;
941                    stage = "shared_taylor_root_probe_input", scope = %scope, branch_index, ?cut,
942                    file.expr = %proxy,
943                    "Residue-root Laurent probe input"
944                );
945                let started = std::time::Instant::now();
946                // The native route keeps its existing probe through at least
947                // zero. The exact audit needs only the requested endpoint:
948                // normalized family substitutions have nonnegative powers, so
949                // an empty root probe cannot acquire a lower-order coefficient.
950                let probe = expand(&proxy, Rational::from(root_probe_depth))?;
951                crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
952                    stage = "shared_taylor_root_probe_done", scope = %scope, branch_index, ?cut,
953                    elapsed_ms = started.elapsed().as_secs_f64() * 1000.0,
954                    absolute_order = %probe.absolute_order(), relative_order = %probe.relative_order(),
955                    coefficient_count = probe.terms().count(),
956                    coefficient_bytes = probe.terms().map(|(_, c)| c.as_view().get_byte_size()).sum::<usize>(),
957                    max_coefficient_bytes = probe.terms().map(|(_, c)| c.as_view().get_byte_size()).max().unwrap_or(0),
958                    "Residue-root Laurent probe completed"
959                );
960                let bound = if probe.is_zero() {
961                    probe.absolute_order()
962                } else {
963                    probe.absolute_order() - probe.relative_order()
964                };
965                extra = extra.max(Rational::from(depth) - &bound);
966                root_probes.insert((branch_index, *cut), (bound, probe));
967            }
968        }
969
970        let mut replacements = Vec::with_capacity(definitions.len());
971        let mut coefficients = BTreeMap::new();
972        for (index, (entry, (independent, dependent, bound, probe))) in
973            definitions.iter().zip(bodies).enumerate()
974        {
975            let requested_order = &bound + &extra;
976            let reused_probe = requested_order < probe.absolute_order();
977            // Native absolute-depth expansion starts at the requested order.
978            // A body t^14 * regular(t) needs only three regular coefficients
979            // through t^16, not a relative-depth-16 expansion of every factor.
980            // A conservative lower bound suffices: divide the exact body, keep
981            // native remainder precision, and restore exponents below.
982            let shift = if !reused_probe && bound > 0 {
983                bound.clone()
984            } else {
985                Rational::from(0)
986            };
987            let expansion_body = &dependent / delta.pow(Atom::num(shift.clone()));
988            let expansion_order = &requested_order - &shift;
989            crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
990                stage = "shared_taylor_body_series_start", scope = %scope, index,
991                lower_bound = %bound, extra_order = %extra, requested_order = %requested_order,
992                normalized_order = %expansion_order, extracted_power = %shift,
993                independent_bytes = independent.as_view().get_byte_size(),
994                dependent_bytes = dependent.as_view().get_byte_size(),
995                "Constructing shared numerator coefficients"
996            );
997            crate::debug_tags!(#generation, #profile, #uv, #numerator, #trace;
998                stage = "shared_taylor_body_series_input", scope = %scope, index,
999                requested_order = %expansion_order, file.expr = %expansion_body,
1000                "Shared numerator series input"
1001            );
1002            let started = std::time::Instant::now();
1003            let mut series = if reused_probe {
1004                probe
1005            } else {
1006                expand(&expansion_body, expansion_order.clone())?
1007            };
1008            // Match Symbolica's inclusive absolute-depth convention on the
1009            // requested rational lattice, without recomputing known coefficients.
1010            series.truncate_absolute_order(
1011                &expansion_order + &(1.into(), expansion_order.denominator()).into(),
1012            );
1013            crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
1014                stage = "shared_taylor_body_series_done", scope = %scope, index, reused_probe,
1015                elapsed_ms = started.elapsed().as_secs_f64() * 1000.0,
1016                coefficient_count = series.terms().count(),
1017                coefficient_bytes = series.terms().map(|(_, c)| c.as_view().get_byte_size()).sum::<usize>(),
1018                max_coefficient_bytes = series.terms().map(|(_, c)| c.as_view().get_byte_size()).max().unwrap_or(0),
1019                "Shared numerator coefficients constructed"
1020            );
1021            let mut retained = Vec::new();
1022            let mut rhs = Atom::Zero;
1023            for (exponent, coefficient) in series.terms() {
1024                let exponent = exponent + &shift;
1025                let coefficient = &independent * coefficient;
1026                if coefficient.is_zero() {
1027                    continue;
1028                }
1029                if scalar_families.contains(&entry.lhs) {
1030                    rhs += scalar_coefficients.retain(&coefficient)
1031                        * delta.pow(Atom::num(&exponent - &bound));
1032                    continue;
1033                }
1034                let tag = symbol!("gammalooprs::uv::numerator_series_tag").call_args([
1035                    scope.clone(),
1036                    Atom::num(index),
1037                    Atom::num(exponent.clone()),
1038                ]);
1039                let lhs = family.call_args(
1040                    std::iter::once(tag.clone()).chain(entry.args.iter().cloned().map(Atom::from)),
1041                );
1042                // The root probe already includes delta^bound. Replacing its
1043                // formal family by N/delta^bound leaves a regular polynomial,
1044                // so no omitted root coefficient can contribute at lower order.
1045                rhs += &lhs * delta.pow(Atom::num(&exponent - &bound));
1046                retained.push(Arc::new(FnMapEntry {
1047                    lhs,
1048                    rhs: coefficient,
1049                    args: entry.args.clone(),
1050                    tags: vec![tag],
1051                    inlining: entry.inlining,
1052                    is_alias: entry.is_alias,
1053                }));
1054            }
1055            replacements.push(
1056                FnMapEntry {
1057                    rhs,
1058                    ..entry.as_ref().clone()
1059                }
1060                .replacement(),
1061            );
1062            coefficients.insert(entry.lhs.clone(), retained);
1063        }
1064        crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
1065            stage = "shared_numerator_taylor_coefficients", scope = %scope,
1066            source_families = definitions.len(),
1067            residue_rows = self.0.len(),
1068            coefficient_families = coefficients.values().map(Vec::len).sum::<usize>(),
1069            source_body_bytes = definitions.iter().map(|entry| entry.rhs.as_view().get_byte_size()).sum::<usize>(),
1070            coefficient_body_bytes = coefficients.values().flatten().map(|entry| entry.rhs.as_view().get_byte_size()).sum::<usize>(),
1071            "Retained shared Taylor coefficients before residue specialization"
1072        );
1073        let result = Self::from_keyed(
1074            sources.iter_keys()
1075                .enumerate()
1076                .map(|(branch_index, (key, integrands))| {
1077                    let roots = integrands
1078                        .iter()
1079                        .map(|(cut, atom)| {
1080                            let (bound, probe) = root_probes
1081                                .remove(&(branch_index, *cut))
1082                                .expect("every residue root was probed");
1083                            let series = if bound > depth {
1084                                Atom::Zero
1085                            } else {
1086                                // Scalar denominators and energies in these
1087                                // coefficients are now Taylor-independent. Only
1088                                // the retained numerator polynomials are composed.
1089                                let probed = probe.to_atom();
1090                                // Native probes established precision. Keep scalar
1091                                // coefficients shared during regular jet composition
1092                                // and across subsequent momentum/mass transformations.
1093                                let started = std::time::Instant::now();
1094                                let retained = scalar_coefficients.retain(&probed);
1095                                let expanded_families = retained.replace_multiple(&replacements);
1096                                crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
1097                                    stage = "shared_taylor_root_series_start", scope = %scope, branch_index, ?cut, depth,
1098                                    input_bytes = atom.as_view().get_byte_size(),
1099                                    reused_probe_bytes = probed.as_view().get_byte_size(),
1100                                    retained_probe_bytes = retained.as_view().get_byte_size(),
1101                                    scalar_coefficients = scalar_coefficients.len(),
1102                                    retention_ms = started.elapsed().as_secs_f64() * 1000.0,
1103                                    substituted_bytes = expanded_families.as_view().get_byte_size(),
1104                                    "Constructing residue-root Taylor coefficients"
1105                                );
1106                                let composed = finish(&expanded_families)?;
1107                                let result = scalar_coefficients.retain(&composed);
1108                                crate::debug_tags!(#generation, #profile, #uv, #numerator, #summary;
1109                                    stage = "shared_taylor_root_series_done", scope = %scope, branch_index, ?cut,
1110                                    elapsed_ms = started.elapsed().as_secs_f64() * 1000.0,
1111                                    result_bytes = result.as_view().get_byte_size(), result_terms = result.nterms(),
1112                                    "Residue-root Taylor coefficients constructed"
1113                                );
1114                                result
1115                            };
1116                            Ok((*cut, series))
1117                        })
1118                        .collect::<Result<Integrands>>()?;
1119                    Ok((
1120                        key.clone(),
1121                        roots.with_numerators(
1122                            integrands
1123                                .numerators()
1124                                .iter()
1125                                .flat_map(|entry| coefficients[&entry.lhs].iter().cloned()),
1126                        )?,
1127                    ))
1128                })
1129                .collect::<Result<Vec<_>>>()?,
1130        )?;
1131        result.with_scalar_definitions(scalar_coefficients.definitions().iter().cloned())
1132    }

1134    /// Materialize residue selectors only at the evaluator boundary.
1135    pub(crate) fn materialize(&self, materialize_key_selector: bool) -> Result<Integrands> {
1136        let mut branches = self.iter_keys();
1137        let (first_key, first) = branches
1138            .next()
1139            .ok_or_else(|| eyre!("direct local-3D residue branches cannot be empty"))?;
1140        let first_selector = first_key.selector(materialize_key_selector);
1141        first
1142            .map(|atom| atom * &first_selector)
1143            .zip_add(branches.map(|(key, integrands)| {
1144                let selector = key.selector(materialize_key_selector);
1145                integrands.map(|atom| atom * &selector)
1146            }))
1147    }
1148}
