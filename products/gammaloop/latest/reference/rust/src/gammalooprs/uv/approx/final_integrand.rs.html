67    pub(crate) fn zip_add(self, others: impl IntoIterator<Item = Self>) -> Result<Self> {
68        let representation = self.representation;
69        if representation == RepresentationMode::Cff {
70            let others = others
71                .into_iter()
72                .map(|other| {
73                    eyre::ensure!(
74                        other.representation == representation,
75                        "cannot sum final integrands from different representations"
76                    );
77                    Ok(other.integrands)
78                })
79                .collect::<Result<Vec<_>>>()?;
80            return Ok(Self {
81                integrands: self.integrands.zip_add(others)?,
82                representation,
83            });
84        }
85
86        // Independent LTD sources can have different Laurent orders at the
87        // same selected physical poles. Missing coefficients are zero in this
88        // final additive sum; source multiplication and CFF keep strict shapes.
89        // Never merge different selected axes or duplicate numerator bodies.
90        let mut axes = None;
91        let mut terms = BTreeMap::<_, Vec<Atom>>::new();
92        let mut numerators = Vec::new();
93        let mut scalars = Vec::new();
94        for source in std::iter::once(self).chain(others) {
95            eyre::ensure!(
96                source.representation == representation,
97                "cannot sum final integrands from different representations"
98            );
99            numerators.extend(source.integrands.numerators);
100            scalars.extend(source.integrands.scalars);
101            for (index, atom) in source.integrands.atoms {
102                let orders = [
103                    index.left_threshold_order,
104                    index.right_threshold_order,
105                    index.lu_cut_order,
106                ];
107                eyre::ensure!(
108                    orders.iter().flatten().all(|order| *order > 0),
109                    "invalid zero residue order in {index:?}"
110                );
111                let selected_axes = orders.map(|order| order.is_some());
112                eyre::ensure!(
113                    *axes.get_or_insert(selected_axes) == selected_axes,
114                    "cannot sum different selected residue axes at {index:?}"
115                );
116                terms.entry(index).or_default().push(atom);
117            }
118        }
119        let integrands: Integrands = terms
120            .into_iter()
121            .map(|(index, terms)| (index, terms.into_iter().sum()))
122            .collect();
123        Ok(Self {
124            integrands: integrands
125                .with_numerators(numerators)?
126                .with_scalar_definitions(scalars)?,
127            representation,
128        })
129    }

155    pub(crate) fn build_direct<S: ForestNodeLike>(
156        &self,
157        graph: &mut Graph,
158        current: &S,
159        local_terms: &Direct3dCts,
160        integrated: &IntegratedCts,
161    ) -> Result<FinalIntegrands> {
162        let reduced = graph
163            .full_filter()
164            .subtract(current.subgraph())
165            .subtract(&graph.initial_state_cut);
166        let full_graph = graph.full_filter();
167
168        let global_num = graph.global_atom();
169        debug_tags!(#generation, #profile, #uv, #graph, #summary;
170            global_num = %global_num.log_display(),
171            "Computed global numerator"
172        );
173
174        // Resolve the cograph's color algebra before residue mapping repeats its
175        // momentum numerator. The final pass still closes attached UV/projector
176        // color indices that remain open at this boundary.
177        let resnum = graph
178            .numerator(&reduced, current.subgraph())
179            .get_single_atom()
180            .expect("graph numerator should be available")
181            .simplify_color_with(ColorSimplifySettings {
182                simplify_non_color: false,
183                ..Default::default()
184            })
185            * global_num;
186        debug_tags!(#generation, #profile, #uv, #numerator, #dump;
187            stage = "final_cograph_numerator_ready",
188            graph = %graph.name,
189            numerator_bytes = resnum.as_view().get_byte_size(),
190            file.atom = %resnum.to_canonical_string(),
191            "Cograph numerator before residue mapping"
192        );
193        let localized_integrated = self
194            .localizer
195            .localize(
196                &integrated.physical_finite_counterterm_atom(),
197                graph,
198                current,
199            )?
200            .combine()?;
201        let localized_integrated = DirectResidueBranches::from_transient(&localized_integrated)?
202            .map(|atom| self.marker.prefix(&full_graph, current.subgraph(), atom));
203        let localized_local = local_terms
204            .branches()?
205            .map(|atom| self.marker.prefix(&full_graph, current.subgraph(), atom));
206        let final_branches = localized_integrated
207            .zip_add(&localized_local)?
208            .multiply_key_mapped(
209                self.localizer.orientation,
210                graph,
211                &resnum,
212                DirectResidueBranches::numerator_scope(),
213            )?;
214
215        // Normalize complete mapped branches before their orientation sum grows
216        // the scalar coefficients seen by color collection. Keep each branch's
217        // numerator and CFF factors intact throughout this linear operation.
218        // The final copy also simplifies arithmetic across energy-call boundaries
219        // (for example even powers), so expose scalar bodies before normalizing.
220        let normalize = Self::final_normalizer(graph, &reduced);
221        let finalized = final_branches
222            .resolved_scalars()?
223            .map_expressions(normalize)?;
224
225        // `DirectResidueBranches` is the sparse, factorization-preserving
226        // representation of sum_k sigma(k) I_k while the Taylor forest is
227        // built. Materialize that opaque residue-map-key selector only at the
228        // evaluator boundary, after every branch-owned numerator map has been
229        // applied. An explicit sum simply replaces every sigma by one; physical
230        // edge directions are separate sign metadata.
231        Ok(FinalIntegrands {
232            integrands: finalized
233                .materialize(!self.localizer.orientation.explicit_orientation_sum_only)?,
234            representation: self.localizer.orientation.cff_options()?.representation,
235        })
236    }

238    #[debug_instrument(
239        graph = %graph.log_display(),

242    pub(crate) fn build_projected<S: ForestNodeLike>(
243        &self,
244        graph: &mut Graph,
245        current: &S,
246        local_terms: &Projected4dCts,
247        integrated: &IntegratedCts,
248    ) -> Result<FinalIntegrands> {
249        if current.subgraph().is_empty() {
250            return Err(eyre::eyre!(
251                "the empty forest must own the complete production CFF, not projected local-4D coefficients"
252            ));
253        }
254
255        let reduced = graph
256            .full_filter()
257            .subtract(current.subgraph())
258            .subtract(&graph.initial_state_cut);
259        let full_graph = graph.full_filter();
260
261        let global_num = graph.global_atom();
262        debug_tags!(#generation, #profile, #uv, #graph, #summary;
263            global_num = %global_num.log_display(),
264            "Computed global numerator"
265        );
266        let resnum = graph
267            .numerator(&reduced, current.subgraph())
268            .get_single_atom()
269            .expect("graph numerator should be available")
270            .simplify_color_with(ColorSimplifySettings {
271                simplify_non_color: false,
272                ..Default::default()
273            })
274            * global_num;
275        let localizer = self.localizer.with_independent_source_sum();
276        debug_tags!(#generation, #profile, #uv, #numerator, #dump;
277            stage = "final_cograph_numerator_ready",
278            graph = %graph.name,
279            numerator_bytes = resnum.as_view().get_byte_size(),
280            file.atom = %resnum.to_canonical_string(),
281            "Cograph numerator before residue mapping"
282        );
283        // Only the projected local-4D route reaches this assembly boundary.
284        // Its child Taylor coefficient deliberately omits the untouched
285        // cograph; choose its outer CFF per independent sector here, converting
286        // only the selected raw routing proposal into physical surfaces.
287        let active_sectors = local_terms.sectors();
288        if active_sectors.is_empty() {
289            return Err(eyre::eyre!(
290                "factorized local term has no active UV sectors"
291            ));
292        }
293        // Restore the production-tree denominators after the outer CFF
294        // contracts its loop-energy dependence.  The exact DDx GL0 UV ray
295        // verifies that this is the full production tree, including the
296        // carrier shared with the factorized self-energy coefficient.
297        let fourddenoms = GS.wrap_tree_denoms(
298            graph.denominator(&graph.tree_edges.subtract(&graph.initial_state_cut), |_| -1),
299        );
300        let allowed_zero: Integrands = localizer
301            .cutset
302            .residue_selector
303            .generate_allowed_keys()
304            .into_iter()
305            .map(|index| (index, Atom::Zero))
306            .collect();
307        let active_edges = graph
308            .iter_edges_of(&reduced)
309            .filter_map(|(pair, edge, data)| {
310                (pair.is_paired()
311                    && !data.data.is_dummy
312                    && !graph.tree_edges.includes(&graph[&edge].1))
313                .then_some(edge)
314            })
315            .collect::<Vec<_>>();
316        let mut selector_free: Option<Vec<Integrands>> = None;
317        for sector in active_sectors {
318            if sector.coefficient.iter().all(|(_, atom)| atom.is_zero()) {
319                // A disabled integrated prefix can deliberately retain a
320                // typed zero sector for later forest replay. Preserve all
321                // allowed cut orders without asking the outer CFF to resolve
322                // a map for an identically zero coefficient.
323                selector_free.get_or_insert_with(Vec::new);
324                continue;
325            }
326            // Keep each ordinary coefficient family visible to both physical
327            // rank analysis and soft routing. Distinct scalar weights prevent
328            // different residue samples from cancelling in this capacity
329            // oracle; they never stand in for hidden energy dependence.
330            let family = symbol!("gammalooprs::uv::numerator_family");
331            let coefficient = symbol!("gammalooprs::uv::numerator_coefficient"; Scalar);
332            if sector.coefficient.numerators().is_empty() {
333                return Err(eyre::eyre!(
334                    "nonzero projected coefficient has no shared numerator family"
335                ));
336            }
337            for entry in sector.coefficient.numerators() {
338                let weight_scope = DirectResidueBranches::numerator_scope().1;
339                let mut weights = BTreeMap::<Atom, Atom>::new();
340                let carrier =
341                    sector
342                        .coefficient
343                        .iter()
344                        .try_fold(Atom::Zero, |sum, (index, root)| {
345                            if *index != crate::cff::CutCFFIndex::new_all_none() {
346                                return Err(eyre::eyre!(
347                                    "projected child coefficient has a production cut key"
348                                ));
349                            }
350                            let carrier = root.replace_map(|view, _, output| {
351                                if let AtomView::Fun(call) = view
352                                    && call.get_symbol() == family
353                                {
354                                    if call.get_nargs() > 0
355                                        && call.get(0) == entry.tags[0].as_view()
356                                    {
357                                        let next = weights.len();
358                                        **output = weights
359                                            .entry(view.to_owned())
360                                            .or_insert_with(|| {
361                                                coefficient.call_args([
362                                                    weight_scope.clone(),
363                                                    Atom::num(next),
364                                                ])
365                                            })
366                                            .clone();
367                                    } else {
368                                        **output = Atom::Zero;
369                                    }
370                                }
371                            });
372                            Ok(sum + carrier)
373                        })?;
374                if carrier.is_zero() {
375                    continue;
376                }
377                // Plan one product while retaining its two ordinary factors.
378                // The chosen occurrence-local route is replayed separately on
379                // the body and the small scalar carrier, never on expanded rows.
380                let (mut factors, localized) = localizer
381                    .projected_cff_from_soft_momentum_proposals(
382                        graph,
383                        current.subgraph(),
384                        &[&entry.rhs * &resnum, carrier],
385                        active_edges.iter().copied(),
386                        CffGenerationContext::EmbeddedCffFactor,
387                    )?;
388                let carrier = factors.pop().expect("routing retains the scalar carrier");
389                let numerator = factors
390                    .pop()
391                    .expect("routing retains the ordinary numerator");
392                // A selected cut can exclude every outer source row even
393                // when the child coefficient is nonzero. Its contribution
394                // keeps the allowed cut zeros without inventing a residue map.
395                if localized.iter_orientations().next().is_none() {
396                    selector_free.get_or_insert_with(Vec::new);
397                    continue;
398                }
399                let localized = DirectResidueBranches::from_transient(
400                    &localized.map(|atom| atom * &fourddenoms),
401                )?;
402                let scope = DirectResidueBranches::numerator_scope();
403                let (rhs, outer_parameters, rows) = localized.prepare_numerator(
404                    localizer.orientation,
405                    graph,
406                    &numerator,
407                    scope.0,
408                )?;
409                let child_parameters = entry
410                    .args
411                    .iter()
412                    .cloned()
413                    .map(Atom::from)
414                    .collect::<Vec<_>>();
415                let parameters = child_parameters
416                    .iter()
417                    .cloned()
418                    .chain(outer_parameters.iter().cloned())
419                    .collect::<Vec<_>>();
420                let prepared = Arc::new(FnMapEntry {
421                    lhs: family.call_args(
422                        std::iter::once(scope.1.clone()).chain(parameters.iter().cloned()),
423                    ),
424                    rhs,
425                    args: parameters
426                        .iter()
427                        .cloned()
428                        .map(Indeterminate::try_from)
429                        .collect::<std::result::Result<Vec<_>, _>>()
430                        .map_err(|error| eyre::eyre!(error))?,
431                    tags: vec![scope.1],
432                    inlining: entry.inlining,
433                    is_alias: entry.is_alias,
434                });
435                // A child's arguments can depend on the entire graph. Bind the
436                // cograph body, child body and carrier with this same outer key
437                // before summing anything; no selector is materialized here.
438                for ((key, integrands), (row_key, outer_arguments)) in
439                    localized.iter_keys().zip(rows)
440                {
441                    eyre::ensure!(*key == row_key, "prepared outer residue order changed");
442                    let replacements = weights
443                        .iter()
444                        .map(|(call, weight)| {
445                            let AtomView::Fun(call) = call.as_view() else {
446                                unreachable!()
447                            };
448                            let arguments = call
449                                .iter()
450                                .skip(entry.tags.len())
451                                .map(|argument| argument.to_owned())
452                                .chain(outer_arguments.iter().cloned());
453                            let call = prepared.lhs.replace_multiple(
454                                parameters.iter().zip(arguments).map(|(parameter, value)| {
455                                    Replacement::new(parameter.to_pattern(), value)
456                                }),
457                            );
458                            Replacement::new(weight.to_pattern(), call)
459                        })
460                        .collect::<Vec<_>>();
461                    let mapped_carrier = key
462                        .map_numerator(localizer.orientation, graph, &carrier)?
463                        .replace_multiple(replacements);
464                    let mapped = integrands
465                        .map(|atom| {
466                            self.marker.prefix(
467                                &full_graph,
468                                current.subgraph(),
469                                &(atom * &mapped_carrier * &sector.frozen_factor),
470                            )
471                        })
472                        .with_numerators(
473                            integrands
474                                .numerators()
475                                .iter()
476                                .cloned()
477                                .chain([Arc::clone(&prepared)]),
478                        )?;
479                    selector_free.get_or_insert_with(Vec::new).push(mapped);
480                }
481            }
482        }
483
484        // The integrated addback is shared with the direct route. Its localizer
485        // hosts the independent cograph sum in both routes, retaining every
486        // source map even when no compatible physical-prefix host survives.
487        let localized_integrated = self
488            .localizer
489            .localize(
490                &integrated.physical_finite_counterterm_atom(),
491                graph,
492                current,
493            )?
494            .combine()?;
495        // An empty integrated zero contributes no source map. The typed local
496        // zero above already retains the allowed cut shape; do not invent an
497        // energy map merely to pass it through the nonempty branch owner.
498        if localized_integrated.iter_orientations().next().is_some() {
499            let localized_integrated =
500                DirectResidueBranches::from_transient(&localized_integrated)?
501                    .multiply_key_mapped(
502                        self.localizer.orientation,
503                        graph,
504                        &resnum,
505                        DirectResidueBranches::numerator_scope(),
506                    )?
507                    .map(|atom| self.marker.prefix(&full_graph, current.subgraph(), atom));
508            // Both legitimate projected maps have now consumed every still-unmapped
509            // numerator factor. Production hosts are mapping metadata only in this
510            // lane: sum them explicitly without ever materializing a selector or
511            // traversing another numerator map.
512            for (_, integrands) in localized_integrated.iter_keys() {
513                selector_free
514                    .get_or_insert_with(Vec::new)
515                    .push(integrands.clone());
516            }
517        }
518        let selector_free = selector_free.ok_or_else(|| {
519            eyre::eyre!("final 3D UV integrand contains no production energy maps")
520        })?;
521        let representation = localizer.orientation.cff_options()?.representation;
522        let summed = FinalIntegrands {
523            integrands: allowed_zero,
524            representation,
525        }
526        .zip_add(selector_free.into_iter().map(|integrands| FinalIntegrands {
527            integrands,
528            representation,
529        }))?;
530        Self::simplify_final(graph, &reduced, summed.integrands, representation)
531    }

548    /// Normalize an already mapped final atom without splitting its factors. This
549    /// tail is deliberately blind to residue maps and cannot map a numerator.
550    fn final_normalizer(graph: &Graph, reduced: &SuBitGraph) -> impl Fn(&Atom) -> Result<Atom> {
551        let energy_replacements = graph
552            .as_ref()
553            .iter_edges_of(reduced)
554            .filter(|(pair, _, _)| pair.is_paired())
555            .map(|(_, edge_id, _)| {
556                let edge_id = usize::from(edge_id) as i64;
557                Replacement::new(function!(GS.energy, edge_id), function!(GS.ose, edge_id))
558            })
559            .collect::<Vec<_>>();
560        move |atom| {
561            // Mass-support tags distinguish equal physical or UV masses while
562            // nested/disconnected local operators are still being composed.
563            // They are mathematically one and disappear only on this final evaluator copy.
564            let mut atom = atom
565                .replace(function!(*LOCAL_3D_MASS_SCOPE, W_.a_))
566                .with(Atom::one())
567                .replace_multiple(&energy_replacements)
568                .replace(function!(GS.ose, W_.mass_, W_.prop_))
569                .with(W_.prop_)
570                // The source local_3d atom still owns its component labels for
571                // outer Taylor operations. This separate final evaluator copy
572                // identifies equal physical energies independently of owner.
573                .replace(function!(GS.on_shell_energy, W_.a_, W_.prop_))
574                .with(function!(GS.on_shell_energy, 0, W_.prop_))
575                .replace_map(|view, _, out| {
576                    if let AtomView::Pow(power) = view {
577                        let (base, exponent) = power.get_base_exp();
578                        if let AtomView::Fun(energy) = base
579                            && energy.get_symbol() == GS.on_shell_energy
580                            && energy.get_nargs() == 2
581                            && let Ok(exponent) = i64::try_from(exponent)
582                            && exponent % 2 == 0
583                        {
584                            let squared = energy.get(1);
585                            **out = squared.pow(exponent / 2);
586                        }
587                    }
588                });
589            // Color and dot simplification are linear, but separating scalar-CFF
590            // summands can destroy their common tensor factors and materialize a
591            // disconnected outer product. The typed source keeps those factors
592            // attached to their tensor network throughout normalization.
593            // Preserve the sum of CFF denominators after residue mapping, just
594            // as the Taylor stage preserves its separate propagator topologies.
595            atom = atom.replace(GS.dim).with(4).simplify_metrics();
596            debug_tags!(#generation, #profile, #uv, #numerator, #dump;
597                stage = "final_integrand_before_color",
598                graph = %graph.name,
599                numerator_bytes = atom.as_view().get_byte_size(),
600                file.atom = %atom.to_canonical_string(),
601                "Mapped factorized integrand before final color simplification"
602            );
603            atom = atom
604                .simplify_color_with(
605                    ColorSimplifySettings {
606                        simplify_non_color: false,
607                        ..Default::default()
608                    }
609                    .with_cof_dimension_invariants(),
610                )
611                .expand_dots()?;
612
613            // Exact production branches have already mapped every owned
614            // numerator fragment. The former coarse export sign replacement
615            // has no role here and must not remap these factors a second time.
616            Ok(atom
617                .replace(GS.m_uv_expansion)
618                .with(GS.m_uv_vacuum)
619                .replace(GS.dim_epsilon)
620                .with(0))
621        }
622    }

807            );

809                FinalIntegrands {
810                    integrands: output.clone(),
811                    representation: RepresentationMode::Cff,
