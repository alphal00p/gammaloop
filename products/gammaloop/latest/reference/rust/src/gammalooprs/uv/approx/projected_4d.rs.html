38/// Projected local-4D Taylor coefficients. They omit the untouched outer CFF,
39/// which is attached only during final assembly. Each child contour has already
40/// summed its own source maps, so coefficients carry no production host or cut key.
41#[derive(Clone, Debug, PartialEq, Eq, Hash)]
42pub(crate) struct Projected4dCts(Vec<Projected4dSector>);

44#[derive(Clone, Debug, PartialEq, Eq, Hash)]
45pub(crate) struct Projected4dSector {
46    pub(crate) coefficient: Integrands,
47    /// Completed-loop localizers are multiplied after outer numerator mapping.
48    pub(crate) frozen_factor: Atom,
49}

51impl Projected4dCts {
52    pub(super) fn new(sectors: Vec<Projected4dSector>) -> Self {
53        Self(sectors)
54    }

56    pub(crate) fn sectors(&self) -> &[Projected4dSector] {
57        &self.0

60    #[cfg(test)]
61    pub(crate) fn map<F: FnMut(&Atom) -> Result<Atom>>(&self, mut f: F) -> Result<Self> {
62        Ok(Self(
63            self.0
64                .iter()
65                .map(|sector| {
66                    Ok(Projected4dSector {
67                        coefficient: sector
68                            .coefficient
69                            .fallible_map(&mut f)?
70                            .map_numerators(&mut f)?,
71                        frozen_factor: sector.frozen_factor.clone(),
72                    })
73                })
74                .collect::<Result<_>>()?,
75        ))
76    }
77}

123impl Local4dProjectionContext {
124    fn canonical_projection(
125        &mut self,
126        sector: FourDSector,
127        graph: &Graph,
128    ) -> Result<Arc<CanonicalUvSector>> {
129        let started = Instant::now();
130        let key = PreparationKey::Canonical(Arc::new(sector));
131        if let Some(PreparationValue::Canonical(sector)) = self.preparations.get(&key) {
132            let sector = Arc::clone(sector);
133            drop(key);
134            self.preparations.cache_time += started.elapsed();
135            return Ok(sector);
136        }
137        let PreparationKey::Canonical(raw) = &key else {
138            unreachable!()
139        };
140        let projection_started = Instant::now();
141        let sector = Arc::new(raw.canonical_projection(graph)?);
142        let projection_time = projection_started.elapsed();
143        let bytes = raw.accounted_bytes()
144            + sector.accounted_bytes()
145            + 2 * std::mem::size_of::<PreparationKey>()
146            + std::mem::size_of::<PreparationValue>()
147            + 128;
148        self.canonical_preparation_builds += 1;
149        self.preparations
150            .insert(key, PreparationValue::Canonical(Arc::clone(&sector)), bytes);
151        self.preparations.cache_time += started.elapsed().saturating_sub(projection_time);
152        Ok(sector)
153    }
154}

176impl Localizer<'_> {
177    /// Project one typed Taylor sector in the independent sub-LMB retained by
178    /// each active component. Denominator ownership selects the component;
179    /// original graph incidence and the stored coordinate LMB remain the sole
180    /// topology and momentum authorities throughout the exact residue chain.
181    pub(super) fn project_factorized_taylor_sector(
182        self,
183        graph: &mut Graph,
184        sector: &CanonicalUvSector,
185        context: &mut Local4dProjectionContext,
186    ) -> Result<Integrands> {
187        if sector.active_components.is_empty() {
188            return Err(eyre!(
189                "typed local-4D projection requires at least one active Taylor component"
190            ));
191        }
192
193        let child_cutset = CutSet::empty(graph.n_hedges());
194        // Each Taylor component is an independently closed contour which is
195        // multiplied into the outer CFF afterwards. EmbeddedCffFactor keeps
196        // every causal term when finite-pole denominators remain. Only for a
197        // connected, nonrepeated, full-rank terminal residue basis does it avoid
198        // reopening equivalent closures: one deterministic Below representative
199        // is then the complete integral.
200        let mut options = self.orientation.cff_options()?;
201        options.cff_generation_context = CffGenerationContext::EmbeddedCffFactor;
202        let excluded_report_owners = graph
203            .iter_edges_of(&graph.initial_state_cut)
204            .chain(graph.iter_edges_of(&graph.tree_edges))
205            .map(|(_, edge, _)| edge)
206            .collect::<std::collections::BTreeSet<_>>();
207        let mut terms = Vec::new();
208        for term in &sector.terms {
209            // Raw physical ownership is diagnostic provenance, separate from
210            // canonical class algebra and the selected native capacities.
211            // Compute it once, before residue states multiply across waves.
212            let raw_degree_started = Instant::now();
213            let raw_reports = sector
214                .active_components
215                .iter()
216                .map(|(owners, _, _)| {
217                    let owners = graph
218                        .iter_edges_of(owners)
219                        .filter_map(|(pair, edge, data)| {
220                            (pair.is_paired()
221                                && !data.data.is_dummy
222                                && !excluded_report_owners.contains(&edge))
223                            .then_some(edge)
224                        });
225                    EnergyPowerAnalyzer::for_physical_emr_edges(owners)
226                        .analyze_atom(&term.source_numerator)
227                })
228                .collect::<Result<Vec<_>, _>>()?;
229            debug_tags!(#generation, #profile, #uv, #local, #four_d;
230                stage = "raw_physical_degree_report",
231                components = raw_reports.len(),
232                elapsed_ms = raw_degree_started.elapsed().as_secs_f64() * 1000.0,
233                "Analyzed raw physical ownership for degree reporting"
234            );
235            let mut component_denominators = vec![Vec::new(); sector.active_components.len()];
236            let mut residual_factor = Atom::one();
237            for (denominator, class) in term.source_witness.iter().zip(&term.source_classes) {
238                if let Some((class, _)) = class {
239                    component_denominators[sector.class(*class).component]
240                        .push(denominator.clone());
241                } else {
242                    // A denominator can be independent of its component's
243                    // quotient loop after a pinch. Its full polynomial remains
244                    // an ordinary factor, with the original physical binding.
245                    residual_factor /= &denominator.full_expr;
246                }
247            }
248            let classes = sector
249                .classes
250                .iter()
251                .filter(|class| term.powers.contains_key(&class.id))
252                .cloned()
253                .collect::<Vec<_>>();
254            terms.push((
255                component_denominators,
256                classes,
257                term.powers.clone(),
258                raw_reports,
259                term.numerator.clone(),
260                Vec::<Atom>::new(),
261                vec![(residual_factor, Vec::<Atom>::new())],
262            ));
263        }
264        // Each component still owns its exact source and occurrence assignment.
265        // Previous residues bind scalar arguments; they never specialize the
266        // ordinary numerator used to prepare the next source. A zero leading
267        // coefficient in one row can therefore overestimate that row's rank,
268        // but cannot understate any expression subsequently sampled.
269        for (component, (_, source_scope, coordinate_lmb)) in
270            sector.active_components.iter().enumerate()
271        {
272            let mut composition_time = Duration::ZERO;
273            // A coefficient position has the same formal name throughout this
274            // wave. Different affine meanings remain visible in the ordinary
275            // body and in each ordered argument row before families can merge.
276            let scope = DirectResidueBranches::numerator_scope().1;
277            let uv_edges = graph
278                .iter_edges_of(source_scope)
279                .filter_map(|(pair, edge_id, edge)| {
280                    (pair.is_paired() && !edge.data.is_dummy).then_some(edge_id)
281                })
282                .collect::<Vec<_>>();
283            // Incidence uses the enclosing source scope, while denominator
284            // owners retain this component's quotient topology and capacities.
285            for (component_denominators, classes, _, raw_reports, numerator, parameters, rows) in
286                &mut terms
287            {
288                if rows.is_empty() {
289                    continue;
290                }
291                let denominators = &component_denominators[component];
292                let active_classes = classes
293                    .iter()
294                    .filter(|class| class.component == component)
295                    .cloned()
296                    .collect::<Vec<_>>();
297                if denominators.is_empty() {
298                    return Err(eyre!(
299                        "active Taylor component has no energy denominator in one 4D term"
300                    ));
301                }
302                debug_tags!(#generation, #uv, #local, #four_d, #cff, #trace;
303                    component,
304                    source_scope = %source_scope.string_label(),
305                    coordinate_lmb = ?coordinate_lmb,
306                    denominators = ?denominators,
307                    log.numerator = &*numerator,
308                    "Preparing shared factorized local-4D child numerator"
309                );
310                context.projected_component_requests += 1;
311                let (mut cff, _) = graph.cff_from_4d_denominators_in_uv_sub_lmb(
312                    denominators,
313                    uv_edges.iter().copied(),
314                    [],
315                    coordinate_lmb,
316                    ExactUvSubLmbFrame::TaylorVacuum,
317                    &child_cutset,
318                    &options,
319                    numerator,
320                    Some(&mut *context),
321                    &active_classes,
322                )?;
323                cff.energy_degree_bound_report.physical_parent_bounds =
324                    raw_reports[component].clone().into_generation_bounds();
325                self.orientation
326                    .record_energy_degree_bound_report(&cff.energy_degree_bound_report);
327                // The sector already carries its forest subtraction sign. Only
328                // the child CFF energy-factor convention changes at this bridge.
329                let production_prefactor = Atom::num(cff.production_prefactor_factor());
330                if cff.terms.is_empty()
331                    || cff.terms.values().all(|term| term.orientations.is_empty())
332                {
333                    rows.clear();
334                    continue;
335                }
336                if cff.terms.len() != 1 {
337                    return Err(eyre!("an uncut UV-child CFF must have one residue index"));
338                }
339                let (index, cff_term) = cff.terms.into_iter().next().unwrap();
340                if index != CutCFFIndex::new_all_none() {
341                    return Err(eyre!(
342                        "an uncut UV-child CFF unexpectedly produced residue index {index}"
343                    ));
344                }
345                let (mapped, new_parameters, argument_rows) = cff_term
346                    .prepare_exact_source_numerator(&scope)
347                    .map_err(|error| {
348                        eyre!(
349                            "{error}; exact UV-child component denominators are {:?}",
350                            denominators
351                        )
352                    })?;
353                let composition_started = Instant::now();
354                let mut next_rows = BTreeMap::<Vec<Atom>, Atom>::new();
355                for (carrier, arguments) in std::mem::take(rows) {
356                    for (orientation, incoming) in cff_term.orientations.iter().zip(&argument_rows)
357                    {
358                        let arguments = arguments
359                            .iter()
360                            .cloned()
361                            .chain(incoming.iter().cloned())
362                            .collect();
363                        *next_rows.entry(arguments).or_insert(Atom::Zero) +=
364                            &carrier * &orientation.expression * &production_prefactor;
365                    }
366                }
367                *rows = next_rows
368                    .into_iter()
369                    .filter_map(|(arguments, carrier)| {
370                        (!carrier.is_zero()).then_some((carrier, arguments))
371                    })
372                    .collect();
373                *numerator = mapped;
374                parameters.extend(new_parameters);
375                // Equal complete argument rows share one summed carrier. Other
376                // Taylor families remain separate ordinary bodies; adding them
377                // must not create another numerator copy per residue sample.
378                composition_time += composition_started.elapsed();
379            }
380            let composition_started = Instant::now();
381            let families_before = terms.len();
382            let mut remaining = BTreeMap::new();
383            for (
384                mut denominators,
385                mut classes,
386                mut powers,
387                mut reports,
388                numerator,
389                parameters,
390                rows,
391            ) in terms
392            {
393                if rows.is_empty() {
394                    continue;
395                }
396                // Completed powers no longer distinguish future source
397                // requests. Equal remaining class IDs retain the same exact
398                // frame and denominator witness; never concatenate witnesses.
399                denominators[component].clear();
400                classes.retain(|class| class.component > component);
401                powers.retain(|class, _| sector.class(*class).component > component);
402                reports[component] = Default::default();
403                let key = (powers.clone(), numerator.clone(), parameters.clone());
404                let (_, _, _, combined_reports, _, _, combined_rows) =
405                    remaining.entry(key).or_insert_with(|| {
406                        (
407                            denominators,
408                            classes,
409                            powers,
410                            vec![EnergyPowerCapMap::default(); reports.len()],
411                            numerator,
412                            parameters,
413                            BTreeMap::<Vec<Atom>, Atom>::new(),
414                        )
415                    });
416                for (combined, report) in combined_reports.iter_mut().zip(reports) {
417                    combined.max_assign(report);
418                }
419                // Merge only exact body/formal equality. Distinct argument
420                // rows, including maps with the same physical host, survive.
421                for (carrier, arguments) in rows {
422                    *combined_rows.entry(arguments).or_insert(Atom::Zero) += carrier;
423                }
424            }
425            terms = remaining
426                .into_values()
427                .map(
428                    |(denominators, classes, powers, reports, numerator, parameters, rows)| {
429                        (
430                            denominators,
431                            classes,
432                            powers,
433                            reports,
434                            numerator,
435                            parameters,
436                            rows.into_iter()
437                                .filter_map(|(arguments, carrier)| {
438                                    (!carrier.is_zero()).then_some((carrier, arguments))
439                                })
440                                .collect(),
441                        )
442                    },
443                )
444                .collect();
445            composition_time += composition_started.elapsed();
446            debug_tags!(#generation, #uv, #local, #four_d, #profile;
447                stage = "component_composition",
448                component,
449                elapsed_ms = composition_time.as_secs_f64() * 1000.0,
450                families_before,
451                remaining_requests = terms.len(),
452                remaining_states = terms.iter().map(|(_, _, _, _, _, _, rows)| rows.len()).sum::<usize>(),
453                "Composed scalar argument rows for shared UV numerator families"
454            );
455            context.numerator_rows.clear();
456            debug_tags!(#generation, #uv, #local, #four_d, #cff, #summary;
457                component,
458                cached_exact_cff_expressions = context.generation_cache.len(),
459                cached_preparations = context.preparations.len(),
460                numerator_template_builds = context.numerator_template_builds,
461                numerator_row_cache_hits = context.numerator_rows.hits,
462                preparation_bytes = context.preparations.retained_bytes(),
463                "Cached exact CFF expressions by topology and capacity in one local-4D component wave"
464            );
465        }
466
467        let composition_started = Instant::now();
468        let mut active = Atom::Zero;
469        let mut definitions = Vec::new();
470        for (_, _, _, _, numerator, parameters, rows) in terms {
471            if rows.is_empty() || numerator.is_zero() {
472                continue;
473            }
474            let mut used = vec![false; parameters.len()];
475            let _ = numerator.replace_map(|view, _, _| {
476                for (index, parameter) in parameters.iter().enumerate() {
477                    used[index] |= parameter.as_view() == view;
478                }
479            });
480            let parameters = parameters
481                .into_iter()
482                .zip(&used)
483                .filter_map(|(parameter, used)| used.then_some(parameter))
484                .collect::<Vec<_>>();
485            let scope = DirectResidueBranches::numerator_scope().1;
486            let lhs = symbol!("gammalooprs::uv::numerator_family")
487                .call_args(std::iter::once(scope.clone()).chain(parameters.iter().cloned()));
488            let entry = Arc::new(FnMapEntry {
489                lhs,
490                rhs: numerator,
491                args: parameters
492                    .iter()
493                    .cloned()
494                    .map(Indeterminate::try_from)
495                    .collect::<std::result::Result<Vec<_>, _>>()
496                    .map_err(|error| eyre!(error))?,
497                tags: vec![scope],
498            });
499            for (carrier, arguments) in rows {
500                let arguments = arguments
501                    .into_iter()
502                    .zip(&used)
503                    .filter_map(|(argument, used)| used.then_some(argument));
504                let call = entry.lhs.replace_multiple(
505                    parameters
506                        .iter()
507                        .zip(arguments)
508                        .map(|(parameter, value)| Replacement::new(parameter.to_pattern(), value)),
509                );
510                active += carrier * call;
511            }
512            definitions.push(entry);
513        }
514        debug_tags!(#generation, #uv, #local, #four_d, #profile;
515            stage = "sector_composition",
516            numerator_families = definitions.len(),
517            elapsed_ms = composition_started.elapsed().as_secs_f64() * 1000.0,
518            "Composed factorized UV sector coefficient"
519        );
520        Integrands::from_iter([(CutCFFIndex::new_all_none(), active)]).with_numerators(definitions)
521    }
522}

524impl Projected4dApproximation<'_> {
525    /// Project an already Taylor-expanded local 4D counterterm in the UV
526    /// child's own energy frame. The contracted cograph is deliberately absent
527    /// here: final assembly attaches it after the independent 4D Taylor
528    /// operation has been converted to CFF.
529    pub(crate) fn project_local_4d(
530        &mut self,
531        local: &Local4dCts,
532        context: &mut Local4dProjectionContext,
533    ) -> Result<Projected4dCts> {
534        let projection_started = Instant::now();
535        let preparation_cache_before = context.preparations.cache_time;
536        let row_cache_before = context.numerator_rows.cache_time;
537        if !self.settings.local_uv_cts_from_expanded_4d_integrands {
538            return Err(eyre!(
539                "the typed local-4D child projection is reserved for local counterterms requested from expanded 4D integrands"
540            ));
541        }
542        if !self.localizer.orientation.explicit_orientation_sum_only {
543            return Err(eyre!(
544                "factorized local-4D-derived UV counterterms currently require `explicit_orientation_sum_only = true`; source-local child residues cannot be broadcast into ordinary orientation-local sectors"
545            ));
546        }
547
548        // Select the typed local sectors directly. Recursive completions are
549        // owned by the separately integrated branch; no cograph is attached here.
550        let normalization_started = Instant::now();
551        let sectors = local.projection_sectors();
552        debug_tags!(#generation, #uv, #local, #four_d, #profile;
553            stage = "sector_grouping",
554            raw_sectors = local.active_sectors().len(),
555            grouped_sectors = sectors.len(),
556            elapsed_ms = normalization_started.elapsed().as_secs_f64() * 1000.0,
557            "Grouped compatible UV projection sectors"
558        );
559        if sectors.is_empty() {
560            // Local4dCts prunes algebraically zero sectors and can also contain
561            // only completed recursive factors. Grouping can cancel nonzero
562            // sectors with identical bindings. The local contribution is then
563            // a typed zero, which needs no child or production energy map.
564            return Ok(Projected4dCts::new(vec![Projected4dSector {
565                coefficient: Integrands::from_iter([(CutCFFIndex::new_all_none(), Atom::Zero)]),
566                frozen_factor: Atom::one(),
567            }]));
568        }
569        if self.localizer.orientation.orientation_ids()?.is_empty() {
570            return Err(eyre!(
571                "orientation pattern selects no production energy maps"
572            ));
573        }
574
575        let mut active_sectors = Vec::new();
576        for sector in sectors {
577            let normalization_started = Instant::now();
578            let cache_time_before = context.preparations.cache_time;
579            let sector = context.canonical_projection(sector, self.graph)?;
580            let cache_time = context.preparations.cache_time - cache_time_before;
581            debug_tags!(#generation, #uv, #local, #four_d, #profile;
582                stage = "canonical_normalization",
583                buckets = sector.terms.len(),
584                denominator_classes = sector.classes.len(),
585                components = sector.active_components.len(),
586                elapsed_ms = normalization_started.elapsed().as_secs_f64() * 1000.0,
587                cache_work_ms = cache_time.as_secs_f64() * 1000.0,
588                "Certified canonical factorized UV algebra"
589            );
590            let frozen_localizer = sector.frozen_lmbs.iter().fold(Atom::one(), |product, lmb| {
591                product * GS.localizing_integrand(lmb)
592            });
593            let active = self
594                .localizer
595                .project_factorized_taylor_sector(self.graph, &sector, context)?;
596
597            active_sectors.push(Projected4dSector {
598                coefficient: active,
599                frozen_factor: frozen_localizer,
600            });
601        }
602
603        let (cff_hits, cff_misses, cff_evictions, retained_cff_bytes) =
604            context.generation_cache.statistics();
605        debug_tags!(#generation, #uv, #local, #four_d, #profile;
606            stage = "local_projection_total",
607            graph = %self.graph.name,
608            elapsed_ms = projection_started.elapsed().as_secs_f64() * 1000.0,
609            component_requests = context.projected_component_requests,
610            template_builds = context.numerator_template_builds,
611            canonical_preparation_builds = context.canonical_preparation_builds,
612            source_preparation_builds = context.source_preparation_builds,
613            preparation_hits = context.preparations.hits,
614            preparation_misses = context.preparations.misses,
615            preparation_evictions = context.preparations.evictions,
616            retained_preparation_bytes = context.preparations.retained_bytes(),
617            preparation_cache_ms = context.preparations.cache_time.as_secs_f64() * 1000.0,
618            preparation_cache_work_ms = (context.preparations.cache_time - preparation_cache_before).as_secs_f64() * 1000.0,
619            cff_hits,
620            cff_misses,
621            cff_evictions,
