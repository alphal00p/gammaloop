142    fn classify_spinney<E: UVE, V, H>(
143        &self,
144        spinney: InternalSubGraph,
145        settings: &UVgenerationSettings,
146        lmb: &LoopMomentumBasis,
147    ) -> Result<Option<Spinney>>
148    where
149        Self: AsRef<HedgeGraph<E, V, H>>,
150    {
151        let dod = self.compute_dod(&spinney.filter);
152        if dod < 0 {
153            return Ok(None);
154        }
155
156        let renormalization_scheme = self.approximation_scheme(&spinney.filter, settings, dod);
157
158        if renormalization_scheme != ApproximationType::Unsubtracted {
159            Spinney::with_scheme(spinney, self, lmb, renormalization_scheme, dod)
160        } else {
161            Ok(None)
162        }
163    }

165    fn classified_spinneys<E: UVE, V, H, S: SubGraphLike<Base = SuBitGraph>>(
166        &self,
167        subgraph: &S,
168        settings: &UVgenerationSettings,
169        lmb: &LoopMomentumBasis,
170    ) -> Result<Vec<Spinney>>
171    where
172        Self: AsRef<HedgeGraph<E, V, H>>,
173    {
174        if !settings.subtract_uv {
175            return Ok(vec![Spinney::empty(self)]);
176        }
177
178        let graph = self.as_ref();
179        let (connected, disconnected): (Vec<_>, Vec<_>) = self
180            .spinneys(subgraph)
181            .into_iter()
182            .partition(|spinney| graph.connected_components(spinney).len() <= 1);
183
184        let mut classified: Vec<_> = connected
185            .into_iter()
186            .filter_map(|spinney| self.classify_spinney(spinney, settings, lmb).transpose())
187            .collect::<Result<_>>()?;
188        let retained_factors: AHashSet<_> = classified
189            .iter()
190            .map(|spinney| spinney.filter().clone())
191            .collect();
192
193        for spinney in disconnected {
194            let components = graph.connected_components(&spinney);
195            if !components.into_iter().all(|component| {
196                let factor = InternalSubGraph::cleaned_filter_optimist(component, graph);
197                retained_factors.contains(&factor.filter)
198            }) {
199                continue;
200            }
201
202            let dod = self.compute_dod(&spinney.filter);
203            if let Some(spinney) =
204                Spinney::with_scheme(spinney, self, lmb, ApproximationType::MUV, dod)?
205            {
206                classified.push(spinney);
207            }
208        }
209        Ok(classified)
210    }

231    fn all_limits<E, V, H, S: SubGraphLike>(
232        &self,
233        subgraph: &S,
234        expr: &Atom,
235        expansion: Symbol,
236        lmb: &LoopMomentumBasis,
237    ) -> Vec<(SubSet<LoopIndex>, Series<AtomField>)>
238    where
239        Self: AsRef<HedgeGraph<E, V, H>>,
240    {
241        let mom_reps = self.normal_emr_replacement(subgraph, lmb, &[W_.x___], |_s| true);
242
243        let ose_reps = self.get_ose_replacements();
244        // for x in &mom_reps {
245        //     println!("LMB replacement: {x}");
246        // }
247
248        // for r in &ose_reps {
249        //     println!("ose{r}");
250        // }
251
252        // This separate analytic copy uses native square-root Laurent series:
253        // after UV rescaling an energy invariant can have a pole, beyond the
254        // compact function's ordinary Taylor derivative domain. Lower energies
255        // after OSE substitution without distributing any numerator factors.
256        let expr = expr
257            .replace_multiple(&ose_reps)
258            .replace_multiple(&mom_reps)
259            .replace(function!(GS.broadcasting_sqrt, W_.a_))
260            .with(Atom::var(W_.a_).sqrt())
261            .replace(function!(GS.on_shell_energy, W_.a_, W_.prop_))
262            .with(Atom::var(W_.prop_).sqrt());
263        // .replace_multiple(&q3_reps);
264        let mut loops = PowersetIterator::<LoopIndex>::new(lmb.loop_edges.len() as u8);
265
266        let mut limits = Vec::new();
267
268        loops.next();
269
270        for ls in loops {
271            let mut expr = expr.clone();
272            for l in ls.included_iter() {
273                let e = usize::from(lmb.loop_edges[l]) as i64;
274                expr = expr
275                    .replace(function!(GS.emr_mom, e, W_.x___))
276                    .with(function!(GS.emr_mom, e, W_.x___) / expansion);
277
278                expr /= Atom::var(expansion).pow(3);
279            }
280
281            let series = expr.series(expansion, Atom::Zero, 0).unwrap();
282
283            // expr = series.to_atom().expand();
284
285            limits.push((ls, series));
286        }
287        limits
288    }

316    fn compute_dod<S: SubGraphLike<Base = SuBitGraph> + SubSetOps>(&self, subgraph: &S) -> i32;
317
318    /// Degree of divergence of the simultaneous loop-energy integrations.
319    ///
320    /// The existing UV DOD already contains the numerator and denominator
321    /// scaling. Replacing the four-dimensional measure by one energy measure
322    /// per loop therefore amounts to subtracting three powers per loop.
323    fn compute_energy_dod<E, V, H, S>(&self, subgraph: &S) -> i32
324    where
325        Self: AsRef<HedgeGraph<E, V, H>>,
326        S: SubGraphLike<Base = SuBitGraph> + SubSetOps,
327    {
328        self.compute_dod(subgraph) - 3 * self.n_loops(subgraph) as i32
329    }

333    fn spinneys<E, V, H, S: SubGraphLike<Base = SuBitGraph>>(
334        &self,
335        subgraph: &S,
336    ) -> AHashSet<InternalSubGraph>
337    where
338        Self: AsRef<HedgeGraph<E, V, H>>,
339    {
340        let ref_graph: &HedgeGraph<E, V, H> = self.as_ref();
341        debug!(subgraph=%ref_graph.dot(subgraph),"Spinneys of subgraph");
342        let _b: SuBitGraph = ref_graph.empty_subgraph();
343
344        if subgraph.is_empty() {
345            let mut spinneys = AHashSet::new();
346            spinneys.insert(ref_graph.empty_subgraph());
347            return spinneys;
348        }
349
350        let cycles = ref_graph.cycle_basis_of(subgraph).0;
351
352        let _init_node = ref_graph.iter_nodes_of(subgraph).next().unwrap().0;
353        let all_subcycles: Vec<_> = Cycle::all_sum_powerset_filter_map(&cycles, &Some)
354            .map(|a| a.into_iter().map(|c| c.internal_graph(ref_graph)).collect())
355            .unwrap();
356
357        let dod_cache = RefCell::new(AHashMap::new());
358
359        let mut spinneys: AHashSet<_> = InternalSubGraph::all_ops_iterative_filter_map(
360            &all_subcycles,
361            &|a, b| a.union(b),
362            &|union| {
363                // println!("{}", self.as_ref().dot(&union));
364                let cached_dod = dod_cache.borrow().get(&union).copied();
365                let keep = match cached_dod {
366                    Some(keep) => keep,
367                    None => {
368                        let keep = self.local_dod(&union) >= 0;
369                        dod_cache.borrow_mut().insert(union.clone(), keep);
370                        keep
371                    }
372                };
373                if keep { Some(union) } else { None }
374            },
375        );
376
377        spinneys.insert(ref_graph.empty_subgraph());
378
379        spinneys
380    }

390    /// Reject source integrals whose loop-energy contours do not vanish at
391    /// infinity. Check the existing unpruned cycle-union inventory so this
392    /// source-level contract does not depend on UV-spinney pruning.
393    pub(crate) fn ensure_energy_convergent_cycles<S>(&self, domain: &S) -> Result<()>
394    where
395        S: SubGraphLike<Base = SuBitGraph>,
396    {
397        let numerator = self.production_numerator_atom_for_full_3d_expression();
398        if numerator.is_zero() {
399            return Ok(());
400        }
401        let mut failures = self
402            .all_cycle_unions(domain)
403            .into_iter()
404            .filter(|cycle| !cycle.is_empty())
405            .filter_map(|cycle| {
406                let loop_count = self.n_loops(&cycle.filter);
407                // `compute_dod` deliberately covers only the edge and vertex
408                // rules used by UV classification. Source convergence instead
409                // sees the complete factorized production numerator, including
410                // graph-global factors and momentum dependence stored outside
411                // this cycle. The production helper excludes initial-cut
412                // factors without changing UV DOD semantics or expanding the
413                // numerator.
414                let lmb = self.lmb_of(&cycle.filter);
415                let integrand = &numerator / self.denominator(&cycle.filter, |_| 1);
416                let four_d_dod = self
417                    .uv_rescaled(cycle.filter.included(), loop_count, &lmb, &lmb, &integrand)
418                    .trailing_exponent();
419                // The rescaled source already contains the four-dimensional
420                // loop measure; retain one energy measure per loop instead.
421                let energy_dod = four_d_dod - 3 * loop_count as i32;
422                (energy_dod >= 0)
423                    .then(|| (cycle.string_label(), loop_count, four_d_dod, energy_dod))
424            })
425            .collect::<Vec<_>>();
426        failures.sort();
427
428        if failures.is_empty() {
429            return Ok(());
430        }
431
432        let details = failures
433            .into_iter()
434            .map(|(cycle, loop_count, four_d_dod, energy_dod)| {
435                format!("cycle {cycle}: L={loop_count}, DOD_4D={four_d_dod}, DOD_E={energy_dod}")
436            })
437            .collect::<Vec<_>>()
438            .join("; ");
439        Err(eyre!(
440            "graph `{}` contains a loop-energy nonconvergent integral ({details}); expected DOD_E < 0 for every cycle union",
441            self.name,
442        ))
443    }
444}

480    fn denominator<S: SubGraphLike, T: Fn(&Edge) -> isize>(
481        &self,
482        subgraph: &S,
483        edge_powers: T,
484    ) -> Atom {
485        let mut den = Atom::num(1);
486
487        for (pair, eid, d) in self.underlying.iter_edges_of(subgraph) {
488            if matches!(pair, HedgePair::Paired { .. }) {
489                let m2 = d.data.mass_atom().pow(2);
490                let edge_power = edge_powers(d.data);
491                let is_power_negative = edge_power < 0;
492                let prop_den = GS.den(
493                    usize::from(eid),
494                    function!(GS.emr_mom, usize::from(eid)),
495                    &m2,
496                    spenso_lor_atom(usize::from(eid) as i32, usize::from(eid), GS.dim)
497                        .pow(2)
498                        .to_dots()
499                        - &m2,
500                );
501                for _i in 0..edge_power.abs() {
502                    if is_power_negative {
503                        den /= prop_den.clone();
504                    } else {
505                        den *= prop_den.clone();
506                    }
507                }
508            }
509        }
510
511        den
512    }

523    fn compute_dod<S: SubGraphLike<Base = SuBitGraph> + SubSetOps>(&self, subgraph: &S) -> i32 {
524        let lmb = self.lmb_of(subgraph);
525        let empty = self.underlying.empty_subgraph();
526        let integrand = self
527            .numerator(subgraph, &empty)
528            .to_d_dim(GS.dim)
529            .get_single_atom()
530            .unwrap()
531            / self.denominator(subgraph, |_| 1);
532        let nloops: usize = self.n_loops(subgraph);
533        self.uv_rescaled(subgraph.included(), nloops, &lmb, &lmb, &integrand)
534            .trailing_exponent()
535    }
536
537    fn local_dod<S: SubGraphLike>(&self, subgraph: &S) -> i32 {
538        let mut dod: i32 = 4 * self.n_loops(subgraph) as i32;
539        for (p, _, e) in self.underlying.iter_edges_of(subgraph) {
540            if p.is_paired() {
541                dod += e.data.dod.deref();
542            }
543        }
544
545        for (_, _, n) in self.underlying.iter_nodes_of(subgraph) {
546            dod += n.dod.deref();
547        }
548
549        dod
550    }
