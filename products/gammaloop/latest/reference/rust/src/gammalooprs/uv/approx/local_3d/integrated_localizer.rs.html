15impl Localizer<'_> {
16    pub(crate) fn localize<S: ForestNodeLike>(
17        &self,
18        expr: &Atom,
19        graph: &mut Graph,
20        integrated_node: &S,
21    ) -> Result<FrozenActiveCt> {
22        if expr.is_zero() {
23            let indices = self.cutset.residue_selector.generate_allowed_keys();
24            return Ok(FrozenActiveCt {
25                active: OrientationIntegrands::from_ids_and_indices(
26                    self.orientation.orientation_ids()?,
27                    &indices,
28                ),
29                frozen_integrands: indices
30                    .into_iter()
31                    .map(|index| (index, Atom::one()))
32                    .collect(),
33            });
34        }
35
36        let to_contract = integrated_node.subgraph();
37        let integrated_loop_count = graph.n_loops(to_contract);
38        // Keep finite addbacks for nested multi-loop entries. Integrated expressions carry
39        // their forest-composition signs where needed, so dropping the localized finite
40        // representative here removes the Tint(T(...)) terms.
41        let finite_ct = expr;
42        debug_tags!(#generation, #profile, #uv, #integrated, #local, #summary;
43            stage = "localize_integrated_ct_forest_overlap",
44            integrated_loop_count,
45            "Applied integrated UV forest-overlap addback rule"
46        );
47        let fourddenoms = GS.wrap_tree_denoms(
48            graph.denominator(&graph.tree_edges.subtract(&graph.initial_state_cut), |_| -1),
49        );
50
51        // CFF capacity belongs to the complete factorized numerator of this
52        // reduced term. The integrated finite CT replaces the numerator of
53        // the contracted spinney; the remaining graph and global factors are
54        // still grown later by their existing owners.
55        let reduced = graph
56            .full_filter()
57            .subtract(to_contract)
58            .subtract(&graph.initial_state_cut);
59        let outside_numerator = graph
60            .numerator(&reduced, to_contract)
61            .get_single_atom()
62            .expect("Graph numerator should be available")
63            * graph.global_atom();
64        let analysis_numerator = expr * &outside_numerator;
65
66        let localizing_integrand = GS.localizing_integrand(integrated_node.lmb());
67        // Integration replaces the contracted spinney by an independent
68        // coefficient. Its cograph residue maps need not extend to a complete
69        // production map, but each still belongs to the finite addback once.
70        let mut active = self.with_independent_source_sum().projected_cff(
71            graph,
72            to_contract,
73            [&analysis_numerator],
74            CffGenerationContext::Standalone,
75        )?;
76        for branch in &mut active.0 {
77            // A source map owns the finite coefficient for every cut order;
78            // attaching it must not repeat the same numerator substitution.
79            let mapped_finite_ct = self.orientation.map_numerator(
80                graph,
81                branch.selector_id,
82                branch.source_edge_energy_map.as_deref(),
83                finite_ct,
84            )?;
85            branch.integrands = branch.integrands.map(|localized| {
86                let localized = localized * &fourddenoms;
87                let localized_cff_byte_size = localized.as_view().get_byte_size();
88                let active_ct = localized * &mapped_finite_ct;
89                debug_tags!(#generation, #profile, #uv, #integrated, #local, #term, #summary;
90                    stage = "localize_integrated_ct_term",
91                    integrated_node = %integrated_node.log_display(),
92                    contracted = %to_contract.string_label(),
93                    reduced = %reduced.string_label(),
94                    residue_map_key = branch.selector_id.0,
95                    localized_cff_byte_size,
96                    active_ct_byte_size = active_ct.as_view().get_byte_size(),
97                    localized_ct_byte_size = (&active_ct * &localizing_integrand).as_view().get_byte_size(),
98                    "Integrated UV CT localization size checkpoint"
99                );
100                active_ct
101            });
102        }
103        let frozen_integrands = self
104            .cutset
105            .residue_selector
106            .generate_allowed_keys()
107            .into_iter()
108            .map(|index| (index, localizing_integrand.clone()))
109            .collect();
110
111        Ok(FrozenActiveCt {
112            active,
113            frozen_integrands,
114        })
115    }
116}
