47fn evaluate_uv_damper<T: FloatLike>(
48    radius: &F<T>,
49    radius_star: &F<T>,
50    e_cm: &F<T>,
51    settings: &UVLocalisationSettings,
52) -> F<T> {
53    if settings.force_uv_dampers_to_one {
54        return radius.one();
55    }
56
57    let normalizing_scale = match settings.dynamic_width {
58        true => radius_star,
59        false => e_cm,
60    };
61
62    let delta_r = radius - radius_star;
63    let sliver_width = F::from_f64(settings.sliver_width) * normalizing_scale;
64
65    if delta_r.abs() > sliver_width || (settings.smooth_sliver && delta_r.abs() == sliver_width) {
66        return radius.zero();
67    }
68
69    let delta_r_sq = &delta_r * &delta_r;
70    let width = F::from_f64(settings.gaussian_width) * normalizing_scale;
71    let width_sq = &width * &width;
72
73    let mut exponent = -&delta_r_sq / width_sq;
74    if settings.smooth_sliver {
75        // This even bump is one at the pole and flat to all orders at the edge.
76        let gap = sliver_width.square() - &delta_r_sq;
77        exponent -= delta_r_sq / gap;
78    }
79    exponent.exp()
80}

82fn evaluate_uv_damper_dual<T: FloatLike>(
83    radius: &HyperDual<F<T>>,
84    radius_star: &HyperDual<F<T>>,
85    e_cm: &F<T>,
86    settings: &UVLocalisationSettings,
87) -> HyperDual<F<T>> {
88    if settings.force_uv_dampers_to_one {
89        return new_constant(radius, &radius.values[0].one());
90    }
91
92    let normalizing_scale = if settings.dynamic_width {
93        radius_star.clone()
94    } else {
95        new_constant(radius, e_cm)
96    };
97
98    let delta_r = radius.clone() - radius_star.clone();
99    let sliver_width =
100        new_constant(radius, &F::from_f64(settings.sliver_width)) * normalizing_scale.clone();
101
102    if delta_r.values[0].abs() > sliver_width.values[0]
103        || (settings.smooth_sliver && delta_r.values[0].abs() == sliver_width.values[0])
104    {
105        return new_constant(radius, &radius.values[0].zero());
106    }
107
108    let delta_r_sq = delta_r.clone() * delta_r;
109    let width = new_constant(radius, &F::from_f64(settings.gaussian_width)) * normalizing_scale;
110    let width_sq = width.clone() * width;
111
112    let mut exponent = -delta_r_sq.clone() / width_sq;
113    if settings.smooth_sliver {
114        let gap = sliver_width.clone() * sliver_width - delta_r_sq.clone();
115        exponent -= delta_r_sq / gap;
116    }
117    exponent.exp()
118}

120fn evaluate_integrated_ct_normalisation<T: FloatLike>(
121    radius: &F<T>,
122    radius_star: &F<T>,
123    _e_cm: &F<T>,
124    settings: &IntegratedCounterTermSettings,
125) -> F<T> {
126    match &settings.range {
127        IntegratedCounterTermRange::Infinite {
128            h_function_settings,
129        } => {
130            // The helper's inverse radial measure leaves dr, so this density must
131            // integrate to one in r: dr / r_star = d(r / r_star).
132            let h = utils::h(&(radius / radius_star), None, None, h_function_settings);
133            h * (radius_star).inv()
134        }
135        IntegratedCounterTermRange::Compact {} => {
136            todo!();
137        }
138    }
139}

177    fn evaluate_alpha<T: FloatLike>(
178        &mut self,
179        input: RstarTDependenceInput<'_, T>,
180        evaluation_metadata: &mut EvaluationMetaData,
181    ) -> HyperDual<F<T>> {
182        let RstarTDependenceInput {
183            t_star,
184            alpha,
185            overlap_center,
186            subspace,
187            unrescaled_momentum_sample,
188            representative_sample,
189            masses,
190            threshold_esurface,
191            lmb,
192            all_lmbs,
193        } = input;
194        debug!("t-star: {}", t_star);
195        debug!("projection alpha: {}", alpha);
196
197        let dual = HyperDual::new(self.dual_shape_for_esurface_evaluation.clone());
198        let dual_alpha = dual.variable(0, alpha.clone());
199        let dual_t = dual.variable(1, t_star.clone());
200
201        let rescale_tstar = unrescaled_momentum_sample
202            .loop_moms()
203            .rescale_with_hyper_dual(&dual_t, None);
204
205        let dualized_externals_three_momenta = unrescaled_momentum_sample
206            .external_moms()
207            .iter()
208            .map(|fm: &FourMomentum<F<T>>| fm.spatial.map_ref(&|x| new_constant(&dual, x)))
209            .collect();
210
211        let mut lmb_transform = rescale_tstar.lmb_transform(
212            lmb,
213            subspace.get_lmb(all_lmbs),
214            &dualized_externals_three_momenta,
215        );
216
217        // The derivative ray comes from raw generation data, while its base is
218        // the actual transformed representative used by the projection solve.
219        // Retain every derivative, aligning the base before differentiating eta.
220        for (momentum, base) in lmb_transform
221            .0
222            .iter_mut()
223            .zip(representative_sample.loop_moms().iter())
224        {
225            momentum.px.values[0] = base.px.clone();
226            momentum.py.values[0] = base.py.clone();
227            momentum.pz.values[0] = base.pz.clone();
228        }
229
230        let dualized_overlap_center = overlap_center
231            .iter()
232            .map(|momentum| momentum.map_ref(&|x| new_constant(&dual, x)))
233            .collect::<crate::momentum::sample::LoopMomenta<_>>();
234
235        let shifted_t_dependent_momenta = lmb_transform
236            .iter()
237            .zip(dualized_overlap_center.iter())
238            .map(|(momentum, center)| momentum.clone() - center.clone())
239            .collect::<crate::momentum::sample::LoopMomenta<_>>();
240
241        let dualized_external_fourmomenta = dualized_externals_three_momenta
242            .into_iter()
243            .zip(unrescaled_momentum_sample.external_moms())
244            .map(|(spatial, fm)| FourMomentum {
245                spatial,
246                temporal: Energy {
247                    value: new_constant(&dual, &fm.temporal.value),
248                },
249            })
250            .collect();
251
252        // The same IFT backend differentiates the dimensionless root. Keep the
253        // raw active displacement so null directions never enter a normalization.
254        let rescale_rstar = shifted_t_dependent_momenta
255            .rescale(&dual_alpha, subspace.as_subspace_simple())
256            .iter()
257            .zip(dualized_overlap_center.iter())
258            .map(|(momentum, center)| momentum.clone() + center.clone())
259            .collect::<crate::momentum::sample::LoopMomenta<_>>();
260
261        let dual_esurface = threshold_esurface.compute_from_dual_momenta(
262            subspace.get_lmb(all_lmbs),
263            masses,
264            &rescale_rstar,
265            &dualized_external_fourmomenta,
266        );
267
268        debug!("Dual e-surface: {}", dual_esurface);
269
270        // HyperDual stores Taylor coefficients, whereas the generated IFT
271        // parameters are ordinary mixed partial derivatives.
272        let params = dual_esurface.values[1..]
273            .iter()
274            .zip(dual_esurface.get_shape().into_iter().skip(1))
275            .map(|(coefficient, orders)| {
276                let derivative = orders
277                    .iter()
278                    .flat_map(|order| 2..=*order)
279                    .fold(coefficient.clone(), |value, factor| {
280                        value * coefficient.from_usize(factor)
281                    });
282                Complex::new_re(derivative)
283            })
284            .collect_vec();
285
286        debug!("Parameters for implicit function theorem: {:#?}", params);
287
288        let result = evaluate_evaluator(
289            self.implicit_function_theorem
290                .as_mut()
291                .expect("alpha(t) evaluator requested without t-derivative support"),
292            &params,
293            evaluation_metadata,
294        )
295        .into_iter()
296        .map(DualOrNot::unwrap_real)
297        .collect_vec();
298
299        debug!("Result from implicit function theorem: {:#?}", result);
300
301        let mut dual_values = vec![alpha.clone()];
302        let mut n_factorial = alpha.one();
303        for (i, result) in result.into_iter().enumerate() {
304            n_factorial *= alpha.from_usize(i + 1);
305            dual_values.push(result.re / &n_factorial);
306        }
307
308        HyperDual::from_values(simple_n_deriv_shape(dual_values.len() - 1), dual_values)
309    }
310}

325// Use the chain rule to express derivatives of the projection root in terms of
326// the mixed derivatives of eta(root(t), t). The neutral symbolic program names
327// that scalar r_star; runtime supplies alpha and forms the physical radius later.
328pub(crate) fn generate_rstar_t_dependence_evaluator(
329    num_t_derivatives: usize,
330) -> Result<RstarTDependenceEvaluator> {
331    if num_t_derivatives == 0 {
332        return Ok(RstarTDependenceEvaluator {
333            dual_shape_for_esurface_evaluation: Vec::new(),
334            implicit_function_theorem: None,
335        });
336    }
337
338    let t = symbol!("t");
339
340    let rstar = parse!("r_star(t)");
341    let e_surface = function!(GS.eta, rstar.clone(), Atom::var(t));
342
343    let mut rstar_derivatives = vec![];
344    for i in 0..num_t_derivatives {
345        if i == 0 {
346            rstar_derivatives.push(rstar.derivative(t));
347        } else {
348            rstar_derivatives.push(rstar_derivatives.last().unwrap().derivative(t));
349        }
350    }
351
352    let mut equations = vec![];
353    for i in 0..num_t_derivatives {
354        if i == 0 {
355            equations.push(e_surface.derivative(t));
356        } else {
357            equations.push(equations.last().unwrap().derivative(t));
358        }
359    }
360
361    let mut solutions = equations
362        .iter()
363        .zip(&rstar_derivatives)
364        .map(|(eq, variable)| {
365            let solutions = Atom::solve(std::slice::from_ref(eq))
366                .wrt_with_exponent::<u8, _>(std::slice::from_ref(variable))
367                .unwrap();
368            let [solution] = solutions.iter().as_slice() else {
369                panic!(
370                    "expected one implicit-function solution, got {} branches",
371                    solutions.len()
372                );
373            };
374            assert!(
375                solution.free_variables().is_empty(),
376                "implicit-function solution is underdetermined"
377            );
378            let variable = PolyVariable::try_from(variable.clone()).unwrap();
379            solution.get(&variable).cloned().unwrap()
380        })
381        .collect_vec();
382
383    for i in 1..solutions.len() {
384        for j in 0..i {
385            solutions[i] = solutions[i]
386                .replace(rstar_derivatives[j].clone())
387                .with(solutions[j].clone());
388        }
389    }
390
391    // dual shape is for e-surface derivatives, implict function theorem should NOT be dualized with this
392    let mut dual_shape = vec![vec![0, 0]];
393    let mut params = vec![];
394    for i in 1..=num_t_derivatives {
395        let mut eta_derivatives_at_this_order = vec![];
396
397        let mut current_r_derivative_counter = 0;
398        let mut current_t_derivative_counter = i;
399
400        loop {
401            dual_shape.push(vec![
402                current_r_derivative_counter,
403                current_t_derivative_counter,
404            ]);
405            let eta_derivative = function!(
406                Symbol::DERIVATIVE,
407                current_r_derivative_counter,
408                current_t_derivative_counter,
409                GS.eta,
410                rstar.clone(),
411                Atom::var(t)
412            );
413
414            eta_derivatives_at_this_order.push(eta_derivative);
415
416            if current_t_derivative_counter == 0 {
417                break;
418            }
419            current_r_derivative_counter += 1;
420            current_t_derivative_counter -= 1;
421        }
422
423        params.extend(eta_derivatives_at_this_order);
424    }
425
426    let fn_map = FunctionMap::new();
427    let fn_map_entries = vec![];
428    let implict_function_theorem = GenericEvaluator::new_from_raw_params(
429        solutions,
430        &params,
431        &fn_map,
432        fn_map_entries,
433        OptimizationSettings::default(),
434        None,
435        &EvaluatorSettings::default(),
436    )?;
437
438    Ok(RstarTDependenceEvaluator {
439        dual_shape_for_esurface_evaluation: dual_shape,
440        implicit_function_theorem: Some(implict_function_theorem),
441    })
442}

455    fn integrated_ct_profiles_preserve_the_radial_residue() {
456        for (function, power) in [
457            (HFunction::Exponential, None),
458            (HFunction::PolyExponential, None),
459            (HFunction::PolyExponential, Some(4)),
460            (HFunction::PolyExponential, Some(16)),
461            // This profile's p=0 normalization is a focused regression
462            // check. Higher powers need log-space quadrature because their
463            // integrable x -> 0 peak is too narrow for a uniform grid.
464            (HFunction::PolyLeftRightExponential, None),
465        ] {
466            for sigma in [0.5, 2.0] {
467                let settings = IntegratedCounterTermSettings {
468                    range: IntegratedCounterTermRange::Infinite {
469                        h_function_settings: HFunctionSettings {
470                            function: function.clone(),
471                            sigma,
472                            power,
473                            ..Default::default()
474                        },
475                    },
476                };
477                for radius_star in [0.3, 3.0] {
478                    // The generated helper cancels r^(3L-1) from the measure.
479                    // Gaussian profiles are negligible after 8 sigma. The
480                    // left/right exponential has only exponential (rather than
481                    // Gaussian) tails, so integrate it over a wider finite
482                    // window before comparing its analytic normalization.
483                    let tail_multiple = if function == HFunction::PolyLeftRightExponential {
484                        64.0
485                    } else {
486                        8.0
487                    };
488                    let step = tail_multiple * sigma * radius_star / 2048.0;
489                    let integral: f64 = (0..2048)
490                        .map(|i| {
491                            evaluate_integrated_ct_normalisation(
492                                &F((i as f64 + 0.5) * step),
493                                &F(radius_star),
494                                &F(1.0),
495                                &settings,
496                            )
497                            .0
498                        })
499                        .sum::<f64>()
500                        * step;
501                    let tolerance = if function == HFunction::PolyLeftRightExponential {
502                        1.0e-8
503                    } else {
504                        2.0e-12
505                    };
506                    assert!(
507                        (integral - 1.0).abs() < tolerance,
508                        "{function:?}, power={power:?}, sigma={sigma}, r_star={radius_star}: {integral}"
509                    );
510                }
511            }
512        }
513    }

643                for (actual, expected) in actual.values.iter().zip(expected) {
644                    assert!((actual.0 - expected).abs() < 2.0e-13 * expected.abs().max(1.0));
645                }
646            }
647            for u in [-0.6_f64, -0.5, -0.4995, 0.4995, 0.5, 0.6] {
648                let radius = new_from_values(&shape, &[F(2.0 + u * scale), F(0.3), F(0.0)]);
649                let actual = evaluate_uv_damper_dual(&radius, &radius_star, &F(4.0), &settings);
650                for coefficient in actual.values {
651                    assert!(coefficient.0.abs() < 1.0e-190);
652                    if u.abs() >= 0.5 {
653                        assert_eq!(coefficient, F(0.0));
654                    }
655                }
656            }
657            let forced = UVLocalisationSettings {
658                force_uv_dampers_to_one: true,
659                ..settings
660            };
661            let radius = new_from_values(&shape, &[F(100.0), F(0.3), F(0.0)]);
662            assert_eq!(
663                evaluate_uv_damper_dual(&radius, &radius_star, &F(4.0), &forced).values,
664                vec![F(1.0), F(0.0), F(0.0)]
665            );
666        }

669    #[test]
670    fn uv_smooth_sliver_helper_preserves_signed_radial_principal_value() {
671        crate::initialisation::test_initialise().unwrap();
672        let settings = UVLocalisationSettings {
673            smooth_sliver: true,
674            sliver_width: 1.5,
675            ..Default::default()
676        };
677        for loop_count in [1, 2] {
678            let (pieces, _) =
679                CrossSectionGraph::single_th_prefactor_helper_atoms(1, loop_count, false, false);
680            let mut helper = GenericEvaluator::new_from_raw_params(
681                [pieces.local],
682                &CrossSectionGraph::single_th_prefactor_helper_params(1, false),
683                &FunctionMap::new(),
684                vec![],
685                OptimizationSettings::default(),
686                None,
687                &EvaluatorSettings::default(),
688            )
689            .unwrap()
690            .into_eager_only();
691            for radius_star in [0.5, 2.0] {
692                // Equal midpoint bins pair around the pole. W > r_star in the
693                // first case requires the helper's negative-radius mirror term.
694                let step = 1.0 / 2048.0;
695                let steps = ((radius_star + settings.sliver_width) / step) as usize;
696                let mut integral = 0.0;
697                for i in 0..steps {
698                    let radius = F((i as f64 + 0.5) * step);
699                    let plus = evaluate_uv_damper(&radius, &F(radius_star), &F(1.0), &settings);
700                    let minus = evaluate_uv_damper(&(-radius), &F(radius_star), &F(1.0), &settings);
701                    let params = [F(1.0), F(1.0), radius, F(radius_star), plus, minus, F(0.0)]
702                        .map(Complex::new_re);
703                    let value = f64::get_evaluator_single(&mut helper)(&params);
704                    integral += value.re.0 * radius.0.powi(3 * loop_count as i32 - 1) * step;
705                }
706                assert!(
707                    integral.abs() < 1.0e-10,
708                    "{loop_count} loops, r_star={radius_star}: PV={integral}"
709                );
710            }
711        }
712    }
