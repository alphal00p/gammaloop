32pub(crate) fn normalize_three_d_expression_cut_support_with_raised_edge_groups<O>(
33    expression: &mut ThreeDExpression<O>,
34    raised_edge_groups: &[Vec<EdgeIndex>],
35) where
36    O: From<usize> + Into<usize>,
37{
38    for orientation in expression.orientations.iter_mut() {
39        for variant in &mut orientation.variants {
40            variant.denominator_edges = normalize_cut_edge_support_with_raised_edge_groups(
41                &variant.denominator_edges,
42                raised_edge_groups,
43            );
44            variant.denominator_edge_support_signs = normalize_cut_edge_support_signs(
45                std::mem::take(&mut variant.denominator_edge_support_signs),
46                raised_edge_groups,
47            );
48        }
49    }
50}

52fn normalize_cut_edge_support_signs(
53    support_signs: BTreeMap<Vec<EdgeIndex>, i64>,
54    raised_edge_groups: &[Vec<EdgeIndex>],
55) -> BTreeMap<Vec<EdgeIndex>, i64> {
56    support_signs
57        .into_iter()
58        .fold(BTreeMap::new(), |mut normalized, (support, sign)| {
59            let support =
60                normalize_cut_edge_support_with_raised_edge_groups(&support, raised_edge_groups);
61            *normalized.entry(support).or_insert(1) *= sign;
62            normalized
63        })
64}

66pub(crate) fn normalize_cut_edge_support_with_raised_edge_groups(
67    edges: &[EdgeIndex],
68    raised_edge_groups: &[Vec<EdgeIndex>],
69) -> Vec<EdgeIndex> {
70    edges
71        .iter()
72        .map(|edge| {
73            raised_edge_groups
74                .iter()
75                .find(|group| group.contains(edge))
76                .and_then(|group| group.first())
77                .copied()
78                .unwrap_or(*edge)
79        })
80        .sorted()
81        .dedup()
82        .collect()
83}

85pub trait GammaLoopCFFVariant {
86    fn to_atom_gs(&self) -> Atom;
87}
88
89impl GammaLoopCFFVariant for CFFVariant {
90    fn to_atom_gs(&self) -> Atom {
91        let half_edge_factor = self
92            .half_edges
93            .iter()
94            .map(|edge_id| Atom::num(1) / (Atom::num(2) * ose_atom_from_index(*edge_id)))
95            .reduce(|acc, factor| acc * factor)
96            .unwrap_or_else(|| Atom::num(1));
97        let scale_factor = if self.uniform_scale_power == 0 {
98            Atom::num(1)
99        } else {
100            Atom::num(1)
101                / Atom::var(GS.numerator_sampling_scale).pow(self.uniform_scale_power as i64)
102        };
103
104        let numerator_surface_factor = self
105            .numerator_surfaces
106            .iter()
107            .map(|surface_id| Atom::from(*surface_id))
108            .reduce(|acc, factor| acc * factor)
109            .unwrap_or_else(|| Atom::num(1));
110
111        Atom::num(self.prefactor.clone())
112            * half_edge_factor
113            * scale_factor
114            * numerator_surface_factor
115            * self.denominator.to_atom_inv()
116    }
117}

119pub trait GammaLoopOrientationExpression {
120    fn to_atom_gs(&self) -> Atom;
121    fn energy_replacements_gs(&self, graph: &Graph) -> Vec<Replacement>;
122}

124impl GammaLoopOrientationExpression for OrientationExpression {
125    fn to_atom_gs(&self) -> Atom {
126        self.variants
127            .iter()
128            .map(GammaLoopCFFVariant::to_atom_gs)
129            .reduce(|acc, atom| acc + atom)
130            .unwrap_or_else(Atom::new)
131    }
132
133    fn energy_replacements_gs(&self, graph: &Graph) -> Vec<Replacement> {
134        energy_map_replacements_gs(
135            self.edge_energy_map
136                .iter()
137                .map(|energy| energy.to_atom_gs(&[])),
138            graph,
139        )
140    }

143pub(crate) fn energy_map_replacements_gs(
144    edge_energy_map: impl IntoIterator<Item = Atom>,
145    graph: &Graph,
146) -> Vec<Replacement> {
147    let edge_energy_map = edge_energy_map.into_iter().collect::<Vec<_>>();
148    let mut replacements = Vec::new();
149    let mink_index = LibraryRep::from(Minkowski {}).to_symbolic([Atom::var(W_.a__)]);
150    let external_edges = graph
151        .underlying
152        .iter_edges()
153        .filter_map(|(pair, edge_id, _)| (!pair.is_paired()).then_some(edge_id))
154        .collect::<BTreeSet<_>>();
155
156    for (edge_id, energy) in edge_energy_map.iter().enumerate() {
157        let edge_id = EdgeIndex(edge_id);
158        // Remapping a generated internal-edge map into GammaLoop's physical
159        // namespace pads unpaired external-edge slots with zero. Those slots
160        // are not residue samples: their EMR momenta must remain the original
161        // external momenta carried by the factorized numerator.
162        if external_edges.contains(&edge_id) {
163            continue;
164        }
165        replacements.push(Replacement::new(
166            GS.emr_mom(edge_id, AIND_SYMBOLS.cind.call(Atom::Zero))
167                .to_pattern(),
168            energy.clone().to_pattern(),
169        ));
170        replacements.push(Replacement::new(
171            GS.emr_mom(edge_id, &mink_index).to_pattern(),
172            (GS.emr_vec_index(edge_id, &mink_index) + energy * GS.energy_delta(&mink_index))
173                .to_pattern(),
174        ));
175    }
176
177    for (loop_id, loop_edge_id) in graph.loop_momentum_basis.loop_edges.iter_enumerated() {
178        let loop_id = usize::from(loop_id);
179        let loop_id_atom = Atom::num(loop_id as i64);
180        let energy = edge_energy_map
181            .get(usize::from(*loop_edge_id))
182            .cloned()
183            .unwrap_or_else(Atom::new);
184        replacements.push(Replacement::new(
185            function!(
186                GS.loop_mom,
187                loop_id_atom.clone(),
188                AIND_SYMBOLS.cind.call(Atom::Zero)
189            )
190            .to_pattern(),
191            energy.clone().to_pattern(),
192        ));
193        for spatial_index in 1..=3 {
194            replacements.push(Replacement::new(
195                function!(
196                    GS.loop_mom,
197                    loop_id_atom.clone(),
198                    AIND_SYMBOLS.cind.call(spatial_index)
199                )
200                .to_pattern(),
201                GS.emr_mom(*loop_edge_id, AIND_SYMBOLS.cind.call(spatial_index))
202                    .to_pattern(),
203            ));
204        }
205        replacements.push(Replacement::new(
206            FunctionBuilder::new(GS.loop_mom)
207                .add_arg(loop_id as i64)
208                .add_arg(mink_index.as_view())
209                .finish()
210                .to_pattern(),
211            (GS.emr_vec_index(*loop_edge_id, &mink_index) + energy * GS.energy_delta(&mink_index))
212                .to_pattern(),
213        ));
214    }
215
216    replacements
217}

229    #[test]
230    fn affine_energy_map_keeps_two_numerator_factors_under_one_map() -> color_eyre::Result<()> {
231        test_initialise()?;
232        let graph: Graph = dot!(
233            digraph affine_map {
234                edge [num=1 mass=0]
235                node [num=1]
236                A -> B [id=0]
237                A -> B [id=1]
238            }
239        )?;
240        let energy_map = LinearEnergyExpr {
241            internal_terms: vec![(EdgeIndex(0), 2.into()), (EdgeIndex(1), (-3).into())],
242            external_terms: vec![(EdgeIndex(2), 5.into())],
243            uniform_scale_coeff: 7.into(),
244            constant: 11.into(),
245        }
246        .canonical();
247        let orientation = OrientationExpression {
248            data: OrientationData::new(EdgeVec::from_iter([
249                Orientation::Undirected,
250                Orientation::Undirected,
251            ])),
252            loop_energy_map: Vec::new(),
253            edge_energy_map: vec![energy_map.clone(), LinearEnergyExpr::zero()],
254            variants: Vec::new(),
255        };
256        let factor_a = Atom::var(symbolica::symbol!("factor_a"));
257        let factor_b = Atom::var(symbolica::symbol!("factor_b"));
258        let energy_component = GS.emr_mom(EdgeIndex(0), AIND_SYMBOLS.cind.call(Atom::Zero));
259        let numerator =
260            (energy_component.clone() + factor_a.clone()) * (energy_component + factor_b.clone());
261
262        let mapped = numerator.replace_multiple(orientation.energy_replacements_gs(&graph));
263        let mapped_energy = Atom::num(2) * ose_atom_from_index(EdgeIndex(0))
264            - Atom::num(3) * ose_atom_from_index(EdgeIndex(1))
265            + Atom::num(5) * external_energy_atom_from_index(EdgeIndex(2))
266            + Atom::num(7) * Atom::var(GS.numerator_sampling_scale)
267            + Atom::num(11);
268        let expected =
269            (mapped_energy.clone() + factor_a.clone()) * (mapped_energy.clone() + factor_b.clone());
270        assert_eq!(mapped, expected);
271        let mixed = (mapped_energy.clone() + factor_a) * (mapped_energy + Atom::num(1) + factor_b);
272        assert_ne!(mapped, mixed);
273        Ok(())
274    }

277    fn padded_external_energy_slots_do_not_erase_factorized_external_momenta()
278    -> color_eyre::Result<()> {
279        test_initialise()?;
280        let graph: Graph = dot!(digraph padded_external_energy_slots {
281            edge [num=1 mass=0]
282            node [num=1]
283            ext [style=invis]
284            v0;
285            v1;
286            v2;
287            ext -> v0 [id=0]
288            ext -> v1 [id=1]
289            v2 -> ext [id=2]
290            v0 -> v2 [id=3 lmb_id=0]
291            v1 -> v0 [id=4]
292            v2 -> v1 [id=5]
293        })?;
294        let mapped_internal_energy = LinearEnergyExpr {
295            internal_terms: vec![(EdgeIndex(3), 2.into())],
296            external_terms: vec![(EdgeIndex(0), (-1).into())],
297            uniform_scale_coeff: 0.into(),
298            constant: 0.into(),
299        }
300        .canonical();
301        let mut edge_energy_map = vec![LinearEnergyExpr::zero(); graph.underlying.n_edges()];
302        edge_energy_map[3] = mapped_internal_energy.clone();
303        let external_energy = GS.emr_mom(EdgeIndex(0), AIND_SYMBOLS.cind.call(Atom::Zero));
304        let internal_energy = GS.emr_mom(EdgeIndex(3), AIND_SYMBOLS.cind.call(Atom::Zero));
305        let spectator = Atom::var(symbolica::symbol!("spectator"));
306        let numerator = external_energy.clone() * (internal_energy + spectator.clone());
307
308        let mapped = numerator.replace_multiple(energy_map_replacements_gs(
309            edge_energy_map.iter().map(|energy| energy.to_atom_gs(&[])),
310            &graph,
311        ));
312        assert_eq!(
313            mapped,
314            external_energy * (mapped_internal_energy.to_atom_gs(&[]) + spectator)
315        );
316        Ok(())
317    }
