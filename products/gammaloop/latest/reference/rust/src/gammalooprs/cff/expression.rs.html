40pub(crate) fn normalize_three_d_expression_cut_support_with_raised_edge_groups<O>(
41    expression: &mut ThreeDExpression<O>,
42    raised_edge_groups: &[Vec<EdgeIndex>],
43) where
44    O: From<usize> + Into<usize>,
45{
46    for orientation in expression.orientations.iter_mut() {
47        for variant in &mut orientation.variants {
48            variant.denominator_edges = normalize_cut_edge_support_with_raised_edge_groups(
49                &variant.denominator_edges,
50                raised_edge_groups,
51            );
52            variant.denominator_edge_support_signs = normalize_cut_edge_support_signs(
53                std::mem::take(&mut variant.denominator_edge_support_signs),
54                raised_edge_groups,
55            );
56        }
57    }
58}

60fn normalize_cut_edge_support_signs(
61    support_signs: BTreeMap<Vec<EdgeIndex>, i64>,
62    raised_edge_groups: &[Vec<EdgeIndex>],
63) -> BTreeMap<Vec<EdgeIndex>, i64> {
64    support_signs
65        .into_iter()
66        .fold(BTreeMap::new(), |mut normalized, (support, sign)| {
67            let support =
68                normalize_cut_edge_support_with_raised_edge_groups(&support, raised_edge_groups);
69            *normalized.entry(support).or_insert(1) *= sign;
70            normalized
71        })
72}

74pub(crate) fn normalize_cut_edge_support_with_raised_edge_groups(
75    edges: &[EdgeIndex],
76    raised_edge_groups: &[Vec<EdgeIndex>],
77) -> Vec<EdgeIndex> {
78    edges
79        .iter()
80        .map(|edge| {
81            raised_edge_groups
82                .iter()
83                .find(|group| group.contains(edge))
84                .and_then(|group| group.first())
85                .copied()
86                .unwrap_or(*edge)
87        })
88        .sorted()
89        .dedup()
90        .collect()
91}

110fn apply_indexed_residue_selection<F>(
111    residues: Vec<(CutCFFIndex, ThreeDExpression<OrientationID>)>,
112    axis: CutCffResidueAxis,
113    mut select: F,
114) -> Vec<(CutCFFIndex, ThreeDExpression<OrientationID>)>
115where
116    F: FnMut(ThreeDExpression<OrientationID>) -> Vec<ThreeDExpression<OrientationID>>,
117{
118    residues
119        .into_iter()
120        .flat_map(|(index, expression)| {
121            select(expression)
122                .into_iter()
123                .enumerate()
124                .map(move |(i, residue)| {
125                    let mut new_index = index;
126                    axis.set_order(&mut new_index, i + 1);
127                    // One causal coordinate has been consumed regardless of
128                    // which raised-order branch `i` labels. Its energy-factor,
129                    // contact, and residue signs remain owned by the selected
130                    // CFF variant; the index records only the residue order.
131                    (new_index, residue)
132                })
133        })
134        .collect()
135}

137pub(super) fn select_indexed_cff_residues(
138    mut cff: ThreeDExpression<OrientationID>,
139    cutset: &CutSet,
140    representation: three_dimensional_reps::RepresentationMode,
141    raised_edge_groups: &[Vec<EdgeIndex>],
142    mut bounds_provider: impl FnMut() -> Result<Vec<(usize, usize)>>,
143) -> Result<Vec<(CutCFFIndex, ThreeDExpression<OrientationID>)>> {
144    if let Some(lu) = cutset.residue_selector.lu.as_ref()
145        && (lu.cut_edge_alternatives.is_empty()
146            || lu.cut_edge_alternatives.iter().any(Vec::is_empty))
147    {
148        return Err(eyre::eyre!(
149            "LU residue selection requires at least one non-empty physical cut alternative"
150        ));
151    }
152    for group in cutset
153        .residue_selector
154        .right_th_cut
155        .iter()
156        .chain(&cutset.residue_selector.left_th_cut)
157        .chain(cutset.residue_selector.lu.iter().map(|lu| &lu.raised_group))
158    {
159        if group.esurface_ids.is_empty() || group.max_occurence == 0 {
160            return Err(eyre::eyre!(
161                "CFF residue selection requires a nonempty surface group and a positive residue order"
162            ));
163        }
164        if let Some(id) = group
165            .esurface_ids
166            .iter()
167            .find(|id| cff.surfaces.esurface_cache.get(**id).is_none())
168        {
169            return Err(eyre::eyre!(
170                "CFF residue selection references missing E-surface {id:?}"
171            ));
172        }
173    }
174    if representation == three_dimensional_reps::RepresentationMode::Ltd {
175        use surface::{HybridSurfaceID, HybridSurfaceRef};
176        // Raised physical edges share one OSE even when the common surface
177        // catalogue names a different occurrence first. Remap energy values
178        // only: numerator-map slots still name their original propagators.
179        let aliases = raised_edge_groups
180            .iter()
181            .filter_map(|group| group.first().map(|first| (group, first)))
182            .flat_map(|(group, first)| group.iter().map(move |edge| (edge.0, first.0)))
183            .collect::<BTreeMap<_, _>>();
184        let canonical_edge = |edge: &mut EdgeIndex| {
185            if let Some(representative) = aliases.get(&edge.0) {
186                *edge = EdgeIndex(*representative);
187            }
188        };
189        for surface in &mut cff.surfaces.esurface_cache {
190            *surface =
191                Graph::normalize_esurface_with_raised_edge_groups(surface, raised_edge_groups);
192        }
193        for surface in &mut cff.surfaces.hsurface_cache {
194            surface
195                .positive_energies
196                .iter_mut()
197                .for_each(&canonical_edge);
198            surface
199                .negative_energies
200                .iter_mut()
201                .for_each(&canonical_edge);
202        }
203        for surface in &mut cff.surfaces.linear_surface_cache {
204            surface.expression = surface.expression.clone().remap_internal_edges(&aliases);
205        }
206        for orientation in &mut cff.orientations {
207            orientation.remap_on_shell_energy_values(&aliases);
208        }
209        let physical_surfaces = cff
210            .surfaces
211            .iter_all_surfaces()
212            .filter_map(|(id, surface)| {
213                let (positive, negative, shift) = match surface {
214                    HybridSurfaceRef::Esurface(surface) => {
215                        (&surface.energies, &[][..], &surface.external_shift)
216                    }
217                    HybridSurfaceRef::Hsurface(surface) => (
218                        &surface.positive_energies,
219                        surface.negative_energies.as_slice(),
220                        &surface.external_shift,
221                    ),
222                    _ => return None,
223                };
224                let expression = positive
225                    .iter()
226                    .map(|edge| LinearEnergyExpr::ose(*edge, 1))
227                    .chain(negative.iter().map(|edge| LinearEnergyExpr::ose(*edge, -1)))
228                    .chain(
229                        shift
230                            .iter()
231                            .map(|(edge, sign)| LinearEnergyExpr::external(*edge, *sign)),
232                    )
233                    .fold(LinearEnergyExpr::zero(), |sum, term| sum + term);
234                Some((id, expression))
235            })
236            .collect::<BTreeMap<_, _>>();
237        let lu_cut_alternatives = cutset.residue_selector.lu.as_ref().map(|lu| {
238            lu.cut_edge_alternatives
239                .iter()
240                .map(|edges| {
241                    normalize_cut_edge_support_with_raised_edge_groups(edges, raised_edge_groups)
242                })
243                .collect::<Vec<_>>()
244        });
245        let mut energy_degree_bounds: Option<Vec<(usize, usize)>> = None;
246        let selections = [
247            (
248                CutCffResidueAxis::RightThreshold,
249                cutset.residue_selector.right_th_cut.as_ref(),
250            ),
251            (
252                CutCffResidueAxis::LeftThreshold,
253                cutset.residue_selector.left_th_cut.as_ref(),
254            ),
255            (
256                CutCffResidueAxis::LuCut,
257                cutset
258                    .residue_selector
259                    .lu
260                    .as_ref()
261                    .map(|lu| &lu.raised_group),
262            ),
263        ];
264        let mut canonical_selections = selections.map(|(axis, group)| {
265            (
266                axis,
267                group.map(|group| {
268                    let selected =
269                        &physical_surfaces[&HybridSurfaceID::Esurface(group.esurface_ids[0])];
270                    esurface::RaisedEsurfaceGroup {
271                        esurface_ids: physical_surfaces
272                            .iter()
273                            .filter_map(|(id, expression)| match id {
274                                HybridSurfaceID::Esurface(id) if expression == selected => {
275                                    Some(*id)
276                                }
277                                _ => None,
278                            })
279                            .collect(),
280                        max_occurence: group.max_occurence,
281                    }
282                }),
283            )
284        });
285        for (_, group) in &mut canonical_selections {
286            let Some(group) = group else { continue };
287            cff.normalize_single_raising(group);
288            group.max_occurence = group.max_occurence.max(
289                cff.orientations
290                    .iter()
291                    .map(|orientation| {
292                        orientation.max_effective_denominator_value_count_on_branch(
293                            &HybridSurfaceID::Esurface(group.esurface_ids[0]),
294                        )
295                    })
296                    .max()
297                    .unwrap_or_default(),
298            );
299        }
300        // Exact denominator sources can carry higher physical poles than the
301        // parent graph's CutSet. Preserve the source's effective order while
302        // leaving the authoritative CFF selection contract unchanged.
303        let selections = canonical_selections
304            .each_ref()
305            .map(|(axis, group)| (*axis, group.as_ref()));
306        let mut residues = vec![(CutCFFIndex::new_all_none(), cff)];
307        for (axis, group) in selections {
308            let Some(group) = group else { continue };
309            let selected =
310                physical_surfaces[&HybridSurfaceID::Esurface(group.esurface_ids[0])].clone();
311            let other_selected = selections
312                .iter()
313                .filter(|(other_axis, _)| *other_axis != axis)
314                .filter_map(|(_, group)| {
315                    group.map(|group| {
316                        physical_surfaces[&HybridSurfaceID::Esurface(group.esurface_ids[0])].clone()
317                    })
318                })
319                .collect::<Vec<_>>();
320            let mut next = Vec::new();
321            for (index, mut expression) in residues {
322                if let CutCffResidueAxis::LuCut = axis {
323                    expression = expression
324                        .restrict_to_cut_alternatives(lu_cut_alternatives.as_ref().unwrap());
325                }
326                let mut surface_expressions = physical_surfaces.clone();
327                surface_expressions.extend(
328                    expression
329                        .surfaces
330                        .linear_surface_cache
331                        .iter_enumerated()
332                        .map(|(id, surface)| {
333                            (HybridSurfaceID::Linear(id), surface.expression.clone())
334                        }),
335                );
336                crate::debug_tags!(#cff, #trace;
337                    stage = "ltd_pole_selection_input",
338                    axis = ?axis,
339                    cut_index = ?index,
340                    selected_surface = ?selected,
341                    scalar_surfaces = ?surface_expressions,
342                    residue_maps = ?expression.orientations,
343                    "Selecting an LTD physical pole"
344                );
345                let coefficients = match expression.select_ltd_residue(
346                    &surface_expressions,
347                    &selected,
348                    &other_selected,
349                    energy_degree_bounds.as_deref()) {
350                    Ok(coefficients) => coefficients,
351                    Err(three_dimensional_reps::generation::GenerationError::LtdResidueRequiresEnergyBounds) => {
352                        let bounds = bounds_provider()?;
353                        let coefficients = expression.select_ltd_residue(&surface_expressions, &selected, &other_selected, Some(&bounds))?;
354                        energy_degree_bounds = Some(bounds);
355                        coefficients
356                    }
357                    Err(error) => return Err(error.into()),
358                };
359                for (order, coefficient) in coefficients.into_iter().enumerate() {
360                    let mut index = index;
361                    axis.set_order(&mut index, order + 1);
362                    crate::debug_tags!(#cff, #trace;
363                        stage = "ltd_pole_selection_coefficient",
364                        axis = ?axis,
365                        cut_index = ?index,
366                        scalar_surfaces = ?coefficient.surfaces.linear_surface_cache,
367                        residue_maps = ?coefficient.orientations,
368                        "Selected LTD physical-pole coefficient"
369                    );
370                    next.push((index, coefficient));
371                }
372            }
373            residues = next;
374        }
375        return Ok(residues);
376    }
377    // Carry selected causal coordinates through the operation pipeline rather
378    // than reconstructing them later from raised-order labels in
379    // `CutCFFIndex`. Energy-factor ownership stays with the generated CFF
380    // variants and is deliberately independent of this indexing operation.
381    let mut residues = vec![(CutCFFIndex::new_all_none(), cff)];
382
383    if let Some(right_threshold) = cutset.residue_selector.right_th_cut.as_ref() {
384        residues = apply_indexed_residue_selection(
385            residues,
386            CutCffResidueAxis::RightThreshold,
387            |expression| expression.select_esurface_residue(right_threshold, representation),
388        );
389    }
390
391    if let Some(left_threshold) = cutset.residue_selector.left_th_cut.as_ref() {
392        residues = apply_indexed_residue_selection(
393            residues,
394            CutCffResidueAxis::LeftThreshold,
395            |expression| expression.select_esurface_residue(left_threshold, representation),
396        );
397    }
398
399    if let Some(lu) = cutset.residue_selector.lu.as_ref() {
400        residues =
401            apply_indexed_residue_selection(residues, CutCffResidueAxis::LuCut, |expression| {
402                // Physical support selection is independent of the 3D
403                // representation. Each representation owns the conversion
404                // from its signed denominator to the selected pole.
405                expression
406                    .restrict_to_cut_alternatives(&lu.cut_edge_alternatives)
407                    .select_esurface_residue(&lu.raised_group, representation)
408            });
409    }
410
411    Ok(residues)
412}

414pub trait GammaLoopCFFVariant {
415    fn to_atom_gs(&self) -> Atom;
416}
417
418impl GammaLoopCFFVariant for CFFVariant {
419    fn to_atom_gs(&self) -> Atom {
420        let half_edge_factor = self
421            .half_edges
422            .iter()
423            .map(|edge_id| Atom::num(1) / (Atom::num(2) * ose_atom_from_index(*edge_id)))
424            .reduce(|acc, factor| acc * factor)
425            .unwrap_or_else(|| Atom::num(1));
426        let scale_factor = if self.uniform_scale_power == 0 {
427            Atom::num(1)
428        } else {
429            Atom::num(1)
430                / Atom::var(GS.numerator_sampling_scale).pow(self.uniform_scale_power as i64)
431        };
432
433        let numerator_surface_factor = self
434            .numerator_surfaces
435            .iter()
436            .map(|surface_id| Atom::from(*surface_id))
437            .reduce(|acc, factor| acc * factor)
438            .unwrap_or_else(|| Atom::num(1));
439
440        Atom::num(self.prefactor.clone())
441            * self.thermal_weight.to_atom()
442            * half_edge_factor
443            * scale_factor
444            * numerator_surface_factor
445            * self.denominator.to_atom_inv()
446    }
447}

449pub trait GammaLoopOrientationExpression {
450    fn to_atom_gs(&self) -> Atom;
451    fn energy_replacements_gs(&self, graph: &Graph) -> Vec<Replacement>;
452}
453
454impl GammaLoopOrientationExpression for OrientationExpression {
455    fn to_atom_gs(&self) -> Atom {
456        let expression = self
457            .variants
458            .iter()
459            .map(GammaLoopCFFVariant::to_atom_gs)
460            .reduce(|acc, atom| acc + atom)
461            .unwrap_or_else(Atom::new);
462        if self
463            .variants
464            .iter()
465            .any(|variant| variant.thermal_weight.medium_mode != MediumMode::Vacuum)
466        {
467            // Each coefficient must carry its own chemical-potential orientation
468            // before an explicit sum removes the residue-map selectors.
469            self.data.orientation.select(expression)
470        } else {
471            expression
472        }
473    }
474
475    fn energy_replacements_gs(&self, graph: &Graph) -> Vec<Replacement> {
476        energy_map_replacements_gs(
477            self.edge_energy_map
478                .iter()
479                .map(|energy| energy.to_atom_gs(&[])),
480            graph,
481        )
482    }
483}

485pub(crate) fn energy_map_replacements_gs(
486    edge_energy_map: impl IntoIterator<Item = Atom>,
487    graph: &Graph,
488) -> Vec<Replacement> {
489    let edge_energy_map = edge_energy_map.into_iter().collect::<Vec<_>>();
490    let mut replacements = Vec::new();
491    let mink_index = LibraryRep::from(Minkowski {}).to_symbolic([Atom::var(W_.a__)]);
492    let external_edges = graph
493        .underlying
494        .iter_edges()
495        .filter_map(|(pair, edge_id, _)| (!pair.is_paired()).then_some(edge_id))
496        .collect::<BTreeSet<_>>();
497
498    for (edge_id, energy) in edge_energy_map.iter().enumerate() {
499        let edge_id = EdgeIndex(edge_id);
500        // Remapping a generated internal-edge map into GammaLoop's physical
501        // namespace pads unpaired external-edge slots with zero. Those slots
502        // are not residue samples: their EMR momenta must remain the original
503        // external momenta carried by the factorized numerator.
504        if external_edges.contains(&edge_id) {
505            continue;
506        }
507        replacements.push(Replacement::new(
508            GS.emr_mom(edge_id, AIND_SYMBOLS.cind.call(Atom::Zero))
509                .to_pattern(),
510            energy.clone().to_pattern(),
511        ));
512        replacements.push(Replacement::new(
513            GS.emr_mom(edge_id, &mink_index).to_pattern(),
514            (GS.emr_vec(edge_id, &mink_index) + energy * GS.energy_delta(&mink_index)).to_pattern(),
515        ));
516    }
517
518    for (loop_id, loop_edge_id) in graph.loop_momentum_basis.loop_edges.iter_enumerated() {
519        let loop_id = usize::from(loop_id);
520        let loop_id_atom = Atom::num(loop_id as i64);
521        let energy = edge_energy_map
522            .get(usize::from(*loop_edge_id))
523            .cloned()
524            .unwrap_or_else(Atom::new);
525        replacements.push(Replacement::new(
526            function!(
527                GS.loop_mom,
528                loop_id_atom.clone(),
529                AIND_SYMBOLS.cind.call(Atom::Zero)
530            )
531            .to_pattern(),
532            energy.clone().to_pattern(),
533        ));
534        for spatial_index in 1..=3 {
535            replacements.push(Replacement::new(
536                function!(
537                    GS.loop_mom,
538                    loop_id_atom.clone(),
539                    AIND_SYMBOLS.cind.call(spatial_index)
540                )
541                .to_pattern(),
542                GS.emr_mom(*loop_edge_id, AIND_SYMBOLS.cind.call(spatial_index))
543                    .to_pattern(),
544            ));
545        }
546        replacements.push(Replacement::new(
547            FunctionBuilder::new(GS.loop_mom)
548                .add_arg(loop_id as i64)
549                .add_arg(mink_index.as_view())
550                .finish()
551                .to_pattern(),
552            (GS.emr_vec(*loop_edge_id, &mink_index) + energy * GS.energy_delta(&mink_index))
553                .to_pattern(),
554        ));
555    }
556
557    replacements
558}

639        let graph: Graph = dot!(digraph thermal_triangle {
640            edge [num=1 mass=0]
641            node [num=1]
642            ext [style=invis]
643            ext -> a [id=3]
644            ext -> b [id=4]
645            c -> ext [id=5]
646            c -> a [id=0 lmb_id=0]
647            a -> b [id=1]
648            b -> c [id=2]
649        })?;

651            MediumMode::ThermodynamicEquilibrium,

655            let mut options = graph.denominator_only_cff_3d_expression_options();

659                .expression;
660            let atom = expression
661                .orientations
662                .iter()

668            // Typed weights emit distributions directly; surface conversion must preserve them.
669            let substituted = expression.surfaces.substitute_energies(&atom, &[]);

671                substituted.contains_symbol(GS.thermal_distribution),
672                "{medium_mode:?} CFF lost its thermal distributions: {substituted}"
673            );
674        }
675        Ok(())

679    fn affine_energy_map_keeps_two_numerator_factors_under_one_map() -> color_eyre::Result<()> {
680        test_initialise()?;
681        let graph: Graph = dot!(
682            digraph affine_map {
683                edge [num=1 mass=0]
684                node [num=1]
685                A -> B [id=0]
686                A -> B [id=1]
687            }
688        )?;
689        let energy_map = LinearEnergyExpr {
690            internal_terms: vec![(EdgeIndex(0), 2.into()), (EdgeIndex(1), (-3).into())],
691            external_terms: vec![(EdgeIndex(2), 5.into())],
692            uniform_scale_coeff: 7.into(),
693            constant: 11.into(),
694        }
695        .canonical();
696        let orientation = OrientationExpression {
697            data: OrientationData::new(EdgeVec::from_iter([
698                Orientation::Undirected,
699                Orientation::Undirected,
700            ])),
701            loop_energy_map: Vec::new(),
702            edge_energy_map: vec![energy_map.clone(), LinearEnergyExpr::zero()],
703            variants: Vec::new(),
704        };
705        let factor_a = Atom::var(symbolica::symbol!("factor_a"));
706        let factor_b = Atom::var(symbolica::symbol!("factor_b"));
707        let energy_component = GS.emr_mom(EdgeIndex(0), AIND_SYMBOLS.cind.call(Atom::Zero));
708        let numerator =
709            (energy_component.clone() + factor_a.clone()) * (energy_component + factor_b.clone());
710
711        let mapped = numerator.replace_multiple(orientation.energy_replacements_gs(&graph));
712        let mapped_energy = Atom::num(2) * ose_atom_from_index(EdgeIndex(0))
713            - Atom::num(3) * ose_atom_from_index(EdgeIndex(1))
714            + Atom::num(5) * external_energy_atom_from_index(EdgeIndex(2))
715            + Atom::num(7) * Atom::var(GS.numerator_sampling_scale)
716            + Atom::num(11);
717        let expected =
718            (mapped_energy.clone() + factor_a.clone()) * (mapped_energy.clone() + factor_b.clone());
719        assert_eq!(mapped, expected);
720        let mixed = (mapped_energy.clone() + factor_a) * (mapped_energy + Atom::num(1) + factor_b);
721        assert_ne!(mapped, mixed);
722        Ok(())
723    }

734            v1;

736            ext -> v0 [id=0]
737            ext -> v1 [id=1]
738            v2 -> ext [id=2]
739            v0 -> v2 [id=3 lmb_id=0]

741            v2 -> v1 [id=5]
742        })?;
743        let mapped_internal_energy = LinearEnergyExpr {
744            internal_terms: vec![(EdgeIndex(3), 2.into())],
745            external_terms: vec![(EdgeIndex(0), (-1).into())],
746            uniform_scale_coeff: 0.into(),
747            constant: 0.into(),
748        }
749        .canonical();
750        let mut edge_energy_map = vec![LinearEnergyExpr::zero(); graph.underlying.n_edges()];
751        edge_energy_map[3] = mapped_internal_energy.clone();
752        let external_energy = GS.emr_mom(EdgeIndex(0), AIND_SYMBOLS.cind.call(Atom::Zero));
753        let internal_energy = GS.emr_mom(EdgeIndex(3), AIND_SYMBOLS.cind.call(Atom::Zero));
754        let spectator = Atom::var(symbolica::symbol!("spectator"));
755        let numerator = external_energy.clone() * (internal_energy + spectator.clone());

758            edge_energy_map.iter().map(|energy| energy.to_atom_gs(&[])),
759            &graph,

761        assert_eq!(
762            mapped,
763            external_energy * (mapped_internal_energy.to_atom_gs(&[]) + spectator)
764        );
765        Ok(())
766    }
