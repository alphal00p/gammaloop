185impl io::Write for LazyJsonWriter {
186    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
187        self.buffer.extend_from_slice(buf);
188
189        while let Some(newline_index) = self.buffer.iter().position(|byte| *byte == b'\n') {
190            let line = self.buffer.drain(..=newline_index).collect::<Vec<_>>();
191            self.write_sanitized_line(&line)?;
192        }
193        Ok(buf.len())
194    }
195
196    fn flush(&mut self) -> io::Result<()> {
197        if !self.buffer.is_empty() {
198            let line = std::mem::take(&mut self.buffer);
199            self.write_sanitized_line(&line)?;
200        }
201        if let Some(file) = self.state.file.lock().unwrap().as_mut() {
202            file.flush()?;
203        }
204        Ok(())
205    }
206}

208impl LazyJsonWriter {
209    fn write_sanitized_line(&self, line: &[u8]) -> io::Result<()> {
210        let mut file = self.state.file.lock().unwrap();
211        if file.is_none() {
212            fs::create_dir_all(&self.state.dir)?;
213            let opened = OpenOptions::new()
214                .create(true)
215                .append(true)
216                .open(self.state.dir.join(&self.state.filename))?;
217            *file = Some(opened);
218        }
219
220        let file = file.as_mut().unwrap();
221        let line = sanitize_json_log_line(line);
222        file.write_all(&line)
223    }
224}

226fn sanitize_json_log_line(line: &[u8]) -> Vec<u8> {
227    let mut trailing_newline = false;
228    let line = if let Some(stripped) = line.strip_suffix(b"\n") {
229        trailing_newline = true;
230        stripped
231    } else {
232        line
233    };
234
235    let sanitized = match serde_json::from_slice::<serde_json::Value>(line) {
236        Ok(mut value) => {
237            strip_ansi_from_json_value(&mut value);
238            serde_json::to_vec(&value).unwrap_or_else(|_| line.to_vec())
239        }
240        Err(_) => line.to_vec(),
241    };
242
243    if trailing_newline {
244        let mut with_newline = sanitized;
245        with_newline.push(b'\n');
246        with_newline
247    } else {
248        sanitized
249    }
250}
251
252fn strip_ansi_from_json_value(value: &mut serde_json::Value) {
253    match value {
254        serde_json::Value::String(string) => {
255            *string = strip_ansi_escape_codes(string);
256        }
257        serde_json::Value::Array(items) => {
258            for item in items {
259                strip_ansi_from_json_value(item);
260            }
261        }
262        serde_json::Value::Object(map) => {
263            for value in map.values_mut() {
264                strip_ansi_from_json_value(value);
265            }
266        }
267        serde_json::Value::Null | serde_json::Value::Bool(_) | serde_json::Value::Number(_) => {}
268    }
269}

303/// Set the file log filter at runtime while preserving required targets.
304pub fn set_file_log_filter(user_spec: impl AsRef<str>) -> Result<()> {
305    let user_spec = if let Some(file) = log_filter_env_override(ENV_FILE_LOG_FILTER) {
306        if std::env::var(ENV_NO_GL_HARD_WARNINGS).is_err() {
307            println!(
308                "WARNING, file log filter is set to {file}, will override settings {}",
309                user_spec.as_ref()
310            );
311        }
312        file
313    } else {
314        user_spec.as_ref().to_string()
315    };
316    let effective_spec = {
317        let mut state = FILE_LOG_SPEC.lock().unwrap();
318        let old_base = state.base_spec.clone();
319        state.base_spec = user_spec;
320        let effective = state.effective_spec_string();
321        if let Some(reason) = state.hard_disabled_reason.clone() {
322            if !directive_is_effectively_off(&effective) {
323                state.base_spec = old_base;
324                return Err(file_log_disabled_error(&reason));
325            }
326        }
327        effective
328    };
329
330    let Some(handles) = FILTER_HANDLES.get() else {
331        return Ok(());
332    };
333    let Some(reload) = handles.file_reload.as_ref() else {
334        return Ok(());
335    };
336    let full_spec = file_filter_from(&effective_spec)?;
337    reload(full_spec)?;
338    Ok(())
339}

443pub(crate) fn init_tracing(dir: impl AsRef<Path>, log_file_name: Option<String>) {
444    symbolica::GLOBAL_SETTINGS
445        .initialize_tracing
446        .store(false, std::sync::atomic::Ordering::Relaxed);
447    FILTER_HANDLES.get_or_init(|| {
448        let file_state = FILE_LOG_SPEC.lock().unwrap().clone();
449        let file_spec = file_state.effective_spec_string();
450        // Keep startup filter semantics aligned with runtime reloads.
451        let file_filter = file_filter_from(&file_spec).unwrap_or_else(|err| {
452            eprintln!(
453                "Invalid initial file log filter '{file_spec}': {err}. Falling back to warn."
454            );
455            file_filter_from("").expect("default file log filter must parse")
456        });
457        // println!(
458        //     "File filter: {file_filter},:{}",
459        //     FILE_LOG_SPEC.lock().unwrap().as_str()
460        // );
461        let stderr_spec = STDERR_LOG_SPEC.lock().unwrap().effective_spec_string();
462        // Keep startup filter semantics aligned with runtime reloads.
463        let stderr_filter = display_filter_from(&stderr_spec).unwrap_or_else(|err| {
464            eprintln!(
465                "Invalid initial display log filter '{stderr_spec}': {err}. Falling back to off."
466            );
467            display_filter_from("").expect("default display log filter must parse")
468        });
469        // println!(
470        //     "Stderr filter: {stderr_filter},:{}",
471        //     STDERR_LOG_SPEC.lock().unwrap().as_str()
472        // );
473
474        let (file_filter_layer, file_handle) = file_filter.clone().reloadable();
475        let (stderr_filter_layer, stderr_handle) = stderr_filter.clone().reloadable();
476
477        // e.g. "2025-08-26T21-05-33.123"
478        let ts = Local::now()
479            .to_rfc3339_opts(SecondsFormat::Millis, true)
480            .replace([':', '+'], "-"); // keep it filename-friendly
481
482        let filename = if let Some(log_name) = log_file_name {
483            format!("gammalog-{log_name}.jsonl")
484        } else {
485            format!("gammalog-{ts}.jsonl")
486        };
487        let file_writer = BoxMakeWriter::new(LazyJsonMakeWriter::new(
488            dir.as_ref().to_path_buf(),
489            filename,
490        ));
491
492        let json = fmt::layer()
493            .event_format(FileJsonFmt)
494            .with_writer(file_writer);
495
496        let indicatif_layer = IndicatifLayer::new()
497            .with_span_field_formatter(IndicatifPbMsgFields{})
498            .with_max_progress_bars(1024, None) .with_progress_style( ProgressStyle::with_template(
499                      "{color_start}{span_child_prefix}{span_fields} -- {wide_msg} {elapsed_subsec}{color_end}",
500                  )
501                  .unwrap()
502                  .with_key(
503                      "elapsed_subsec",
504                      elapsed_subsec,
505                  )
506                  .with_key(
507                      "color_start",
508                      |state: &ProgressState, writer: &mut dyn std::fmt::Write| {
509                          let elapsed = state.elapsed();
510
511                          if elapsed > Duration::from_secs(60) {
512                              // Red
513                              let _ = write!(writer, "\x1b[{}m", 1 + 30);
514                          } else if elapsed > Duration::from_secs(4) {
515                              // Yellow
516                              let _ = write!(writer, "\x1b[{}m", 3 + 30);
517                          }
518                      },
519                  )
520                  .with_key(
521                      "color_end",
522                      |state: &ProgressState, writer: &mut dyn std::fmt::Write| {
523                          if state.elapsed() > Duration::from_secs(4) {
524                              let _ = write!(writer, "\x1b[0m");
525                          }
526                      },
527                  ),
528              )
529              .with_span_child_prefix_symbol("↳ ")
530              .with_span_child_prefix_indent(" ");
531
532        // Pretty status layer for stderr - only show status events
533        let status_layer = fmt::layer()
534            .with_target(false)
535            .event_format(GammaDisplayFormat::dynamic(|| {
536                LOG_STYLE_STATE.lock().unwrap().effective_style()
537            }))
538            .with_writer(indicatif_layer.get_stderr_writer());
539
540        let subscriber = tracing_subscriber::registry()
541            .with(Filtered::new(status_layer, stderr_filter_layer))
542            .with(indicatif_layer.with_filter(IndicatifFilter::new(false)));
543
544        let init_result = if file_state.hard_disabled() {
545            subscriber.try_init()
546        } else {
547            subscriber
548                .with(Filtered::new(json, file_filter_layer))
549                .try_init()
550        };
551        if init_result.is_err() {
552            return detached_filter_handles(!file_state.hard_disabled());
553        }
554
555        let file_reload = (!file_state.hard_disabled()).then_some(Box::new(move |filter| {
556            file_handle.reload(filter);
557            Ok(())
558        }) as ReloadFilterFn);
559        let stderr_reload = Box::new(move |filter| {
560            stderr_handle.reload(filter);
561            Ok(())
562        }) as ReloadFilterFn;
563
564        FilterHandles {
565            file_reload,
566            stderr_reload,
567        }
568    });
569}

601impl<'writer> FormatFields<'writer> for IndicatifPbMsgFields {
602    fn format_fields<R: RecordFields>(
603        &self,
604        mut writer: Writer<'writer>,
605        fields: R,
606    ) -> std::fmt::Result {
607        struct Visitor {
608            msg: Option<String>,
609        }
610
611        impl tracing::field::Visit for Visitor {
612            fn record_debug(&mut self, f: &tracing::field::Field, v: &dyn std::fmt::Debug) {
613                if f.name() == "indicatif.pb_msg" {
614                    self.msg = Some(format!("{v:?}"));
615                }
616            }
617
618            fn record_str(&mut self, f: &tracing::field::Field, v: &str) {
619                if f.name() == "indicatif.pb_msg" {
620                    self.msg = Some(v.to_string());
621                }
622            }
623        }
624
625        let mut v = Visitor { msg: None };
626        fields.record(&mut v);
627        if let Some(msg) = v.msg {
628            write!(writer, "{msg}")?;
629        }
630        Ok(())
631    }
632}

635struct FileJsonFmt;
636
637impl<S, N> FormatEvent<S, N> for FileJsonFmt
638where
639    S: Subscriber + for<'a> LookupSpan<'a>,
640    N: for<'writer> FormatFields<'writer> + 'static,
641{
642    fn format_event(
643        &self,
644        ctx: &FmtContext<'_, S, N>,
645        mut w: Writer<'_>,
646        event: &Event<'_>,
647    ) -> std::fmt::Result {
648        let now = Local::now().to_rfc3339_opts(SecondsFormat::Millis, true);
649        let meta = event.metadata();
650
651        let mut visitor = RoutedFieldsVisitor::for_sink(LogSink::File);
652        event.record(&mut visitor);
653
654        let mut payload = serde_json::Map::new();
655        payload.insert("timestamp".to_string(), serde_json::Value::String(now));
656        payload.insert(
657            "level".to_string(),
658            serde_json::Value::String(meta.level().to_string()),
659        );
660        payload.insert(
661            "target".to_string(),
662            serde_json::Value::String(meta.target().to_string()),
663        );
664
665        if let Some(module_path) = meta.module_path() {
666            payload.insert(
667                "module_path".to_string(),
668                serde_json::Value::String(module_path.to_string()),
669            );
670        }
671        if let Some(file) = meta.file() {
672            payload.insert(
673                "file".to_string(),
674                serde_json::Value::String(file.to_string()),
675            );
676        }
677        if let Some(line) = meta.line() {
678            payload.insert("line".to_string(), line.into());
679        }
680        if let Some(message) = visitor.message {
681            payload.insert("message".to_string(), serde_json::Value::String(message));
682        }
683
684        for (key, value) in visitor.fields {
685            payload.insert(key, value);
686        }
687
688        if let Some(span) = ctx.lookup_current() {
689            let mut current_span = serde_json::Map::new();
690            current_span.insert(
691                "name".to_string(),
692                serde_json::Value::String(span.name().to_string()),
693            );
694            if let Some(fields) = span.extensions().get::<FormattedFields<N>>() {
695                let formatted = fields.to_string();
696                if !formatted.is_empty() {
697                    current_span.insert("fields".to_string(), serde_json::Value::String(formatted));
698                }
699            }
700            payload.insert("span".to_string(), serde_json::Value::Object(current_span));
701        }
702
703        let line = serde_json::to_string(&payload).map_err(|_| std::fmt::Error)?;
704        w.write_str(&line)?;
705        w.write_char('\n')
706    }
707}
