36    /// Get all app template paths written under drawings/templates.

38        GammaLoopTemplateAssets::iter().map(|path| path.to_string())

41    /// Extract all drawing templates to drawings/templates relative to target_dir.
42    pub fn extract_templates<P: AsRef<Path>>(target_dir: P) -> Result<()> {
43        let target = target_dir.as_ref().join("drawings/templates");
44
45        fs::create_dir_all(&target)?;
46
47        extract_package::<GammaLoopTemplateAssets>(&target)?;
48        extract_package::<LinnestPackageAssets>(&target.join(LINNEST_PACKAGE_DIR))?;
49        extract_package::<KurvstPackageAssets>(&target.join(KURVST_PACKAGE_DIR))?;
50
51        Ok(())
52    }
53
54    /// Extract the Justfile to the target directory
55    pub fn extract_justfile<P: AsRef<Path>>(target_dir: P) -> Result<()> {
56        let justfile_path = target_dir.as_ref().join("justfile");
57
58        if let Some(file) = Self::get("drawing/justfile") {
59            fs::write(&justfile_path, file.data)?;
60        }
61
62        Ok(())
63    }
64}
65
66fn extract_package<E: RustEmbed>(target: &Path) -> Result<()> {
67    fs::create_dir_all(target)?;
68    for package_path in E::iter() {
69        if let Some(file) = E::get(package_path.as_ref()) {
70            let target_path = target.join(package_path.as_ref());
71
72            if let Some(parent) = target_path.parent() {
73                fs::create_dir_all(parent)?;
74            }
75
76            fs::write(&target_path, file.data)?;
77        }
78    }
79
80    Ok(())
81}
