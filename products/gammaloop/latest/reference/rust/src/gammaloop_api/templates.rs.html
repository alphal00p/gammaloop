34impl Assets {
35    /// Get all app template paths written under drawings/templates.
36    pub fn template_paths() -> impl Iterator<Item = String> {
37        GammaLoopTemplateAssets::iter().map(|path| path.to_string())
38    }
39
40    /// Extract all drawing templates to drawings/templates relative to target_dir.
41    pub fn extract_templates<P: AsRef<Path>>(target_dir: P) -> Result<()> {
42        let target = target_dir.as_ref().join("drawings/templates");
43
44        fs::create_dir_all(&target)?;
45
46        extract_package::<GammaLoopTemplateAssets>(&target)?;
47        extract_package::<LinnestPackageAssets>(&target.join(LINNEST_PACKAGE_DIR))?;
48        extract_package::<KurvstPackageAssets>(&target.join(KURVST_PACKAGE_DIR))?;
49
50        Ok(())
51    }
52
53    /// Extract the Justfile to the target directory
54    pub fn extract_justfile<P: AsRef<Path>>(target_dir: P) -> Result<()> {
55        let justfile_path = target_dir.as_ref().join("justfile");
56
57        if let Some(file) = Self::get("drawing/justfile") {
58            fs::write(&justfile_path, file.data)?;
59        }
60
61        Ok(())
62    }
63}
64
65fn extract_package<E: RustEmbed>(target: &Path) -> Result<()> {
66    fs::create_dir_all(target)?;
67    for package_path in E::iter() {
68        if let Some(file) = E::get(package_path.as_ref()) {
69            let target_path = target.join(package_path.as_ref());
70
71            if let Some(parent) = target_path.parent() {
72                fs::create_dir_all(parent)?;
73            }
74
75            fs::write(&target_path, file.data)?;
76        }
77    }
78
79    Ok(())
80}

116        assert!(templates
