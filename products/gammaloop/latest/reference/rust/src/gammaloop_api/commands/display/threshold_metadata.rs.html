27pub(super) fn render(
28    state: &State,
29    process_id: usize,
30    integrand_name: &str,
31    requested_graphs: &[String],
32    format: MetadataDisplayFormat,
33) -> Result<String> {
34    if format == MetadataDisplayFormat::Toml && requested_graphs.len() != 1 {
35        return Err(eyre!(
36            "Threshold TOML output requires exactly one master graph selected with --graph"
37        ));
38    }
39    let process = state
40        .process_list
41        .processes
42        .get(process_id)
43        .ok_or_else(|| eyre!("Unknown process #{process_id}"))?;
44    let integrand = process.get_integrand(integrand_name)?;
45    integrand.require_generated()?;
46    let graphs = match &process.collection {
47        ProcessCollection::Amplitudes(amplitudes) => {
48            let amplitude = &amplitudes[&integrand.canonical_name];
49            amplitude
50                .graph_group_structure
51                .iter()
52                .map(|group| {
53                    let graph = &amplitude.graphs[group
54                        .into_iter()
55                        .next()
56                        .expect("graph groups have a master")];
57                    (
58                        &graph.graph,
59                        graph.derived_data.resolved_threshold_counterterms.as_ref(),
60                        graph
61                            .derived_data
62                            .threshold_topology
63                            .as_ref()
64                            .map(|(_, lmbs)| lmbs)
65                            .or(graph.derived_data.lmbs.as_ref()),
66                    )
67                })
68                .collect_vec()
69        }
70        ProcessCollection::CrossSections(cross_sections) => {
71            let cross_section = &cross_sections[&integrand.canonical_name];
72            cross_section
73                .graph_group_structure
74                .iter()
75                .map(|group| {
76                    let graph = &cross_section.supergraphs[group
77                        .into_iter()
78                        .next()
79                        .expect("graph groups have a master")];
80                    (
81                        &graph.graph,
82                        graph.derived_data.resolved_threshold_counterterms.as_ref(),
83                        graph.derived_data.lmbs.as_ref(),
84                    )
85                })
86                .collect_vec()
87        }
88    };
89    for requested in requested_graphs {
90        if !graphs.iter().any(|(graph, _, _)| graph.name == *requested) {
91            return Err(eyre!(
92                "Unknown master graph '{}' for integrand '{}'. Available master graphs: {}",
93                requested,
94                integrand.canonical_name,
95                graphs
96                    .iter()
97                    .map(|(graph, _, _)| graph.name.as_str())
98                    .join(", "),
99            ));
100        }
101    }
102    graphs
103        .into_iter()
104        .filter(|(graph, _, _)| {
105            requested_graphs.is_empty() || requested_graphs.contains(&graph.name)
106        })
107        .map(|(graph, resolved, lmbs)| render_graph(graph, resolved, lmbs, format))
108        .collect::<Result<Vec<_>>>()
109        .map(|reports| reports.join("\n\n"))
110}

112fn render_graph(
113    graph: &Graph,
114    resolved: Option<&ResolvedThresholdCounterterms>,
115    lmbs: Option<&TiVec<LmbIndex, LoopMomentumBasis>>,
116    format: MetadataDisplayFormat,
117) -> Result<String> {
118    let generation = resolved
119        .map(|resolved| {
120            let lmbs = lmbs.ok_or_else(|| {
121                eyre!(
122                    "Graph '{}' has resolved threshold metadata without its generation LMBs",
123                    graph.name,
124                )
125            })?;
126            Ok::<_, eyre::Report>((resolved, lmbs))
127        })
128        .transpose()?;
129    let spec = generation.map_or_else(
130        || graph.threshold_counterterms.value.clone(),
131        |(resolved, lmbs)| resolved.materialized_spec(&graph.threshold_counterterms, lmbs),
132    );
133    match format {
134        MetadataDisplayFormat::Toml => {
135            if generation.is_none() {
136                return Err(eyre!(
137                    "Threshold subtraction was disabled during generation of graph '{}'. Its generation toggle cannot be represented by a threshold_counterterms DOT attribute",
138                    graph.name,
139                ));
140            }
141            render_toml(&spec)
142        }
143        MetadataDisplayFormat::Pretty => render_pretty(&graph.name, &spec, generation),
144    }
145}

147fn render_toml(spec: &ThresholdCountertermSpec) -> Result<String> {
148    let toml = {
149        let _defaults = ShowDefaultsGuard::new(true);
150        toml::to_string(spec).context("Serializing threshold counterterm settings")?
151    };
152    let mut attributes = GlobalData::from(());
153    attributes
154        .statements
155        .insert("threshold_counterterms".to_string(), format!("\n{toml}"));
156    Ok(format!("{attributes:width$}", width = 0))
157}

159fn render_pretty(
160    graph_name: &str,
161    spec: &ThresholdCountertermSpec,
162    generation: Option<(
163        &ResolvedThresholdCounterterms,
164        &TiVec<LmbIndex, LoopMomentumBasis>,
165    )>,
166) -> Result<String> {
167    let mut output = format!(
168        "{}\n",
169        format!("Threshold subtraction · {graph_name}")
170            .bold()
171            .blue()
172    );
173    let generated = generation.map_or(0, |(resolved, _)| {
174        resolved
175            .variants
176            .iter()
177            .flat_map(|variant| &variant.associations)
178            .filter(|association| association.eligible)
179            .count()
180    });
181    if generation.is_none() {
182        writeln!(output, "{}", "Threshold subtraction was disabled during generation. Supplied declarations are shown below.".yellow())?;
183    } else {
184        writeln!(
185            output,
186            "{generated} generated threshold associations; static generation metadata"
187        )?;
188    }
189
190    let edges = |edges: &[EdgeIndex]| format!("[{}]", edges.iter().map(|edge| edge.0).join(", "));
191    let mut table = Builder::new();
192    table.push_record(
193        [
194            "Cut / threshold",
195            "Variant / generation",
196            "Solve basis / signed cycles",
197            "Multiplier",
198        ]
199        .map(|heading| heading.bold().blue().to_string()),
200    );
201    let mut row_count = 0;
202    for cut in &spec.cuts {
203        for threshold in &cut.thresholds {
204            if threshold.counterterms.is_empty() {
205                table.push_record([
206                    format!(
207                        "cut {}\nthreshold {}",
208                        edges(&cut.edges).cyan(),
209                        edges(&threshold.edges).yellow()
210                    ),
211                    "default\nskipped / unresolved".yellow().to_string(),
212                    "automatic\nsubspace: unresolved\nparent: unresolved"
213                        .dimmed()
214                        .to_string(),
215                    "1".dimmed().to_string(),
216                ]);
217                row_count += 1;
218            }
219            for declaration in &threshold.counterterms {
220                let name = declaration.name.as_deref().unwrap_or("default");
221                let resolved = generation.and_then(|(resolved, lmbs)| {
222                    resolved
223                        .variants
224                        .iter()
225                        .filter(|variant| variant.name == name)
226                        .find_map(|variant| {
227                            variant
228                                .associations
229                                .iter()
230                                .find(|association| {
231                                    association.cut_edges == cut.edges
232                                        && association.threshold_edges == threshold.edges
233                                })
234                                .map(|association| (association, variant.side, lmbs))
235                        })
236                });
237                let status = if declaration.disable {
238                    "disabled".red()
239                } else if let Some((association, _, _)) = resolved {
240                    if association.eligible {
241                        match association.origin {
242                            ThresholdCountertermOrigin::Explicit => "generated · explicit".green(),
243                            ThresholdCountertermOrigin::Autogenerated => {
244                                "generated · default".green()
245                            }
246                        }
247                    } else {
248                        "skipped".yellow()
249                    }
250                } else {
251                    "skipped / unresolved".yellow()
252                };
253                let side = resolved
254                    .map(|(_, side, _)| match side {
255                        ThresholdCountertermSide::Amplitude => "amplitude".cyan(),
256                        ThresholdCountertermSide::Left => "left".green(),
257                        ThresholdCountertermSide::Right => "right".yellow(),
258                    })
259                    .unwrap_or_else(|| "unresolved".dimmed());
260                let mut solve = format!(
261                    "{}\nsubspace: {}\nparent: {}",
262                    declaration
263                        .group_id
264                        .map(|id| format!("group_id = {id}").yellow().to_string())
265                        .unwrap_or_else(|| "automatic group".dimmed().to_string()),
266                    declaration
267                        .subspace
268                        .as_deref()
269                        .map(|value| edges(value).yellow().to_string())
270                        .unwrap_or_else(|| "maximal (unresolved)".dimmed().to_string()),
271                    declaration
272                        .parent_lmb
273                        .as_deref()
274                        .map(|value| edges(value).cyan().to_string())
275                        .unwrap_or_else(|| "unresolved".dimmed().to_string()),
276                );
277                if let Some((association, _, lmbs)) = resolved {
278                    for (edge, cycle) in association.subspace.solve_signature(lmbs) {
279                        let cycle = cycle
280                            .iter()
281                            .map(|(edge, sign)| {
282                                let sign = match sign {
283                                    SignOrZero::Plus => "+".green(),
284                                    SignOrZero::Minus => "-".red(),
285                                    SignOrZero::Zero => "0".dimmed(),
286                                };
287                                format!("{sign}e{}", edge.0)
288                            })
289                            .join(" ");
290                        write!(
291                            solve,
292                            "\n{} → {cycle}",
293                            format!("e{}", edge.0).bold().cyan()
294                        )?;
295                    }
296                }
297                let multiplier = if let Some(multiplier) = &declaration.multiplier {
298                    let mut text = format!(
299                        "{}\nsymmetrize = {}\nopaque_derivatives = {}",
300                        multiplier.expression.magenta(),
301                        multiplier.symmetrize.to_string().yellow(),
302                        multiplier.opaque_derivatives.to_string().yellow(),
303                    );
304                    for (name, definition) in &multiplier.function_map {
305                        write!(
306                            text,
307                            "\nlocal {} = {}",
308                            name.bold().cyan(),
309                            definition.magenta()
310                        )?;
311                    }
312                    text
313                } else {
314                    "1".dimmed().to_string()
315                };
316                table.push_record([
317                    format!(
318                        "cut {}\nthreshold {}\n{side}",
319                        edges(&cut.edges).cyan(),
320                        edges(&threshold.edges).yellow()
321                    ),
322                    format!("{}\n{status}", name.bold().cyan()),
323                    solve,
324                    multiplier,
325                ]);
326                row_count += 1;
327            }
328        }
329    }
330    if row_count == 0 {
331        writeln!(output, "No threshold declarations.")?;
332    } else {
333        let mut table = table.build();
334        table
335            .with(Style::rounded())
336            .with(Alignment::left())
337            .with(Alignment::top());
338        for (column, width) in [26, 24, 38, 48].into_iter().enumerate() {
339            table.modify(Columns::one(column), Width::wrap(width).keep_words(true));
340        }
341        writeln!(output, "{table}")?;
342        writeln!(
343            output,
344            "{} {} → {}e3 {}e14 means varying basis momentum e3 adds to edge 3 and subtracts from edge 14.",
345            "Cycles:".bold().blue(),
346            "e3".bold().cyan(),
347            "+".green(),
348            "-".red()
349        )?;
350        writeln!(output, "{}", "Signs follow edge orientation. Selected edges + signed cycles define automatic solve groups; group_id can separate them.".dimmed())?;
351    }
352    if !spec.function_map.is_empty() {
353        writeln!(
354            output,
355            "{}",
356            "Shared function definitions (local definitions override these)"
357                .bold()
358                .blue()
359        )?;
360        let mut definitions = Builder::new();
361        definitions.push_record(
362            ["Function", "Definition"].map(|heading| heading.bold().blue().to_string()),
363        );
364        for (name, definition) in &spec.function_map {
365            definitions.push_record([
366                name.bold().cyan().to_string(),
367                definition.magenta().to_string(),
368            ]);
369        }
370        let mut definitions = definitions.build();
371        definitions
372            .with(Style::rounded())
373            .with(Alignment::left())
374            .modify(Columns::one(0), Width::wrap(32).keep_words(true))
375            .modify(Columns::one(1), Width::wrap(100).keep_words(true));
376        writeln!(output, "{definitions}")?;
377    }
378    Ok(output)
379}

386    #[test]
387    fn gl638_dot_metadata_roundtrips_with_verbose_flags_and_function_scopes() {
388        let graph: DotGraph = DotGraph::from_string(include_str!(
389            "../../../../../examples/cli/epem_a_ttxh/NNLO/graphs/GL638.dot"
390        ))
391        .unwrap();
392        let mut spec: ThresholdCountertermSpec =
393            toml::from_str(&graph.global_data.statements["threshold_counterterms"]).unwrap();
394        let variant = &mut spec.cuts[0].thresholds[0].counterterms[0];
395        variant.name = Some("quoted \"variant\" λ\\line\nend\\".into());
396        variant
397            .multiplier
398            .as_mut()
399            .unwrap()
400            .function_map
401            .insert("wQ()".into(), "local_override()".into());
402        let attribute = render_toml(&spec).unwrap();
403        assert!(attribute.starts_with("threshold_counterterms = \"\n"));
404        assert!(!attribute.contains('\u{1b}'));
405        assert!(attribute.contains("disable = false"));
406        assert!(attribute.contains("symmetrize = false"));
407        assert!(attribute.contains("opaque_derivatives = true"));
408        assert!(attribute.contains("parent_lmb = [3, 6, 7, 10]"));
409        let reparsed: DotGraph =
410            DotGraph::from_string(format!("digraph G {{\n{attribute}\n}}")).unwrap();
411        let roundtrip: ThresholdCountertermSpec =
412            toml::from_str(&reparsed.global_data.statements["threshold_counterterms"]).unwrap();
413        assert_eq!(roundtrip, spec);
414        assert_eq!(roundtrip.function_map["wQ()"], "P(0,cind(0))+P(1,cind(0))");
415    }

446    #[test]
447    fn empty_enabled_thresholds_are_distinct_from_disabled_generation() -> Result<()> {
448        use gammalooprs::{initialisation::test_initialise, utils::load_generic_model};
449
450        test_initialise()?;
451        let spec = ThresholdCountertermSpec::default();
452        let resolved = ResolvedThresholdCounterterms {
453            legacy_equivalent: true,
454            variants: TiVec::new(),
455            cross_section_cut_groups: TiVec::new(),
456        };
457        let lmbs = TiVec::new();
458        let enabled = render_pretty("empty", &spec, Some((&resolved, &lmbs))).unwrap();
459        assert!(enabled.contains("0 generated threshold associations"));
460        assert!(!enabled.contains("disabled during generation"));
461        let disabled = render_pretty("empty", &spec, None).unwrap();
462        assert!(disabled.contains("disabled during generation"));
463        let attribute = render_toml(&spec).unwrap();
464        assert!(attribute.contains("cuts = []"));
465
466        let model = load_generic_model("scalars");
467        let graph = Graph::from_string(
468            include_str!("../../../../../tests/resources/graphs/scalar_bubble.dot"),
469            &model,
470        )?
471        .remove(0);
472        let lmbs = TiVec::from(vec![graph.loop_momentum_basis.clone()]);
473        let before = graph.threshold_counterterms.clone();
474        let enabled = render_graph(
475            &graph,
476            Some(&resolved),
477            Some(&lmbs),
478            MetadataDisplayFormat::Toml,
479        )?;
480        assert_eq!(enabled, attribute);
481        let disabled =
482            render_graph(&graph, None, Some(&lmbs), MetadataDisplayFormat::Toml).unwrap_err();
483        assert!(disabled.to_string().contains("disabled during generation"));
484        let missing_lmbs =
485            render_graph(&graph, Some(&resolved), None, MetadataDisplayFormat::Toml).unwrap_err();
486        assert!(missing_lmbs
487            .to_string()
488            .contains("without its generation LMBs"));
489        assert_eq!(graph.threshold_counterterms, before);
490        Ok(())
491    }
