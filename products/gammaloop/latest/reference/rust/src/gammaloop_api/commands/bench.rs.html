181        if !self.momentum_space && (self.graph_id.is_some() || self.orientation_id.is_some()) {
182            return Err(eyre!(

190        &self,
191        state: &mut State,
192        process_id: usize,
193        integrand_name: &str,
194    ) -> Result<Option<String>> {
195        let Some(graph_id) = self.graph_id else {

198        let integrand = state
199            .process_list

202            eyre!(

204                graph_id,
205                integrand_name,

213        &self,

218        let integrand = state

232    fn repeated_point_array(&self, rows: usize) -> Result<Array2<f64>> {

248    fn evaluate_samples_request<'a>(
249        &'a self,
250        process_id: usize,
251        integrand_name: String,
252        points: &'a Array2<f64>,
253        discrete_dims: &'a Array2<usize>,
254        graph_name: &Option<String>,
255        return_generated_events: bool,
256    ) -> EvaluateSamples<'a> {
257        let batch_len = points.nrows();
258        EvaluateSamples {
259            process_id: Some(process_id),
260            integrand_name: Some(integrand_name),
261            use_arb_prec: self.use_arb_prec,
262            minimal_output: false,
263            return_generated_events: Some(return_generated_events),
264            momentum_space: self.momentum_space,
265            points: points.view(),
266            integrator_weights: None,
267            discrete_dims: Some(discrete_dims.view()),
268            graph_names: graph_name
269                .as_ref()
270                .map(|name| vec![Some(name.clone()); batch_len]),
271            orientations: self
272                .orientation_id
273                .map(|orientation| vec![Some(orientation); batch_len]),
274        }
275    }

277    fn displayed_result_from_evaluation(
278        &self,
279        evaluation: &EvaluationResultOutput,
280    ) -> Result<(Option<f64>, Complex<f64>)> {
281        let raw_result = evaluation.integrand_result;
282        let jacobian = evaluation.parameterization_jacobian.map(|jac| jac.0);
283        let displayed_result = if self.momentum_space {
284            raw_result
285        } else {
286            let jacobian = jacobian.ok_or_else(|| {
287                eyre!("An x-space benchmark requires a parameterization jacobian, but none was returned")
288            })?;
289            raw_result.map(|entry| entry * F(jacobian))
290        };
291        Ok((jacobian, displayed_result.map(|entry| entry.0)))
292    }

294    fn write_json_output<T: Serialize>(&self, value: &T) -> Result<()> {
295        let Some(path) = self.json_output.as_deref() else {
296            return Ok(());
297        };
298        let path = resolve_json_output_path(path)?;
299        if let Some(parent) = path.parent() {
300            fs::create_dir_all(parent)
301                .with_context(|| format!("Could not create {}", parent.display()))?;
302        }
303        fs::write(&path, serde_json::to_vec_pretty(value)?)
304            .with_context(|| format!("Could not write benchmark JSON output {}", path.display()))?;
305        info!(
306            "Wrote benchmark JSON output to {}",
307            path.display().to_string().green()
308        );
309        Ok(())
310    }

312    fn run_benchmark(
313        &self,
314        state: &mut State,
315        process_id: usize,
316        integrand_name: String,
317        graph_name: Option<String>,
318    ) -> Result<(Option<f64>, Complex<f64>)> {
319        let target = parse_bench_target_duration(&self.duration)?;
320        let n_batches = self.n_batches.unwrap_or(BENCH_DEFAULT_BATCHES);
321        if n_batches == 0 {
322            return Err(eyre!("bench requires at least one batch"));
323        }
324
325        let warmup_points = self.repeated_point_array(BENCH_WARMUP_SAMPLES)?;
326        let warmup_discrete_dims = self.repeated_discrete_dim_array(BENCH_WARMUP_SAMPLES)?;
327        let warmup_start = Instant::now();
328        let warmup = evaluate_samples(
329            state,
330            &self.evaluate_samples_request(
331                process_id,
332                integrand_name.clone(),
333                &warmup_points,
334                &warmup_discrete_dims,
335                &graph_name,
336                false,
337            ),
338        )?;
339        let warmup_timing = warmup_start.elapsed();
340        let warmup_samples = warmup.samples.len().max(1);
341        let warmup_per_sample = warmup_timing.as_secs_f64() / warmup_samples as f64;
342        let total_samples = if warmup_per_sample > 0.0 {
343            ((target.as_secs_f64() / warmup_per_sample).ceil() as usize).max(n_batches)
344        } else {
345            n_batches
346        };
347
348        let batch_sizes = split_samples_into_batches(total_samples, n_batches);
349        let progress_bar = bench_progress_bar(n_batches);
350        progress_bar.set_message(
351            format!(
352                "{} {}",
353                "running benchmark batches".cyan(),
354                format!("0/{n_batches}").blue()
355            )
356            .to_string(),
357        );
358
359        let batch_timings = (|| -> Result<Vec<_>> {
360            let mut batch_timings = Vec::with_capacity(n_batches);
361            for (batch_index, batch_size) in batch_sizes.into_iter().enumerate() {
362                let points = self.repeated_point_array(batch_size)?;
363                let discrete_dims = self.repeated_discrete_dim_array(batch_size)?;
364                let batch_start = Instant::now();
365                let batch = evaluate_samples(
366                    state,
367                    &self.evaluate_samples_request(
368                        process_id,
369                        integrand_name.clone(),
370                        &points,
371                        &discrete_dims,
372                        &graph_name,
373                        false,
374                    ),
375                )?;
376                let total_timing = batch_start.elapsed();
377                batch_timings.push(bench_batch_timing(total_timing, &batch)?);
378                update_bench_progress(
379                    &progress_bar,
380                    batch_index + 1,
381                    n_batches,
382                    &batch_timings,
383                    total_samples,
384                );
385            }
386            Ok(batch_timings)
387        })();
388        progress_bar.finish_and_clear();
389        let batch_timings = batch_timings?;
390
391        let warmup_evaluation = warmup
392            .samples
393            .first()
394            .ok_or_else(|| eyre!("bench warmup returned no sample"))?;
395        let (jacobian, displayed_result) =
396            self.displayed_result_from_evaluation(&warmup_evaluation.evaluation)?;
397        let bench_output = BenchOutput {
398            process_id,
399            integrand_name: integrand_name.clone(),
400            momentum_space: self.momentum_space,
401            point: self.point.clone(),
402            discrete_dim: self.discrete_dim.clone(),
403            graph_name: graph_name.clone(),
404            orientation_id: self.orientation_id,
405            target_seconds: target.as_secs_f64(),
406            warmup_samples,
407            warmup_seconds: warmup_timing.as_secs_f64(),
408            total_samples,
409            n_batches,
410            minimal_integrand: self.minimal_integrand,
411            parameterization_jacobian: jacobian,
412            displayed_result,
413            batches: batch_timings.clone(),
414            summary: bench_summary_rows(&batch_timings),
415        };
416        self.write_json_output(&bench_output)?;
417
418        info!(
419            "\n{}\n{}",
420            format!(
421                "Benchmark for integrand '{}' at fixed input point: target {}, warmup {} sample(s), profiled {} sample(s) in {} batch(es).",
422                integrand_name.green(),
423                format_duration_smart(target).yellow(),
424                warmup_samples.to_string().blue(),
425                total_samples.to_string().blue(),
426                n_batches.to_string().blue(),
427            ),
428            render_bench_table(&batch_timings)
429        );
430
431        Ok((jacobian, displayed_result))
432    }

434    pub fn run(&self, state: &mut State) -> Result<()> {
435        let (process_id, integrand_name) =
436            state.find_integrand_ref(self.process.as_ref(), self.integrand_name.as_ref())?;
437        self.validate_selector_mode()?;
438        if !self.momentum_space {
439            self.validate_x_space_point(state, process_id, &integrand_name)?;
440        }
441        let graph_name = self.resolve_graph_name(state, process_id, &integrand_name)?;
442        let integrand_name_for_restore = integrand_name.clone();
443        self.with_optional_minimal_integrand_settings(
444            state,
445            process_id,
446            &integrand_name_for_restore,
447            |state| self.run_benchmark(state, process_id, integrand_name, graph_name),
448        )
449        .map(drop)
450    }
451
452    fn with_optional_minimal_integrand_settings<R>(
453        &self,
454        state: &mut State,
455        process_id: usize,
456        integrand_name: &str,
457        run: impl FnOnce(&mut State) -> Result<R>,
458    ) -> Result<R> {
459        if !self.minimal_integrand {
460            return run(state);
461        }
462
463        let original_settings = {
464            let integrand = state
465                .process_list
466                .get_integrand_mut(process_id, integrand_name)?;
467            let original_settings = integrand.get_settings().clone();
468            apply_minimal_integrand_settings(integrand.get_mut_settings());
469            original_settings
470        };
471
472        let result = run(state);
473        let integrand = state
474            .process_list
475            .get_integrand_mut(process_id, integrand_name)?;
476        *integrand.get_mut_settings() = original_settings;
477        result
478    }
479}

481fn apply_minimal_integrand_settings(settings: &mut RuntimeSettings) {
482    settings.general.enable_cache = false;
483    settings.general.debug_cache = false;
484    settings.general.generate_events = false;
485    settings.general.store_additional_weights_in_event = false;
486    settings.observables = Default::default();
487    settings.selectors = Default::default();
488    settings.stability.rotation_axis = Vec::new();
489    settings.stability.levels = vec![StabilityLevelSetting::default_double()];
490    settings.stability.check_on_norm = false;
491    settings.stability.escalate_if_exact_zero = false;
492    settings.stability.loop_momenta_norm_escalation_factor = -1.0;
493    settings.stability.recording = None;
494}

496fn parse_bench_target_duration(input: &str) -> Result<Duration> {
497    let trimmed = input.trim();
498    if trimmed.is_empty() {
499        return Err(eyre!("bench requires a non-empty duration"));
500    }
501
502    let lower = trimmed.to_ascii_lowercase();
503    let (number, multiplier) = if let Some(number) = lower.strip_suffix("microseconds") {
504        (number, 1.0e-6)
505    } else if let Some(number) = lower.strip_suffix("microsecond") {
506        (number, 1.0e-6)
507    } else if let Some(number) = lower.strip_suffix("usec") {
508        (number, 1.0e-6)
509    } else if let Some(number) = lower.strip_suffix("us") {
510        (number, 1.0e-6)
511    } else if let Some(number) = lower.strip_suffix("µs") {
512        (number, 1.0e-6)
513    } else if let Some(number) = lower.strip_suffix("milliseconds") {
514        (number, 1.0e-3)
515    } else if let Some(number) = lower.strip_suffix("millisecond") {
516        (number, 1.0e-3)
517    } else if let Some(number) = lower.strip_suffix("msec") {
518        (number, 1.0e-3)
519    } else if let Some(number) = lower.strip_suffix("ms") {
520        (number, 1.0e-3)
521    } else if let Some(number) = lower.strip_suffix("seconds") {
522        (number, 1.0)
523    } else if let Some(number) = lower.strip_suffix("second") {
524        (number, 1.0)
525    } else if let Some(number) = lower.strip_suffix("sec") {
526        (number, 1.0)
527    } else if let Some(number) = lower.strip_suffix('s') {
528        (number, 1.0)
529    } else {
530        (trimmed, 1.0)
531    };
532    let seconds = number
533        .trim()
534        .parse::<f64>()
535        .with_context(|| format!("Could not parse bench duration '{input}'"))?
536        * multiplier;
537    if !seconds.is_finite() || seconds <= 0.0 {
538        return Err(eyre!("bench duration must be positive and finite"));
539    }
540    Duration::try_from_secs_f64(seconds)
541        .map_err(|_| eyre!("bench duration is too large to represent"))
542}

544fn parse_positive_usize(input: &str) -> std::result::Result<usize, String> {
545    let value = input
546        .parse::<usize>()
547        .map_err(|err| format!("expected a positive integer: {err}"))?;
548    if value == 0 {
549        return Err("expected a positive integer".to_string());
550    }
551    Ok(value)
552}

562fn split_samples_into_batches(total_samples: usize, batches: usize) -> Vec<usize> {
563    let base = total_samples / batches;
564    let remainder = total_samples % batches;
565    (0..batches)
566        .map(|index| base + usize::from(index < remainder))
567        .collect()
568}

570fn bench_batch_timing(
571    total_timing: Duration,
572    batch: &gammalooprs::integrands::evaluation::BatchSampleEvaluationResult,
573) -> Result<BenchBatchTiming> {
574    let sample_count = batch.samples.len();
575    if sample_count == 0 {
576        return Err(eyre!("bench batch returned no samples"));
577    }
578
579    let mut parameterization = 0.0;
580    let mut integrand_inclusive = 0.0;
581    let mut event_processing = 0.0;
582    let mut evaluator = 0.0;
583    for sample in &batch.samples {
584        let metadata = sample
585            .evaluation
586            .evaluation_metadata
587            .as_ref()
588            .ok_or_else(|| eyre!("bench requires evaluation metadata"))?;
589        parameterization += metadata.parameterization_time.as_secs_f64();
590        integrand_inclusive += metadata.integrand_evaluation_time.as_secs_f64();
591        event_processing += metadata.event_processing_time.as_secs_f64();
592        evaluator += metadata.evaluator_evaluation_time.as_secs_f64();
593    }
594
595    let inv_sample_count = 1.0 / sample_count as f64;
596    let parameterization = parameterization * inv_sample_count;
597    let evaluator = evaluator * inv_sample_count;
598    // integrand_evaluation_time is inclusive of evaluator calls. Keep the
599    // displayed rows disjoint by showing the non-evaluator integrand residual.
600    let integrand = (integrand_inclusive * inv_sample_count - evaluator).max(0.0);
601    let event_processing = event_processing * inv_sample_count;
602    // Keep Total as the wall-clock time around evaluate_samples. The metadata
603    // rows below are subtracted from that wrapper timing so other/overhead
604    // captures unclassified evaluation work plus evaluate_samples overhead.
605    let total = total_timing.as_secs_f64() * inv_sample_count;
606    let known = parameterization + integrand + event_processing + evaluator;
607    let other = (total - known).max(0.0);
608
609    Ok(BenchBatchTiming {
610        parameterization,
611        integrand,
612        event_processing,
613        evaluator,
614        other,
615        total,
616    })
617}

619fn bench_summary_rows(batch_timings: &[BenchBatchTiming]) -> Vec<BenchSummaryRow> {
620    let rows = bench_rows(batch_timings);
621    let total_values = batch_timings
622        .iter()
623        .map(|timing| timing.total)
624        .collect::<Vec<_>>();
625    let (total_mean, total_standard_error) = mean_and_standard_error(&total_values);
626
627    let mut summary = rows
628        .into_iter()
629        .map(|row| {
630            let (mean, standard_error) = row.mean_and_standard_error();
631            BenchSummaryRow {
632                category: row.category.to_string(),
633                mean_seconds_per_sample: mean,
634                standard_error_seconds_per_sample: standard_error,
635                percentage_of_total: (total_mean > 0.0).then_some(mean / total_mean * 100.0),
636            }
637        })
638        .collect::<Vec<_>>();
639    summary.push(BenchSummaryRow {
640        category: "Total".to_string(),
641        mean_seconds_per_sample: total_mean,
642        standard_error_seconds_per_sample: total_standard_error,
643        percentage_of_total: (total_mean > 0.0).then_some(100.0),
644    });
645    summary
646}

658fn update_bench_progress(
659    progress_bar: &ProgressBar,
660    completed_batches: usize,
661    n_batches: usize,
662    batch_timings: &[BenchBatchTiming],
663    total_samples: usize,
664) {
665    let total_values = batch_timings
666        .iter()
667        .map(|timing| timing.total)
668        .collect::<Vec<_>>();
669    let (total_mean, total_standard_error) = mean_and_standard_error(&total_values);
670    let projected_total = Duration::try_from_secs_f64((total_mean * total_samples as f64).max(0.0))
671        .unwrap_or(Duration::MAX);
672    let message = format!(
673        "{} {} | {} {} | {} {}",
674        "batch".cyan(),
675        format!("{completed_batches}/{n_batches}").blue(),
676        "total/sample".cyan(),
677        format_timing_with_uncertainty(total_mean, total_standard_error).yellow(),
678        "projected total".cyan(),
679        format_duration_smart(projected_total).yellow(),
680    );
681    progress_bar.set_position(completed_batches as u64);
682    progress_bar.set_message(message);
683}

686    let mut rows = vec![
687        BenchRow {
688            category: "parameterization",
689            timings: batch_timings
690                .iter()
691                .map(|timing| timing.parameterization)
692                .collect(),
693        },
694        BenchRow {
695            category: "integrand",
696            timings: batch_timings
697                .iter()
698                .map(|timing| timing.integrand)
699                .collect(),
700        },
701        BenchRow {
702            category: "event processing",
703            timings: batch_timings
704                .iter()
705                .map(|timing| timing.event_processing)
706                .collect(),
707        },
708    ];
709
710    let total_values = batch_timings
711        .iter()
712        .map(|timing| timing.total)
713        .collect::<Vec<_>>();
714    let (total_mean, _) = mean_and_standard_error(&total_values);
715    let other_values = batch_timings
716        .iter()
717        .map(|timing| timing.other)
718        .collect::<Vec<_>>();
719    let (other_mean, _) = mean_and_standard_error(&other_values);
720    if other_mean
721        > (total_mean * BENCH_OTHER_RELATIVE_THRESHOLD).max(BENCH_OTHER_ABSOLUTE_THRESHOLD_SECONDS)
722    {
723        rows.push(BenchRow {
724            category: "other/overhead",
725            timings: other_values,
726        });
727    }
728
729    rows.push(BenchRow {
730        category: "evaluators",
731        timings: batch_timings
732            .iter()
733            .map(|timing| timing.evaluator)
734            .collect(),
735    });
736    rows
737}

739fn render_bench_table(batch_timings: &[BenchBatchTiming]) -> String {
740    let rows = bench_rows(batch_timings);
741    let total_values = batch_timings
742        .iter()
743        .map(|timing| timing.total)
744        .collect::<Vec<_>>();
745    let (total_mean, _) = mean_and_standard_error(&total_values);
746
747    let total_row = BenchRow {
748        category: "Total",
749        timings: total_values,
750    };
751    let total_record_index = rows.len() + 1;
752
753    let mut builder = Builder::new();
754    builder.push_record([
755        "category".bold().cyan().to_string(),
756        "time / sample".bold().cyan().to_string(),
757        "% total".bold().cyan().to_string(),
758    ]);
759    for row in &rows {
760        let (mean, standard_error) = row.mean_and_standard_error();
761        builder.push_record([
762            row.category.cyan().to_string(),
763            format_timing_with_uncertainty(mean, standard_error)
764                .yellow()
765                .to_string(),
766            format_percentage_cell(mean, total_mean).blue().to_string(),
767        ]);
768    }
769
770    let (total_mean, total_standard_error) = total_row.mean_and_standard_error();
771    builder.push_record([
772        total_row.category.bold().green().to_string(),
773        format_timing_with_uncertainty(total_mean, total_standard_error)
774            .bold()
775            .green()
776            .to_string(),
777        "100.0%".bold().green().to_string(),
778    ]);
779
780    let mut table = builder.build();
781    table.with(
782        Style::rounded().horizontals([
783            (
784                1,
785                HorizontalLine::new('─')
786                    .intersection('┼')
787                    .left('├')
788                    .right('┤'),
789            ),
790            (
791                total_record_index,
792                HorizontalLine::new('─')
793                    .intersection('┼')
794                    .left('├')
795                    .right('┤'),
796            ),
797        ]),
798    );
799    table.with(Modify::new(Rows::first()).with(Alignment::center()));
800    table.with(Modify::new(Rows::new(1..).intersect(Columns::one(0))).with(Alignment::left()));
801    table.with(Modify::new(Rows::new(1..).intersect(Columns::new(1..))).with(Alignment::right()));
802    table.to_string()
803}
804
805fn mean_and_standard_error(values: &[f64]) -> (f64, f64) {
806    if values.is_empty() {
807        return (0.0, 0.0);
808    }
809    let mean = values.iter().sum::<f64>() / values.len() as f64;
810    if values.len() < 2 {
811        return (mean, 0.0);
812    }
813    let variance = values
814        .iter()
815        .map(|value| (value - mean).powi(2))
816        .sum::<f64>()
817        / (values.len() - 1) as f64;
818    (mean, variance.sqrt() / (values.len() as f64).sqrt())
819}
820
821fn format_percentage_cell(value: f64, total: f64) -> String {
822    if total <= 0.0 {
823        return "-".to_string();
824    }
825    format!("{:.1}%", value / total * 100.0)
826}
827
828fn format_duration_smart(duration: Duration) -> String {
829    let (scaled, unit) = scale_seconds_for_display(duration.as_secs_f64());
830    format!("{} {unit}", format_significant(scaled, 3))
831}
832
833fn format_timing_with_uncertainty(mean_seconds: f64, standard_error_seconds: f64) -> String {
834    let (scaled_mean, unit) = scale_seconds_for_display(mean_seconds);
835    let scale = unit_scale(unit);
836    let scaled_error = standard_error_seconds * scale;
837    format!(
838        "{}±{} {unit}",
839        format_significant(scaled_mean, 3),
840        format_significant(scaled_error, 2)
841    )
842}
843
844fn scale_seconds_for_display(seconds: f64) -> (f64, &'static str) {
845    let abs_seconds = seconds.abs();
846    if abs_seconds < 1.0e-6 {
847        (seconds * 1.0e9, "ns")
848    } else if abs_seconds < 1.0e-3 {
849        (seconds * 1.0e6, "µs")
850    } else if abs_seconds < 1.0 {
851        (seconds * 1.0e3, "ms")
852    } else {
853        (seconds, "s")
854    }
855}
856
857fn unit_scale(unit: &str) -> f64 {
858    match unit {
859        "ns" => 1.0e9,
860        "µs" => 1.0e6,
861        "ms" => 1.0e3,
862        "s" => 1.0,
863        _ => 1.0,
864    }
865}
866
867fn format_significant(value: f64, significant_digits: usize) -> String {
868    if !value.is_finite() {
869        return value.to_string();
870    }
871    if value == 0.0 {
872        return format!("{:.*}", significant_digits.saturating_sub(1), 0.0);
873    }
874    let decimals = (significant_digits as i32 - 1 - value.abs().log10().floor() as i32).max(0);
875    format!("{:.*}", decimals as usize, value)
876}
