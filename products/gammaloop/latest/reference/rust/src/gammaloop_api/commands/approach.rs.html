288    pub fn run(&self, state: &mut State, cli_settings: &CLISettings) -> Result<PathBuf> {
289        self.validate_cli()?;
290        let (process_id, integrand_name) =
291            state.find_integrand_ref(self.process.as_ref(), self.integrand_name.as_ref())?;
292        let process_name = state.process_list.processes[process_id]
293            .definition
294            .folder_name
295            .clone();
296        let model = state.resolve_model_for_integrand(process_id, &integrand_name)?;
297        let base_integrand = state
298            .process_list
299            .get_integrand(process_id, &integrand_name)?
300            .require_generated()?
301            .clone();
302
303        if !self.momentum_space && (self.graph_id.is_some() || self.orientation_id.is_some()) {
304            return Err(eyre!(
305                "Graph and orientation selectors are only supported in momentum-space approach."
306            ));
307        }
308        if !self.momentum_space {
309            let expected_dimension =
310                base_integrand.expected_x_space_dimension(&self.discrete_dim)?;
311            if self.point.len() != expected_dimension {
312                return Err(eyre!(
313                    "Expected {} x-space coordinates for this integrand selection, got {}.",
314                    expected_dimension,
315                    self.point.len()
316                ));
317            }
318        }
319
320        let graph_name = self.resolve_graph_name(&base_integrand, &integrand_name)?;
321        let axes = self.parse_axes()?;
322        let spacing = self.spacing()?;
323        let t_values = self.t_values(spacing)?;
324        let jobs = self.build_jobs(&axes, &t_values);
325        let n_cores = self
326            .n_cores
327            .unwrap_or(cli_settings.global.n_cores.integrate);
328        if n_cores == 0 {
329            return Err(eyre!("--n-cores must be at least 1."));
330        }
331
332        let total_jobs = jobs.len();
333        let progress = approach_progress_bar(total_jobs as u64);
334        let pool = ThreadPoolBuilder::new()
335            .num_threads(n_cores)
336            .build()
337            .context("failed to build approach evaluation thread pool")?;
338        let worker_count = n_cores.min(total_jobs.max(1));
339        let worker_inputs = (0..worker_count)
340            .map(|worker_index| {
341                let worker_jobs = jobs
342                    .iter()
343                    .skip(worker_index)
344                    .step_by(worker_count)
345                    .cloned()
346                    .collect::<Vec<_>>();
347                let mut integrand = base_integrand.clone();
348                force_event_output(&mut integrand);
349                (integrand, model.clone(), worker_jobs)
350            })
351            .collect::<Vec<_>>();
352        let completed_jobs = Arc::new(AtomicUsize::new(0));
353        let completed_workers = Arc::new(AtomicUsize::new(0));
354        let results = pool.install(|| {
355            worker_inputs
356                .into_par_iter()
357                .map(
358                    |(mut integrand, model, worker_jobs)| -> Result<Vec<ApproachPointRecord>> {
359                        integrand
360                            .warm_up(&model)
361                            .context("failed to warm up approach worker integrand")?;
362                        let mut records = Vec::with_capacity(worker_jobs.len());
363                        for job in worker_jobs {
364                            let record = self.evaluate_job(
365                                &job,
366                                &mut integrand,
367                                &model,
368                                graph_name.as_deref(),
369                                axes.len(),
370                                t_values.len(),
371                            );
372                            let completed_total =
373                                completed_jobs.fetch_add(1, Ordering::Relaxed) + 1;
374                            progress.inc(1);
375                            progress.set_message(progress_message(
376                                job.axis_index,
377                                axes.len(),
378                                job.axis_point_index,
379                                t_values.len(),
380                                completed_total,
381                                total_jobs,
382                            ));
383                            records.push(record?);
384                        }
385                        let completed_worker_count =
386                            completed_workers.fetch_add(1, Ordering::Relaxed) + 1;
387                        progress.set_message(format!(
388                            "{} {}",
389                            "collecting worker results".bright_magenta().bold(),
390                            format!("{completed_worker_count:>2}/{worker_count:<2}")
391                                .bright_white()
392                                .bold()
393                        ));
394                        if completed_worker_count == worker_count {
395                            progress.println(
396                                "Approach evaluations complete; collecting worker results"
397                                    .bright_blue()
398                                    .bold()
399                                    .to_string(),
400                            );
401                        }
402                        Ok(records)
403                    },
404                )
405                .collect::<Result<Vec<_>>>()
406        });
407        progress.finish_with_message(
408            "approach evaluations complete; finalizing output"
409                .bright_green()
410                .bold()
411                .to_string(),
412        );
413
414        info!(
415            "{} {}",
416            "Finalizing approach results".bright_blue().bold(),
417            "(sorting records and preparing JSON)".bright_black()
418        );
419        let mut point_records = results?
420            .into_iter()
421            .flatten()
422            .collect::<Vec<ApproachPointRecord>>();
423        point_records.sort_by_key(|record| record.index);
424
425        let evaluated_points = point_records
426            .iter()
427            .filter(|record| record.status == "evaluated")
428            .count();
429        let skipped_points = point_records.len() - evaluated_points;
430        let output = ApproachOutput {
431            schema_version: 1,
432            command: ApproachCommandJson {
433                name: "approach",
434                use_arb_prec: self.use_arb_prec,
435                graph_id: self.graph_id,
436                orientation_id: self.orientation_id,
437                discrete_dim: self.discrete_dim.clone(),
438            },
439            process: ApproachProcessJson {
440                id: process_id,
441                name: process_name,
442            },
443            integrand: ApproachIntegrandJson {
444                name: integrand_name,
445                kind: base_integrand.kind_name().to_string(),
446            },
447            space: if self.momentum_space {
448                "momentum".to_string()
449            } else {
450                "coordinate".to_string()
451            },
452            base_point: self.point.clone(),
453            axes,
454            spacing: ApproachSpacingJson {
455                kind: spacing,
456                n_points: self.n_points,
457                min_abs_t: matches!(spacing, ApproachSpacing::Logarithmic)
458                    .then_some(self.min_abs_t),
459                t_values,
460            },
461            n_cores,
462            points_per_axis: 2 * self.n_points + 1,
463            evaluated_points,
464            skipped_points,
465            points: point_records,
466        };
467
468        let output_path = self
469            .output_results
470            .clone()
471            .unwrap_or_else(|| PathBuf::from("approach_result.json"));
472        let relative_output = relative_path_display(&output_path);
473        info!(
474            "{} {}",
475            "Writing approach JSON to".bright_blue().bold(),
476            relative_output.bright_cyan()
477        );
478        if let Some(parent) = output_path.parent() {
479            if !parent.as_os_str().is_empty() {
480                fs::create_dir_all(parent).with_context(|| {
481                    format!(
482                        "failed to create approach output directory '{}'",
483                        parent.display()
484                    )
485                })?;
486            }
487        }
488        let file = fs::File::create(&output_path).with_context(|| {
489            format!(
490                "failed to create approach output file '{}'",
491                output_path.display()
492            )
493        })?;
494        serde_json::to_writer_pretty(file, &output).with_context(|| {
495            format!(
496                "failed to serialize approach output to '{}'",
497                output_path.display()
498            )
499        })?;
500
501        let pdf_path = output_path.with_extension("pdf");
502        let relative_pdf = relative_path_display(&pdf_path);
503        let plot_command = format!(
504            "python3 assets/plot_approach_result.py {} --output {}",
505            shell_quote(&relative_output),
506            shell_quote(&relative_pdf)
507        );
508        info!(
509            "{} {}",
510            "Approach results written to".bright_green().bold(),
511            relative_output.bright_cyan()
512        );
513        info!(
514            "{} {}",
515            "Plot with:".bright_green().bold(),
516            plot_command.bright_cyan()
517        );
518
519        Ok(output_path)
520    }

522    fn validate_cli(&self) -> Result<()> {
523        if self.point.is_empty() {
524            return Err(eyre!("approach requires a midpoint supplied with --point."));
525        }
526        if self.approach_axes.is_empty() {
527            return Err(eyre!(
528                "approach requires at least one --approach-axis value."
529            ));
530        }
531        if self.n_points == 0 {
532            return Err(eyre!("--n-points must be at least 1."));
533        }
534        if self.logarithmic && !(self.min_abs_t > 0.0 && self.min_abs_t <= 1.0) {
535            return Err(eyre!(
536                "--min-abs-t must be greater than 0 and at most 1 for logarithmic spacing."
537            ));
538        }
539        if self.n_cores == Some(0) {
540            return Err(eyre!("--n-cores must be at least 1."));
541        }
542        Ok(())
543    }

545    fn spacing(&self) -> Result<ApproachSpacing> {
546        if self.linear && self.logarithmic {

548        }

550            ApproachSpacing::Logarithmic
551        } else {
552            ApproachSpacing::Linear
553        })

582    fn t_values(&self, spacing: ApproachSpacing) -> Result<Vec<f64>> {
583        let magnitudes = match spacing {
584            ApproachSpacing::Linear => (1..=self.n_points)
585                .map(|index| index as f64 / self.n_points as f64)
586                .collect::<Vec<_>>(),
587            ApproachSpacing::Logarithmic => {
588                if self.n_points == 1 {
589                    vec![1.0]
590                } else {
591                    let log_min = self.min_abs_t.ln();
592                    let denom = (self.n_points - 1) as f64;
593                    (0..self.n_points)
594                        .map(|index| (log_min * (1.0 - index as f64 / denom)).exp())
595                        .collect()
596                }
597            }
598        };
599        let mut t_values = magnitudes
600            .iter()
601            .rev()
602            .map(|value| -*value)
603            .collect::<Vec<_>>();
604        t_values.push(0.0);
605        t_values.extend(magnitudes);
606        Ok(t_values)
607    }

609    fn build_jobs(&self, axes: &[Vec<f64>], t_values: &[f64]) -> Vec<ApproachJob> {
610        let mut jobs = Vec::with_capacity(axes.len() * t_values.len());
611        for (axis_index, axis) in axes.iter().enumerate() {
612            for (axis_point_index, t) in t_values.iter().copied().enumerate() {
613                let point = self
614                    .point
615                    .iter()
616                    .zip(axis)
617                    .map(|(center, direction)| center + t * direction)
618                    .collect::<Vec<_>>();
619                let skip_reason = (!self.momentum_space)
620                    .then(|| coordinate_skip_reason(&point))
621                    .flatten();
622                jobs.push(ApproachJob {
623                    index: jobs.len(),
624                    axis_index,
625                    axis_point_index,
626                    t,
627                    point,
628                    skip_reason,
629                });
630            }
631        }
632        jobs
633    }

635    fn evaluate_job(
636        &self,
637        job: &ApproachJob,
638        integrand: &mut ProcessIntegrand,
639        model: &Model,
640        graph_name: Option<&str>,
641        _axis_count: usize,
642        _axis_point_count: usize,
643    ) -> Result<ApproachPointRecord> {
644        if let Some(skip_reason) = &job.skip_reason {
645            return Ok(ApproachPointRecord {
646                index: job.index,
647                axis_index: job.axis_index,
648                axis_point_index: job.axis_point_index,
649                t: job.t,
650                point: job.point.clone(),
651                status: "skipped",
652                skip_reason: Some(skip_reason.clone()),
653                evaluation: None,
654            });
655        }
656
657        let mut samples = if self.momentum_space {
658            let input = build_momentum_input(
659                integrand,
660                &job.point,
661                1.0,
662                &self.discrete_dim,
663                graph_name,
664                self.orientation_id,
665            )?;
666            integrand
667                .evaluate_momentum_configurations_raw(model, &[input], self.use_arb_prec)?
668                .samples
669        } else {
670            let sample = build_havana_sample(integrand, &job.point, &self.discrete_dim, 1.0)?;
671            integrand
672                .evaluate_samples_raw(
673                    model,
674                    &[sample],
675                    1,
676                    self.use_arb_prec,
677                    false,
678                    Complex::new_zero(),
679                )?
680                .samples
681        };
682        let evaluation = samples
683            .pop()
684            .ok_or_else(|| eyre!("approach evaluation did not return a sample"))?;
685        Ok(ApproachPointRecord {
686            index: job.index,
687            axis_index: job.axis_index,
688            axis_point_index: job.axis_point_index,
689            t: job.t,
690            point: job.point.clone(),
691            status: "evaluated",
692            skip_reason: None,
693            evaluation: Some(approach_evaluation_record(integrand, evaluation)?),
694        })
695    }

698fn parse_axis(raw_axis: &str, expected_dimension: usize) -> Result<Vec<f64>> {
699    let trimmed = raw_axis
700        .trim()
701        .trim_start_matches('[')
702        .trim_end_matches(']');
703    let axis = trimmed
704        .split(',')
705        .map(str::trim)
706        .filter(|component| !component.is_empty())
707        .map(|component| {
708            component.parse::<f64>().map_err(|err| {
709                eyre!(
710                    "Could not parse approach-axis component '{}' as a float: {}",
711                    component,
712                    err
713                )
714            })
715        })
716        .collect::<Result<Vec<_>>>()?;
717    if axis.len() != expected_dimension {
718        return Err(eyre!(
719            "Approach axis has {} components, but the point has {} components.",
720            axis.len(),
721            expected_dimension
722        ));
723    }
724    if axis.iter().all(|component| *component == 0.0) {
725        return Err(eyre!("Approach axis must not be the zero vector."));
726    }
727    Ok(axis)
728}

730fn coordinate_skip_reason(point: &[f64]) -> Option<String> {
731    let invalid = point
732        .iter()
733        .enumerate()
734        .filter(|(_, value)| !(0.0..=1.0).contains(*value))
735        .map(|(index, value)| format!("{index}:{value:+.16e}"))
736        .collect::<Vec<_>>();
737    (!invalid.is_empty()).then(|| {
738        format!(
739            "coordinate-space point lies outside the unit hypercube at component(s) {}",
740            invalid.join(", ")
741        )
742    })
743}

751fn approach_evaluation_record(
752    integrand: &ProcessIntegrand,
753    evaluation: EvaluationResult,
754) -> Result<ApproachEvaluationRecord> {
755    let parameterization_jacobian = evaluation.parameterization_jacobian.map(|value| value.0);
756    let integrator_weight = evaluation.integrator_weight.0;
757    let total_scale = parameterization_jacobian.unwrap_or(1.0) * integrator_weight;
758    let total_weight = Complex::new(
759        F(evaluation.integrand_result.re.0 * total_scale),
760        F(evaluation.integrand_result.im.0 * total_scale),
761    );
762
763    let mut event_weight_sum = Complex::new(F(0.0), F(0.0));
764    let mut additional_weight_sums = BTreeMap::<String, Complex<F<f64>>>::new();
765    let mut contribution_sums = BTreeMap::<ContributionKey, Complex<F<f64>>>::new();
766    let mut events = Vec::new();
767    let parameterization_settings = integrand
768        .get_settings()
769        .sampling
770        .get_parameterization_settings()
771        .unwrap_or_default();
772
773    for (event_group_index, event_group) in evaluation.event_groups.iter().enumerate() {
774        for (event_index, event) in event_group.iter().enumerate() {
775            event_weight_sum += event.weight;
776            let graph_id = event.cut_info.graph_id;
777            let graph_name = integrand
778                .graph_name_by_id(graph_id)
779                .map(ToString::to_string);
780            let graph_group_id = integrand.graph_group_id_by_graph_id(graph_id);
781            let cut_id = event.cut_info.cut_id;
782            let cut_edges = integrand.cut_edge_ids(graph_id, cut_id).unwrap_or_default();
783            let lmb_channel_id = event.cut_info.lmb_channel_id;
784            let lmb_sample_id = lmb_channel_id
785                .map(|channel_id| {
786                    integrand.lmb_sample_id_for_channel(
787                        graph_id,
788                        channel_id,
789                        &parameterization_settings,
790                    )
791                })
792                .transpose()?
793                .flatten();
794
795            let event_key = ContributionKey {
796                contribution: "event_weight".to_string(),
797                graph_id,
798                graph_name: graph_name.clone(),
799                graph_group_id,
800                orientation_id: event.cut_info.orientation_id,
801                cut_id,
802                cut_edges: cut_edges.clone(),
803                lmb_sample_id,
804            };
805            add_contribution(&mut contribution_sums, event_key, event.weight);
806
807            let mut additional_weights = BTreeMap::new();
808            for (key, value) in &event.additional_weights.weights {
809                let label = additional_weight_key_label(*key);
810                *additional_weight_sums
811                    .entry(label.clone())
812                    .or_insert_with(zero_complex) += *value;
813                additional_weights.insert(label.clone(), complex_json(*value));
814                let contribution_key = ContributionKey {
815                    contribution: label,
816                    graph_id,
817                    graph_name: graph_name.clone(),
818                    graph_group_id,
819                    orientation_id: event.cut_info.orientation_id,
820                    cut_id,
821                    cut_edges: cut_edges.clone(),
822                    lmb_sample_id,
823                };
824                add_contribution(&mut contribution_sums, contribution_key, *value);
825            }
826
827            events.push(EventRecord {
828                event_group_index,
829                event_index,
830                graph_id,
831                graph_name,
832                graph_group_id,
833                orientation_id: event.cut_info.orientation_id,
834                cut_id,
835                cut_edges,
836                lmb_channel_id,
837                lmb_sample_id,
838                weight: complex_json(event.weight),
839                additional_weights,
840            });
841        }
842    }
843
844    Ok(ApproachEvaluationRecord {
845        integrand_result: complex_json(evaluation.integrand_result),
846        parameterization_jacobian,
847        integrator_weight,
848        total_weight: complex_json(total_weight),
849        event_weight_sum: complex_json(event_weight_sum),
850        additional_weight_sums: additional_weight_sums
851            .into_iter()
852            .map(|(key, value)| (key, complex_json(value)))
853            .collect(),
854        contributions: contribution_sums
855            .into_iter()
856            .map(|(key, weight)| contribution_record(key, weight))
857            .collect(),
858        events,
859        metadata: EvaluationMetadataJson {
860            generated_event_count: evaluation.evaluation_metadata.generated_event_count,
861            accepted_event_count: evaluation.evaluation_metadata.accepted_event_count,
862            is_nan: evaluation.evaluation_metadata.is_nan,
863            total_time_seconds: evaluation.evaluation_metadata.total_timing.as_secs_f64(),
864            parameterization_time_seconds: evaluation
865                .evaluation_metadata
866                .parameterization_time
867                .as_secs_f64(),
868            integrand_evaluation_time_seconds: evaluation
869                .evaluation_metadata
870                .integrand_evaluation_time
871                .as_secs_f64(),
872            evaluator_evaluation_time_seconds: evaluation
873                .evaluation_metadata
874                .evaluator_evaluation_time
875                .as_secs_f64(),
876            event_processing_time_seconds: evaluation
877                .evaluation_metadata
878                .event_processing_time
879                .as_secs_f64(),
880        },
881    })
882}

884fn add_contribution(
885    contribution_sums: &mut BTreeMap<ContributionKey, Complex<F<f64>>>,
886    key: ContributionKey,
887    value: Complex<F<f64>>,
888) {
889    *contribution_sums.entry(key).or_insert_with(zero_complex) += value;
890}

892fn contribution_record(key: ContributionKey, weight: Complex<F<f64>>) -> ContributionRecord {
893    ContributionRecord {
894        label: contribution_label(&key),
895        contribution: key.contribution,
896        graph_id: key.graph_id,
897        graph_name: key.graph_name,
898        graph_group_id: key.graph_group_id,
899        orientation_id: key.orientation_id,
900        cut_id: key.cut_id,
901        cut_edges: key.cut_edges,
902        lmb_sample_id: key.lmb_sample_id,
903        weight: complex_json(weight),
904    }
905}
906
907fn contribution_label(key: &ContributionKey) -> String {
908    let graph = key
909        .graph_name
910        .as_deref()
911        .map(ToString::to_string)
912        .unwrap_or_else(|| format!("#{}", key.graph_id));
913    let edge_label = if key.cut_edges.is_empty() {
914        "[]".to_string()
915    } else {
916        format!(
917            "[{}]",
918            key.cut_edges
919                .iter()
920                .map(ToString::to_string)
921                .collect::<Vec<_>>()
922                .join(",")
923        )
924    };
925    let mut parts = vec![
926        key.contribution.clone(),
927        format!("graph={graph}"),
928        format!("cut={} edges={edge_label}", key.cut_id),
929    ];
930    if let Some(orientation_id) = key.orientation_id {
931        parts.push(format!("orientation={orientation_id}"));
932    }
933    if let Some(lmb_sample_id) = key.lmb_sample_id {
934        parts.push(format!("lmb_sample={lmb_sample_id}"));
935    }
936    parts.join(" ")
937}

939fn additional_weight_key_label(key: AdditionalWeightKey) -> String {
940    match key {
941        AdditionalWeightKey::FullMultiplicativeFactor => "full_multiplicative_factor".to_string(),
942        AdditionalWeightKey::Original => "original".to_string(),
943        AdditionalWeightKey::ThresholdCounterterm { subset_index } => {
944            format!("threshold_counterterm_{subset_index}")
945        }
946        AdditionalWeightKey::AmplitudeThresholdCounterterm {
947            esurface_id,
948            overlap_group,
949        } => format!("ct_{esurface_id}_{overlap_group}"),
950    }
951}

1012        .chars()
1013        .all(|ch| ch.is_ascii_alphanumeric() || matches!(ch, '/' | '.' | '_' | '-' | '='))

1016    } else {
1017        format!("'{}'", value.replace('\'', "'\\''"))
