51impl Import {
52    pub fn run(self, state: &mut State, cli_settings: &CLISettings) -> Result<()> {
53        match self {
54            Import::Graphs {
55                path,
56                process,
57                integrand_name,
58                overwrite,
59                append,
60            } => {
61                let resolved_path =
62                    Self::resolve_graph_import_path(&path, &cli_settings.state.folder)?;
63                let default_process_name = resolved_path
64                    .file_stem()
65                    .ok_or_else(|| {
66                        eyre!(
67                            "Could not derive a process name from graph path '{}'",
68                            resolved_path.display()
69                        )
70                    })?
71                    .to_string_lossy()
72                    .into_owned();
73                let (process_name, process_id) = match process {
74                    Some(ProcessRef::Id(id)) => (None, Some(id)),
75                    Some(ProcessRef::Name(name)) => (Some(name), None),
76                    Some(ProcessRef::Unqualified(value)) => {
77                        let name_match = state
78                            .process_list
79                            .processes
80                            .iter()
81                            .position(|p| p.definition.folder_name == value);
82                        if let Ok(id) = value.parse::<usize>() {
83                            if name_match.is_some() {
84                                return Err(eyre!(
85                                    "Ambiguous process reference '{}'. Use '#{}' or 'name:{}' to disambiguate.",
86                                    value,
87                                    id,
88                                    value
89                                ));
90                            }
91                            (None, Some(id))
92                        } else {
93                            (Some(value), None)
94                        }
95                    }
96                    None => (Some(default_process_name), None),
97                };
98
99                info!(
100                    "Loading graphs from '{}'",
101                    Self::display_graph_import_path(&resolved_path).display()
102                );
103                let graphs = Graph::from_path(&resolved_path, &state.model)?;
104                state.import_graphs(
105                    graphs,
106                    process_name,
107                    process_id,
108                    integrand_name,
109                    overwrite,
110                    append,
111                )
112            }
113            Import::Model(im) => im.run(state),
114        }
115    }

117    fn resolve_graph_import_path(path: &Path, state_folder: &Path) -> Result<PathBuf> {
118        let cwd = env::current_dir().wrap_err(
119            "Failed to query the current working directory while resolving graph import path",
120        )?;
121
122        if path.is_absolute() {
123            let normalized = Self::normalize_path_lexically(path);
124            return normalized
125                .exists()
126                .then_some(normalized)
127                .ok_or_else(|| eyre!("Graph file '{}' does not exist.", path.display()));
128        }
129
130        let state_root = state_folder.parent().unwrap_or(state_folder);
131        let state_root_candidate;
132        let absolute_state_root = Self::normalize_path_lexically(if state_root.is_absolute() {
133            state_root
134        } else {
135            state_root_candidate = cwd.join(state_root);
136            &state_root_candidate
137        });
138        let state_candidate = Self::normalize_path_lexically(&absolute_state_root.join(path));
139        if state_candidate.exists() {
140            return Ok(state_candidate);
141        }
142
143        let cwd_candidate = Self::normalize_path_lexically(&cwd.join(path));
144        if cwd_candidate.exists() {
145            return Ok(cwd_candidate);
146        }
147
148        Err(eyre!(
149            "Could not find graph file '{}'. Tried '{}' (active state root) and '{}' (current working directory).",
150            path.display(),
151            state_candidate.display(),
152            cwd_candidate.display()
153        ))
154    }
155
156    fn display_graph_import_path(path: &Path) -> PathBuf {
157        path.canonicalize()
158            .unwrap_or_else(|_| Self::normalize_path_lexically(path))
159    }
160
161    fn normalize_path_lexically(path: &Path) -> PathBuf {
162        use std::path::Component;
163
164        let mut normalized = PathBuf::new();
165        for component in path.components() {
166            match component {
167                Component::CurDir => {}
168                Component::ParentDir => {
169                    if normalized.components().next_back().is_some_and(|last| {
170                        !matches!(last, Component::RootDir | Component::Prefix(_))
171                    }) {
172                        normalized.pop();
173                    } else if !path.is_absolute() {
174                        normalized.push(component.as_os_str());
175                    }
176                }
177                Component::Normal(part) => normalized.push(part),
178                Component::RootDir | Component::Prefix(_) => {
179                    normalized.push(component.as_os_str());
180                }
181            }
182        }
183        normalized
184    }
