62#[derive(Debug, Clone)]
63struct KindedTemplate<T> {
64    kind: String,
65    settings: T,
66    root: JsonValue,
67}
68
69fn build_kinded_template<T>(settings: T, tag_key: &str, context: &str) -> KindedTemplate<T>
70where
71    T: Clone + Serialize,
72{
73    let root = serialize_settings_with_defaults(&settings, context)
74        .expect("template settings must serialize with defaults");
75    let kind = root
76        .get(tag_key)
77        .and_then(JsonValue::as_str)
78        .unwrap_or_else(|| panic!("missing tag '{tag_key}' in {context}"))
79        .to_string();
80    KindedTemplate {
81        kind,
82        settings,
83        root,
84    }
85}

309fn serialize_named_settings_map<T>(
310    map: &BTreeMap<String, T>,
311    context: &str,
312) -> Result<BTreeMap<String, JsonValue>>
313where
314    T: Serialize,
315{
316    map.iter()
317        .map(|(name, settings)| {
318            serialize_settings_with_defaults(settings, context).map(|root| (name.clone(), root))
319        })
320        .collect()
321}

348pub(crate) fn summarize_quantity(settings: &QuantitySettings) -> String {
349    match settings {
350        QuantitySettings::Particle(settings) => summarize_particle_quantity(settings),
351        QuantitySettings::Jet(settings) => summarize_jet_quantity(settings),
352        QuantitySettings::AFB {} => "forward/backward asymmetry".to_string(),

354        QuantitySettings::GraphId {} => "graph id metadata".to_string(),
355        QuantitySettings::GraphGroupId {} => "graph-group id metadata".to_string(),

361fn summarize_particle_quantity(settings: &ParticleQuantitySettings) -> String {
362    let QuantitySettings::Particle(settings) =
363        QuantitySettings::Particle(settings.clone()).normalized()
364    else {
365        unreachable!("particle quantity normalization must preserve its variant");
366    };
367    format!(
368        "{} pdgs={}",
369        summarize_quantity_computation(&settings.computation),
370        format_integer_list(&settings.pdgs)
371    )
372}
373
374fn summarize_jet_quantity(settings: &JetQuantitySettings) -> String {
375    let QuantitySettings::Jet(settings) = QuantitySettings::Jet(settings.clone()).normalized()
376    else {
377        unreachable!("jet quantity normalization must preserve its variant");
378    };
379    format!(
380        "{} algorithm={:?} dR={} min_jpt={} clustered_pdgs={}",
381        summarize_quantity_computation(&settings.computation),
382        settings.clustering.algorithm,
383        settings.clustering.dR,
384        settings.clustering.min_jpt,
385        settings
386            .clustering
387            .clustered_pdgs
388            .as_ref()
389            .map(|pdgs| format!("{pdgs:?}"))
390            .unwrap_or_else(|| "None (model default)".to_string())
391    )
392}

394fn summarize_quantity_computation(settings: &QuantityComputationSettings) -> String {
395    match settings.computation {
396        QuantityComputation::Scalar => format!(
397            "computation=scalar quantity={} ordering={} order={}",
398            settings.quantity.unwrap_or(FilterQuantity::PT),
399            settings.ordering.unwrap_or(QuantityOrdering::Quantity),
400            settings.order
401        ),
402        QuantityComputation::Count => "computation=count".to_string(),
403        QuantityComputation::Pair => format!(
404            "computation=pair pair_quantity={} pairing={} ordering={} order={}",
405            settings.pair_quantity.unwrap_or(PairQuantity::DeltaR),
406            settings.pairing.unwrap_or_default(),
407            settings.ordering.unwrap_or(QuantityOrdering::Quantity),
408            settings.order
409        ),
410    }
411}

413pub(crate) fn summarize_observable(settings: &ObservableSettings) -> String {
414    let selection_summary = if settings.selections.is_empty() {
415        "observable_selections=[]".to_string()
416    } else {
417        format!("observable_selections={:?}", settings.selections)
418    };
419    let histogram_summary = match &settings.histogram {
420        HistogramSettings::Continuous(histogram) => format!(
421            "continuous bins={} range=[{}, {}] log_x_axis={} title={} type_description={}",
422            histogram.n_bins,
423            histogram.x_min,
424            histogram.x_max,
425            histogram.log_x_axis,
426            histogram
427                .title
428                .clone()
429                .unwrap_or_else(|| "<observable name>".to_string()),
430            histogram.type_description,
431        ),
432        HistogramSettings::Discrete(histogram) => format!(
433            "discrete domain={:?} ordering={:?} labels={:?} title={} type_description={}",
434            histogram.domain,
435            histogram.ordering,
436            histogram.labels,
437            histogram
438                .title
439                .clone()
440                .unwrap_or_else(|| "<observable name>".to_string()),
441            histogram.type_description,
442        ),
443    };
444    format!(
445        "quantity={} selection={} {} transform={:?} phase={:?} {}",
446        format_reference(&settings.quantity),
447        format_entry_selection(settings.entry_selection, settings.entry_index),
448        selection_summary,
449        settings.value_transform,
450        settings.phase,
451        histogram_summary,
452    )
453}

455pub(crate) fn summarize_selector(settings: &SelectorSettings) -> String {
456    let selector_summary = match &settings.selector {
457        SelectorDefinitionSettings::ValueRange(selector) => format!(
458            "range=[{}, {}] reduction={:?}",
459            selector
460                .min
461                .map(|value| value.to_string())
462                .unwrap_or_else(|| "-inf".to_string()),
463            selector
464                .max
465                .map(|value| value.to_string())
466                .unwrap_or_else(|| "inf".to_string()),
467            selector.reduction
468        ),
469        SelectorDefinitionSettings::DiscreteRange(selector) => format!(
470            "discrete_range=[{}, {}]",
471            selector
472                .min
473                .map(|value| value.to_string())
474                .unwrap_or_else(|| "-inf".to_string()),
475            selector
476                .max
477                .map(|value| value.to_string())
478                .unwrap_or_else(|| "inf".to_string()),
479        ),
480        SelectorDefinitionSettings::CountRange(selector) => format!(
481            "count=[{}, {}]",
482            selector.min_count,
483            selector
484                .max_count
485                .map(|value| value.to_string())
486                .unwrap_or_else(|| "inf".to_string())
487        ),
488    };
489
490    format!(
491        "quantity={} active={} selection={} {selector_summary}",
492        format_reference(&settings.quantity),
493        settings.active,
494        format_entry_selection(settings.entry_selection, settings.entry_index),
495    )
496}

498fn format_entry_selection(selection: EntrySelection, entry_index: usize) -> String {
499    match selection {
500        EntrySelection::All => "all".to_string(),
501        EntrySelection::LeadingOnly => "leading_only".to_string(),
502        EntrySelection::NthOnly => format!("nth_only[{entry_index}]"),
503    }
504}
505
506fn format_reference(name: &str) -> String {
507    if name.is_empty() {
508        "<unset>".to_string()
509    } else {
510        name.to_string()
511    }
512}
513
514fn format_integer_list(values: &[isize]) -> String {
515    if values.is_empty() {
516        "[]".to_string()
517    } else {
518        format!(
519            "[{}]",
520            values
521                .iter()
522                .map(|value| value.to_string())
523                .collect::<Vec<_>>()
524                .join(",")
525        )
526    }
