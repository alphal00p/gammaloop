22/// Evaluate one or more integrand inputs through the ordinary `f64` result boundary.
23///
24/// Each row of [`Self::points`] has one of two layouts:
25///
26/// - with [`Self::momentum_space`] set to `false`, the row contains integration-space
27///   coordinates and must have the dimension selected by the integrand and
28///   [`Self::discrete_dims`];
29/// - with [`Self::momentum_space`] set to `true`, the row contains exactly one
30///   `(px, py, pz)` triplet per independent loop momentum. The flattened order is
31///   `[k1x, k1y, k1z, k2x, k2y, k2z, ...]`. Energies and external momenta are not
32///   inputs to this interface.
33///
34/// A momentum-space row whose length is not a multiple of three is rejected before
35/// evaluation. The selected integrand, graph, or graph group determines the required
36/// number of loop momenta and rejects the wrong number of complete triplets.
37///
38/// [`Self::use_arb_prec`] is a legacy compatibility flag: it selects the configured
39/// `f128` stability level, falling back to arbitrary precision only when no `f128`
40/// level exists. It does not change this type's `f64` output contract. Use
41/// [`EvaluateSamplesPrecise`] when the caller must retain an `f64`, `f128`, or
42/// arbitrary-precision result.
43///
44/// The [GammaLoop sample-evaluation contract](https://alphal00p.github.io/gammaloop/products/gammaloop/latest/reference/interfaces/#sample-evaluation-contract)
45/// connects this request to the CLI and Python interfaces and to the maintained
46/// checked example.
47///
48/// # Example
49///
50/// ```rust
51/// use gammaloop_api::commands::evaluate_samples::EvaluateSamples;
52/// use ndarray::arr2;
53///
54/// let loop_momenta = arr2(&[[
55///     0.11, -0.07, 0.19, // k1 = (px, py, pz)
56///     -0.13, 0.05, 0.29, // k2 = (px, py, pz)
57/// ]]);
58/// let request = EvaluateSamples {
59///     process_id: Some(0),
60///     integrand_name: None,
61///     use_arb_prec: true,
62///     minimal_output: true,
63///     return_generated_events: None,
64///     momentum_space: true,
65///     points: loop_momenta.view(),
66///     integrator_weights: None,
67///     discrete_dims: None,
68///     graph_names: None,
69///     orientations: None,
70/// };
71///
72/// assert_eq!(request.points.ncols(), 2 * 3);
73/// assert!(request.momentum_space);
74/// ```
75#[derive(Debug, Clone)]
76pub struct EvaluateSamples<'a> {
77    /// Process containing the integrand; required when process selection is ambiguous.
78    pub process_id: Option<usize>,
79    /// Integrand to evaluate; required when the selected process has multiple candidates.
80    pub integrand_name: Option<String>,
81    /// Select configured f128, or Arb when f128 is unavailable, instead of the full ladder.
82    pub use_arb_prec: bool,
83    /// Omit optional evaluation metadata from each returned sample.
84    pub minimal_output: bool,
85    /// Temporary event-generation override restored after this evaluation.
86    pub return_generated_events: Option<bool>,
87    /// Interpret rows as consecutive spatial loop-momentum `(px, py, pz)` triplets.
88    pub momentum_space: bool,
89    /// Two-dimensional inputs with one integration or momentum-space sample per row.
90    pub points: ArrayView2<'a, f64>,
91    /// Optional weight for each sample row; omitted weights default to one.
92    pub integrator_weights: Option<ArrayView1<'a, f64>>,
93    /// Optional discrete integration coordinates with one row per sample.
94    pub discrete_dims: Option<ArrayView2<'a, usize>>,
95    /// Optional momentum-space graph selection for each sample.
96    pub graph_names: Option<Vec<Option<String>>>,
97    /// Optional momentum-space orientation selection for each sample.
98    pub orientations: Option<Vec<Option<usize>>>,
99}

101impl<'a> EvaluateSamples<'a> {
102    pub fn run(&self, state: &mut State) -> Result<BatchSampleEvaluationResult> {
103        let process_ref = self.process_id.map(ProcessRef::Id);
104        let (process_id, integrand_name) =
105            state.find_integrand_ref(process_ref.as_ref(), self.integrand_name.as_ref())?;
106        let model = state.resolve_model_for_integrand(process_id, &integrand_name)?;
107
108        let integrand = state
109            .process_list
110            .get_integrand_mut(process_id, &integrand_name)?;
111        let run_result = with_return_generated_events_override(
112            integrand,
113            self.return_generated_events,
114            |integrand| {
115                integrand.warm_up(&model)?;
116
117                let batch_len = self.points.nrows();
118                let integrator_weights =
119                    normalize_integrator_weights(self.integrator_weights.as_ref(), batch_len)?;
120                let graph_names = normalize_optional_per_sample(
121                    self.graph_names.as_ref(),
122                    batch_len,
123                    "graph_names",
124                )?;
125                let orientations = normalize_optional_per_sample(
126                    self.orientations.as_ref(),
127                    batch_len,
128                    "orientations",
129                )?;
130
131                if !self.momentum_space
132                    && (graph_names.iter().any(Option::is_some)
133                        || orientations.iter().any(Option::is_some))
134                {
135                    return Err(eyre!(
136                    "Graph and orientation selection are only supported in momentum-space evaluation."
137                ));
138                }
139
140                if let Some(discrete_dims) = &self.discrete_dims {
141                    if discrete_dims.nrows() != batch_len {
142                        return Err(eyre!(
143                            "Expected {} rows in discrete_dims, got {}.",
144                            batch_len,
145                            discrete_dims.nrows()
146                        ));
147                    }
148                }
149
150                let samples = if self.momentum_space {
151                    let inputs = build_momentum_inputs(
152                        integrand,
153                        &self.points,
154                        &integrator_weights,
155                        self.discrete_dims.as_ref(),
156                        &graph_names,
157                        &orientations,
158                    )?;
159                    integrand
160                        .evaluate_momentum_configurations_raw(&model, &inputs, self.use_arb_prec)?
161                        .samples
162                } else {
163                    let samples = build_havana_samples(
164                        integrand,
165                        &self.points,
166                        &integrator_weights,
167                        self.discrete_dims.as_ref(),
168                    )?;
169                    integrand
170                        .evaluate_samples_raw(
171                            &model,
172                            &samples,
173                            1,
174                            self.use_arb_prec,
175                            false,
176                            Complex::new_zero(),
177                        )?
178                        .samples
179                };
180
181                let observables = integrand
182                    .observable_snapshot_bundle()
183                    .unwrap_or_else(ObservableSnapshotBundle::default);
184
185                Ok((samples, observables))
186            },
187        );
188
189        let (samples, observables) = run_result?;
190
191        Ok(BatchSampleEvaluationResult {
192            samples: samples
193                .into_iter()
194                .map(|evaluation| SampleEvaluationResult {
195                    evaluation: evaluation.into_output(self.minimal_output),
196                })
197                .collect(),
198            observables,
199        })
200    }
201}

203/// Evaluate inputs while preserving the numeric type used by the final stability level.
204///
205/// Coordinate layouts and validation are identical to [`EvaluateSamples`]. With
206/// [`Self::use_arb_prec`] set to `false`, the configured stability ladder may return
207/// an `f64`, `f128`, or arbitrary-precision result. Setting it to `true` selects the
208/// configured `f128` level, or arbitrary precision when `f128` is unavailable. Unlike
209/// [`EvaluateSamples`], this request retains that active result type.
210#[derive(Debug, Clone)]
211pub struct EvaluateSamplesPrecise<'a> {
212    /// Process containing the integrand; required when process selection is ambiguous.
213    pub process_id: Option<usize>,
214    /// Integrand to evaluate; required when the selected process has multiple candidates.
215    pub integrand_name: Option<String>,
216    /// Select configured f128, or Arb when f128 is unavailable, instead of the full ladder.
217    pub use_arb_prec: bool,
218    /// Omit optional evaluation metadata from each returned sample.
219    pub minimal_output: bool,
220    /// Temporary event-generation override restored after this evaluation.
221    pub return_generated_events: Option<bool>,
222    /// Interpret rows as consecutive spatial loop-momentum `(px, py, pz)` triplets.
223    pub momentum_space: bool,
224    /// Two-dimensional inputs with one integration or momentum-space sample per row.
225    pub points: ArrayView2<'a, f64>,
226    /// Optional weight for each sample row; omitted weights default to one.
227    pub integrator_weights: Option<ArrayView1<'a, f64>>,
228    /// Optional discrete integration coordinates with one row per sample.
229    pub discrete_dims: Option<ArrayView2<'a, usize>>,
230    /// Optional momentum-space graph selection for each sample.
231    pub graph_names: Option<Vec<Option<String>>>,
232    /// Optional momentum-space orientation selection for each sample.
233    pub orientations: Option<Vec<Option<usize>>>,
234}

239        let (process_id, integrand_name) =

242
243        let integrand = state
244            .process_list
245            .get_integrand_mut(process_id, &integrand_name)?;
246        let (raw_samples, observables) = with_return_generated_events_override(

248            self.return_generated_events,
249            |integrand| {

251
252                let batch_len = self.points.nrows();
253                let integrator_weights =
254                    normalize_integrator_weights(self.integrator_weights.as_ref(), batch_len)?;
255                let graph_names = normalize_optional_per_sample(
256                    self.graph_names.as_ref(),
257                    batch_len,
258                    "graph_names",
259                )?;
260                let orientations = normalize_optional_per_sample(
261                    self.orientations.as_ref(),
262                    batch_len,
263                    "orientations",
264                )?;
265
266                if !self.momentum_space
267                    && (graph_names.iter().any(Option::is_some)
268                        || orientations.iter().any(Option::is_some))
269                {
270                    return Err(eyre!(
271                        "Graph and orientation selection are only supported in momentum-space evaluation."
272                    ));
273                }
274
275                if let Some(discrete_dims) = &self.discrete_dims {
276                    if discrete_dims.nrows() != batch_len {
277                        return Err(eyre!(
278                            "Expected {} rows in discrete_dims, got {}.",
279                            batch_len,
280                            discrete_dims.nrows()
281                        ));
282                    }
283                }
284
285                let mut raw_samples = if self.momentum_space {
286                    let inputs = build_momentum_inputs(
287                        integrand,
288                        &self.points,
289                        &integrator_weights,
290                        self.discrete_dims.as_ref(),
291                        &graph_names,
292                        &orientations,
293                    )?;
294                    integrand
295                        .evaluate_momentum_configurations_precise_raw(
296                            &model,
297                            &inputs,
298                            self.use_arb_prec,
299                        )?
300                        .samples
301                } else {
302                    let samples = build_havana_samples(
303                        integrand,
304                        &self.points,
305                        &integrator_weights,
306                        self.discrete_dims.as_ref(),
307                    )?;
308                    integrand
309                        .evaluate_samples_precise_raw(
310                            &model,
311                            &samples,
312                            self.use_arb_prec,
313                            Complex::new_zero(),
314                        )?
315                        .samples
316                };
317
318                let observables = merge_precise_observables(integrand, &mut raw_samples)?;
319                Ok((raw_samples, observables))
320            },
321        )?;
322
323        Ok(PreciseBatchSampleEvaluationResult {
324            samples: raw_samples
325                .into_iter()
326                .map(|evaluation| PreciseSampleEvaluationResult {
327                    evaluation: evaluation.into_output(self.minimal_output),
328                })
329                .collect(),
330            observables,
331        })
332    }
333}

336///
337/// See [`EvaluateSamplesPrecise`] for the coordinate, validation, and precision contract.
338pub fn evaluate_samples_precise<'a>(
339    state: &mut State,
340    request: &EvaluateSamplesPrecise<'a>,
341) -> Result<PreciseBatchSampleEvaluationResult> {
342    request.run(state)

345/// Evaluate exactly one sample while retaining its active numeric result type.
346///
347/// This rejects requests whose [`EvaluateSamplesPrecise::points`] view does not have
348/// exactly one row. See [`EvaluateSamplesPrecise`] for the remaining contract.
349pub fn evaluate_sample_precise<'a>(
350    state: &mut State,
351    request: &EvaluateSamplesPrecise<'a>,
352) -> Result<PreciseSingleSampleEvaluationResult> {
353    let mut results = request.run(state)?;
354    if results.samples.len() != 1 {
355        return Err(eyre!(
356            "evaluate_sample_precise expects exactly one sample, got {}.",
357            results.samples.len()
358        ));
359    }
360    Ok(PreciseSingleSampleEvaluationResult {
361        sample: results.samples.remove(0),
362        observables: results.observables,
363    })
364}

397fn with_return_generated_events_override<R>(
398    integrand: &mut gammalooprs::integrands::process::ProcessIntegrand,
399    return_generated_events: Option<bool>,
400    run: impl FnOnce(&mut gammalooprs::integrands::process::ProcessIntegrand) -> Result<R>,
401) -> Result<R> {
402    let previous_generate_events = integrand.get_settings().general.generate_events;
403    if let Some(return_generated_events) = return_generated_events {
404        integrand.get_mut_settings().general.generate_events = return_generated_events;
405    }
406    let result = run(integrand);
407    if integrand.get_settings().general.generate_events != previous_generate_events {
408        integrand.get_mut_settings().general.generate_events = previous_generate_events;
409    }
410    result
411}

413fn normalize_optional_per_sample<T: Clone>(
414    values: Option<&Vec<Option<T>>>,
415    batch_len: usize,
416    label: &str,
417) -> Result<Vec<Option<T>>> {
418    match values {
419        Some(values) => {
420            if values.len() != batch_len {
421                return Err(eyre!(
422                    "Expected {} entries in {}, got {}.",
423                    batch_len,
424                    label,
425                    values.len()
426                ));
427            }
428            Ok(values.clone())
429        }
430        None => Ok(vec![None; batch_len]),
431    }
432}

434fn normalize_integrator_weights(
435    integrator_weights: Option<&ArrayView1<'_, f64>>,
436    batch_len: usize,
437) -> Result<Vec<f64>> {
438    match integrator_weights {
439        Some(weights) => {
440            if weights.len() != batch_len {
441                return Err(eyre!(
442                    "Expected {} entries in integrator_weights, got {}.",
443                    batch_len,
444                    weights.len()
445                ));
446            }
447            Ok(weights.iter().copied().collect())
448        }
449        None => Ok(vec![1.0; batch_len]),
450    }
451}

453pub(crate) fn build_havana_sample(
454    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
455    point: &[f64],
456    discrete_dim: &[usize],
457    integrator_weight: f64,
458) -> Result<Sample<F<f64>>> {
459    let expected_dimension = integrand.expected_x_space_dimension(discrete_dim)?;
460    if point.len() != expected_dimension {
461        return Err(eyre!(
462            "Expected {} x-space coordinates for this integrand selection, got {}.",
463            expected_dimension,
464            point.len()
465        ));
466    }
467
468    Ok(havana_sample(
469        point.iter().map(|&x| F(x)).collect::<Vec<F<f64>>>(),
470        discrete_dim,
471        F(integrator_weight),
472    ))
473}

475fn build_havana_samples(
476    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
477    points: &ArrayView2<'_, f64>,
478    integrator_weights: &[f64],
479    discrete_dims: Option<&ArrayView2<'_, usize>>,
480) -> Result<Vec<Sample<F<f64>>>> {
481    (0..points.nrows())
482        .map(|sample_index| {
483            let discrete_dim: Vec<usize> = discrete_dims
484                .as_ref()
485                .map(|dims| dims.row(sample_index).iter().copied().collect())
486                .unwrap_or_default();
487            build_havana_sample(
488                integrand,
489                points
490                    .row(sample_index)
491                    .as_slice()
492                    .expect("point rows are contiguous"),
493                &discrete_dim,
494                integrator_weights[sample_index],
495            )
496        })
497        .collect()
498}

500pub(crate) fn build_momentum_input(
501    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
502    point: &[f64],
503    integrator_weight: f64,
504    discrete_dim: &[usize],
505    graph_name: Option<&str>,
506    orientation: Option<usize>,
507) -> Result<MomentumSpaceEvaluationInput> {
508    if !point.len().is_multiple_of(3) {
509        return Err(eyre!(
510            "Momentum-space evaluation expects flattened (px, py, pz) triplets, so the coordinate count must be a multiple of 3; got {}.",
511            point.len()
512        ));
513    }
514
515    let loop_momenta = point
516        .chunks_exact(3)
517        .map(|coords| ThreeMomentum {
518            px: F(coords[0]),
519            py: F(coords[1]),
520            pz: F(coords[2]),
521        })
522        .collect();
523
524    match &integrand.get_settings().sampling {
525        gammalooprs::settings::runtime::SamplingSettings::Default(_)
526        | gammalooprs::settings::runtime::SamplingSettings::MultiChanneling(_) => {
527            if !discrete_dim.is_empty() {
528                return Err(eyre!(
529                    "This integrand does not use discrete graph sampling; expected no discrete dimensions, got {:?}.",
530                    discrete_dim
531                ));
532            }
533
534            if orientation.is_some() && graph_name.is_none() {
535                return Err(eyre!(
536                    "A specific orientation requires a specific graph in momentum-space evaluation."
537                ));
538            }
539
540            let graph_id = graph_name
541                .map(|graph_name| {
542                    integrand.find_graph_id_by_name(graph_name).ok_or_else(|| {
543                        eyre!(
544                            "Unknown graph '{}' in momentum-space evaluation.",
545                            graph_name
546                        )
547                    })
548                })
549                .transpose()?;
550
551            if let (Some(graph_id), Some(orientation)) = (graph_id, orientation) {
552                let orientation_count =
553                    integrand.graph_orientation_count(graph_id).ok_or_else(|| {
554                        eyre!(
555                            "Graph id {} is invalid for momentum-space evaluation.",
556                            graph_id
557                        )
558                    })?;
559                if orientation >= orientation_count {
560                    return Err(eyre!(
561                        "Orientation {} is out of range for graph '{}'; the graph has {} orientations.",
562                        orientation,
563                        graph_name.expect("graph name should exist when graph_id exists"),
564                        orientation_count
565                    ));
566                }
567            }
568
569            Ok(MomentumSpaceEvaluationInput {
570                loop_momenta,
571                integrator_weight: F(integrator_weight),
572                graph_id,
573                group_id: None,
574                orientation,
575                channel_id: None,
576            })
577        }
578        gammalooprs::settings::runtime::SamplingSettings::DiscreteGraphs(settings) => {
579            if !discrete_dim.is_empty() && (graph_name.is_some() || orientation.is_some()) {
580                return Err(eyre!(
581                    "Discrete dimensions are mutually exclusive with explicit graph/orientation selection in momentum-space evaluation."
582                ));
583            }
584
585            if let Some(graph_name) = graph_name {
586                let graph_id = integrand.find_graph_id_by_name(graph_name).ok_or_else(|| {
587                    eyre!(
588                        "Unknown graph '{}' in momentum-space evaluation.",
589                        graph_name
590                    )
591                })?;
592
593                if let Some(orientation) = orientation {
594                    let orientation_count =
595                        integrand.graph_orientation_count(graph_id).ok_or_else(|| {
596                            eyre!(
597                                "Graph id {} is invalid for momentum-space evaluation.",
598                                graph_id
599                            )
600                        })?;
601                    if orientation >= orientation_count {
602                        return Err(eyre!(
603                            "Orientation {} is out of range for graph '{}'; the graph has {} orientations.",
604                            orientation,
605                            graph_name,
606                            orientation_count
607                        ));
608                    }
609                }
610
611                return Ok(MomentumSpaceEvaluationInput {
612                    loop_momenta,
613                    integrator_weight: F(integrator_weight),
614                    graph_id: Some(graph_id),
615                    group_id: None,
616                    orientation,
617                    channel_id: None,
618                });
619            }
620
621            let (group_from_dims, orientation_from_dims, channel_id) = if discrete_dim.is_empty() {
622                (None, None, None)
623            } else {
624                integrand.resolve_discrete_selection(discrete_dim)?
625            };
626
627            let group_from_name = graph_name
628                .map(|graph_name| integrand.resolve_group_id_by_master_name(graph_name))
629                .transpose()?;
630
631            let group_id = match (group_from_dims, group_from_name) {
632                (Some(group_id), Some(group_from_name)) if group_id != group_from_name => {
633                    return Err(eyre!(
634                        "Discrete dimensions select graph group {}, but graph '{}' resolves to group {}.",
635                        group_id.0,
636                        graph_name.expect("graph name must exist"),
637                        group_from_name.0
638                    ));
639                }
640                (Some(group_id), _) => Some(group_id),
641                (None, Some(group_id)) => Some(group_id),
642                (None, None) => None,
643            };
644
645            let orientation = match (orientation_from_dims, orientation) {
646                (Some(from_dims), Some(from_arg)) if from_dims != from_arg => {
647                    return Err(eyre!(
648                        "Discrete dimensions select orientation {}, but the explicit momentum-space orientation is {}.",
649                        from_dims,
650                        from_arg
651                    ));
652                }
653                (Some(from_dims), _) => Some(from_dims),
654                (None, from_arg) => from_arg,
655            };
656
657            if group_id.is_none() && orientation.is_some() {
658                return Err(eyre!(
659                    "A specific orientation requires a specific graph group in momentum-space evaluation."
660                ));
661            }
662
663            if settings.sample_orientations && group_id.is_some() && orientation.is_none() {
664                return Err(eyre!(
665                    "Momentum-space evaluation for this integrand requires an orientation selection."
666                ));
667            }
668
669            Ok(MomentumSpaceEvaluationInput {
670                loop_momenta,
671                integrator_weight: F(integrator_weight),
672                graph_id: None,
673                group_id,
674                orientation,
675                channel_id,
676            })
677        }
678    }
679}

681fn build_momentum_inputs(
682    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
683    points: &ArrayView2<'_, f64>,
684    integrator_weights: &[f64],
685    discrete_dims: Option<&ArrayView2<'_, usize>>,
686    graph_names: &[Option<String>],
687    orientations: &[Option<usize>],
688) -> Result<Vec<MomentumSpaceEvaluationInput>> {
689    (0..points.nrows())
690        .map(|sample_index| {
691            let discrete_dim: Vec<usize> = discrete_dims
692                .as_ref()
693                .map(|dims| dims.row(sample_index).iter().copied().collect())
694                .unwrap_or_default();
695            build_momentum_input(
696                integrand,
697                points
698                    .row(sample_index)
699                    .as_slice()
700                    .expect("point rows are contiguous"),
701                integrator_weights[sample_index],
702                &discrete_dim,
703                graph_names[sample_index].as_deref(),
704                orientations[sample_index],
705            )
706        })
707        .collect()
708}

710fn clear_precise_event_groups(
711    result: &mut gammalooprs::integrands::evaluation::PreciseEvaluationResult,
712) {
713    match result {
714        gammalooprs::integrands::evaluation::PreciseEvaluationResult::Double(result) => {
715            result.event_groups.clear()
716        }
717        gammalooprs::integrands::evaluation::PreciseEvaluationResult::Quad(result) => {
718            result.event_groups.clear()
719        }
720        gammalooprs::integrands::evaluation::PreciseEvaluationResult::Arb(result) => {
721            result.event_groups.clear()
722        }
723    }
724}

726fn merge_precise_observables(
727    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
728    results: &mut [gammalooprs::integrands::evaluation::PreciseEvaluationResult],
729) -> Result<ObservableSnapshotBundle> {
730    let mut observables = ObservableSnapshotBundle::default();
731    let mut have_observables = false;
732
733    for evaluation in results.iter_mut() {
734        let sample_observables = integrand
735            .build_observable_snapshots_for_precise_result(evaluation)
736            .unwrap_or_default();
737        if !sample_observables.histograms.is_empty() {
738            if have_observables {
739                observables.merge_in_place(&sample_observables)?;
740            } else {
741                observables = sample_observables;
742                have_observables = true;
743            }
744        }
745        if !integrand.get_settings().general.generate_events {
746            clear_precise_event_groups(evaluation);
747        }
748    }
749
750    Ok(observables)
751}

753fn set_top_level_sample_weight(sample: &mut Sample<F<f64>>, integrator_weight: F<f64>) {
754    match sample {
755        Sample::Continuous(weight, _)
756        | Sample::Discrete(weight, _, _)
757        | Sample::Uniform(weight, _, _) => *weight = integrator_weight,
758    }
759}

761fn havana_sample(
762    cont: Vec<F<f64>>,
763    discrete_dimensions: &[usize],
764    integrator_weight: F<f64>,
765) -> Sample<F<f64>> {
766    let mut sample = Sample::Continuous(F(1.0), cont);
767
768    for &discrete_dimension in discrete_dimensions.iter().rev() {
769        sample = Sample::Discrete(F(1.0), discrete_dimension, Some(Box::new(sample)));
770    }
771
772    set_top_level_sample_weight(&mut sample, integrator_weight);
773    sample
774}
