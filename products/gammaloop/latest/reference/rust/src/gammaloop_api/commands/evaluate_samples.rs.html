37impl<'a> EvaluateSamples<'a> {
38    pub fn run(&self, state: &mut State) -> Result<BatchSampleEvaluationResult> {
39        let process_ref = self.process_id.map(ProcessRef::Id);
40        let (process_id, integrand_name) =
41            state.find_integrand_ref(process_ref.as_ref(), self.integrand_name.as_ref())?;
42        let model = state.resolve_model_for_integrand(process_id, &integrand_name)?;
43
44        let integrand = state
45            .process_list
46            .get_integrand_mut(process_id, &integrand_name)?;
47        let run_result = with_return_generated_events_override(
48            integrand,
49            self.return_generated_events,
50            |integrand| {
51                integrand.warm_up(&model)?;
52
53                let batch_len = self.points.nrows();
54                let integrator_weights =
55                    normalize_integrator_weights(self.integrator_weights.as_ref(), batch_len)?;
56                let graph_names = normalize_optional_per_sample(
57                    self.graph_names.as_ref(),
58                    batch_len,
59                    "graph_names",
60                )?;
61                let orientations = normalize_optional_per_sample(
62                    self.orientations.as_ref(),
63                    batch_len,
64                    "orientations",
65                )?;
66
67                if !self.momentum_space
68                    && (graph_names.iter().any(Option::is_some)
69                        || orientations.iter().any(Option::is_some))
70                {
71                    return Err(eyre!(
72                    "Graph and orientation selection are only supported in momentum-space evaluation."
73                ));
74                }
75
76                if let Some(discrete_dims) = &self.discrete_dims {
77                    if discrete_dims.nrows() != batch_len {
78                        return Err(eyre!(
79                            "Expected {} rows in discrete_dims, got {}.",
80                            batch_len,
81                            discrete_dims.nrows()
82                        ));
83                    }
84                }
85
86                let samples = if self.momentum_space {
87                    let inputs = build_momentum_inputs(
88                        integrand,
89                        &self.points,
90                        &integrator_weights,
91                        self.discrete_dims.as_ref(),
92                        &graph_names,
93                        &orientations,
94                    )?;
95                    integrand
96                        .evaluate_momentum_configurations_raw(&model, &inputs, self.use_arb_prec)?
97                        .samples
98                } else {
99                    let samples = build_havana_samples(
100                        integrand,
101                        &self.points,
102                        &integrator_weights,
103                        self.discrete_dims.as_ref(),
104                    )?;
105                    integrand
106                        .evaluate_samples_raw(
107                            &model,
108                            &samples,
109                            1,
110                            self.use_arb_prec,
111                            false,
112                            Complex::new_zero(),
113                        )?
114                        .samples
115                };
116
117                let observables = integrand
118                    .observable_snapshot_bundle()
119                    .unwrap_or_else(ObservableSnapshotBundle::default);
120
121                Ok((samples, observables))
122            },
123        );
124
125        let (samples, observables) = run_result?;
126
127        Ok(BatchSampleEvaluationResult {
128            samples: samples
129                .into_iter()
130                .map(|evaluation| SampleEvaluationResult {
131                    evaluation: evaluation.into_output(self.minimal_output),
132                })
133                .collect(),
134            observables,
135        })
136    }
137}

301fn with_return_generated_events_override<R>(
302    integrand: &mut gammalooprs::integrands::process::ProcessIntegrand,
303    return_generated_events: Option<bool>,
304    run: impl FnOnce(&mut gammalooprs::integrands::process::ProcessIntegrand) -> Result<R>,
305) -> Result<R> {
306    let previous_generate_events = integrand.get_settings().general.generate_events;
307    if let Some(return_generated_events) = return_generated_events {
308        integrand.get_mut_settings().general.generate_events = return_generated_events;
309    }
310    let result = run(integrand);
311    if integrand.get_settings().general.generate_events != previous_generate_events {
312        integrand.get_mut_settings().general.generate_events = previous_generate_events;
313    }
314    result
315}

317fn normalize_optional_per_sample<T: Clone>(
318    values: Option<&Vec<Option<T>>>,
319    batch_len: usize,
320    label: &str,
321) -> Result<Vec<Option<T>>> {
322    match values {
323        Some(values) => {
324            if values.len() != batch_len {
325                return Err(eyre!(
326                    "Expected {} entries in {}, got {}.",
327                    batch_len,
328                    label,
329                    values.len()
330                ));
331            }
332            Ok(values.clone())
333        }
334        None => Ok(vec![None; batch_len]),
335    }
336}

338fn normalize_integrator_weights(
339    integrator_weights: Option<&ArrayView1<'_, f64>>,
340    batch_len: usize,
341) -> Result<Vec<f64>> {
342    match integrator_weights {
343        Some(weights) => {
344            if weights.len() != batch_len {
345                return Err(eyre!(
346                    "Expected {} entries in integrator_weights, got {}.",
347                    batch_len,
348                    weights.len()
349                ));
350            }
351            Ok(weights.iter().copied().collect())
352        }
353        None => Ok(vec![1.0; batch_len]),
354    }
355}

357pub(crate) fn build_havana_sample(
358    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
359    point: &[f64],
360    discrete_dim: &[usize],
361    integrator_weight: f64,
362) -> Result<Sample<F<f64>>> {
363    let expected_dimension = integrand.expected_x_space_dimension(discrete_dim)?;
364    if point.len() != expected_dimension {
365        return Err(eyre!(
366            "Expected {} x-space coordinates for this integrand selection, got {}.",
367            expected_dimension,
368            point.len()
369        ));
370    }
371
372    Ok(havana_sample(
373        point.iter().map(|&x| F(x)).collect::<Vec<F<f64>>>(),
374        discrete_dim,
375        F(integrator_weight),
376    ))
377}

379fn build_havana_samples(
380    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
381    points: &ArrayView2<'_, f64>,
382    integrator_weights: &[f64],
383    discrete_dims: Option<&ArrayView2<'_, usize>>,
384) -> Result<Vec<Sample<F<f64>>>> {
385    (0..points.nrows())
386        .map(|sample_index| {
387            let discrete_dim: Vec<usize> = discrete_dims
388                .as_ref()
389                .map(|dims| dims.row(sample_index).iter().copied().collect())
390                .unwrap_or_default();
391            build_havana_sample(
392                integrand,
393                points
394                    .row(sample_index)
395                    .as_slice()
396                    .expect("point rows are contiguous"),
397                &discrete_dim,
398                integrator_weights[sample_index],
399            )
400        })
401        .collect()
402}

404pub(crate) fn build_momentum_input(
405    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
406    point: &[f64],
407    integrator_weight: f64,
408    discrete_dim: &[usize],
409    graph_name: Option<&str>,
410    orientation: Option<usize>,
411) -> Result<MomentumSpaceEvaluationInput> {
412    if !point.len().is_multiple_of(3) {
413        return Err(eyre!(
414            "Momentum-space evaluation expects a multiple of 3 coordinates, got {}.",
415            point.len()
416        ));
417    }
418
419    let loop_momenta = point
420        .chunks_exact(3)
421        .map(|coords| ThreeMomentum {
422            px: F(coords[0]),
423            py: F(coords[1]),
424            pz: F(coords[2]),
425        })
426        .collect();
427
428    match &integrand.get_settings().sampling {
429        gammalooprs::settings::runtime::SamplingSettings::Default(_)
430        | gammalooprs::settings::runtime::SamplingSettings::MultiChanneling(_) => {
431            if !discrete_dim.is_empty() {
432                return Err(eyre!(
433                    "This integrand does not use discrete graph sampling; expected no discrete dimensions, got {:?}.",
434                    discrete_dim
435                ));
436            }
437
438            if orientation.is_some() && graph_name.is_none() {
439                return Err(eyre!(
440                    "A specific orientation requires a specific graph in momentum-space evaluation."
441                ));
442            }
443
444            let graph_id = graph_name
445                .map(|graph_name| {
446                    integrand.find_graph_id_by_name(graph_name).ok_or_else(|| {
447                        eyre!(
448                            "Unknown graph '{}' in momentum-space evaluation.",
449                            graph_name
450                        )
451                    })
452                })
453                .transpose()?;
454
455            if let (Some(graph_id), Some(orientation)) = (graph_id, orientation) {
456                let orientation_count =
457                    integrand.graph_orientation_count(graph_id).ok_or_else(|| {
458                        eyre!(
459                            "Graph id {} is invalid for momentum-space evaluation.",
460                            graph_id
461                        )
462                    })?;
463                if orientation >= orientation_count {
464                    return Err(eyre!(
465                        "Orientation {} is out of range for graph '{}'; the graph has {} orientations.",
466                        orientation,
467                        graph_name.expect("graph name should exist when graph_id exists"),
468                        orientation_count
469                    ));
470                }
471            }
472
473            Ok(MomentumSpaceEvaluationInput {
474                loop_momenta,
475                integrator_weight: F(integrator_weight),
476                graph_id,
477                group_id: None,
478                orientation,
479                channel_id: None,
480            })
481        }
482        gammalooprs::settings::runtime::SamplingSettings::DiscreteGraphs(settings) => {
483            if !discrete_dim.is_empty() && (graph_name.is_some() || orientation.is_some()) {
484                return Err(eyre!(
485                    "Discrete dimensions are mutually exclusive with explicit graph/orientation selection in momentum-space evaluation."
486                ));
487            }
488
489            if let Some(graph_name) = graph_name {
490                let graph_id = integrand.find_graph_id_by_name(graph_name).ok_or_else(|| {
491                    eyre!(
492                        "Unknown graph '{}' in momentum-space evaluation.",
493                        graph_name
494                    )
495                })?;
496
497                if let Some(orientation) = orientation {
498                    let orientation_count =
499                        integrand.graph_orientation_count(graph_id).ok_or_else(|| {
500                            eyre!(
501                                "Graph id {} is invalid for momentum-space evaluation.",
502                                graph_id
503                            )
504                        })?;
505                    if orientation >= orientation_count {
506                        return Err(eyre!(
507                            "Orientation {} is out of range for graph '{}'; the graph has {} orientations.",
508                            orientation,
509                            graph_name,
510                            orientation_count
511                        ));
512                    }
513                }
514
515                return Ok(MomentumSpaceEvaluationInput {
516                    loop_momenta,
517                    integrator_weight: F(integrator_weight),
518                    graph_id: Some(graph_id),
519                    group_id: None,
520                    orientation,
521                    channel_id: None,
522                });
523            }
524
525            let (group_from_dims, orientation_from_dims, channel_id) = if discrete_dim.is_empty() {
526                (None, None, None)
527            } else {
528                integrand.resolve_discrete_selection(discrete_dim)?
529            };
530
531            let group_from_name = graph_name
532                .map(|graph_name| integrand.resolve_group_id_by_master_name(graph_name))
533                .transpose()?;
534
535            let group_id = match (group_from_dims, group_from_name) {
536                (Some(group_id), Some(group_from_name)) if group_id != group_from_name => {
537                    return Err(eyre!(
538                        "Discrete dimensions select graph group {}, but graph '{}' resolves to group {}.",
539                        group_id.0,
540                        graph_name.expect("graph name must exist"),
541                        group_from_name.0
542                    ));
543                }
544                (Some(group_id), _) => Some(group_id),
545                (None, Some(group_id)) => Some(group_id),
546                (None, None) => None,
547            };
548
549            let orientation = match (orientation_from_dims, orientation) {
550                (Some(from_dims), Some(from_arg)) if from_dims != from_arg => {
551                    return Err(eyre!(
552                        "Discrete dimensions select orientation {}, but the explicit momentum-space orientation is {}.",
553                        from_dims,
554                        from_arg
555                    ));
556                }
557                (Some(from_dims), _) => Some(from_dims),
558                (None, from_arg) => from_arg,
559            };
560
561            if group_id.is_none() && orientation.is_some() {
562                return Err(eyre!(
563                    "A specific orientation requires a specific graph group in momentum-space evaluation."
564                ));
565            }
566
567            if settings.sample_orientations && group_id.is_some() && orientation.is_none() {
568                return Err(eyre!(
569                    "Momentum-space evaluation for this integrand requires an orientation selection."
570                ));
571            }
572
573            Ok(MomentumSpaceEvaluationInput {
574                loop_momenta,
575                integrator_weight: F(integrator_weight),
576                graph_id: None,
577                group_id,
578                orientation,
579                channel_id,
580            })
581        }
582    }
583}

585fn build_momentum_inputs(
586    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
587    points: &ArrayView2<'_, f64>,
588    integrator_weights: &[f64],
589    discrete_dims: Option<&ArrayView2<'_, usize>>,
590    graph_names: &[Option<String>],
591    orientations: &[Option<usize>],
592) -> Result<Vec<MomentumSpaceEvaluationInput>> {
593    (0..points.nrows())
594        .map(|sample_index| {
595            let discrete_dim: Vec<usize> = discrete_dims
596                .as_ref()
597                .map(|dims| dims.row(sample_index).iter().copied().collect())
598                .unwrap_or_default();
599            build_momentum_input(
600                integrand,
601                points
602                    .row(sample_index)
603                    .as_slice()
604                    .expect("point rows are contiguous"),
605                integrator_weights[sample_index],
606                &discrete_dim,
607                graph_names[sample_index].as_deref(),
608                orientations[sample_index],
609            )
610        })
611        .collect()
612}

614fn clear_precise_event_groups(
615    result: &mut gammalooprs::integrands::evaluation::PreciseEvaluationResult,
616) {
617    match result {
618        gammalooprs::integrands::evaluation::PreciseEvaluationResult::Double(result) => {
619            result.event_groups.clear()
620        }
621        gammalooprs::integrands::evaluation::PreciseEvaluationResult::Quad(result) => {
622            result.event_groups.clear()
623        }
624        gammalooprs::integrands::evaluation::PreciseEvaluationResult::Arb(result) => {
625            result.event_groups.clear()
626        }
627    }

630fn merge_precise_observables(
631    integrand: &gammalooprs::integrands::process::ProcessIntegrand,
632    results: &mut [gammalooprs::integrands::evaluation::PreciseEvaluationResult],
633) -> Result<ObservableSnapshotBundle> {
634    let mut observables = ObservableSnapshotBundle::default();
635    let mut have_observables = false;
636
637    for evaluation in results.iter_mut() {
638        let sample_observables = integrand
639            .build_observable_snapshots_for_precise_result(evaluation)
640            .unwrap_or_default();
641        if !sample_observables.histograms.is_empty() {
642            if have_observables {
643                observables.merge_in_place(&sample_observables)?;
644            } else {
645                observables = sample_observables;
646                have_observables = true;
647            }
648        }
649        if !integrand.get_settings().general.generate_events {
650            clear_precise_event_groups(evaluation);
651        }
652    }
653
654    Ok(observables)
655}

657fn set_top_level_sample_weight(sample: &mut Sample<F<f64>>, integrator_weight: F<f64>) {
658    match sample {
659        Sample::Continuous(weight, _)
660        | Sample::Discrete(weight, _, _)
661        | Sample::Uniform(weight, _, _) => *weight = integrator_weight,
662    }

665fn havana_sample(
666    cont: Vec<F<f64>>,
667    discrete_dimensions: &[usize],
668    integrator_weight: F<f64>,
669) -> Sample<F<f64>> {
670    let mut sample = Sample::Continuous(F(1.0), cont);
671
672    for &discrete_dimension in discrete_dimensions.iter().rev() {
673        sample = Sample::Discrete(F(1.0), discrete_dimension, Some(Box::new(sample)));
674    }
675
676    set_top_level_sample_weight(&mut sample, integrator_weight);
677    sample
