27pub struct Renormalize {
28    /// Process reference: `#<id>`, `name:<name>`, or `<id>/<name>`
29    #[arg(
30        short = 'p',
31        long = "process",
32        value_name = "PROCESS",
33        completion_process_selector(crate::completion::SelectorKind::Amplitude)
34    )]
35    pub process: Option<ProcessRef>,
36
37    #[arg(short = 'a', long = "align-to-rqft", default_value_t = false)]
38    pub align_to_rqft: bool,
39
40    #[arg(short = 'f', long = "print-namespaces", default_value_t = false)]
41    pub print_namespaces: bool,
42
43    /// The name of the process to inspect
44    /// The amplitude name to renormalize
45    #[arg(
46        short = 'i',
47        long = "integrand-name",
48        value_name = "NAME",
49        completion_integrand_selector(crate::completion::SelectorKind::Amplitude)
50    )]
51    pub integrand_name: Option<String>,
52
53    /// The directory to store per-graph results in
54    #[arg(short = 'o', long, value_hint = clap::ValueHint::FilePath)]
55    pub result_path: Option<PathBuf>,
56}
57
58impl Renormalize {
59    pub fn run(&self, state: &mut State, global_cli_settings: &CLISettings) -> Result<Vec<Atom>> {
60        let (process_id, _) =
61            state.find_integrand_ref(self.process.as_ref(), self.integrand_name.as_ref())?;
62        let amplitude = state
63            .process_list
64            .get_amplitude_mut_ref(self.process.as_ref(), self.integrand_name.as_ref())?;
65
66        let mut settings = global_cli_settings.global.generation.uv.clone();
67
68        settings.generate_integrated = true;
69        let prescription = &mut settings.renormalization_prescription;
70        for scheme in [
71            &mut prescription.log_divergent,
72            &mut prescription.massive_power_divergent,
73            &mut prescription.massless_power_divergent,
74        ] {
75            if *scheme == ApproximationType::MUV {
76                *scheme = ApproximationType::PolePart;
77            }
78        }
79        for rule in &mut prescription.overrides {
80            if rule.prescription == ApproximationType::MUV {
81                rule.prescription = ApproximationType::PolePart;
82            }
83        }
84
85        let mut renormalization_part: Vec<Atom> = Vec::new();
86        let output_dir = if let Some(path) = self.result_path.clone() {
87            fs::create_dir_all(&path)?;
88            Some(path)
89        } else {
90            None
91        };
92
93        const MAX_PRINT_CHARS: usize = 4000;
94        let clip_expr = |expr: &Atom, print_namespaces: bool| -> (String, bool) {
95            let expr_str = if print_namespaces {
96                format!("{:#}", expr)
97            } else {
98                format!("{}", expr)
99            };
100            if expr_str.len() > MAX_PRINT_CHARS {
101                (format!("{}...", &expr_str[..MAX_PRINT_CHARS]), true)
102            } else {
103                (expr_str, false)
104            }
105        };
106
107        for (index, graph_term) in amplitude.graphs.iter_mut().enumerate() {
108            let mut part = graph_term.renormalization_part(&settings)?.expression;
109
110            part = state
111                .model
112                .apply_parameter_replacement_rules(&state.model.apply_coupling_replacement_rules(
113                    &part.simplify_color().expand().simplify_metrics().to_dots(),
114                ))
115                .collect_factors();
116
117            if self.align_to_rqft {
118                part = (part
119                    .replace(CS.tr)
120                    .with(Atom::num((1, 2)))
121                    .replace(CS.nc)
122                    .with(CS.ca)
123                    .replace(parse!("UFO::aS"))
124                    .with(parse!("gs").pow(2) / (Atom::var(Symbol::PI) * 4))
125                    / 8)
126                .expand_num()
127            }
128
129            let (expr_to_print, clipped) = clip_expr(&part, self.print_namespaces);
130            if clipped {
131                info!(
132                    "Renormalization part for graph '{}' (clipped):\n{}",
133                    graph_term.graph.name, expr_to_print
134                );
135            } else {
136                info!(
137                    "Renormalization part for graph '{}':\n{}",
138                    graph_term.graph.name, expr_to_print
139                );
140            }
141            if let Some(dir) = &output_dir {
142                let sanitized_name = graph_term
143                    .graph
144                    .name
145                    .chars()
146                    .map(|c| if c.is_ascii_alphanumeric() { c } else { '_' })
147                    .collect::<String>();
148                let file_name = format!("{:03}_{}.txt", index + 1, sanitized_name);
149                let typst_name = format!("{:03}_{}.typ", index + 1, sanitized_name);
150                let mut path = File::create(dir.join(file_name))?;
151                write!(path, "{}", part.printer(PrintOptions::file()))?;
152                let mut path = File::create(dir.join(typst_name))?;
153                write!(
154                    path,
155                    "$ {} $",
156                    part.printer(SpensoPrintSettings::typst_options())
157                )?;
158            }
159            renormalization_part.push(part);
160        }
161
162        if let Some(dir) = &output_dir {
163            info!(
164                "Saved renormalization part per graph for process {} in directory: {}",
165                state.process_list.processes[process_id]
166                    .definition
167                    .folder_name
168                    .green(),
169                dir.display()
170            );
171        }
172
173        Ok(renormalization_part)
174    }
