27#[derive(Subcommand, Debug, Serialize, Deserialize, Clone, JsonSchema, PartialEq)]
28pub enum Profile {
29    /// Ultraviolet profile analysis
30    UltraViolet(#[command(flatten)] UltraVioletProfile),

34}

36#[derive(Args, Debug, Serialize, Deserialize, Clone, JsonSchema, PartialEq)]
37pub struct UltraVioletProfile {
38    /// Process reference: `#<id>`, `name:<name>`, or `<id>/<name>`
39    #[arg(
40        short = 'p',
41        long = "process",
42        value_name = "PROCESS",
43        completion_process_selector(crate::completion::SelectorKind::Any)
44    )]
45    pub process: Option<ProcessRef>,
46
47    /// The integrand name to inspect
48    #[arg(
49        short = 'i',
50        long = "integrand-name",
51        value_name = "NAME",
52        completion_integrand_selector(crate::completion::SelectorKind::Any)
53    )]
54    pub integrand_name: Option<String>,
55
56    /// Number of scaling points to sample
57    #[arg(long = "n-points", default_value_t = 20)]
58    pub n_points: usize,
59
60    /// Minimum scaling factor
61    #[arg(long = "min-scaling", default_value_t = 3.0)]
62    pub min_scale_exponent: f64,
63
64    /// Maximum scaling factor
65    #[arg(long = "max-scaling", default_value_t = 6.0)]
66    pub max_scale_exponent: f64,
67
68    /// Use f128 precision for evaluation
69    #[arg(long = "use_f128")]
70    pub use_f128: bool,
71
72    #[arg(long = "analyse_analytically")]
73    pub analyse_analytically: bool,
74
75    /// Profile each visible orientation separately and include per-orientation results
76    #[arg(long = "per-orientation")]
77    pub per_orientation: bool,
78
79    /// Random seed for momentum sampling
80    #[arg(long = "seed")]
81    pub seed: Option<u64>,
82
83    /// Fixed UV ray directions as flattened 3-vectors, repeated for all rays if one direction is supplied
84    #[arg(
85        long = "uv-ray-directions",
86        num_args = 1..,
87        value_delimiter = ',',
88        allow_negative_numbers = true
89    )]
90    pub uv_ray_directions: Vec<f64>,
91
92    /// Fixed UV ray starting norms, repeated for all rays if one norm is supplied
93    #[arg(
94        long = "uv-ray-norms",
95        num_args = 1..,
96        value_delimiter = ',',
97        allow_negative_numbers = true,
98        requires = "uv_ray_directions"
99    )]
100    pub uv_ray_norms: Vec<f64>,
101
102    /// Output file for results (optional)
103    #[arg(short = 'o', long = "output", value_hint = clap::ValueHint::FilePath)]
104    pub output_file: Option<PathBuf>,
105}

195pub enum ProfileResult {
196    UltraViolet(UVProfileAnalysis),
197    InfraRed(IrLimitTestReport),
198}
199
200impl ProfileResult {
201    pub fn unwrap_uv(self) -> UVProfileAnalysis {
202        match self {
203            ProfileResult::UltraViolet(uv_analyis) => uv_analyis,
204            _ => panic!("result does not contain uv profilel analysis data"),
205        }
206    }
207
208    pub fn unwrap_ir(self) -> IrLimitTestReport {
209        match self {
210            ProfileResult::InfraRed(ir_analyis) => ir_analyis,
211            _ => panic!("result does not contain ir profile analysis data"),
212        }
213    }
214}

216impl Profile {
217    #[instrument(skip_all)]
218    #[allow(clippy::needless_update)]
219    pub fn run(
220        &self,
221        state: &mut State,
222        _global_cli_settings: &CLISettings,
223    ) -> Result<ProfileResult> {
224        match self {
225            Profile::UltraViolet(UltraVioletProfile {
226                process,
227                integrand_name,
228                n_points,
229                min_scale_exponent,
230                max_scale_exponent,
231                use_f128,
232                seed,
233                uv_ray_directions,
234                uv_ray_norms,
235                analyse_analytically,
236                per_orientation,
237                output_file,
238            }) => {
239                let (process_id, integrand_name) =
240                    state.find_integrand_ref(process.as_ref(), integrand_name.as_ref())?;
241                let model = state.resolve_model_for_integrand(process_id, &integrand_name)?;
242                let default_uv_ray_norm = {
243                    let process = &mut state.process_list.processes[process_id];
244                    match &mut process.collection {
245                        ProcessCollection::Amplitudes(amplitudes) => {
246                            let amplitude =
247                                amplitudes.get_mut(&integrand_name).ok_or_else(|| {
248                                    eyre!(
249                                        "No amplitude named '{}' in process '{}'",
250                                        integrand_name,
251                                        process.definition.folder_name
252                                    )
253                                })?;
254                            let integrand = amplitude.integrand.as_mut().ok_or(eyre!(
255                                "Integrand {} has not yet been generated, but exists",
256                                amplitude.name
257                            ))?;
258                            integrand.warm_up(&model)?;
259                            integrand.get_settings().kinematics.e_cm
260                        }
261                        ProcessCollection::CrossSections(cross_sections) => {
262                            let cross_section =
263                                cross_sections.get_mut(&integrand_name).ok_or_else(|| {
264                                    eyre!(
265                                        "No cross section named '{}' in process '{}'",
266                                        integrand_name,
267                                        process.definition.folder_name
268                                    )
269                                })?;
270                            let integrand = cross_section.integrand.as_mut().ok_or(eyre!(
271                                "Integrand {} has not yet been generated, but exists",
272                                cross_section.name
273                            ))?;
274                            integrand.warm_up(&model)?;
275                            integrand.get_settings().kinematics.e_cm
276                        }
277                    }
278                };
279
280                let fixed_uv_ray = if uv_ray_directions.is_empty() {
281                    None
282                } else {
283                    let uv_ray_norms = if uv_ray_norms.is_empty() {
284                        vec![default_uv_ray_norm]
285                    } else {
286                        uv_ray_norms.clone()
287                    };
288                    Some(UVProfileFixedRay::from_flat_components(
289                        uv_ray_directions,
290                        &uv_ray_norms,
291                    )?)
292                };
293
294                let profile_settings = ProfileSettings {
295                    n_points: *n_points,
296                    min_scale_exponent: *min_scale_exponent,
297                    max_scale_exponent: *max_scale_exponent,
298                    seed: (*seed).unwrap_or(42),
299                    use_f128: *use_f128,
300                    analyse_analytically: *analyse_analytically,
301                    orientation_mode: if *per_orientation {
302                        OrientationProfileMode::PerOrientation
303                    } else {
304                        OrientationProfileMode::Summed
305                    },
306                    fixed_uv_ray,
307                    ..Default::default()
308                };
309                let profile_res = {
310                    let process = &mut state.process_list.processes[process_id];
311                    match &mut process.collection {
312                        ProcessCollection::Amplitudes(amplitudes) => amplitudes
313                            .get_mut(&integrand_name)
314                            .ok_or_else(|| {
315                                eyre!(
316                                    "No amplitude named '{}' in process '{}'",
317                                    integrand_name,
318                                    process.definition.folder_name
319                                )
320                            })?
321                            .profile(&model, &profile_settings)?,
322                        ProcessCollection::CrossSections(cross_sections) => cross_sections
323                            .get_mut(&integrand_name)
324                            .ok_or_else(|| {
325                                eyre!(
326                                    "No cross section named '{}' in process '{}'",
327                                    integrand_name,
328                                    process.definition.folder_name
329                                )
330                            })?
331                            .profile(&model, &profile_settings)?,
332                    }
333                }
334                .analyse();
335
336                for t in profile_res.tables_per_graph(-0.9) {
337                    info!("\n{}", t);
338                }
339
340                for t in profile_res.analytic_tables_per_graph() {
341                    let Some(t) = t else {
342                        continue;
343                    };
344                    info!("\n{}", t);
345                }
346
347                for t in profile_res.per_orientation_tables_per_graph(-0.9) {
348                    let Some(t) = t else {
349                        continue;
350                    };
351                    info!("\n{}", t);
352                }
353
354                if let Some(file) = output_file {
355                    profile_res.write_profile_data(file)?
356                }
357
358                Ok(ProfileResult::UltraViolet(profile_res))
359            }
360            Profile::InfraRed(InfraRedProfile {
361                process,
362                integrand_name,
363                n_points,
364                min_scale_exponent,
365                max_scale_exponent,
366                seed,
367                output_file: _,
368                select,
369                per_orientation,
370                show_per_cut_info,
371            }) => {
372                let ir_profile_settings = IRProfileSetting {
373                    lambda_exp_start: *min_scale_exponent,
374                    lambda_exp_end: *max_scale_exponent,
375                    steps: *n_points,
376                    seed: seed.unwrap_or(420),
377                    select_limits_and_graphs: select.clone(),
378                    orientation_mode: if *per_orientation {
379                        OrientationProfileMode::PerOrientation
380                    } else {
381                        OrientationProfileMode::Summed
382                    },
383                    show_per_cut_info: *show_per_cut_info,
384                };
385
386                let (process_id, integrand_name) =
387                    state.find_integrand_ref(process.as_ref(), integrand_name.as_ref())?;
388
389                let model = state.resolve_model_for_integrand(process_id, &integrand_name)?;
390
391                let process = &mut state.process_list.processes[process_id];
392                let integrand = match &mut process.collection {
393                    ProcessCollection::Amplitudes(amplitudes) => amplitudes
394                        .get_mut(&integrand_name)
395                        .ok_or_else(|| {
396                            eyre!(
397                                "No amplitude named '{}' in process '{}'",
398                                integrand_name,
399                                process.definition.folder_name
400                            )
401                        })?
402                        .integrand
403                        .as_mut(),
404                    ProcessCollection::CrossSections(xs) => xs
405                        .get_mut(&integrand_name)
406                        .ok_or_else(|| {
407                            eyre!(
408                                "No xs named '{}' in process '{}'",
409                                integrand_name,
410                                process.definition.folder_name
411                            )
412                        })?
413                        .integrand
414                        .as_mut(),
415                };
416
417                let profile_result = match integrand.ok_or(eyre!(
418                    "Integrand {} has not yet been generated",
419                    integrand_name
420                ))? {
421                    ProcessIntegrand::CrossSection(cross_section_integrand) => {
422                        cross_section_integrand.test_ir(&ir_profile_settings, &model)?
423                    }
424                    ProcessIntegrand::Amplitude(amplitude_integrand) => {
425                        amplitude_integrand.test_ir(&ir_profile_settings, &model)?
426                    }
427                };
428
429                info!("\n{}", profile_result);
430                Ok(ProfileResult::InfraRed(profile_result))
431            }
432        }
433    }
