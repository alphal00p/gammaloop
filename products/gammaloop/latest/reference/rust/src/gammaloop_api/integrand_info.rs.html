72pub enum IntegrandThresholdStatus {
73    NoRadialDependence,
74    AlwaysPinched,
75    CanBecomePinched,
76    ProvenNonExisting,
77    PotentiallyExisting,
78}
79
80impl IntegrandThresholdStatus {
81    pub fn as_str(self) -> &'static str {
82        match self {
83            Self::NoRadialDependence => "no_radial_dependence",
84            Self::AlwaysPinched => "always_pinched",
85            Self::CanBecomePinched => "can_become_pinched",
86            Self::ProvenNonExisting => "proven_non_existing",
87            Self::PotentiallyExisting => "potentially_existing",
88        }
89    }
90
91    pub fn is_currently_viable(self) -> bool {
92        matches!(self, Self::CanBecomePinched | Self::PotentiallyExisting)
93    }
94
95    pub fn can_become_pinched(self) -> bool {
96        self == Self::CanBecomePinched
97    }
98}

106            ThresholdCountertermStatus::ProvenNonExisting => Self::ProvenNonExisting,
107            ThresholdCountertermStatus::PotentiallyExisting => Self::PotentiallyExisting,
108        }
109    }
110}

190impl IntegrandCutInfo {
191    fn from_cross_section_cut(
192        graph_term: &gammalooprs::integrands::process::cross_section::CrossSectionGraphTerm,
193        cut_id: CutId,
194        cut: &CrossSectionCut,
195        raising_power: usize,
196        model: &Model,
197        param_builder: &ParamBuilder,
198        settings: &RuntimeSettings,
199    ) -> Self {
200        let associations = &graph_term.cut_threshold_associations[cut_id];
201        let threshold_info = |association| {
202            IntegrandCutThresholdInfo::from_association(
203                association,
204                &graph_term.graph,
205                cut,
206                model,
207                param_builder,
208                settings,
209            )
210        };
211
212        Self {
213            // Retain `CutId` for internal indexing and convert only at the API boundary.
214            cut_id: usize::from(cut_id),
215            edge_ids: cut_edge_ids(&graph_term.graph, cut),
216            raising_power,
217            left_thresholds: associations.left.iter().map(&threshold_info).collect(),
218            right_thresholds: associations.right.iter().map(threshold_info).collect(),
219        }
220    }
221
222    fn active_threshold_cut(&self, esurface_id: usize) -> Option<IntegrandActiveThresholdCutInfo> {
223        let mut is_currently_viable = false;
224        let mut can_become_pinched = false;
225
226        for threshold in self
227            .left_thresholds
228            .iter()
229            .chain(self.right_thresholds.iter())
230            .filter(|threshold| threshold.esurface_id == esurface_id)
231        {
232            is_currently_viable |= threshold.status.is_currently_viable();
233            can_become_pinched |= threshold.status.can_become_pinched();
234        }
235
236        is_currently_viable.then_some(IntegrandActiveThresholdCutInfo {
237            cut_id: self.cut_id,
238            can_become_pinched,
239        })
240    }
241}

283pub(crate) fn collect_integrand_info(
284    state: &State,
285    process_id: usize,
286    integrand_name: &str,
287) -> Result<IntegrandInfo> {
288    let process = &state.process_list.processes[process_id];
289    let resolved = process.get_integrand(integrand_name)?;
290    let generated = resolved.require_generated()?;
291    let generation_compilation = generated.frozen_compilation().clone();
292    let active_f64_backend = generated.active_f64_backend();
293
294    match (generated, &process.collection) {
295        (
296            gammalooprs::integrands::process::ProcessIntegrand::Amplitude(integrand),
297            ProcessCollection::Amplitudes(amplitudes),
298        ) => {
299            let amplitude = amplitudes.get(&resolved.canonical_name).ok_or_else(|| {
300                eyre!(
301                    "Could not resolve amplitude record '{}' in process #{} ({})",
302                    resolved.canonical_name,
303                    process.definition.process_id,
304                    process.definition.folder_name
305                )
306            })?;
307            let model = state.resolve_model_for_integrand(
308                process.definition.process_id,
309                &resolved.canonical_name,
310            )?;
311            Ok(IntegrandInfo {
312                process_id: process.definition.process_id,
313                process_name: process.definition.folder_name.clone(),
314                integrand_name: resolved.canonical_name,
315                kind: IntegrandKind::Amplitude,
316                generation_compilation: generation_compilation.clone(),
317                active_f64_backend,
318                graph_count: amplitude.graphs.len(),
319                graph_group_count: amplitude.graph_group_structure.len(),
320                record_size_bytes: integrand_record_size_from_amplitude(amplitude)?,
321                graph_groups: amplitude_graph_groups(integrand, &model)?,
322            })
323        }
324        (
325            gammalooprs::integrands::process::ProcessIntegrand::CrossSection(integrand),
326            ProcessCollection::CrossSections(cross_sections),
327        ) => {
328            let cross_section = cross_sections
329                .get(&resolved.canonical_name)
330                .ok_or_else(|| {
331                    eyre!(
332                        "Could not resolve cross-section record '{}' in process #{} ({})",
333                        resolved.canonical_name,
334                        process.definition.process_id,
335                        process.definition.folder_name
336                    )
337                })?;
338            let model = state.resolve_model_for_integrand(
339                process.definition.process_id,
340                &resolved.canonical_name,
341            )?;
342            Ok(IntegrandInfo {
343                process_id: process.definition.process_id,
344                process_name: process.definition.folder_name.clone(),
345                integrand_name: resolved.canonical_name,
346                kind: IntegrandKind::CrossSection,
347                generation_compilation,
348                active_f64_backend,
349                graph_count: cross_section.supergraphs.len(),
350                graph_group_count: cross_section.graph_group_structure.len(),
351                record_size_bytes: integrand_record_size_from_cross_section(cross_section)?,
352                graph_groups: cross_section_graph_groups(integrand, &model)?,
353            })
354        }
355        _ => Err(eyre!(
356            "Process/integrand type mismatch for '{}' in process #{} ({})",
357            resolved.canonical_name,
358            process.definition.process_id,
359            process.definition.folder_name
360        )),
361    }
362}

364fn orientation_signature(orientation: &EdgeVec<Orientation>) -> Vec<i8> {
365    orientation
366        .iter()
367        .map(|(_, orientation)| match *orientation {
368            Orientation::Default => 1,
369            Orientation::Reversed => -1,
370            Orientation::Undirected => 0,
371        })
372        .collect()
373}

375fn orientation_edge_ids(orientation: &EdgeVec<Orientation>) -> Vec<usize> {
376    orientation.iter().map(|(edge_id, _)| edge_id.0).collect()
377}

379fn cut_edge_ids(graph: &Graph, cut: &CrossSectionCut) -> Vec<usize> {
380    cut.cut
381        .iter_edges(&graph.underlying)
382        .map(|(_, edge)| {
383            graph
384                .underlying
385                .iter_edges()
386                .find_map(|(_, edge_id, edge_data)| {
387                    (edge_data.data.name == edge.data.name).then_some(edge_id.0)
388                })
389                .expect("cut edge should resolve in graph")
390        })
391        .collect()
392}

394fn cut_raising_powers(
395    graph: &gammalooprs::integrands::process::cross_section::CrossSectionGraphTerm,
396) -> typed_index_collections::TiVec<CutId, usize> {
397    let mut raising_powers: typed_index_collections::TiVec<CutId, usize> =
398        vec![1; graph.cuts.len()].into();
399    for cut_group in graph.cut_group_data.cut_groups.iter() {
400        let raising_power = cut_group.related_esurface_group.max_occurence;
401        for cut_id in &cut_group.cuts {
402            raising_powers[*cut_id] = raising_power;
403        }
404    }
405    raising_powers
406}

408fn lmb_channel_ids(
409    lmbs: &typed_index_collections::TiVec<
410        gammalooprs::graph::LmbIndex,
411        gammalooprs::graph::LoopMomentumBasis,
412    >,
413    multi_channeling_setup: &LmbMultiChannelingSetup,
414    graph_name: &str,
415    parameterization_settings: &ParameterizationSettings,
416) -> Result<Vec<Option<usize>>> {
417    let mut channel_ids = vec![None; lmbs.len()];
418    for (channel_id, lmb_index) in multi_channeling_setup
419        .effective_channels(graph_name, parameterization_settings)?
420        .into_iter()
421        .enumerate()
422    {
423        channel_ids[usize::from(lmb_index)] = Some(channel_id);
424    }
425    Ok(channel_ids)
426}

428fn amplitude_graph_groups(
429    integrand: &gammalooprs::integrands::process::amplitude::AmplitudeIntegrand,
430    model: &Model,
431) -> Result<Vec<IntegrandGraphGroupInfo>> {
432    let parameterization_settings = integrand
433        .settings
434        .sampling
435        .get_parameterization_settings()
436        .unwrap_or_default();
437    let external_momenta = integrand
438        .settings
439        .kinematics
440        .externals
441        .get_dependent_externals::<f64>(DependentMomentaConstructor::Amplitude(
442            &integrand.data.external_signature,
443        ))
444        .context("While constructing amplitude external momenta for integrand information")?;
445    let e_cm = F(integrand.settings.kinematics.e_cm);
446    let existence_tolerance = F(integrand.settings.subtraction.esurface_existence_threshold);
447    let real_mass_vectors = integrand
448        .data
449        .graph_terms
450        .iter()
451        .map(|graph_term| graph_term.graph.get_real_mass_vector::<f64>(model))
452        .collect::<Vec<_>>();
453    integrand
454        .data
455        .graph_group_structure
456        .iter_enumerated()
457        .map(|(group_id, group)| {
458            let master_graph_id = group
459                .into_iter()
460                .next()
461                .expect("graph group should not be empty");
462            let master_graph = &integrand.data.graph_terms[master_graph_id];
463            let channel_ids = lmb_channel_ids(
464                &master_graph.lmbs,
465                &master_graph.multi_channeling_setup,
466                &master_graph.graph.name,
467                &parameterization_settings,
468            )?;
469            // Amplitude overlap centers are solved from the canonical union across all members
470            // of the graph group. Report that same union rather than the master graph's local
471            // raised-surface numbering.
472            let threshold_esurfaces = integrand.data.group_derived_data[group_id]
473                .esurface_map
474                .iter_enumerated()
475                .filter_map(|(group_esurface_id, raised_esurface_map)| {
476                    let (representative_graph_id, raised_esurface_id) = raised_esurface_map
477                        .iter_enumerated()
478                        .filter_map(|(graph_group_position, raised_esurface_id)| {
479                            raised_esurface_id.map(|raised_esurface_id| {
480                                let graph_id = group[graph_group_position];
481                                (graph_id, raised_esurface_id)
482                            })
483                        })
484                        .find(|(graph_id, raised_esurface_id)| {
485                            integrand.data.graph_terms[*graph_id]
486                                .threshold_counterterm
487                                .generated_mask[*raised_esurface_id]
488                        })?;
489                    let graph_term = &integrand.data.graph_terms[representative_graph_id];
490                    let local_esurface_id =
491                        graph_term.threshold_counterterm.raised_data.raised_groups
492                            [raised_esurface_id]
493                            .esurface_ids[0];
494
495                    let mut classification = IntegrandEsurfaceClassification::NonExisting;
496                    for (graph_group_position, raised_esurface_id) in raised_esurface_map
497                        .iter_enumerated()
498                        .filter_map(|(graph_group_position, raised_esurface_id)| {
499                            raised_esurface_id.map(|raised_esurface_id| {
500                                (graph_group_position, raised_esurface_id)
501                            })
502                        })
503                    {
504                        let graph_id = group[graph_group_position];
505                        let candidate = &integrand.data.graph_terms[graph_id];
506                        let candidate_esurface_id =
507                            candidate.threshold_counterterm.raised_data.raised_groups
508                                [raised_esurface_id]
509                                .esurface_ids[0];
510                        let candidate_status = candidate.esurfaces[candidate_esurface_id]
511                            .existence_status(
512                                &external_momenta,
513                                &candidate.graph.loop_momentum_basis,
514                                &real_mass_vectors[graph_id],
515                                &e_cm,
516                                &existence_tolerance,
517                            );
518                        match candidate_status {
519                            EsurfaceExistenceStatus::Existing => {
520                                classification = IntegrandEsurfaceClassification::Existing;
521                                break;
522                            }
523                            EsurfaceExistenceStatus::Pinched => {
524                                classification = IntegrandEsurfaceClassification::Pinched;
525                            }
526                            EsurfaceExistenceStatus::NonExisting => {}
527                        }
528                    }
529
530                    Some(IntegrandThresholdEsurfaceInfo {
531                        esurface_id: group_esurface_id.0,
532                        representative_graph_id,
533                        edge_ids: threshold_esurface_edge_ids(
534                            &graph_term.esurfaces,
535                            local_esurface_id.0,
536                        ),
537                        classification: Some(classification),
538                        active_cuts: Vec::new(),
539                    })
540                })
541                .collect::<Vec<_>>();
542            let threshold_esurface_ids = threshold_esurfaces
543                .iter()
544                .map(|threshold| threshold.esurface_id)
545                .collect();
546            Ok(IntegrandGraphGroupInfo {
547                group_id: usize::from(group_id),
548                graphs: group
549                    .into_iter()
550                    .map(|graph_id| IntegrandGraphInfo {
551                        graph_id,
552                        name: integrand.data.graph_terms[graph_id].graph.name.clone(),
553                        is_master: graph_id == master_graph_id,
554                    })
555                    .collect(),
556                orientation_edge_ids: master_graph
557                    .orientations
558                    .first()
559                    .map(orientation_edge_ids)
560                    .unwrap_or_default(),
561                orientations: master_graph
562                    .orientations
563                    .iter()
564                    .enumerate()
565                    .map(|(orientation_id, orientation)| IntegrandOrientationInfo {
566                        orientation_id,
567                        signature: orientation_signature(orientation),
568                    })
569                    .collect(),
570                loop_momentum_bases: master_graph
571                    .lmbs
572                    .iter_enumerated()
573                    .map(|(basis_id, lmb)| IntegrandLoopMomentumBasisInfo {
574                        basis_id: usize::from(basis_id),
575                        channel_id: channel_ids[usize::from(basis_id)],
576                        edge_ids: lmb.loop_edges.iter().map(|edge_id| edge_id.0).collect(),
577                        matches_generation_basis: lmb.loop_edges.len()
578                            == master_graph.graph.loop_momentum_basis.loop_edges.len()
579                            && lmb.loop_edges.iter().all(|edge_id| {
580                                master_graph
581                                    .graph
582                                    .loop_momentum_basis
583                                    .loop_edges
584                                    .contains(edge_id)
585                            }),
586                    })
587                    .collect(),
588                threshold_esurface_ids,
589                threshold_esurfaces,
590                cuts: Vec::new(),
591            })
592        })
593        .collect()
594}

596fn cross_section_graph_groups(
597    integrand: &gammalooprs::integrands::process::cross_section::CrossSectionIntegrand,
598    model: &Model,
599) -> Result<Vec<IntegrandGraphGroupInfo>> {
600    let parameterization_settings = integrand
601        .settings
602        .sampling
603        .get_parameterization_settings()
604        .unwrap_or_default();
605    integrand
606        .data
607        .graph_group_structure
608        .iter()
609        .enumerate()
610        .map(|(group_id, group)| {
611            let master_graph_id = group
612                .into_iter()
613                .next()
614                .expect("graph group should not be empty");
615            let master_graph = &integrand.data.graph_terms[master_graph_id];
616            let mut active_model_param_builder: ParamBuilder =
617                master_graph.graph.param_builder.clone();
618            active_model_param_builder.update_model_values(model);
619            let channel_ids = lmb_channel_ids(
620                &master_graph.lmbs,
621                &master_graph.multi_channeling_setup,
622                &master_graph.graph.name,
623                &parameterization_settings,
624            )?;
625            let cut_raising_powers = cut_raising_powers(master_graph);
626
627            let cuts = master_graph
628                .cuts
629                .iter_enumerated()
630                .map(|(cut_id, cut)| {
631                    IntegrandCutInfo::from_cross_section_cut(
632                        master_graph,
633                        cut_id,
634                        cut,
635                        cut_raising_powers[cut_id],
636                        model,
637                        &active_model_param_builder,
638                        &integrand.settings,
639                    )
640                })
641                .collect::<Vec<_>>();
642
643            let threshold_esurface_ids = master_graph
644                .threshold_candidate_esurface_ids
645                .iter()
646                .map(|esurface_id| esurface_id.0)
647                .collect::<Vec<_>>();
648
649            let threshold_esurfaces = threshold_esurface_ids
650                .iter()
651                .copied()
652                .map(|esurface_id| {
653                    let active_cuts = cuts
654                        .iter()
655                        .filter_map(|cut| cut.active_threshold_cut(esurface_id))
656                        .collect();
657
658                    IntegrandThresholdEsurfaceInfo {
659                        esurface_id,
660                        representative_graph_id: master_graph_id,
661                        edge_ids: threshold_esurface_edge_ids(
662                            &master_graph.graph.surface_cache.esurface_cache,
663                            esurface_id,
664                        ),
665                        classification: None,
666                        active_cuts,
667                    }
668                })
669                .collect::<Vec<_>>();
670
671            Ok(IntegrandGraphGroupInfo {
672                group_id,
673                graphs: group
674                    .into_iter()
675                    .map(|graph_id| IntegrandGraphInfo {
676                        graph_id,
677                        name: integrand.data.graph_terms[graph_id].graph.name.clone(),
678                        is_master: graph_id == master_graph_id,
679                    })
680                    .collect(),
681                orientation_edge_ids: master_graph
682                    .orientations
683                    .first()
684                    .map(orientation_edge_ids)
685                    .unwrap_or_default(),
686                orientations: master_graph
687                    .orientations
688                    .iter()
689                    .enumerate()
690                    .map(|(orientation_id, orientation)| IntegrandOrientationInfo {
691                        orientation_id,
692                        signature: orientation_signature(orientation),
693                    })
694                    .collect(),
695                loop_momentum_bases: master_graph
696                    .lmbs
697                    .iter_enumerated()
698                    .map(|(basis_id, lmb)| IntegrandLoopMomentumBasisInfo {
699                        basis_id: usize::from(basis_id),
700                        channel_id: channel_ids[usize::from(basis_id)],
701                        edge_ids: lmb.loop_edges.iter().map(|edge_id| edge_id.0).collect(),
702                        matches_generation_basis: lmb.loop_edges.len()
703                            == master_graph.graph.loop_momentum_basis.loop_edges.len()
704                            && lmb.loop_edges.iter().all(|edge_id| {
705                                master_graph
706                                    .graph
707                                    .loop_momentum_basis
708                                    .loop_edges
709                                    .contains(edge_id)
710                            }),
711                    })
712                    .collect(),
713                threshold_esurface_ids,
714                threshold_esurfaces,
715                cuts,
716            })
717        })
718        .collect()
719}

721fn integrand_record_size_from_amplitude(amplitude: &Amplitude) -> Result<usize> {
722    let mut record = amplitude.clone();
723    record.integrand = None;
724    let encoded =
725        bincode::encode_to_vec(&record, bincode::config::standard()).with_context(|| {
726            format!(
727                "While serializing amplitude '{}' for integrand info",
728                amplitude.name
729            )
730        })?;
731    Ok(encoded.len())
732}
