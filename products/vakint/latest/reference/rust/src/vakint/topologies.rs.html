231    pub fn count_propagators_in_integral(integral: AtomView) -> usize {
232        let mut n_props: usize = 0;
233        let p = vk_parse!("prop(args__)").unwrap().to_pattern();
234        let rhs_p = S.one.to_pattern();
235        let matcher = integral.replace(&p);
236        let mut match_iter = matcher.iter(&rhs_p);
237        while match_iter.next().is_some() {
238            n_props += 1;
239        }
240        n_props
241    }

243    pub fn generate_topologies_with_contractions(
244        canonical_expression: AtomView,
245        short_expression: AtomView,
246        contractions: TopologyContractions,
247        applicable_evaluation_methods: EvaluationOrder,
248    ) -> Result<Self, VakintError> {
249        let n_tot_props = Topologies::count_propagators_in_integral(canonical_expression);
250        if n_tot_props == 0 {
251            return Err(VakintError::InvalidIntegralFormat(format!(
252                "No propagators found in integral '{}'",
253                canonical_expression
254            )));
255        }
256        let mut topologies = vec![];
257
258        let unique_contractions = match contractions {
259            TopologyContractions::Custom(c) => c,
260            TopologyContractions::Automatic => {
261                let master_topology = Topology::generate_topology_with_contraction(
262                    n_tot_props,
263                    canonical_expression,
264                    short_expression,
265                    vec![],
266                    applicable_evaluation_methods.clone(),
267                )?;
268                let cs = master_topology
269                    .get_integral()
270                    .graph
271                    .find_unique_contractions();
272                topologies.push(master_topology);
273                cs
274            }
275        };
276
277        for contracted_prop_indices in unique_contractions {
278            topologies.push(Topology::generate_topology_with_contraction(
279                n_tot_props,
280                canonical_expression,
281                short_expression,
282                contracted_prop_indices,
283                applicable_evaluation_methods.clone(),
284            )?);
285        }
286        Ok(Topologies(topologies))
287    }

289    pub fn match_topologies_to_user_input(
290        &self,
291        input: AtomView,
292        allow_unknown_integrals: bool,
293    ) -> Result<Option<ReplacementRules>, VakintError> {
294        for topology in self.0.iter() {
295            if matches!(topology, Topology::Unknown(_)) && !allow_unknown_integrals {
296                continue;
297            }
298            if let Some(replacement_rules) = topology.match_topology_to_user_input(input)? {
299                return Ok(Some(replacement_rules));
300            }
301        }
302        Ok(None)
303    }

341    pub fn generate_topology_with_contraction(
342        n_tot_props: usize,
343        canonical_expression: AtomView,
344        short_expression: AtomView,
345        contraction: Vec<usize>,
346        applicable_evaluation_methods: EvaluationOrder,
347    ) -> Result<Topology, VakintError> {
348        let mut contracted_canonical_expression = canonical_expression.to_owned();
349        let mut contracted_short_expression = short_expression.to_owned();
350        let mut nodes_to_merge = vec![];
351
352        for prop_id in contraction.iter() {
353            if let Some(m) = get_prop_with_id(contracted_canonical_expression.as_view(), *prop_id) {
354                let (left_node_id, right_node_id) = get_node_ids(&m).unwrap();
355                if right_node_id > left_node_id {
356                    nodes_to_merge.push((right_node_id, left_node_id));
357                } else {
358                    nodes_to_merge.push((left_node_id, right_node_id));
359                }
360            } else {
361                return Err(VakintError::InvalidIntegralFormat(format!(
362                    "Cannot contract propagatore id {} as it is not found in integral: {}",
363                    prop_id, contracted_canonical_expression
364                )));
365            }
366            contracted_canonical_expression = contracted_canonical_expression
367                .replace(
368                    vk_parse!(format!("prop({},args__)", prop_id).as_str())
369                        .unwrap()
370                        .to_pattern(),
371                )
372                .with(S.one.to_pattern());
373
374            for a in [&mut contracted_short_expression] {
375                if !a.is_zero() {
376                    *a = a
377                        .replace(
378                            vk_parse!(format!("pow({})", prop_id).as_str())
379                                .unwrap()
380                                .to_pattern(),
381                        )
382                        .with(S.zero.to_pattern());
383                }
384            }
385        }
386
387        let mut old_contracted_canonical_expression = contracted_canonical_expression.clone();
388        'replace_contracted_nodes: loop {
389            for (old_node_id, new_node_id) in nodes_to_merge.iter() {
390                for (lhs, rhs) in [
391                    (
392                        format!("edge({},nr_)", old_node_id),
393                        format!("edge({},nr_)", new_node_id),
394                    ),
395                    (
396                        format!("edge(nl_,{})", old_node_id),
397                        format!("edge(nl_,{})", new_node_id),
398                    ),
399                ] {
400                    contracted_canonical_expression = contracted_canonical_expression
401                        .replace(vk_parse!(lhs.as_str()).unwrap().to_pattern())
402                        .with(vk_parse!(rhs.as_str()).unwrap().to_pattern());
403                }
404            }
405            if old_contracted_canonical_expression == contracted_canonical_expression {
406                break 'replace_contracted_nodes;
407            }
408            old_contracted_canonical_expression = contracted_canonical_expression.clone();
409        }
410
411        // let tt: Topology = Integral::new(
412        //     n_tot_props,
413        //     Some(contracted_canonical_expression.clone()),
414        //     Some(contracted_short_expression.clone()),
415        //     applicable_evaluation_methods.clone(),
416        // )?
417        // .into();
418        // println!("tt={}", tt);
419        contracted_canonical_expression =
420            Topology::force_an_lmb(contracted_canonical_expression.as_view(), n_tot_props)?;
421        // if contracted_short_expression.to_canonical_string()
422        //     == String::from("I3L(msq(1),0,pow(2),pow(3),pow(4),pow(5),0)")
423        // {
424        //     println!("after={}", tt);
425        //     panic!("TTTT");
426        // }
427        if !contraction.is_empty() {
428            contracted_short_expression = contracted_short_expression
429                .replace(vk_parse!("TopoName_(args___)").unwrap().to_pattern())
430                .when(Condition::from((
431                    vk_symbol!("TopoName_"),
432                    function_condition(),
433                )))
434                .with_map(move |match_in| {
435                    let topo_name = if let Match::FunctionName(f) =
436                        match_in.get(vk_symbol!("TopoName_")).unwrap()
437                    {
438                        f.to_string()
439                    } else {
440                        panic!("Topology name must correspond to a function name.");
441                    };
442                    let args = if let Match::Multiple(SliceType::Arg, aviews) =
443                        match_in.get(vk_symbol!("args___")).unwrap()
444                    {
445                        aviews
446                    } else {
447                        panic!("Topology arguments must be wrapped in Symbolica Arg construction.");
448                    };
449                    let mut res = FunctionBuilder::new(vk_symbol!(
450                        format!(
451                            "{}_pinch_{}",
452                            topo_name,
453                            contraction
454                                .iter()
455                                .map(|i| i.to_string())
456                                .collect::<Vec<_>>()
457                                .join("_")
458                        )
459                        .as_str()
460                    ));
461                    for a in args.iter() {
462                        res = a.to_owned().add_arg_to_function_builder(res);
463                    }
464                    res.finish()
465                });
466        }
467        Ok(Integral::new(
468            n_tot_props,
469            Some(contracted_canonical_expression),
470            Some(contracted_short_expression),
471            applicable_evaluation_methods,
472        )?
473        .into())
474    }

476    pub fn force_an_lmb(
477        input_canonical_expression: AtomView,
478        n_props: usize,
479    ) -> Result<Atom, VakintError> {
480        let mut loop_mom_ids: HashSet<i64> = HashSet::new();
481        let mut momenta = vec![];
482        let mut prop_ids = vec![];
483        for prop_id in 1..=n_props {
484            if let Some(m) = get_prop_with_id(input_canonical_expression, prop_id) {
485                prop_ids.push(prop_id);
486                let q = m.get(&vk_symbol!("q_")).unwrap().to_owned();
487                for (_s, (_a_id, mom_id)) in get_individual_momenta_from_atom(q.as_view())?.iter() {
488                    loop_mom_ids.insert(*mom_id);
489                }
490                momenta.push((prop_id, q));
491            }
492        }
493        if loop_mom_ids != HashSet::from_iter(1..=loop_mom_ids.len() as i64) {
494            return Err(VakintError::InvalidIntegralFormat(format!(
495                "Loop momenta must have IDs ranging from 1 to the number of loop momenta: {:?}",
496                loop_mom_ids
497            )));
498        }
499        let n_loops = loop_mom_ids.len();
500
501        let mut need_to_force_lmb = false;
502        for i_loop in 1..=n_loops {
503            if !momenta.iter().any(|(_i_prop, k)| {
504                k.as_view() == function!(S.k, Atom::num(i_loop as i64)).as_view()
505            }) {
506                need_to_force_lmb = true;
507                break;
508            }
509        }
510        // if !need_to_force_lmb && False {
511        if !need_to_force_lmb {
512            return Ok(input_canonical_expression.to_owned());
513        }
514
515        let g = Graph::new_from_atom(input_canonical_expression, n_props)?;
516
517        let lmb = g.get_one_lmb()?;
518        // println!("input_canonical_expression={}", input_canonical_expression);
519        // println!("lmb: {:?}", lmb);
520        if lmb.len() != n_loops {
521            return Err(VakintError::InvalidIntegralFormat(format!(
522                "Loop momentum basis identified does not have the same loop count ({}) as detected for this topology ({}).",
523                lmb.len(),
524                n_loops
525            )));
526        }
527
528        let mom_vecs_to_symbols = Arc::new(
529            (1..=n_loops)
530                .map(|i| {
531                    (
532                        function!(S.k, Atom::num(i as i64)).to_pattern(),
533                        (
534                            Atom::var(vk_symbol!(format!("k{}", i))).to_pattern(),
535                            (
536                                vk_symbol!(format!("k{}", i)),
537                                vk_symbol!(format!("krotated{}", i)),
538                            ),
539                        ),
540                    )
541                })
542                .collect::<Vec<_>>(),
543        );
544
545        let mut system = vec![];
546        for (i_lmb, lmb_edge_id) in lmb.iter().enumerate() {
547            if let Some(m) = get_prop_with_id(input_canonical_expression, *lmb_edge_id as usize) {
548                let mut q = m.get(&vk_symbol!("q_")).unwrap().to_owned();
549                for (src, (trgt, (_trgt_symbol, _trgt_rotated_symbol))) in
550                    mom_vecs_to_symbols.iter()
551                {
552                    q = q.replace(src).with(trgt.clone());
553                }
554                q -= Atom::var(mom_vecs_to_symbols[i_lmb].1.1.1);
555                system.push(q);
556            }
557        }
558        // println!(
559        //     "system: {:?}",
560        //     system.iter().map(|a| a.to_string()).collect::<Vec<_>>()
561        // );
562        let variables = mom_vecs_to_symbols
563            .iter()
564            .map(|(_src, (_trgt, (trgt_symbol, _trgt_rotated_symbol)))| Atom::var(*trgt_symbol))
565            .collect::<Vec<_>>();
566        // println!(
567        //     "variables: {:?}",
568        //     variables.iter().map(|a| a.to_string()).collect::<Vec<_>>()
569        // );
570        let basis_change = Arc::new(
571            match Atom::solve_linear_system::<u8, _, _>(
572                system
573                    .iter()
574                    .map(|a| a.as_view())
575                    .collect::<Vec<_>>()
576                    .as_slice(),
577                variables.as_slice(),
578            ) {
579                Ok(b) => b,
580                Err(e) => {
581                    return Err(VakintError::InvalidIntegralFormat(format!(
582                        "Could not solve the linear system to force the loop momentum basis: {:?}",
583                        e
584                    )));
585                }
586            },
587        );
588        // println!(
589        //     "basis_change: {:?}",
590        //     basis_change
591        //         .iter()
592        //         .map(|a| a.to_string())
593        //         .collect::<Vec<_>>()
594        // );
595
596        let mut rotated_result = input_canonical_expression.to_owned();
597        for prop_id in prop_ids {
598            rotated_result = rotated_result
599                .replace(
600                    vk_parse!(format!("prop({},edges_,q_,mUVsq_,pow_)", prop_id).as_str())
601                        .unwrap()
602                        .to_pattern(),
603                )
604                .with_map({
605                    let mom_vecs_to_symbols = mom_vecs_to_symbols.clone();
606                    let basis_change = basis_change.clone();
607                    move |match_in| {
608                        let mut q = match_in.get(vk_symbol!("q_")).unwrap().to_atom();
609                        for ((src, (_trgt, _trgt_symbol)), rotated_expr) in
610                            mom_vecs_to_symbols.iter().zip(basis_change.iter())
611                        {
612                            q = q.replace(src).with(rotated_expr.to_pattern());
613                        }
614                        // Map back symbols to k(i) atoms
615                        for (src, (_trgt, (_trgt_symbol, trgt_rotated_symbol))) in
616                            mom_vecs_to_symbols.iter()
617                        {
618                            q = q
619                                .replace(Atom::var(*trgt_rotated_symbol).to_pattern())
620                                .with(src.clone());
621                        }
622                        q = q.expand();
623                        function!(
624                            vk_symbol!("prop"),
625                            Atom::num(prop_id as i64),
626                            match_in.get(vk_symbol!("edges_")).unwrap().to_atom(),
627                            q,
628                            match_in.get(vk_symbol!("mUVsq_")).unwrap().to_atom(),
629                            match_in.get(vk_symbol!("pow_")).unwrap().to_atom()
630                        )
631                    }
632                });
633        }
634
635        // println!("Rotated topology: {}", rotated_result);
636
637        Ok(rotated_result)
638    }

640    pub fn get_integral(&self) -> &Integral {
641        match self {
642            Topology::OneLoop(i)
643            | Topology::TwoLoop(i)
644            | Topology::ThreeLoop(i)
645            | Topology::FourLoop(i)
646            | Topology::Unknown(i) => i,
647        }
648    }

652            Topology::OneLoop(i)
653            | Topology::TwoLoop(i)
654            | Topology::ThreeLoop(i)
655            | Topology::FourLoop(i)

660    pub fn to_canonical(
661        &self,
662        integral: AtomView,
663        replacement_rules: &ReplacementRules,
664        short_form: bool,
665    ) -> Atom {
666        match self {
667            Topology::Unknown(_) => integral
668                .replace(vk_parse!("topo(props_)").unwrap().to_pattern())
669                .with(vk_parse!("topo(UNKNOWN(props_))").unwrap().to_pattern()),
670            t => t.get_integral().to_canonical(replacement_rules, short_form),
671        }
672    }
673
674    fn match_topology_to_user_input(
675        &self,
676        input: AtomView,
677    ) -> Result<Option<ReplacementRules>, VakintError> {
678        match self {
679            Topology::Unknown(unknown_integral) => {
680                // println!(
681                //     "\n>>>Trying to match with the unknown integral pattern {}\n<<<",
682                //     self
683                // );
684                let undirected_input = input
685                    .replace(vk_parse!("edge(x_,y_)").unwrap().to_pattern())
686                    .with(vk_parse!("uedge(x_,y_)").unwrap().to_pattern());
687                if undirected_input
688                    .pattern_match(
689                        &unknown_integral.generic_pattern.pattern,
690                        Some(&unknown_integral.generic_pattern.conditions),
691                        Some(&unknown_integral.generic_pattern.match_settings),
692                    )
693                    .next()
694                    .is_some()
695                {
696                    // Make sure all propagators have IDs starting at 1 until n_props
697                    let mut test_input = input.to_owned();
698                    let mut prop_id = 1;
699                    let mut prop_pattern = vk_parse!(format!("prop({},args__)", prop_id).as_str())
700                        .unwrap()
701                        .to_pattern();
702                    while test_input
703                        .pattern_match(&prop_pattern, Some(&Condition::default()), None)
704                        .next()
705                        .is_some()
706                    {
707                        test_input = test_input
708                            .replace(&prop_pattern)
709                            .with(Atom::num(1).to_pattern());
710                        prop_id += 1;
711                        prop_pattern = vk_parse!(format!("prop({},args__)", prop_id).as_str())
712                            .unwrap()
713                            .to_pattern();
714                    }
715                    test_input = test_input
716                        .replace(vk_parse!("UNKNOWN(args__)").unwrap().to_pattern())
717                        .with(vk_parse!("args__").unwrap().to_pattern());
718                    if test_input != vk_parse!("topo(1)").unwrap() {
719                        return Err(VakintError::InvalidIntegralFormat(format!(
720                            "UNKNOWN integrals must have propagator ids ranging from 1 to their maximal number of propagators: {}",
721                            test_input
722                        )));
723                    }
724
725                    let mut loop_momenta_ids = HashSet::new();
726                    for i_prop in 1..=prop_id - 1 {
727                        for (mom_symbol, (_, mom_id)) in get_individual_momenta(
728                            get_prop_with_id(input, i_prop)
729                                .unwrap()
730                                .get(&vk_symbol!("q_"))
731                                .unwrap()
732                                .as_view(),
733                        )?
734                        .iter()
735                        {
736                            if *mom_symbol != S.k {
737                                return Err(VakintError::InvalidIntegralFormat(format!(
738                                    "Unknown integrals must have only loop momenta with symbol {}: {}",
739                                    S.k, mom_symbol
740                                )));
741                            }
742                            if *mom_id < 1 {
743                                return Err(VakintError::InvalidIntegralFormat(format!(
744                                    "Unknown integrals must have loop momenta ids starting from 1, not {}",
745                                    mom_id
746                                )));
747                            }
748                            loop_momenta_ids.insert(*mom_id);
749                        }
750                    }
751                    let mut sorted_loop_momenta_ids = loop_momenta_ids
752                        .iter()
753                        .map(|mom_id| *mom_id as usize)
754                        .collect::<Vec<_>>();
755                    sorted_loop_momenta_ids.sort();
756                    if sorted_loop_momenta_ids
757                        != (1..=sorted_loop_momenta_ids.len()).collect::<Vec<_>>()
758                    {
759                        return Err(VakintError::InvalidIntegralFormat(format!(
760                            "Unknown integrals must have loop momenta ids ranging from 1 to their maximal number of loop momenta, not {:?}",
761                            sorted_loop_momenta_ids
762                        )));
763                    }
764                    let n_loops = *sorted_loop_momenta_ids.last().unwrap();
765                    //println!("Found a match!");
766                    let mut replacement_rules = ReplacementRules {
767                        canonical_topology: Topology::Unknown(unknown_integral.clone()),
768                        ..ReplacementRules::default()
769                    };
770                    replacement_rules.canonical_expression_substitutions.insert(
771                        vk_parse!("integral").unwrap(),
772                        input
773                            .replace(vk_parse!("UNKNOWN(int_)").unwrap().to_pattern())
774                            .with(vk_parse!("int_").unwrap().to_pattern()),
775                    );
776                    replacement_rules.canonical_expression_substitutions.insert(
777                        vk_parse!("n_props").unwrap(),
778                        Atom::num((prop_id - 1) as i64),
779                    );
780                    replacement_rules
781                        .canonical_expression_substitutions
782                        .insert(vk_parse!("n_loops").unwrap(), Atom::num(n_loops as i64));
783                    Ok(Some(replacement_rules))
784                } else {
785                    //println!("Does not match!");
786                    Ok(None)
787                }
788            }
789            t => {
790                // println!("\n>>>Trying to match: {}\n<<<", self);
791                if let Some(mut replacement_rule) =
792                    t.get_integral().match_integral_to_user_input(input)?
793                {
794                    replacement_rule.canonical_topology = t.clone();
795                    // println!("Found a match! ->\n{}", replacement_rule);
796                    Ok(Some(replacement_rule))
797                } else {
798                    Ok(None)
799                }
800            }
801        }
802    }
