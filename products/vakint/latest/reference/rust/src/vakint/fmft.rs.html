39    pub fn process_fmft_form_output(
40        &self,
41        processed_form_output: Atom,
42    ) -> Result<Atom, VakintError> {
43        let mut res = processed_form_output.to_owned();
44        res = res
45            .replace(vk_parse!("d").unwrap().to_pattern())
46            .with(vk_parse!("4-2*ep").unwrap().to_pattern());
47        // Temporarily work with the variable "ep" instead of the UTF-8 symbol for epsilon
48        res = res
49            .replace(Atom::var(vk_symbol!(self.settings.epsilon_symbol.as_str())).to_pattern())
50            .with(vk_parse!("ep").unwrap().to_pattern());
51        res = res
52            .replace(
53                function!(S.vkdot, function!(S.p, S.id1_a), function!(S.p, S.id2_a)).to_pattern(),
54            )
55            .when(
56                &(Condition::from((S.id1_, number_condition()))
57                    & Condition::from((S.id2_, number_condition()))),
58            )
59            .with(function!(S.dot, function!(S.p, S.id1_a), function!(S.p, S.id2_a)).to_pattern());
60        Ok(res)
61    }

63    pub fn substitute_gam_functions(&self, result: AtomView) -> Atom {
64        let mut res = result.to_owned();
65        res = res
66            .replace(vk_parse!("Gam(x_,y_)").unwrap().to_pattern())
67            .with(
68                vk_parse!("exp(ep*y_*EulerGamma)*Gamma(x_+ep*y_)")
69                    .unwrap()
70                    .to_pattern(),
71            );
72        res = res
73            .replace(vk_parse!("iGam(x_,y_)").unwrap().to_pattern())
74            .with(
75                vk_parse!("exp(-ep*y_*EulerGamma)/Gamma(x_+ep*y_)")
76                    .unwrap()
77                    .to_pattern(),
78            );
79        res
80    }

82    pub fn expand_masters(&self, result: AtomView) -> Result<Atom, VakintError> {
83        let mut r = result.to_owned();
84        r.repeat_map(Box::new(move |av: AtomView| {
85            let mut res = av.to_owned();
86            res = self.substitute_gam_functions(res.as_view());
87            for (src, (trgt, restriction)) in MASTERS_EXPANSION.iter() {
88                res = res
89                    .replace(src.to_pattern())
90                    .when(restriction)
91                    .with(trgt.to_pattern());
92            }
93            res
94        }));
95        if let Some(m) = r
96            .pattern_match(&vk_parse!("GammaArgs(x_,y_)").unwrap().into(), None, None)
97            .next()
98        {
99            return Err(VakintError::FMFTError(format!(
100                "FMFT result contains a Gamma function whose numerical evaluation is not implemented in vakint: Gamma({}+{}*{})",
101                m.get(&S.x_).unwrap(),
102                m.get(&S.y_).unwrap(),
103                self.settings.epsilon_symbol
104            )));
105        }
106        Ok(r)
107    }

109    pub fn substitute_masters(&self, result: AtomView) -> Result<Atom, VakintError> {
110        let processed_constants = MASTERS_NUMERIC_SUBSTITUTIONS
111            .iter()
112            .map(|(src, (trgt, condition))| {
113                (
114                    src,
115                    (
116                        set_precision_in_polynomial_atom(
117                            trgt.as_view(),
118                            vk_symbol!("ep"),
119                            &self.settings,
120                        ),
121                        condition.clone(),
122                    ),
123                )
124            })
125            .collect::<Vec<_>>();
126        // for (a, (b, _c)) in processed_constants.clone() {
127        //     println!("{} -> {}", a, b);
128        // }
129        let mut r = result.to_owned();
130        r.repeat_map(Box::new(move |av: AtomView| {
131            let mut res = av.to_owned();
132            for (src, (trgt, matching_condition)) in processed_constants.iter() {
133                res = res
134                    .replace(src.to_pattern())
135                    .when(matching_condition)
136                    .with(trgt.to_pattern());
137            }
138            res
139        }));
140        // println!("DONE! {}", r);
141        Ok(r)
142    }

144    pub fn substitute_poly_gamma(&self, result: AtomView) -> Result<Atom, VakintError> {
145        let processed_constants = POLY_GAMMA_SUBSTITUTIONS
146            .iter()
147            .map(|(src, trgt)| {
148                (
149                    src,
150                    set_precision_in_float_atom(trgt.as_view(), &self.settings),
151                )
152            })
153            .collect::<Vec<_>>();
154        let mut r = result.to_owned();
155
156        r = r
157            .replace(vk_parse!("Gamma(n_)").unwrap().to_pattern())
158            .when(Condition::from((S.n_, gt_condition(0))))
159            .with_map(move |match_in| {
160                let n = get_integer_from_atom(match_in.get(S.n_).unwrap().to_atom().as_view())
161                    .unwrap() as u32;
162                Atom::num(Integer::factorial(n - 1))
163            });
164        r.repeat_map(Box::new(move |av: AtomView| {
165            let mut res = av.to_owned();
166            for (src, trgt) in processed_constants.iter() {
167                res = res.replace(src.to_pattern()).with(trgt.to_pattern());
168            }
169            res
170        }));
171
172        if let Some(m) = r
173            .pattern_match(&vk_parse!("PolyGamma(x_,y_)").unwrap().into(), None, None)
174            .next()
175        {
176            return Err(VakintError::FMFTError(format!(
177                "FMFT result contains a PolyGamma function whose numerical evaluation is not implemented in vakint: PolyGamma({},{})",
178                m.get(&S.x_).unwrap(),
179                m.get(&S.y_).unwrap()
180            )));
181        }
182        Ok(r)
183    }

187            .iter()

189                (
190                    src,
191                    set_precision_in_float_atom(trgt.as_view(), &self.settings),
192                )
193            })
194            .collect::<Vec<_>>();
195        let mut r = result.to_owned();
196        r.repeat_map(Box::new(move |av: AtomView| {
197            let mut res = av.to_owned();
198            for (src, trgt) in processed_constants.iter() {
199                res = res.replace(src.to_pattern()).with(trgt.to_pattern());
200            }
201            res
202        }));
203
204        Ok(r)
205    }

208impl Vakint {
209    pub fn fmft_evaluate(
210        &self,
211        settings: &VakintSettings,
212        input_numerator: AtomView,
213        integral_specs: &ReplacementRules,
214        options: &FMFTOptions,
215    ) -> Result<Atom, VakintError> {
216        let integral = integral_specs.canonical_topology.get_integral();
217
218        debug!(
219            "Processing the following integral with {}:\n{}",
220            "FMFT".green(),
221            integral
222        );
223
224        let err = Err(VakintError::InvalidGenericExpression(format!(
225            "Could not find the shorthand integral name in the FMFT expression: {}",
226            integral
227                .short_expression
228                .as_ref()
229                .map(|a| a.to_canonical_string())
230                .unwrap_or("None".to_string())
231        )));
232
233        let fmft = FMFT::with_settings(settings.clone());
234
235        let integral_name = if let Some(short_expression) = integral.short_expression.as_ref() {
236            if let Some(m) = short_expression
237                .pattern_match(
238                    &function!(S.fun_, S.any_a___).to_pattern(),
239                    Some(&Condition::from((S.fun_, symbol_condition()))),
240                    None,
241                )
242                .next()
243            {
244                AtomPrinter::new_with_options(
245                    m.get(&S.fun_).unwrap().as_atom_view(),
246                    PrintOptions::file_no_namespace(),
247                )
248                .to_string()
249            } else {
250                return err;
251            }
252        } else {
253            return err;
254        };
255
256        let (muv_atom, muv_sq_atom) =
257            Vakint::identify_uv_mass_symbols(integral.canonical_expression.as_ref().unwrap())?;
258
259        // Here we map the propagators in the correct order for the definition of the topology in FMFT
260        let vakint_to_fmft_edge_map = match integral_name.as_str().split("_pinch_").next().unwrap()
261        {
262            "I4L_H" => vec![2, 3, 4, 5, 6, 7, 8, 9, 1],
263            "I4L_X" => vec![2, 3, 4, 5, 6, 7, 8, 9, 10],
264            "I4L_BMW" => vec![3, 4, 5, 6, 7, 8, 9, 10],
265            "I4L_FG" => vec![1, 3, 4, 5, 6, 7, 8, 9],
266            _ => {
267                return Err(VakintError::InvalidGenericExpression(format!(
268                    "Integral {} is not supported by FMFT.",
269                    integral_name
270                )));
271            }
272        };
273
274        let mut numerator = Vakint::convert_to_dot_notation(input_numerator);
275
276        if utils::could_match(
277            &function!(S.dot, function!(S.p, S.id1_a), function!(S.k, S.id2_a)).to_pattern(),
278            numerator.as_view(),
279        ) {
280            return Err(VakintError::InvalidNumerator(format!(
281                "Make sure the numerator has been tensor-reduced before being processed by FMFT : {}",
282                numerator
283            )));
284        }
285
286        // Now map all exterior dot products into a special function `vkdot` so that it does not interfere with FMFT
287        // Do not forget to normalize by the dimensionality, which is muv^2 in this case.
288        numerator = numerator
289            .replace(
290                function!(S.dot, function!(S.p, S.id1_a), function!(S.p, S.id2_a)).to_pattern(),
291            )
292            .when(
293                &(Condition::from((S.id1_, number_condition()))
294                    & Condition::from((S.id2_, number_condition()))),
295            )
296            .with(
297                function!(S.vkdot, function!(S.p, S.id1_a), function!(S.p, S.id2_a)).to_pattern(),
298            );
299
300        // And finally map all interior products into the form p<i>.p<i> expected by fmft, where <i> is the edge id carrying momentum k<i>.
301        let momenta: HashMap<usize, Atom> = integral_specs.get_propagator_property_list("q_");
302        let mut lmb_prop_indices = vec![];
303        for i_loop in 1..=integral.n_loops {
304            if let Some((i_edge, _)) = momenta.iter().find(|(_i_prop, k)| {
305                k.as_view() == function!(S.k, Atom::num(i_loop as i64)).as_view()
306            }) {
307                lmb_prop_indices.push(*i_edge as isize);
308            } else {
309                lmb_prop_indices.push(-1);
310            }
311        }
312
313        let vakint_to_fmft_edge_map_copy = vakint_to_fmft_edge_map.clone();
314        let muv_sq_atom_clone = muv_sq_atom.clone();
315        numerator = numerator.replace(
316                function!(S.dot, function!(S.k, S.id1_a), function!(S.k, S.id2_a))
317            .to_pattern()).when(&(Condition::from((S.id1_, number_condition()))
318            & Condition::from((S.id2_, number_condition())))).with_map(
319                move |match_in| {
320                    let id1 = lmb_prop_indices[(get_integer_from_atom(match_in.get(S.id1_).unwrap().to_atom().as_view()).unwrap()-1) as usize];
321                    if id1 < 0 {
322                        panic!(
323                            "Could not find LMB edge for momentum k({}) in a topology supported by FMFT and used in numerator.",
324                            get_integer_from_atom(match_in.get(S.id1_).unwrap().to_atom().as_view()).unwrap()
325                        );
326                    }
327                    let id2 = lmb_prop_indices[(get_integer_from_atom(match_in.get(S.id2_).unwrap().to_atom().as_view()).unwrap()-1) as usize];
328                    if id2 < 0 {
329                        panic!(
330                            "Could not find LMB edge for momentum k({}) in a topology supported by FMFT and used in numerator.",
331                            get_integer_from_atom(match_in.get(S.id2_).unwrap().to_atom().as_view()).unwrap()
332                        );
333                    }
334                    let i_edge1 =
335                        vakint_to_fmft_edge_map_copy[(id1 as usize) - 1];
336                    let i_edge2 =
337                        vakint_to_fmft_edge_map_copy[(id2 as usize) - 1];
338                    // in FMFT, the loop momenta dot products need to be written p<i>.p<i>
339                    // The outter dot will be converted to an inner dot in the next step
340                    // Again we must normalize by the dimensionality muv^2
341                    vk_parse!(format!("dot(p{},p{})", i_edge1, i_edge2).as_str()).unwrap()
342                        * (muv_sq_atom_clone.clone())
343                }
344            );
345
346        // Substitute eps by (4-d)/2
347        numerator = numerator
348            .replace(vk_parse!(&settings.epsilon_symbol).unwrap().to_pattern())
349            .with(vk_parse!("(4-d)/2").unwrap().to_pattern());
350
351        let muv_symbol = match (muv_atom.as_view(), muv_sq_atom.as_view()) {
352            (AtomView::Var(s), _) => s.get_symbol(),
353            (_, AtomView::Var(s2)) => s2.get_symbol(),
354            _ => {
355                return Err(VakintError::MalformedGraph(
356                    "Could not identify the UV mass symbol in the integral.".into(),
357                ));
358            }
359        };
360
361        let (form_header_additions, expression_str, indices) =
362            self.sanitize_user_expressions(settings, numerator.as_view(), true, &[muv_symbol])?;
363
364        // let expression_str =
365        //     &AtomPrinter::new_with_options(numerator.as_view(), PrintOptions::file_no_namespace())
366        //         .to_string();
367
368        let dot_produce_replacer =
369            Regex::new(r"dot\((?<vecA>[\w|\d]+),(?<vecB>[\w|\d]+)\)").unwrap();
370        let numerator_string = dot_produce_replacer
371            .replace_all(&expression_str, "($vecA.$vecB)")
372            .to_string();
373
374        let powers = (1..=integral.n_props)
375            .map(|i_prop| {
376                integral_specs
377                    .canonical_expression_substitutions
378                    .get(&function!(S.pow, Atom::num(i_prop as i64)))
379                    .map(|a| a.try_into().unwrap())
380                    .unwrap_or(0)
381            })
382            .collect::<Vec<_>>();
383
384        let integral_string = powers
385            .iter()
386            .zip(vakint_to_fmft_edge_map)
387            .map(|(pwr, fmft_edge_index)| format!("d{}^{}", fmft_edge_index, -pwr))
388            .collect::<Vec<_>>()
389            .join("*");
390
391        // Replace functions with 1 and get all remaining symbols
392        // let mut numerator_additional_symbols = input_numerator
393        //     .replace(vk_parse!("f_(args__)").unwrap().to_pattern())
394        //     .with(vk_parse!("1").unwrap().to_pattern())
395        //     .get_all_symbols(false);
396        // let eps_symbol = vk_symbol!(settings.epsilon_symbol.clone());
397        // numerator_additional_symbols.retain(|&s| s != eps_symbol);
398        let numerator_additional_symbols: std::collections::HashSet<
399            symbolica::atom::Symbol,
400            ahash::RandomState,
401        > = std::collections::HashSet::default();
402
403        let template = Template::parse_template(TEMPLATES.get("run_fmft.txt").unwrap()).unwrap();
404        let mut vars: HashMap<String, String> = HashMap::new();
405        vars.insert("numerator".into(), numerator_string);
406        vars.insert("integral".into(), integral_string);
407        // match (muv_atom.as_view(), muv_sq_atom.as_view()) {
408        //     (AtomView::Var(s), _) => vars.insert("symbols".into(), s.get_symbol().to_string()),
409        //     (_, AtomView::Var(s2)) => vars.insert("symbols".into(), s2.get_symbol().to_string()),
410        //     _ => {
411        //         return Err(VakintError::MalformedGraph(
412        //             "Could not identify the UV mass symbol in the integral.".into(),
413        //         ));
414        //     }
415        // };
416        vars.insert("symbols".into(), "NOPREDEDUBEDVAKINTSYMBOL".into());
417
418        let mut additional_symbols_str = if !numerator_additional_symbols.is_empty() {
419            format!(
420                "Auto S {};\n",
421                numerator_additional_symbols
422                    .iter()
423                    .map(|item| item.get_stripped_name())
424                    .collect::<Vec<_>>()
425                    .join(", "),
426            )
427        } else {
428            "".into()
429        };
430        additional_symbols_str = format!(
431            "{}\nCF {};",
432            additional_symbols_str,
433            undress_vakint_symbols(&get_full_name(&S.g))
434        );
435        if !form_header_additions.is_empty() {
436            if !additional_symbols_str.is_empty() {
437                additional_symbols_str =
438                    format!("{}\n{}", form_header_additions, additional_symbols_str);
439            } else {
440                additional_symbols_str = form_header_additions;
441            }
442        }
443        vars.insert("additional_symbols".into(), additional_symbols_str);
444
445        let rendered = template
446            .render(&RenderOptions {
447                variables: vars,
448                ..Default::default()
449            })
450            .unwrap();
451
452        let form_result = self.run_form(
453            settings,
454            &["fmft.frm".into()],
455            ("run_fmft.frm".into(), rendered),
456            vec![],
457            settings.clean_tmp_dir,
458            settings.temporary_directory.clone(),
459        )?;
460
461        let processed_form_result =
462            self.process_form_output(settings, form_result, indices, BTreeMap::new())?;
463        let mut evaluated_integral = fmft.process_fmft_form_output(processed_form_result)?;
464        debug!(
465            "{}: raw result from FORM:\n{}",
466            "FMFT".green(),
467            evaluated_integral
468        );
469
470        // Restore dimensionality now
471
472        // Offset dimensionality by the denominators
473        let muv_sq_dimension = 2 * (integral.n_loops as i64) - powers.iter().sum::<i64>();
474
475        evaluated_integral *= vk_parse!(
476            format!(
477                "({})^{}",
478                muv_sq_atom.to_canonical_string(),
479                muv_sq_dimension
480            )
481            .as_str()
482        )
483        .unwrap();
484
485        let fmft_normalization_correction = vk_parse!(
486            format!(
487                "(
488                (𝑖*(𝜋^((4-2*{eps})/2)))\
489              * (exp(-EulerGamma))^({eps})\
490              * (exp(-logmUVmu-log_mu_sq))^({eps})\
491             )^{n_loops}",
492                eps = settings.epsilon_symbol,
493                n_loops = integral.n_loops
494            )
495            .as_str()
496        )
497        .unwrap();
498
499        // Adjust normalization factor
500        let mut complete_normalization = fmft_normalization_correction
501            * settings
502                .get_integral_normalization_factor_atom()?
503                .replace(S.n_loops.to_pattern())
504                .with(Atom::num(integral.n_loops as i64).to_pattern());
505        complete_normalization = complete_normalization
506            .replace(Atom::var(vk_symbol!(settings.epsilon_symbol.as_str())).to_pattern())
507            .with(vk_parse!("ep").unwrap().to_pattern());
508
509        evaluated_integral *= complete_normalization;
510
511        if options.expand_masters {
512            let expansion_depth =
513                settings.number_of_terms_in_epsilon_expansion - (integral.n_loops as i64) - 1;
514            debug!(
515                "{}: Expanding master integrals with terms up to and including {}^{} ...",
516                "FMFT".green(),
517                settings.epsilon_symbol,
518                expansion_depth
519            );
520            evaluated_integral = fmft.expand_masters(evaluated_integral.as_view())?;
521
522            debug!(
523                "{}: Series expansion of the result up to and including terms of order {}^{} ...",
524                "FMFT".green(),
525                settings.epsilon_symbol,
526                expansion_depth
527            );
528            evaluated_integral = match evaluated_integral.series(
529                vk_symbol!("ep"),
530                Atom::Zero.as_view(),
531                Rational::from(expansion_depth),
532            ) {
533                Ok(a) => a,
534                Err(e) => return Err(VakintError::SymbolicaError(e.to_string())),
535            }
536            .to_atom();
537
538            // Sanity check
539            if let Some(m) = evaluated_integral
540                .pattern_match(&vk_parse!("Oep(x_,y_)").unwrap().to_pattern(), None, None)
541                .next()
542            {
543                return Err(VakintError::FMFTError(format!(
544                    "FMFT expansion yielded terms beyond expansion depth supported: Oep({},{})",
545                    m.get(&S.x_).unwrap(),
546                    m.get(&S.y_).unwrap(),
547                )));
548            }
549
550            if options.susbstitute_masters {
551                debug!(
552                    "{}: Substituting master integrals coefficient with their numerical evaluations...",
553                    "FMFT".green()
554                );
555                evaluated_integral = fmft.substitute_masters(evaluated_integral.as_view())?;
556                debug!(
557                    "{}: Substituting PolyGamma and period constants...",
558                    "FMFT".green()
559                );
560                evaluated_integral = evaluated_integral.expand();
561                evaluated_integral = fmft.substitute_poly_gamma(evaluated_integral.as_view())?;
562                evaluated_integral =
563                    fmft.substitute_additional_constants(evaluated_integral.as_view())?;
564                // Sanity check
565                if let Some(m) = evaluated_integral
566                    .pattern_match(&vk_parse!("Oep(x_,y_)").unwrap().to_pattern(), None, None)
567                    .next()
568                {
569                    return Err(VakintError::FMFTError(format!(
570                        "FMFT expansion yielded terms beyond expansion depth supported: Oep({},{})",
571                        m.get(&S.x_).unwrap(),
572                        m.get(&S.y_).unwrap(),
573                    )));
574                }
575            }
576        }
577
578        evaluated_integral = evaluated_integral
579            .replace(vk_parse!("ep").unwrap().to_pattern())
580            .with(Atom::var(vk_symbol!(settings.epsilon_symbol.as_str())).to_pattern());
581
582        if !settings.use_dot_product_notation {
583            evaluated_integral = Vakint::convert_from_dot_notation(evaluated_integral.as_view());
584        }
585
586        let log_muv_mu_sq = function!(
587            Symbol::LOG,
588            muv_sq_atom / Atom::var(vk_symbol!(settings.mu_r_sq_symbol.as_str()))
589        );
590
591        let log_mu_sq = function!(
592            Symbol::LOG,
593            Atom::var(vk_symbol!(settings.mu_r_sq_symbol.as_str()))
594        );
595
596        evaluated_integral = evaluated_integral
597            .replace(vk_parse!("logmUVmu").unwrap().to_pattern())
598            .with((log_muv_mu_sq).to_pattern());
599        evaluated_integral = evaluated_integral
