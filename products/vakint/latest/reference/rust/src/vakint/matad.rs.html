172            res = self.substitute_gam_functions(res.as_view());
173            for (src, (trgt, matching_condition)) in processed_constants.iter() {
174                res = res
175                    .replace(src.to_pattern())
176                    .when(matching_condition)
177                    .with(trgt.to_pattern());
178            }
179            res
180        }));

182        // println!("DONE! {}", r);
183        Ok(r)
184    }

187        let mut r = result.to_owned();

189            let mut res = av.to_owned();

192                res = res
193                    .replace(src.to_pattern())
194                    .when(restriction)
195                    .with(trgt.to_pattern());

321impl Vakint {
322    pub fn matad_evaluate(
323        &self,
324        settings: &VakintSettings,
325        input_numerator: AtomView,
326        integral_specs: &ReplacementRules,
327        options: &MATADOptions,
328    ) -> Result<Atom, VakintError> {
329        let integral = integral_specs.canonical_topology.get_integral();
330
331        debug!(
332            "Processing the following integral with {}:\n{}",
333            "MATAD".green(),
334            integral
335        );
336
337        let err = Err(VakintError::InvalidGenericExpression(format!(
338            "Could not find the shorthand integral name in the MATAD expression: {}",
339            integral
340                .short_expression
341                .as_ref()
342                .map(|a| a.to_canonical_string())
343                .unwrap_or("None".to_string())
344        )));
345
346        let matad = MATAD::with_settings(settings.clone());
347
348        let integral_name = if let Some(short_expression) = integral.short_expression.as_ref() {
349            if let Some(m) = short_expression
350                .pattern_match(
351                    &function!(S.fun_, S.any_a___).to_pattern(),
352                    Some(&Condition::from((S.fun_, symbol_condition()))),
353                    None,
354                )
355                .next()
356            {
357                AtomPrinter::new_with_options(
358                    m.get(&S.fun_).unwrap().as_atom_view(),
359                    PrintOptions::file_no_namespace(),
360                )
361                .to_string()
362            } else {
363                return err;
364            }
365        } else {
366            return err;
367        };
368
369        let (muv_atom, muv_sq_atom) =
370            Vakint::identify_uv_mass_symbols(integral.canonical_expression.as_ref().unwrap())?;
371
372        // Here we map the propagators in the correct order for the definition of the topology in MATAD
373        let vakint_to_matad_edge_map = match integral_name.as_str().split("_pinch_").next().unwrap()
374        {
375            "I1L" => vec![1],
376            "I2L" => vec![2, 3, 1],
377            "I3L" => vec![4, 5, 6, 1, 2, 3],
378            _ => {
379                return Err(VakintError::InvalidGenericExpression(format!(
380                    "Integral {} is not supported by MATAD.",
381                    integral_name
382                )));
383            }
384        };
385
386        let mut numerator = Vakint::convert_to_dot_notation(input_numerator);
387
388        // println!("Numerator before processing: {}", numerator);
389        numerator = numerator.replace_multiple(&[
390            Replacement::new(
391                muv_sq_atom.to_pattern(),
392                vk_parse!("M^2").unwrap().to_pattern(),
393            ),
394            Replacement::new(muv_atom.to_pattern(), vk_parse!("M").unwrap().to_pattern()),
395        ]);
396
397        if utils::could_match(
398            &function!(S.dot, function!(S.p, S.id1_a), function!(S.k, S.id2_a)).to_pattern(),
399            numerator.as_view(),
400        ) {
401            return Err(VakintError::InvalidNumerator(format!(
402                "Make sure the numerator has been tensor-reduced before being processed by MATAD : {}",
403                numerator
404            )));
405        }
406
407        // Now map all exterior dot products into a special function `vkdot` so that it does not interfere with MATAD
408        numerator = numerator
409            .replace(
410                function!(S.dot, function!(S.p, S.id1_a), function!(S.p, S.id2_a)).to_pattern(),
411            )
412            .when(
413                &(Condition::from((S.id1_, number_condition()))
414                    & Condition::from((S.id2_, number_condition()))),
415            )
416            .with(
417                function!(S.vkdot, function!(S.p, S.id1_a), function!(S.p, S.id2_a)).to_pattern(),
418            );
419
420        // And finally map all interior products into the form p<i>.p<i> expected by matad, where <i> is the edge id carrying momentum k<i>.
421        let momenta = integral_specs.get_propagator_property_list("q_");
422        let mut lmb_prop_indices = vec![];
423        for i_loop in 1..=integral.n_loops {
424            if let Some((i_edge, _)) = momenta.iter().find(|(_i_prop, k)| {
425                k.as_view() == function!(S.k, Atom::num(i_loop as i64)).as_view()
426            }) {
427                lmb_prop_indices.push(*i_edge as isize);
428            } else {
429                lmb_prop_indices.push(-1);
430            }
431        }
432
433        let vakint_to_matad_edge_map_copy = vakint_to_matad_edge_map.clone();
434
435        numerator = numerator.replace(
436                function!(S.dot, function!(S.k, S.id1_a), function!(S.k, S.id2_a))
437            .to_pattern()).when(&(Condition::from((S.id1_, number_condition()))
438            & Condition::from((S.id2_, number_condition())))).with_map(
439                move |match_in| {
440                    let id1 = lmb_prop_indices[(get_integer_from_atom(match_in.get(S.id1_).unwrap().to_atom().as_view()).unwrap()-1) as usize];
441                    if id1 < 0 {
442                        panic!(
443                            "Could not find LMB edge for momentum k({}) in a topology supported by MATAD and used in numerator.",
444                            get_integer_from_atom(match_in.get(S.id1_).unwrap().to_atom().as_view()).unwrap()
445                        );
446                    }
447                    let id2 = lmb_prop_indices[(get_integer_from_atom(match_in.get(S.id2_).unwrap().to_atom().as_view()).unwrap()-1) as usize];
448                    if id2 < 0 {
449                        panic!(
450                            "Could not find LMB edge for momentum k({}) in a topology supported by MATAD and used in numerator.",
451                            get_integer_from_atom(match_in.get(S.id2_).unwrap().to_atom().as_view()).unwrap()
452                        );
453                    }
454                    let i_edge1 =
455                        vakint_to_matad_edge_map_copy[(id1 as usize) - 1];
456                    let i_edge2 =
457                        vakint_to_matad_edge_map_copy[(id2 as usize) - 1];
458                    // in MATAD, the loop momenta dot products need to be written p<i>.p<i>
459                    // The outter dot will be converted to an inner dot in the next step
460                    // Also keep in mind that in MATAD propagators are euclidean propagators so
461                    // a minus sign is necessary here.
462                    vk_parse!(format!("-dot(p{},p{})", i_edge1, i_edge2).as_str()).unwrap()
463                });
464
465        // Substitute eps by (4-d)/2
466        numerator = numerator
467            .replace(vk_parse!(&settings.epsilon_symbol).unwrap().to_pattern())
468            .with(vk_parse!("(4-d)/2").unwrap().to_pattern());
469
470        let muv_symbol = match (muv_atom.as_view(), muv_sq_atom.as_view()) {
471            (AtomView::Var(s), _) => s.get_symbol(),
472            (_, AtomView::Var(s2)) => s2.get_symbol(),
473            _ => {
474                return Err(VakintError::MalformedGraph(
475                    "Could not identify the UV mass symbol in the integral.".into(),
476                ));
477            }
478        };
479
480        // let indices = Vakint::identify_vector_indices(numerator.as_view())?;
481        // for (i_index, user_i) in indices.iter().enumerate() {
482        //     numerator = numerator.replace(user_i.to_pattern()).with(Atom::num(
483        //         (FORM_REPLACEMENT_INDEX_SHIFT + 1 + (i_index as u64)) as i64,
484        //     ));
485        // }
486        let (form_header_additions, expression_str, indices) =
487            self.sanitize_user_expressions(settings, numerator.as_view(), true, &[muv_symbol])?;
488
489        // let expression_str =
490        //     &AtomPrinter::new_with_options(numerator.as_view(), PrintOptions::file_no_namespace())
491        //         .to_string();
492
493        let dot_produce_replacer =
494            Regex::new(r"dot\((?<vecA>[\w|\d]+),(?<vecB>[\w|\d]+)\)").unwrap();
495        let numerator_string = dot_produce_replacer
496            .replace_all(&expression_str, "($vecA.$vecB)")
497            .to_string();
498        let powers = (1..=integral.n_props)
499            .map(|i_prop| {
500                integral_specs
501                    .canonical_expression_substitutions
502                    .get(&function!(S.pow, Atom::num(i_prop as i64)))
503                    .map(|a| a.try_into().unwrap())
504                    .unwrap_or(0)
505            })
506            .collect::<Vec<_>>();
507
508        let integral_string = powers
509            .iter()
510            .zip(vakint_to_matad_edge_map)
511            .map(|(pwr, matag_edge_index)| format!("s{}m^{}", matag_edge_index, pwr))
512            .collect::<Vec<_>>()
513            .join("*");
514
515        //println!("MATAD input string: {}", format!("({})*({})", numerator_string, integral_string));
516
517        // Replace functions with 1 and get all remaining symbols
518        // let mut numerator_additional_symbols: std::collections::HashSet<
519        //     symbolica::atom::Symbol,
520        //     ahash::RandomState,
521        // > = input_numerator
522        //     .replace(vk_parse!("f_(args__)").unwrap().to_pattern())
523        //     .with(vk_parse!("1").unwrap().to_pattern())
524        //     .get_all_symbols(false);
525        // let eps_symbol = vk_symbol!(settings.epsilon_symbol.clone());
526        // numerator_additional_symbols.retain(|&s| s != eps_symbol);
527        let numerator_additional_symbols: std::collections::HashSet<
528            symbolica::atom::Symbol,
529            ahash::RandomState,
530        > = HashSet::default();
531
532        let template = Template::parse_template(TEMPLATES.get("run_matad.txt").unwrap()).unwrap();
533        let mut vars: HashMap<String, String> = HashMap::new();
534        vars.insert("numerator".into(), numerator_string);
535
536        // match (muv_atom.as_view(), muv_sq_atom.as_view()) {
537        //     (AtomView::Var(s), _) => {
538        //         vars.insert("symbols".into(), get_full_name(&s.get_symbol()).into())
539        //     }
540        //     (_, AtomView::Var(s2)) => {
541        //         vars.insert("symbols".into(), get_full_name(&s2.get_symbol()).into())
542        //     }
543        //     _ => {
544        //         return Err(VakintError::MalformedGraph(
545        //             "Could not identify the UV mass symbol in the integral.".into(),
546        //         ));
547        //     }
548        // };
549        vars.insert("symbols".into(), "NOPREDEDUBEDVAKINTSYMBOL".into());
550        vars.insert("integral".into(), integral_string);
551        vars.insert("n_loops".into(), format!("{}", integral.n_loops));
552        let mut additional_symbols_str = if !numerator_additional_symbols.is_empty() {
553            format!(
554                "Auto S {};\n",
555                numerator_additional_symbols
556                    .iter()
557                    .map(|item| item.get_stripped_name())
558                    .collect::<Vec<_>>()
559                    .join(", "),
560            )
561        } else {
562            "".into()
563        };
564        additional_symbols_str = format!(
565            "{}\nCF {};",
566            additional_symbols_str,
567            undress_vakint_symbols(&get_full_name(&S.g))
568        );
569        if !form_header_additions.is_empty() {
570            if !additional_symbols_str.is_empty() {
571                additional_symbols_str =
572                    format!("{}\n{}", form_header_additions, additional_symbols_str);
573            } else {
574                additional_symbols_str = form_header_additions;
575            }
576        }
577        vars.insert("additional_symbols".into(), additional_symbols_str);
578
579        let rendered = template
580            .render(&RenderOptions {
581                variables: vars,
582                ..Default::default()
583            })
584            .unwrap();
585
586        let form_result = self.run_form(
587            settings,
588            &["matad-ng.hh".into()],
589            ("run_matad.frm".into(), rendered),
590            vec![],
591            settings.clean_tmp_dir,
592            settings.temporary_directory.clone(),
593        )?;
594
595        let processed_form_result =
596            self.process_form_output(settings, form_result, indices, BTreeMap::new())?;
597        let mut evaluated_integral = matad.process_matad_form_output(processed_form_result)?;
598        debug!(
599            "{}: raw result from FORM:\n{}",
600            "MATAD".green(),
601            evaluated_integral
602        );
603        evaluated_integral = evaluated_integral
604            .replace(vk_parse!("M^pow_").unwrap().to_pattern())
605            .when(Condition::from((S.pow_, even_condition())))
606            .with(
607                vk_parse!(format!("({})^(pow_/2)", muv_sq_atom.to_canonical_string()).as_str())
608                    .unwrap()
609                    .to_pattern(),
610            );
611        // Use may have "stupidly" also kept odd powers in the numerator
612        evaluated_integral = evaluated_integral
613            .replace(vk_parse!("M^pow_").unwrap().to_pattern())
614            .when(Condition::from((S.pow_, odd_condition())))
615            .with(
616                vk_parse!(format!("({})^pow_", muv_atom.to_canonical_string()).as_str())
617                    .unwrap()
618                    .to_pattern(),
619            );
620        evaluated_integral = evaluated_integral
621            .replace(vk_parse!("M").unwrap().to_pattern())
622            .when(Condition::from((S.pow_, odd_condition())))
623            .with(
624                vk_parse!(muv_atom.to_canonical_string().as_str())
625                    .unwrap()
626                    .to_pattern(),
627            );
628        let mut matad_normalization_correction = vk_parse!(
629            format!(
630                "(
631                    (𝑖*(𝜋^((4-2*{eps})/2)))\
632                  * (exp(-EulerGamma))^({eps})\
633                  * (exp(-logmUVmu-log_mu_sq))^({eps})\
634                 )^{n_loops}",
635                eps = settings.epsilon_symbol,
636                n_loops = integral.n_loops
637            )
638            .as_str()
639        )
640        .unwrap();
641
642        // Since MATAD uses euclidean denominator, we must adjust the overall sign by (-1) per quadratic denominator with power one.
643        matad_normalization_correction *=
644            vk_parse!(format!("((-1)^{})", powers.iter().sum::<i64>()).as_str()).unwrap();
645
646        // Adjust normalization factor
647        let mut complete_normalization = matad_normalization_correction
648            * settings
649                .get_integral_normalization_factor_atom()?
650                .replace(S.n_loops.to_pattern())
651                .with(Atom::num(integral.n_loops as i64).to_pattern());
652        complete_normalization = complete_normalization
653            .replace(Atom::var(vk_symbol!(settings.epsilon_symbol.as_str())).to_pattern())
654            .with(vk_parse!("ep").unwrap().to_pattern());
655
656        evaluated_integral *= complete_normalization;
657
658        if options.expand_masters {
659            let expansion_depth =
660                settings.number_of_terms_in_epsilon_expansion - (integral.n_loops as i64) - 1;
661            if options.direct_numerical_substition {
662                debug!(
663                    "{}: Substitute masters directly with their numerical values, with terms up to and including {}^{} ...",
664                    "MATAD".green(),
665                    settings.epsilon_symbol,
666                    expansion_depth
667                );
668                evaluated_integral =
669                    matad.substitute_masters_directly(evaluated_integral.as_view())?;
670            } else {
671                debug!(
672                    "{}: Expanding masters and including terms up to and including {}^{} ...",
673                    "MATAD".green(),
674                    settings.epsilon_symbol,
675                    expansion_depth
676                );
677                evaluated_integral = matad.expand_matad_masters(evaluated_integral.as_view())?;
678            }
679            // Temporary work around for series bug in Symbolica
680            // evaluated_integral = vk_parse!("(any___)^-1").unwrap().replace_all(
681            //     evaluated_integral.as_view(),
682            //     &PatternOrMap::Map(Box::new(move |match_in| {
683            //         Atom::num(1) / match_in.get(S.any___).unwrap().to_atom().expand()
684            //     })),
685            //     None,
686            //     None,
687            // );
688            debug!(
689                "{}: Series expansion of the result up to and including terms of order {}^{} ...",
690                "MATAD".green(),
691                settings.epsilon_symbol,
692                expansion_depth
693            );
694            evaluated_integral = match evaluated_integral.series(
695                vk_symbol!("ep"),
696                Atom::Zero.as_view(),
697                Rational::from(expansion_depth),
698            ) {
699                Ok(a) => a,
700                Err(e) => return Err(VakintError::SymbolicaError(e.to_string())),
701            }
702            .to_atom();
703
704            // Sanity check
705            if let Some(m) = evaluated_integral
706                .pattern_match(&vk_parse!("Oep(x_,y_)").unwrap().to_pattern(), None, None)
707                .next()
708            {
709                return Err(VakintError::MATADError(format!(
710                    "MATAD expansion yielded terms beyond expansion depth supported: Oep({},{})",
711                    m.get(&S.x_).unwrap(),
712                    m.get(&S.y_).unwrap(),
713                )));
714            }
715            if options.direct_numerical_substition {
716                debug!(
717                    "{}: Substituting PolyGamma and period constants...",
718                    "MATAD".green()
719                );
720                evaluated_integral = evaluated_integral.expand();
721                evaluated_integral = matad.substitute_poly_gamma(evaluated_integral.as_view())?;
722                evaluated_integral =
723                    matad.substitute_additional_constants(evaluated_integral.as_view())?;
724            } else if options.susbstitute_masters {
725                debug!("{}: Substituting masters with HPLs ...", "MATAD".green());
726                evaluated_integral = matad.substitute_masters(evaluated_integral.as_view())?;
727                if options.substitute_hpls {
728                    // Expanding here is important to improve efficiency and avoid symbolica bugs with floating point coefficients
729                    evaluated_integral = evaluated_integral.expand();
730                    debug!("{}: Substituting HPLs with numerics ...", "MATAD".green());
731                    evaluated_integral = matad.substitute_hpls(evaluated_integral.as_view())?;
732                    evaluated_integral =
733                        matad.substitute_poly_gamma(evaluated_integral.as_view())?;
734                    evaluated_integral =
735                        matad.substitute_additional_constants(evaluated_integral.as_view())?;
736                }
737            }
738            // Sanity check
739            if let Some(m) = evaluated_integral
740                .pattern_match(&vk_parse!("Oep(x_,y_)").unwrap().to_pattern(), None, None)
741                .next()
742            {
743                return Err(VakintError::FMFTError(format!(
744                    "FMFT expansion yielded terms beyond expansion depth supported: Oep({},{})",
745                    m.get(&S.x_).unwrap(),
746                    m.get(&S.y_).unwrap(),
747                )));
748            }
749        }
750
751        evaluated_integral = evaluated_integral
752            .replace(vk_parse!("ep").unwrap().to_pattern())
753            .with(Atom::var(vk_symbol!(settings.epsilon_symbol.as_str())).to_pattern());
754
755        if !settings.use_dot_product_notation {
756            evaluated_integral = Vakint::convert_from_dot_notation(evaluated_integral.as_view());
757        }

759        let log_muv_mu_sq = function!(
