1//! Vacuum projection through universal tensor kernels and opaque coefficients.
2//!
3//! Source coefficients remain Symbolica ASTs while loop degrees and tensor
4//! slots are reduced. The analytic numerator's epsilon dependence is then
5//! resolved before truncation; FORM receives coefficient-one tensor monomials.

20/// A coefficient ring over scalar-product monomials. This same owning
21/// representation is used for projected vacuum outputs and for scalar
22/// nonvacuum inputs; the latter may have dot(k,p) in their monomial keys.
23#[derive(Clone, Default)]
24pub(crate) struct ScalarTerms(pub(crate) BTreeMap<Atom, Atom>);

26impl ScalarTerms {
27    /// Resolve the supported Laurent dependence in the analytically integrated
28    /// numerator. Opaque tensor functions cannot supply an epsilon derivative;
29    /// ordinary mathematical functions use Symbolica's existing series algebra.
30    pub(crate) fn epsilon_series(
31        input: AtomView<'_>,
32        settings: &VakintSettings,
33        depth: symbolica::poly::series::SeriesDepth,
34    ) -> Result<symbolica::poly::series::Series<symbolica::domains::atom::AtomField>, VakintError>
35    {
36        fn check(view: AtomView<'_>, epsilon: Symbol) -> Result<(), VakintError> {
37            if !view.get_all_symbols(true).contains(&epsilon) {
38                return Ok(());
39            }
40            match view {
41                AtomView::Fun(function)
42                    if [Symbol::LOG, Symbol::EXP, Symbol::SIN, Symbol::COS]
43                        .contains(&function.get_symbol())
44                        && function.get_nargs() == 1 =>
45                {
46                    check(function.get(0), epsilon)
47                }
48                AtomView::Fun(_) => Err(VakintError::InvalidNumerator(
49                    "epsilon-dependent opaque functions must be resolved before Laurent expansion"
50                        .into(),
51                )),
52                AtomView::Add(sum) => sum.iter().try_for_each(|term| check(term, epsilon)),
53                AtomView::Mul(product) => {
54                    product.iter().try_for_each(|factor| check(factor, epsilon))
55                }
56                AtomView::Pow(power) => {
57                    let (base, exponent) = power.get_base_exp();
58                    check(base, epsilon)?;
59                    check(exponent, epsilon)
60                }
61                _ => Ok(()),
62            }
63        }
64        let epsilon = vk_symbol!(&settings.epsilon_symbol);
65        check(input, epsilon)?;
66        let series = input
67            .series(epsilon, 0, depth)
68            .map_err(|error| VakintError::SymbolicaError(error.to_string()))?;
69        if series.terms().any(|(power, coefficient)| {
70            !power.is_integer() || coefficient.get_all_symbols(true).contains(&epsilon)
71        }) {
72            return Err(VakintError::InvalidNumerator(
73                "the numerator requires integer Laurent powers with epsilon-independent coefficients".into(),
74            ));
75        }
76        Ok(series)
77    }

79    fn insert(&mut self, key: Atom, value: Atom) {
80        if value.is_zero() {
81            return;
82        }
83        let coefficient = self.0.entry(key.clone()).or_insert_with(Atom::zero);
84        *coefficient += value;
85        // Extracting loop monomials can leave a zero external coefficient in
86        // a factored form, also inside a nonzero coefficient. Certify additive
87        // subtrees before their parents, replacing only proven zeros. Keep all
88        // collection on scratch candidates so surviving factors stay intact.
89        // Collection can cycle between equivalent forms; never expand them.
90        *coefficient = coefficient.replace_map_bottom_up(|term, _, out| {
91            if !matches!(term, AtomView::Add(_)) {
92                return;
93            }
94            let mut candidate = term.to_owned();
95            let mut seen = BTreeSet::new();
96            while !candidate.is_zero() && seen.insert(candidate.clone()) {
97                // Numerical-content GCDs are safe only in the exact real
98                // rational domain. Functions stay opaque, as in collect_num;
99                // other powers can create complex or infinite coefficients.
100                let mut rational = true;
101                candidate.visitor(&mut |part| {
102                    match part {
103                        AtomView::Fun(_) => return false,
104                        AtomView::Num(number) => {
105                            let number = number.get_coeff_view();
106                            rational &= matches!(
107                                number,
108                                CoefficientView::Natural(..) | CoefficientView::Large(..)
109                            ) && number.is_real();
110                        }
111                        AtomView::Pow(power) => {
112                            rational &= i64::try_from(power.get_base_exp().1)
113                                .is_ok_and(|exponent| exponent > 0);
114                        }
115                        _ => {}
116                    }
117                    rational
118                });
119                candidate = if rational {
120                    candidate.collect_num()
121                } else {
122                    candidate.collect_by_coefficient()
123                };
124            }
125            if candidate.is_zero() {
126                **out = Atom::zero();
127            }
128        });
129        if coefficient.is_zero() {
130            self.0.remove(&key);
131        }
132    }

134    pub(crate) fn parse(view: AtomView<'_>) -> Result<Self, VakintError> {
135        // Select only complete loop-dot atoms. Symbolica's existing AtomField
136        // collector retains every external scalar/tensor AST as a coefficient;
137        // it never converts the whole numerator into a polynomial over Q(i).
138        fn inventory(view: AtomView<'_>, dots: &mut BTreeSet<Atom>) -> Result<i32, VakintError> {
139            if !view.get_all_symbols(true).contains(&S.k) {
140                return Ok(0);
141            }
142            let overflow = || {
143                VakintError::InvalidNumerator("total loop scalar-product degree exceeds i32".into())
144            };
145            match view {
146                AtomView::Fun(dot) if dot.get_symbol() == S.dot && dot.get_nargs() == 2 => {
147                    for vector in dot {
148                        let AtomView::Fun(vector) = vector else {
149                            return Err(VakintError::InvalidNumerator(
150                                "expected elementary vector".into(),
151                            ));
152                        };
153                        if ![S.k, S.p].contains(&vector.get_symbol())
154                            || vector.get_nargs() != 1
155                            || i64::try_from(vector.get(0)).is_err()
156                        {
157                            return Err(VakintError::InvalidNumerator(
158                                "expected elementary k(i) or p(i)".into(),
159                            ));
160                        }
161                    }
162                    dots.insert(view.to_owned());
163                    Ok(1)
164                }
165                AtomView::Add(sum) => sum
166                    .iter()
167                    .try_fold(0i32, |degree, term| Ok(degree.max(inventory(term, dots)?))),
168                AtomView::Mul(product) => product.iter().try_fold(0i32, |degree, factor| {
169                    degree
170                        .checked_add(inventory(factor, dots)?)
171                        .ok_or_else(overflow)
172                }),
173                AtomView::Pow(power) => {
174                    let (base, exponent) = power.get_base_exp();
175                    let Ok(exponent @ 0..=2147483646) = i64::try_from(exponent) else {
176                        return Err(VakintError::InvalidNumerator(
177                            "loop dependence requires a nonnegative integer power below i32::MAX"
178                                .into(),
179                        ));
180                    };
181                    inventory(base, dots)?
182                        .checked_mul(exponent as i32)
183                        .ok_or_else(overflow)
184                }
185                _ => Err(VakintError::InvalidNumerator(format!(
186                    "expected polynomial loop scalar products after contraction: {view}"
187                ))),
188            }
189        }
190        let mut dots = BTreeSet::new();
191        inventory(view, &mut dots)?;
192        let mut result = Self::default();
193        if dots.is_empty() {
194            result.insert(Atom::one(), view.to_owned());
195            return Ok(result);
196        }
197        let dots = dots.into_iter().collect::<Vec<_>>();
198        let mut used = view.get_all_symbols(true);
199        let mut serial = 0;
200        let mut aliases = BTreeMap::<Atom, Atom>::new();
201        // AtomField retains products of sums, but its polynomial collector
202        // can rewrite an unselected c^(a+b) as c^a*c^b. Hide maximal external
203        // composite coefficients too, so their complete AST survives intact.
204        let protected = view.replace_map(|term, _, out| {
205            if term.get_all_symbols(true).contains(&S.k) {
206                if matches!(term, AtomView::Fun(dot) if dot.get_symbol() == S.dot) {
207                    // A selected complete dot is one variable: do not visit
208                    // its momentum arguments or replace their integer IDs.
209                    **out = term.to_owned();
210                }
211            } else if !matches!(term, AtomView::Num(_) | AtomView::Var(_)) {
212                let alias = aliases.entry(term.to_owned()).or_insert_with(|| {
213                    loop {
214                        let symbol = vk_symbol!(format!("scalar_coefficient_{serial}"));
215                        serial += 1;
216                        if used.insert(symbol) {
217                            break Atom::var(symbol);
218                        }
219                    }
220                });
221                **out = alias.clone();
222            }
223        });
224        for (monomial, mut coefficient) in protected.coefficient_list::<i32>(&dots) {
225            for (original, alias) in &aliases {
226                coefficient = coefficient
227                    .replace(alias.to_pattern())
228                    .with(original.to_pattern());
229            }
230            if coefficient.get_all_symbols(true).contains(&S.k) {
231                return Err(VakintError::InvalidNumerator(
232                    "loop dependence remains outside a polynomial scalar-product monomial".into(),
233                ));
234            }
235            result.insert(monomial, coefficient);
236        }
237        Ok(result)
238    }

240    pub(crate) fn expression(&self) -> Atom {
241        self.0
242            .iter()
243            .fold(Atom::zero(), |sum, (monomial, coefficient)| {
244                sum + coefficient * monomial.clone()
245            })
246    }

248    /// Batch one synthetic scalar kernel, restoring each coefficient only
249    /// after the backend returns. Callers must reserve topology symbols too.
250    /// Resolve epsilon dependence before choosing the required backend order;
251    /// opaque tensor functions must already have epsilon-independent bodies.
252    #[allow(clippy::type_complexity)]
253    pub(crate) fn backend_kernel(
254        &self,
255        settings: &VakintSettings,
256        reserved: &[Atom],
257    ) -> Result<(Atom, Vec<(Atom, Atom)>, i64), VakintError> {
258        let mut used = self.expression().get_all_symbols(true);
259        for expression in reserved {
260            used.extend(expression.get_all_symbols(true));
261        }
262        let mut kernel = Atom::zero();
263        let mut aliases = Vec::new();
264        let mut extra_orders = 0;
265        let mut serial = 0;
266        for (monomial, coefficient) in &self.0 {
267            if coefficient.is_zero() {
268                continue;
269            }
270            let pole_order = Self::epsilon_pole_order(coefficient.as_view(), settings)?;
271            extra_orders = extra_orders.max(pole_order);
272            let symbol = loop {
273                let symbol = vk_symbol!(format!("vacuum_coefficient_{serial}"));
274                serial += 1;
275                if used.insert(symbol) {
276                    break symbol;
277                }
278            };
279            let alias = Atom::var(symbol);
280            kernel += &alias * monomial.clone();
281            aliases.push((alias, coefficient.clone()));
282        }
283        Ok((kernel, aliases, extra_orders))
284    }

286    /// Keep the conservative order for regular or evanescent coefficients;
287    /// only genuine Laurent poles require additional backend terms.
288    pub(crate) fn epsilon_pole_order(
289        input: AtomView<'_>,
290        settings: &VakintSettings,
291    ) -> Result<i64, VakintError> {
292        // Absolute order zero includes every pole without searching for a
293        // nonzero leading term in an exactly cancelling coefficient. Retain
294        // the complete coefficient for the independent final Laurent series.
295        let series = Self::epsilon_series(
296            input,
297            settings,
298            symbolica::poly::series::SeriesDepth::absolute(0),
299        )?;
300        series
301            .get_trailing_exponent()
302            .numerator()
303            .to_i64()
304            .and_then(i64::checked_neg)
305            .map(|order| order.max(0))
306            .ok_or_else(|| VakintError::InvalidNumerator("epsilon pole order exceeds i64".into()))
307    }

309    /// Every coefficient is evaluated at the supplied point before one
310    /// numerical integral is constructed; ERROR therefore stays correlated.
311    /// The owner supplies the inclusive coefficient order: the requested
312    /// answer's order plus a valid pole bound for its denominator class.
313    pub(crate) fn numerical_kernel(
314        &self,
315        vakint: &Vakint,
316        settings: &VakintSettings,
317        options: &PySecDecOptions,
318        coefficient_order: i64,
319        reserved: &[Atom],
320    ) -> Result<(Atom, PySecDecOptions, i64), VakintError> {
321        use symbolica::{
322            domains::{
323                float::{Complex, RealLike},
324                rational::Rational,
325            },
326            poly::series::SeriesDepth,
327        };
328
329        let _ = vakint.scalar_numerator(settings, self.expression().as_view())?;
330        let params = vakint.params_from_complex_f64(settings, &options.numerical_parameters);
331        let externals = options
332            .numerical_external_momenta
333            .iter()
334            .map(|(name, value)| {
335                let id = name
336                    .strip_prefix('p')
337                    .and_then(|id| id.parse::<usize>().ok())
338                    .ok_or_else(|| {
339                        VakintError::EvaluationError(format!(
340                            "external momentum name must be p followed by an integer: {name}"
341                        ))
342                    })?;
343                Ok((id, *value))
344            })
345            .collect::<Result<std::collections::HashMap<_, _>, VakintError>>()?;
346        let externals = vakint.externals_from_f64(settings, &externals);
347        let mut values = Vec::new();
348        let mut common_power = 0;
349        for (monomial, coefficient) in &self.0 {
350            let series = Self::epsilon_series(
351                coefficient.as_view(),
352                settings,
353                SeriesDepth::absolute(coefficient_order),
354            )?;
355            for (power, coefficient) in series.terms() {
356                if !power.is_integer() {
357                    return Err(VakintError::InvalidNumerator(
358                        "the numerator requires integer Laurent powers of epsilon".into(),
359                    ));
360                }
361                let power = power.numerator().to_i64().ok_or_else(|| {
362                    VakintError::InvalidNumerator("epsilon power exceeds i64".into())
363                })?;
364                // Evaluate each epsilon-independent coefficient. Passing the
365                // full Laurent polynomial would unnecessarily narrow powers
366                // through the numeric helper's i8 coefficient collector.
367                let evaluated = Vakint::full_numerical_evaluation_without_error(
368                    settings,
369                    coefficient.as_view(),
370                    &Default::default(),
371                    &params,
372                    Some(&externals),
373                )?;
374                let mut value = Complex::new(0.0, 0.0);
375                for (coefficient_power, term) in evaluated.0 {
376                    if coefficient_power != 0 {
377                        return Err(VakintError::InvalidNumerator(
378                            "contract Lorentz traces before constructing scalar coefficient sectors".into(),
379                        ));
380                    }
381                    value.re += term.re.to_f64();
382                    value.im += term.im.to_f64();
383                }
384                if !value.re.is_finite() || !value.im.is_finite() {
385                    return Err(VakintError::EvaluationError(
386                        "a numerical numerator coefficient is not finite".into(),
387                    ));
388                }
389                if value.re == 0. && value.im == 0. {
390                    continue;
391                }
392                common_power = common_power.min(power);
393                values.push((monomial.clone(), power, value));
394            }
395        }
396        let mut used = self.expression().get_all_symbols(true);
397        for expression in reserved {
398            used.extend(expression.get_all_symbols(true));
399        }
400        let mut options = options.clone();
401        let mut kernel = Atom::zero();
402        let epsilon = Atom::var(vk_symbol!(&settings.epsilon_symbol));
403        let mut serial = 0;
404        for (monomial, power, value) in values {
405            let shifted_power = power.checked_sub(common_power).ok_or_else(|| {
406                VakintError::InvalidNumerator("epsilon power shift exceeds i64".into())
407            })?;
408            if !settings.project_onto_tensor_integrals {
409                // Numerical integration receives one complete polynomial with
410                // actual complex coefficients, rather than synthetic aliases.
411                // PySecDec forwards these literals to FORM, which requires
412                // exact rationals. Preserve every bit of the evaluated point.
413                let coefficient = Atom::num(Complex::new(
414                    Rational::try_from(value.re)
415                        .map_err(|error| VakintError::EvaluationError(error.into()))?,
416                    Rational::try_from(value.im)
417                        .map_err(|error| VakintError::EvaluationError(error.into()))?,
418                ));
419                kernel += coefficient * epsilon.clone().pow(shifted_power) * monomial;
420                continue;
421            }
422            let (alias, name) = loop {
423                let symbol = vk_symbol!(format!("vacuum_numerical_coefficient_{serial}"));
424                serial += 1;
425                let alias = Atom::var(symbol);
426                let name = crate::utils::undress_vakint_symbols(&alias.to_canonical_string());
427                if used.insert(symbol) && !options.numerical_parameters.contains_key(&name) {
428                    break (alias, name);
429                }
430            };
431            options.numerical_parameters.insert(name, value);
432            kernel += alias * epsilon.clone().pow(Atom::num(shifted_power)) * monomial.clone();
433        }
434        Ok((kernel, options, common_power))
435    }
436}

438/// One instance per VakintExpression operation; its slot family and cache
439/// never escape the owning operation. Include every expression's numerator
440/// when choosing the family, so a cache hit cannot collide with user indices.
441pub(crate) struct VacuumProjection {
442    slot: Symbol,
443    dimension: Atom,
444    kernels: BTreeMap<Degrees, ScalarTerms>,
445}
446
447impl VacuumProjection {
448    pub(crate) fn new(settings: &VakintSettings, inputs: &[Atom]) -> Self {
449        let mut serial = 0;
450        let slot = loop {
451            let candidate = vk_symbol!(format!("vacuum_projection_slot_{serial}"));
452            if !candidate.is_symmetric()
453                && !candidate.is_antisymmetric()
454                && !candidate.is_cyclesymmetric()
455                && !candidate.is_linear()
456                && candidate.get_normalization_function().is_none()
457                && inputs
458                    .iter()
459                    .all(|input| !input.get_all_symbols(true).contains(&candidate))
460            {
461                break candidate;
462            }
463            serial += 1;
464        };
465        Self {
466            slot,
467            dimension: Atom::num(4)
468                - Atom::num(2) * Atom::var(vk_symbol!(&settings.epsilon_symbol)),
469            kernels: BTreeMap::new(),
470        }
471    }
472
473    fn slot(&self, loop_id: i64, ordinal: u32) -> Atom {
474        FunctionBuilder::new(S.tensor_index)
475            .add_arg(Atom::var(self.slot))
476            .add_arg(Atom::num(loop_id))
477            .add_arg(Atom::num(i64::from(ordinal)))
478            .finish()
479    }
480
481    fn metric(left: Atom, right: Atom) -> Atom {
482        FunctionBuilder::new(S.g)
483            .add_arg(left)
484            .add_arg(right)
485            .finish()
486    }
487
488    fn loop_id(vector: AtomView<'_>) -> Option<i64> {
489        let AtomView::Fun(vector) = vector else {
490            return None;
491        };
492        (vector.get_symbol() == S.k && vector.get_nargs() == 1)
493            .then(|| i64::try_from(vector.get(0)).ok())
494            .flatten()
495    }
496
497    fn angular_dependence(view: AtomView<'_>) -> bool {
498        match view {
499            AtomView::Fun(dot)
500                if dot.get_symbol() == S.dot
501                    && dot.get_nargs() == 2
502                    && dot.iter().all(|vector| Self::loop_id(vector).is_some()) =>
503            {
504                false
505            }
506            AtomView::Fun(function) => {
507                function.get_symbol() == S.k || function.iter().any(Self::angular_dependence)
508            }
509            AtomView::Add(sum) => sum.iter().any(Self::angular_dependence),
510            AtomView::Mul(product) => product.iter().any(Self::angular_dependence),
511            AtomView::Pow(power) => {
512                let (base, exponent) = power.get_base_exp();
513                Self::angular_dependence(base) || Self::angular_dependence(exponent)
514            }
515            _ => false,
516        }
517    }
518
519    fn product(&self, left: &TensorTerms, right: &TensorTerms) -> Result<TensorTerms, VakintError> {
520        let mut result = TensorTerms::new();
521        for (a, coefficient_a) in left {
522            for (b, coefficient_b) in right {
523                let mut degrees = a.clone();
524                let mut shifted = coefficient_b.clone();
525                for (loop_id, rank) in b {
526                    let offset = *a.get(loop_id).unwrap_or(&0);
527                    // Descending substitution is simultaneous for a positive
528                    // shift: no newly-created target can be a later source.
529                    for ordinal in (0..*rank).rev() {
530                        shifted = shifted
531                            .replace(self.slot(*loop_id, ordinal).to_pattern())
532                            .with(
533                                self.slot(
534                                    *loop_id,
535                                    ordinal.checked_add(offset).ok_or_else(|| {
536                                        VakintError::InvalidNumerator(
537                                            "tensor degree overflows u32".into(),
538                                        )
539                                    })?,
540                                )
541                                .to_pattern(),
542                            );
543                    }
544                    degrees.insert(
545                        *loop_id,
546                        offset.checked_add(*rank).ok_or_else(|| {
547                            VakintError::InvalidNumerator("tensor degree overflows u32".into())
548                        })?,
549                    );
550                }
551                *result.entry(degrees).or_insert_with(Atom::zero) += coefficient_a * shifted;
552            }
553        }
554        result.retain(|_, coefficient| !coefficient.is_zero());
555        Ok(result)
556    }
557
558    fn tensor_terms(&self, view: AtomView<'_>) -> Result<TensorTerms, VakintError> {
559        if !Self::angular_dependence(view) {
560            return Ok(BTreeMap::from([(Degrees::new(), view.to_owned())]));
561        }
562        match view {
563            AtomView::Add(sum) => {
564                let mut result = TensorTerms::new();
565                for term in sum {
566                    for (degree, coefficient) in self.tensor_terms(term)? {
567                        *result.entry(degree).or_insert_with(Atom::zero) += coefficient;
568                    }
569                }
570                result.retain(|_, coefficient| !coefficient.is_zero());
571                Ok(result)
572            }
573            AtomView::Mul(product) => {
574                let mut result = BTreeMap::from([(Degrees::new(), Atom::one())]);
575                for factor in product {
576                    result = self.product(&result, &self.tensor_terms(factor)?)?;
577                }
578                Ok(result)
579            }
580            AtomView::Pow(power) => {
581                let (base, exponent) = power.get_base_exp();
582                let exponent = u32::try_from(i64::try_from(exponent).map_err(|_| {
583                    VakintError::InvalidNumerator("nonpolynomial loop tensor".into())
584                })?)
585                .map_err(|_| VakintError::InvalidNumerator("nonpolynomial loop tensor".into()))?;
586                let mut base = self.tensor_terms(base)?;
587                let mut exponent = exponent;
588                let mut result = BTreeMap::from([(Degrees::new(), Atom::one())]);
589                while exponent != 0 {
590                    if exponent % 2 == 1 {
591                        result = self.product(&result, &base)?;
592                    }
593                    exponent /= 2;
594                    if exponent != 0 {
595                        base = self.product(&base, &base)?;
596                    }
597                }
598                Ok(result)
599            }
600            AtomView::Fun(vector) if vector.get_symbol() == S.k && vector.get_nargs() == 2 => {
601                let loop_id = i64::try_from(vector.get(0)).map_err(|_| {
602                    VakintError::InvalidNumerator("loop momentum id must be an integer".into())
603                })?;
604                Ok(BTreeMap::from([(
605                    BTreeMap::from([(loop_id, 1)]),
606                    Self::metric(vector.get(1).to_owned(), self.slot(loop_id, 0)),
607                )]))
608            }
609            AtomView::Fun(dot) if dot.get_symbol() == S.dot && dot.get_nargs() == 2 => {
610                let (loop_id, external) = if let Some(loop_id) = Self::loop_id(dot.get(0)) {
611                    (loop_id, dot.get(1))
612                } else if let Some(loop_id) = Self::loop_id(dot.get(1)) {
613                    (loop_id, dot.get(0))
614                } else {
615                    return Err(VakintError::InvalidNumerator(
616                        "unsupported loop scalar product".into(),
617                    ));
618                };
619                let AtomView::Fun(external) = external else {
620                    return Err(VakintError::InvalidNumerator(
621                        "expected elementary external vector".into(),
622                    ));
623                };
624                if external.get_symbol() != S.p
625                    || external.get_nargs() != 1
626                    || i64::try_from(external.get(0)).is_err()
627                {
628                    return Err(VakintError::InvalidNumerator(
629                        "expected elementary external vector".into(),
630                    ));
631                }
632                Ok(BTreeMap::from([(
633                    BTreeMap::from([(loop_id, 1)]),
634                    FunctionBuilder::new(S.p)
635                        .add_arg(external.get(0))
636                        .add_arg(self.slot(loop_id, 0))
637                        .finish(),
638                )]))
639            }
640            _ => Err(VakintError::InvalidNumerator(format!(
641                "unsupported polynomial loop-tensor factor: {view}"
642            ))),
643        }
644    }
645
646    /// The owner must certify vacuum denominators before calling this
647    /// angular projector. Nonvacuum scalar inputs use ScalarTerms directly.
648    pub(crate) fn project(
649        &mut self,
650        input: AtomView<'_>,
651        mut universal_projector: impl FnMut(Atom) -> Result<Atom, VakintError>,
652    ) -> Result<ScalarTerms, VakintError> {
653        // This is a contraction, never a call to FORM on the input. It also
654        // normalizes closed tensor powers and checks literal dummy incidence.
655        let input = LorentzTensor::parse(input, &self.dimension)?.expression;
656        let mut result = ScalarTerms::default();
657        for (degrees, coefficient) in self.tensor_terms(input.as_view())? {
658            let odd = degrees
659                .values()
660                .fold(false, |odd, rank| odd ^ (rank % 2 == 1));
661            if odd {
662                continue;
663            }
664            if !self.kernels.contains_key(&degrees) {
665                let mut monomial = Atom::one();
666                for (loop_id, rank) in &degrees {
667                    for ordinal in 0..*rank {
668                        monomial *= FunctionBuilder::new(S.k)
669                            .add_arg(Atom::num(*loop_id))
670                            .add_arg(self.slot(*loop_id, ordinal))
671                            .finish();
672                    }
673                }
674                let kernel = if degrees.is_empty() {
675                    Atom::one()
676                } else {
677                    universal_projector(monomial)?
678                };
679                self.kernels
680                    .insert(degrees.clone(), ScalarTerms::parse(kernel.as_view())?);
681            }
682            for (kernel_monomial, metric_coefficient) in &self.kernels[&degrees].0 {
683                let coefficient = LorentzTensor::parse(
684                    (&coefficient * metric_coefficient).as_view(),
685                    &self.dimension,
686                )?
687                .expression;
688                // Scalar invariants retained above now enter only their
689                // monomial key. This also deduplicates before backend calls.
690                for (coefficient_monomial, coefficient) in
691                    ScalarTerms::parse(coefficient.as_view())?.0
692                {
693                    result.insert(kernel_monomial * coefficient_monomial, coefficient);
694                }
695            }
696        }
697        Ok(result)
698    }

1207        let spectator = atom("(a+b)*(c+d)*tensor(p(1,mu),mu)");
1208        // This independent scalar master keeps the terms required by a

1223                ScalarTerms::epsilon_pole_order(coefficient.as_view(), &settings).unwrap(),
