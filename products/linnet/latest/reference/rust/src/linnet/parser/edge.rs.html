13pub struct DotEdgeData {
14    pub statements: BTreeMap<String, String>,
15    pub local_statements: BTreeMap<String, String>,
16    pub edge_id: Option<EdgeIndex>,
17}
18
19impl FromIterator<(String, String)> for DotEdgeData {
20    fn from_iter<T: IntoIterator<Item = (String, String)>>(iter: T) -> Self {
21        let mut edge_id = None;
22        let statements: BTreeMap<String, String> = iter
23            .into_iter()
24            .filter_map(|(k, v)| match k.as_str() {
25                "id" => {
26                    edge_id = Some(EdgeIndex::from(strip_quotes(&v).parse::<usize>().unwrap()));
27                    None
28                }
29                _ => Some((k, strip_quotes(&v).to_string())),
30            })
31            .collect();
32
33        DotEdgeData {
34            local_statements: BTreeMap::new(),
35            statements,
36
37            edge_id,
38        }
39    }
40}

42impl Display for DotEdgeData {
43    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
44        let mut first = true;
45        for (key, value) in &self.statements {
46            if !first {
47                write!(f, " ")?;
48            }
49            write!(f, "{key}=\"{value}\"")?;
50            first = false;
51        }
52
53        // if let Some(id) = self.edge_id {

55        // }
56
57        Ok(())
58    }

61impl DotEdgeData {
62    pub fn remove_common(&mut self, common: &GlobalData) {
63        self.statements.retain(|k, v| {
64            if let Some(common_value) = common.edge_statements.get(k) {
65                v != common_value
66            } else {
67                true
68            }
69        });
70    }
71    pub fn empty() -> Self {
72        DotEdgeData {
73            statements: BTreeMap::new(),
74            local_statements: BTreeMap::new(),
75            edge_id: None,
76        }
77    }
78
79    pub fn extend(&mut self, other: Self) {
80        self.statements.extend(other.statements);
81    }
82
83    pub fn format(&self, template: impl AsRef<str>) -> String {
84        let mut result = template.as_ref().to_owned();
85
86        // Find all occurrences of {key} in the template
87        for (key, value) in &self.statements {
88            let placeholder = format!("{{{key}}} ");
89            result = result.replace(&placeholder, value);
90        }
91
92        // Remove any remaining {whatever} patterns
93        while let Some(start) = result.find('{') {
94            if let Some(end) = result[start..].find('}') {
95                result.replace_range(start..=start + end, "");
96            } else {
97                break;
98            }
99        }
100
101        result
102    }
103
104    pub fn add_statement(&mut self, key: impl ToString, value: impl ToString) {
105        self.statements.insert(key.to_string(), value.to_string());
106    }
107
108    pub fn get<Q: Ord + ?Sized, F: FromStr>(&self, key: &Q) -> Option<Result<F, F::Err>>
109    where
110        String: Borrow<Q>,
111    {
112        self.statements.get(key).map(|s| s.parse())
113    }
114}

122impl DotEdgeData {
123    pub fn from_parser(
124        edge: Edge,
125        map: &BTreeMap<String, NodeIdOrDangling>,
126        is_digraph: impl Into<Orientation>,
127        global_data: &GlobalData,
128    ) -> (
129        Self,
130        Orientation,
131        HedgeData<DotHedgeData>,
132        Either<HedgeData<DotHedgeData>, Flow>,
133    ) {
134        let mut orientation = is_digraph.into();
135        let mut source_data = DotHedgeData::from(edge.source_port());
136        let mut sink_data = DotHedgeData::from(edge.sink_port());
137        let local_statements = edge.attr.clone().into_iter().collect();
138        let mut statements = global_data.edge_statements.clone();
139        statements.extend(edge.attr.into_iter().filter_map(|(key, value)| {
140            let stripped_value = strip_quotes(&value);
141            match key.as_str() {
142                "dir" => match stripped_value {
143                    "forward" => orientation = Orientation::Default,
144                    "back" => orientation = Orientation::Reversed,
145                    "none" => orientation = Orientation::Undirected,
146                    _ => panic!("Invalid edge direction"),
147                },
148                "source" => {
149                    source_data.statement = Some(stripped_value.to_string());
150                }
151                "sink" => {
152                    sink_data.statement = Some(stripped_value.to_string());
153                }
154                _ => {
155                    return Some((key, stripped_value.to_string()));
156                }
157            }
158            None
159        }));
160
161        let source = map[&edge.from.id].clone();
162
163        let target = map[&edge.to.id].clone();
164
165        let (edge, source, target) = match (source, target) {
166            (NodeIdOrDangling::Id(source), NodeIdOrDangling::Id(target)) => {
167                //Full edge
168                let mut dot_edge: DotEdgeData = statements.into_iter().collect();
169                dot_edge.local_statements = local_statements;
170                (
171                    dot_edge,
172                    source.add_data(source_data),
173                    Either::Left(target.add_data(sink_data)),
174                )
175            }
176            (NodeIdOrDangling::Id(source), NodeIdOrDangling::Dangling { statements: states }) => {
177                statements.extend(
178                    states
179                        .into_iter()
180                        .filter(|(a, _)| !(a.as_str() == "shape" || a.as_str() == "label")),
181                );
182                let mut dot_edge: DotEdgeData = statements.into_iter().collect();
183                dot_edge.local_statements = local_statements;
184                if !sink_data.is_none() {
185                    panic!("Sink edge cannot have data:{sink_data}");
186                }
187                (
188                    dot_edge,
189                    source.add_data(source_data),
190                    Either::Right(Flow::Source),
191                )
192            }
193            (NodeIdOrDangling::Dangling { statements: states }, NodeIdOrDangling::Id(sink)) => {
194                statements.extend(
195                    states
196                        .into_iter()
197                        .filter(|(a, _)| !(a.as_str() == "shape" || a.as_str() == "label")),
198                );
199                let mut dot_edge: DotEdgeData = statements.into_iter().collect();
200                dot_edge.local_statements = local_statements;
201                if !source_data.is_none() {
202                    panic!("Source edge cannot have data:{source_data}");
203                }
204
205                (
206                    dot_edge,
207                    sink.add_data(sink_data),
208                    Either::Right(Flow::Sink),
209                )
210            }
211            _ => panic!("Cannot connect an edge to two external nodes"),
212        };
213
214        (edge, orientation, source, target)
215    }
