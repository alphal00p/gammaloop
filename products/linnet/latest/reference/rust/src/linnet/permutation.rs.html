1//! # Permutations
2//!
3//! This module provides a `Permutation` struct and associated functionalities
4//! for representing and working with permutations of a sequence of integers `0..n`.
5//!
6//! ## Key Features:
7//!
8//! - **Representation**: A `Permutation` is stored by its direct mapping (`map[i]` is
9//!   the image of `i`) and its inverse mapping.
10//! - **Construction**:
11//!   - Identity permutation: `Permutation::id(n)`.
12//!   - From a mapping vector: `Permutation::from_map(vec![...])`.
13//!   - From disjoint cycles: `Permutation::from_disjoint_cycles(&[vec![...]])`.
14//!   - From potentially overlapping cycles with specified composition order:
15//!     `Permutation::from_cycles_ordered(&[vec![...]], order)`.
16//! - **Basic Operations**:
17//!   - Inverse: `p.inverse()`.
18//!   - Composition: `p1.compose(&p2)` (applies `p2` then `p1`).
19//!   - Apply to slices: `p.apply_slice(data)` (returns a new `Vec`),
20//!     `p.apply_slice_in_place(data_mut)`.
21//!   - Power: `p.pow(k)`.
22//!   - Sign: `p.sign()` (+1 for even, -1 for odd).
23//!   - Check for identity: `p.is_identity()`.
24//! - **Cycle Utilities**:
25//!   - Find cycle decomposition: `p.find_cycles()`.
26//!   - Convert to transpositions: `p.transpositions()`.
27//! - **Iterators**:
28//!   - `p.iter_slice(data)`: Iterates over `data` in the order defined by `p.map`.
29//!   - `p.iter_slice_inv(data)`: Iterates over `data` in the order defined by `p.inv`.
30//!   - Mutable versions: `p.iter_slice_mut(data_mut)`, `p.iter_slice_inv_mut(data_mut)`.
31//! - **Ranking and Unranking**:
32//!   - Myrvold & Ruskey's ranking/unranking algorithms (`myrvold_ruskey_rank1`, `myrvold_ruskey_unrank1`, etc.).
33//! - **Sorting**:
34//!   - `Permutation::sort(slice)`: Returns the permutation that sorts the slice.
35//! - **Graph Integration**:
36//!   - The `HedgeGraphExt` and `PermutationExt` traits suggest integration with
37//!     graph structures, likely for tasks like finding graph automorphisms or
38//!     canonical labellings by permuting vertices/hedges.
39//!
40//! The module is designed to be comprehensive for permutation-related tasks,
41//! including those relevant to combinatorial algorithms and graph theory.

57/// A permutation of `0..n`, with the ability to apply itself (or its inverse) to slices.
58///
59/// # Examples
60///
61/// ```
62/// use linnet::permutation::Permutation;
63///
64/// // Create a permutation that maps 0->2, 1->0, 2->1, 3->3
65/// let p = Permutation::from_map(vec![2, 0, 1, 3]);
66///
67/// // Apply the permutation to a slice
68/// let data = vec![10, 20, 30, 40];
69/// let permuted = p.apply_slice(&data);
70/// assert_eq!(permuted, vec![20, 30, 10, 40]);
71/// ```
72#[derive(Debug, Clone, PartialEq, Eq, Hash)]
73#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
74#[cfg_attr(feature = "bincode", derive(bincode::Encode, bincode::Decode))]
75pub struct Permutation {
76    map: Vec<usize>,
77    inv: Vec<usize>,
78}

108    /// Creates a permutation from a mapping vector.
109    /// The `map` vector states where index `i` is sent: `map[i]` is the image of `i`.
110    ///
111    /// # Examples
112    ///
113    /// ```
114    /// # use linnet::permutation::Permutation;
115    /// let p = Permutation::from_map(vec![2, 0, 1]);
116    /// assert_eq!(p.apply_slice(&[10,20,30]), vec![20,30,10]);
117    /// ```
118    pub fn from_map(map: Vec<usize>) -> Self {
119        let mut inv = vec![0; map.len()];
120        for (i, &j) in map.iter().enumerate() {
121            inv[j] = i;
122        }
123        Permutation { map, inv }
124    }

136    /// Creates a permutation from (from,to) pairs, minimizing swaps among all valid completions.
137    pub fn from_mappings<I>(mappings: I, size: usize) -> Result<Self, String>
138    where
139        I: IntoIterator<Item = (usize, usize)>,
140    {
141        if size == 0 {
142            return Ok(Self::id(0));
143        }
144
145        // Identity to start from
146        let mut map: Vec<usize> = (0..size).collect();
147
148        // forced_sources[s] = true iff s has an explicitly forced mapping
149        let mut forced_sources = vec![false; size];
150
151        // owner_of_target[t] = Some(s) if *any* assignment currently owns t (forced or identity we keep)
152        let mut owner_of_target: Vec<Option<usize>> = vec![None; size];
153
154        // forced_owner_of_target[t] = Some(s) iff a *forced* mapping s -> t exists
155        let mut forced_owner_of_target: Vec<Option<usize>> = vec![None; size];
156
157        // 1) Apply & validate forced mappings in O(1) each
158        for (from, to) in mappings {
159            if from >= size || to >= size {
160                return Err(format!(
161                    "Index out of bounds: mapping ({from}, {to}) for size {size}",
162                ));
163            }
164            if let Some(prev_from) = forced_owner_of_target[to] {
165                if prev_from != from {
166                    return Err(format!(
167                        "Target {to} is mapped from multiple sources ({prev_from} and {from})",
168                    ));
169                }
170            }
171            if forced_sources[from] && map[from] != to {
172                return Err(format!(
173                    "Source {} has conflicting mappings ({} vs {})",
174                    from, map[from], to
175                ));
176            }
177
178            map[from] = to;
179            forced_sources[from] = true;
180            forced_owner_of_target[to] = Some(from);
181            owner_of_target[to] = Some(from);
182        }
183
184        // 2) Keep identities whenever possible; collect displaced (unforced whose identity is taken)
185        let mut displaced_sources: Vec<usize> = Vec::new();
186        for s in 0..size {
187            if forced_sources[s] {
188                continue; // fixed by input
189            }
190            if owner_of_target[s].is_none() {
191                // identity slot s is free -> keep s->s
192                map[s] = s;
193                owner_of_target[s] = Some(s);
194            } else {
195                // identity slot s is already taken by a forced mapping -> s is displaced
196                displaced_sources.push(s);
197            }
198        }
199
200        // 3) For each displaced source s, assign it to the unique free target reachable
201        //    by walking through forced owners: t0 = s; while forced_owner_of_target[t] = Some(u) { t = u; }
202        //    Use a small memo to avoid re-walking shared tails.
203        let mut sink_cache: Vec<Option<usize>> = vec![None; size];
204
205        for s in displaced_sources {
206            let mut path: Vec<usize> = Vec::new();
207            let mut t = s;
208
209            // Walk until we hit a cached sink or a free target
210            loop {
211                if let Some(sink) = sink_cache[t] {
212                    // propagate cached sink to the path and assign
213                    for x in path.drain(..) {
214                        sink_cache[x] = Some(sink);
215                    }
216                    map[s] = sink;
217                    owner_of_target[sink] = Some(s);
218                    break;
219                }
220                path.push(t);
221
222                match forced_owner_of_target[t] {
223                    Some(owner) => {
224                        // advance along the forced-owner chain
225                        t = owner;
226                        continue;
227                    }
228                    None => {
229                        // t is a free target (no forced owner) -> this is the sink
230                        for x in path.drain(..) {
231                            sink_cache[x] = Some(t);
232                        }
233                        map[s] = t;
234                        owner_of_target[t] = Some(s);
235                        break;
236                    }
237                }
238            }
239        }
240
241        // 4) Final uniqueness check
242        let mut seen = vec![false; size];
243        for &to in &map {
244            if to >= size || std::mem::replace(&mut seen[to], true) {
245                return Err("Failed to create valid permutation".to_string());
246            }
247        }
248
249        Ok(Self::from_map(map))
250    }

252    /// Creates a permutation from a list of sources mapping to the prefix targets 0..k-1.
253    ///
254    /// The j-th element of `sources` is forced to map to target `j`.
255    /// Remaining indices are chosen to (1) keep identities when possible and
256    /// (2) minimize the number of swaps (maximize the number of cycles).
257    ///
258    /// # Arguments
259    /// * `sources` - iterator of distinct indices in `0..size`, length = k
260    /// * `size`    - total permutation size (n)
261    ///
262    /// # Returns
263    /// A swaps-optimal completion satisfying `sources[j] -> j`.
264    ///
265    /// # Example
266    /// ```
267    /// # use linnet::permutation::Permutation;
268    /// let p = Permutation::from_prefix_sources([5, 0, 3], 6).unwrap();
269    /// // forced: 5->0, 0->1, 3->2  (targets 0,1,2)
270    /// assert_eq!(p[5], 0);
271    /// assert_eq!(p[0], 1);
272    /// assert_eq!(p[3], 2);
273    /// // others chosen swaps-optimally
274    /// ```
275    pub fn from_prefix_sources<I>(sources: I, size: usize) -> Result<Self, String>
276    where
277        I: IntoIterator<Item = usize>,
278    {
279        if size == 0 {
280            return Ok(Self::id(0));
281        }
282
283        // Collect sources and validate in O(n+k)
284        let src_vec: Vec<usize> = sources.into_iter().collect();
285        let k = src_vec.len();
286        if k > size {
287            return Err(format!("Too many sources: k={k} > size={size}"));
288        }
289
290        let mut seen_source = vec![false; size];
291        for &s in &src_vec {
292            if s >= size {
293                return Err(format!("Source {s} out of bounds for size {size}"));
294            }
295            if std::mem::replace(&mut seen_source[s], true) {
296                return Err(format!("Duplicate source index {s}"));
297            }
298        }
299
300        // Identity base
301        let mut map: Vec<usize> = (0..size).collect();
302
303        // Bookkeeping
304        let mut forced_sources = vec![false; size]; // s is in src_vec
305        let mut owner_of_target: Vec<Option<usize>> = vec![None; size]; // who currently owns target t
306
307        // forced_owner_of_target[t] = Some(s) only for t in 0..k with s = src_vec[t]
308        // (This defines the fixed "owner graph" we walk to find sinks.)
309        let mut forced_owner_of_target: Vec<Option<usize>> = vec![None; size];
310
311        // 1) Apply forced prefix mappings: src_vec[j] -> j
312        for (t, &s) in (0..k).zip(src_vec.iter()) {
313            // note: t is guaranteed < size since k <= size
314            if let Some(prev) = owner_of_target[t] {
315                // Should never happen here because each prefix target appears exactly once
316                if prev != s {
317                    return Err(format!("Target {t} already owned by {prev}"));
318                }
319            }
320            map[s] = t;
321            forced_sources[s] = true;
322            owner_of_target[t] = Some(s);
323            forced_owner_of_target[t] = Some(s);
324        }
325
326        // 2) Keep identities whenever possible; collect displaced
327        let mut displaced: Vec<usize> = Vec::new();
328        for i in 0..size {
329            if forced_sources[i] {
330                continue;
331            }
332            if owner_of_target[i].is_none() {
333                // identity target i is free -> keep i->i
334                map[i] = i;
335                owner_of_target[i] = Some(i);
336            } else {
337                // identity target taken by some forced mapping -> i is displaced
338                displaced.push(i);
339            }
340        }
341
342        // 3) Route each displaced source to its unique sink via forced-owner chains.
343        //    Cache sinks so shared tails are walked once.
344        let mut sink_cache: Vec<Option<usize>> = vec![None; size];
345
346        for s in displaced {
347            let mut path: Vec<usize> = Vec::new();
348            let mut t = s;
349
350            loop {
351                if let Some(cached) = sink_cache[t] {
352                    // Assign cached sink; mark ownership
353                    for x in path.drain(..) {
354                        sink_cache[x] = Some(cached);
355                    }
356                    // Safety: cached is free by construction of earlier assignments
357                    map[s] = cached;
358                    owner_of_target[cached] = Some(s);
359                    break;
360                }
361
362                path.push(t);
363
364                match forced_owner_of_target[t] {
365                    Some(owner) => {
366                        // Follow the forced owner chain
367                        t = owner;
368                    }
369                    None => {
370                        // We reached a target not in the forced prefix (or outside 0..k)
371                        // This is the unique sink for the whole component.
372                        // It is guaranteed free at this point because:
373                        // - If t == i for some identity i kept earlier, that i cannot lie on the
374                        //   forced-owner chain (no forced owner on t), hence we would not be
375                        //   displacing s into an already-owned t in this component.
376                        for x in path.drain(..) {
377                            sink_cache[x] = Some(t);
378                        }
379                        map[s] = t;
380                        owner_of_target[t] = Some(s);
381                        break;
382                    }
383                }
384            }
385        }
386
387        // 4) Verify permutation validity
388        let mut seen = vec![false; size];
389        for &to in &map {
390            if to >= size || std::mem::replace(&mut seen[to], true) {
391                return Err("Construction failed: duplicate or OOB target".into());
392            }
393        }
394
395        Ok(Self::from_map(map))
396    }

483    /// Applies `self` in-place to the provided slice by using transpositions
484    /// derived from the cycle decomposition.
485    ///
486    /// # Examples
487    ///
488    /// ```
489    /// # use linnet::permutation::Permutation;
490    /// let p = Permutation::from_map(vec![2, 0, 1]);
491    /// let mut data = vec![10, 20, 30];
492    /// p.apply_slice_in_place(&mut data);
493    /// assert_eq!(data, vec![20, 30, 10]);
494    /// ```
495    pub fn apply_slice_in_place<T: Clone, S>(&self, slice: &mut S)
496    where
497        S: AsMut<[T]>,
498    {
499        let transpositions = self.transpositions();
500        for (i, j) in transpositions.iter().rev() {
501            slice.as_mut().swap(*i, *j);
502        }
503    }

674    /// Returns the cycle decomposition of `self` as a `Vec` of cycles,
675    /// each cycle represented as a `Vec<usize>`.
676    /// Each cycle lists the indices of a single cycle, e.g. `[0, 2, 1]` means `0->2, 2->1, 1->0`.
677    ///
678    /// # Examples
679    ///
680    /// ```
681    /// # use linnet::permutation::Permutation;
682    /// let p = Permutation::from_map(vec![2, 0, 1, 3]);
683    /// let cycles = p.find_cycles();
684    /// // cycles might be [[0, 2, 1], [3]]
685    /// assert_eq!(cycles.len(), 2);
686    /// ```
687    pub fn find_cycles(&self) -> Vec<Vec<usize>> {
688        let mut visited = vec![false; self.map.len()];
689        let mut cycles = Vec::new();
690        for i in 0..self.map.len() {
691            if visited[i] {
692                continue;
693            }
694            let mut cycle = Vec::new();
695            let mut j = i;
696            while !visited[j] {
697                visited[j] = true;
698                cycle.push(j);
699                j = self.map[j];
700            }
701            if !cycle.is_empty() {
702                cycles.push(cycle);
703            }
704        }
705        cycles
706    }

752    /// Computes the rank of the permutation in the Myrvold & Ruskey "Rank1" ordering.
753    ///
754    /// This is a recursive implementation. For permutations of size `n`, it removes
755    /// the position of `n-1` from the permutation, multiplies the result by `n`,
756    /// and adds the index of `n-1`.
757    ///
758    /// # Examples
759    ///
760    /// ```
761    /// # use linnet::permutation::Permutation;
762    /// let p = Permutation::from_map(vec![2, 1, 3, 0]);
763    /// assert_eq!(p.myrvold_ruskey_rank1(), 12);
764    /// ```
765    pub fn myrvold_ruskey_rank1(mut self) -> usize {
766        let n = self.map.len();
767        if self.map.len() == 1 {
768            return 0;
769        }
770
771        let s = self.map[n - 1];
772        self.map.swap_remove(self.inv[n - 1]);
773        self.inv.swap_remove(s);
774
775        s + n * self.myrvold_ruskey_rank1()
776    }

778    /// Unranks a permutation of size `n` from its Myrvold & Ruskey "Rank1" index.
779    ///
780    /// # Examples
781    ///
782    /// ```
783    /// # use linnet::permutation::Permutation;
784    /// let p = Permutation::myrvold_ruskey_unrank1(4, 12);
785    /// assert_eq!(p.map(), &[2, 1, 3, 0]);
786    /// ```
787    pub fn myrvold_ruskey_unrank1(n: usize, mut rank: usize) -> Self {
788        let mut p = (0..n).collect::<Vec<_>>();
789        for i in (1..=n).rev() {
790            let j = rank % i;
791            rank /= i;
792            p.swap(i - 1, j);
793        }
794        Permutation::from_map(p)
795    }

801    /// Computes the rank of the permutation in the Myrvold & Ruskey "Rank2" ordering.
802    ///
803    /// # Examples
804    ///
805    /// ```
806    /// # use linnet::permutation::Permutation;
807    /// let p = Permutation::from_map(vec![2, 1, 3, 0]);
808    /// // Suppose it has rank = R. We can test we get p back by unranking R.

810    /// let q = Permutation::myrvold_ruskey_unrank2(4, rank);
811    /// assert_eq!(q, p);
812    /// ```
813    pub fn myrvold_ruskey_rank2(mut self) -> usize {
814        let n = self.map.len();

817        }
818        let s = self.map[n - 1];
819        self.map.swap_remove(self.inv[n - 1]);
820        self.inv.swap_remove(s);
821        s * Self::factorial(n - 1) + self.myrvold_ruskey_rank2()
822    }

826    /// # Examples

892    /// For k = 0, it returns the identity of the same size.
893    ///
894    /// # Examples
895    ///
896    /// ```
897    /// # use linnet::permutation::Permutation;
898    /// let p = Permutation::from_map(vec![1, 2, 0]);

901    /// assert_eq!(p2.map(), &[2, 0, 1]);
902    /// ```
903    // -- ADDED
904    pub fn pow(&self, k: usize) -> Self {
905        if k == 0 {
906            return Permutation::id(self.map.len());
907        }
908        let mut result = Permutation::id(self.map.len());
909        let mut base = self.clone();
910        let mut exp = k;
911
912        while exp > 0 {
913            if exp % 2 == 1 {
914                result = result.compose(&base);
915            }
916            base = base.compose(&base);
917            exp /= 2;
918        }
919        result
920    }

972    /// Creates a permutation from a set of disjoint cycles.
973    /// Each cycle should be a `Vec<usize>` representing indices in the cycle.
974    /// The cycles must be disjoint (no shared elements).
975    ///
976    /// # Examples
977    ///
978    /// ```
979    /// # use linnet::permutation::Permutation;
980    /// let cycles = vec![vec![0, 1, 2], vec![3, 4]];
981    /// let p = Permutation::from_disjoint_cycles(&cycles).unwrap();
982    /// assert_eq!(p.map(), &[1, 2, 0, 4, 3]);
983    ///
984    /// // Error if cycles are not disjoint
985    /// let invalid = vec![vec![0, 1], vec![1, 2]];
986    /// assert!(Permutation::from_disjoint_cycles(&invalid).is_err());
987    /// ```
988    pub fn from_disjoint_cycles(cycles: &[Vec<usize>]) -> Result<Self, String> {
989        // Find the size of the permutation (maximum index + 1)
990        let n = cycles
991            .iter()
992            .flat_map(|cycle| cycle.iter())
993            .max()
994            .map(|&max| max + 1)
995            .unwrap_or(0);
996
997        // Check for duplicates across cycles
998        let mut seen = vec![false; n];
999        for cycle in cycles {
1000            for &idx in cycle {
1001                if seen[idx] {
1002                    return Err("Cycles are not disjoint".to_string());
1003                }
1004                seen[idx] = true;
1005            }
1006        }
1007
1008        // Create the permutation map
1009        let mut map = (0..n).collect::<Vec<_>>();
1010        for cycle in cycles {
1011            if cycle.len() <= 1 {
1012                continue;
1013            }
1014
1015            // Map each element to the next element in the cycle
1016            for i in 0..cycle.len() {
1017                let from = cycle[i];
1018                let to = cycle[(i + 1) % cycle.len()];
1019                map[from] = to;
1020            }
1021        }
1022
1023        Ok(Permutation::from_map(map))
1024    }
1025    /// Creates a permutation from any set of cycles with specified reading order.
1026    pub fn from_cycles_ordered(cycles: &[Vec<usize>], order: CycleOrder) -> Self {
1027        // Find the size of the permutation
1028        let n = cycles
1029            .iter()
1030            .flat_map(|cycle| cycle.iter())
1031            .max()
1032            .map(|&max| max + 1)
1033            .unwrap_or(0);
1034
1035        // Start with identity permutation
1036        let mut result = Permutation::id(n);
1037
1038        // Choose iteration order based on CycleOrder
1039        let cycle_iter: Box<dyn Iterator<Item = &Vec<usize>>> = match order {
1040            CycleOrder::LastFirst => Box::new(cycles.iter().rev()),
1041            CycleOrder::FirstFirst => Box::new(cycles.iter()),
1042        };
1043
1044        // Apply cycles in specified order
1045        for cycle in cycle_iter {
1046            if cycle.len() <= 1 {
1047                continue;
1048            }
1049
1050            // Create a single cycle permutation
1051            let mut cycle_map = (0..n).collect::<Vec<_>>();
1052            for i in 0..cycle.len() {
1053                let from = cycle[i];
1054                let to = cycle[(i + 1) % cycle.len()];
1055                cycle_map[from] = to;
1056            }
1057            let cycle_perm = Permutation::from_map(cycle_map);
1058            result = cycle_perm.compose(&result);
1059        }
1060
1061        result
1062    }

1066    ///
1067    /// # Examples
1068    ///
1069    /// ```
1070    /// # use linnet::permutation::Permutation;
1071    /// let cycles = vec![vec![0, 1, 2], vec![1, 2]];

1073    /// assert_eq!(p.map(), &[1, 0, 2]);
1074    /// ```
1075    pub fn from_cycles(cycles: &[Vec<usize>]) -> Self {

1077    }

1083    pub fn generate_all(generators: &[Permutation]) -> Result<Vec<Permutation>, PermutationError> {
1084        let size = if let Some(generator) = generators.first() {
1085            generator.length()
1086        } else {
1087            return Err(PermutationError::EmptyGenerators);
1088        };
1089        let mut all = AHashSet::new();
1090        let mut stack = vec![Permutation::id(size)];
1091        for g in generators {
1092            if g.length() != size {
1093                return Err(PermutationError::InvalidGeneratorLength);
1094            }
1095        }
1096
1097        while let Some(current) = stack.pop() {
1098            for g in generators {
1099                let next = current.compose(g);
1100                if all.insert(next.clone()) {
1101                    stack.push(next);
1102                }
1103            }
1104        }
1105
1106        Ok(all.drain().collect())
1107    }

1332pub struct PermutationMapIterMut<'a, T: 'a> {
1333    ptr: NonNull<T>,
1334    map_indices: &'a [usize],
1335    current_map_idx: usize,
1336    len: usize,
1337    _marker: PhantomData<&'a mut T>,
1338}
1339
1340impl<'a, T: 'a> PermutationMapIterMut<'a, T> {
1341    fn new(slice: &'a mut [T], map_indices: &'a [usize]) -> Self {
1342        let slice_len = slice.len();
1343        // This check also guarantees that every index in `map_indices` is valid for `slice`,
1344        // assuming the permutation map itself does not contain duplicated or otherwise invalid entries.
1345        if let Some(max_index) = max_out_of_bounds_index(map_indices, slice_len) {
1346            panic!(
1347                "Slice length ({slice_len}) is too short for all indices in permutation map. Max index in map: Some({max_index})"
1348            );
1349        }
1350        PermutationMapIterMut {
1351            ptr: NonNull::new(slice.as_mut_ptr()).expect("Slice pointer cannot be null"),
1352            map_indices,
1353            current_map_idx: 0,
1354            len: map_indices.len(),
1355            _marker: PhantomData,
1356        }
1357    }
1358}
1359
1360impl<'a, T> Iterator for PermutationMapIterMut<'a, T> {
1361    type Item = &'a mut T;
1362
1363    fn next(&mut self) -> Option<Self::Item> {
1364        if self.current_map_idx >= self.map_indices.len() {
1365            None
1366        } else {
1367            let perm_idx = self.map_indices[self.current_map_idx];
1368            self.current_map_idx += 1;
1369            self.len -= 1;
1370            // SAFETY:
1371            // - `ptr` is valid and non-null for the lifetime 'a.
1372            // - `perm_idx` is an index from the permutation's map.
1373            // - `PermutationMapIterMut::new` asserts that `slice.len()` is sufficient,
1374            //   and that all `map_indices` are valid for a slice of length `map_indices.len()`.
1375            // - Each call to `next` uses a unique index from `map_indices`,
1376            //   yielding a mutable reference to a distinct element.
1377            unsafe { Some(&mut *self.ptr.as_ptr().add(perm_idx)) }
1378        }
1379    }
1380
1381    fn size_hint(&self) -> (usize, Option<usize>) {
1382        (self.len, Some(self.len))
1383    }
1384}

1387    fn len(&self) -> usize {
1388        self.len
1389    }
1390}

2397    /// Return the minimum number of swaps to realize permutation p from identity:
2398    /// n - number_of_cycles.
2399    fn min_swaps_of_perm(p: &Permutation) -> usize {
2400        let n = p.length();
2401        let cycles = p.find_cycles();
2402        let num_cycles = cycles.len();
2403        n - num_cycles
2404    }

2406    /// Check whether permutation p satisfies a set of forced mappings.
2407    fn satisfies_forced(p: &Permutation, forced: &[(usize, usize)]) -> bool {
2408        forced.iter().all(|&(f, t)| p[f] == t)
2409    }

2411    /// Simple next_permutation over a slice of usize in-place; returns false when wrapped.
2412    fn next_permutation(a: &mut [usize]) -> bool {
2413        // Find longest non-increasing suffix
2414        if a.len() < 2 {
2415            return false;
2416        }
2417        let mut i = a.len() - 2;
2418        while a[i] >= a[i + 1] {
2419            if i == 0 {
2420                return false;
2421            }
2422            i -= 1;
2423        }
2424        // Find rightmost successor to pivot
2425        let mut j = a.len() - 1;
2426        while a[j] <= a[i] {
2427            j -= 1;
2428        }
2429        a.swap(i, j);
2430        // Reverse suffix
2431        a[i + 1..].reverse();
2432        true
2433    }
2434
2435    /// Exhaustively compute the minimum possible swap-count over all permutations
2436    /// of size `n` that satisfy `forced`. (Brute force = (n!) scan; only use for small n.)
2437    fn brute_min_swaps_given_forced(
2438        n: usize,
2439        forced: &[(usize, usize)],
2440    ) -> Option<(usize, Vec<usize>)> {
2441        if n == 0 {
2442            return Some((0, vec![]));
2443        }
2444        let mut map: Vec<usize> = (0..n).collect(); // identity
2445        let mut best: Option<(usize, Vec<usize>)> = None;
2446
2447        loop {
2448            // candidate permutation
2449            let cand = Permutation::from_map(map.clone());
2450            if satisfies_forced(&cand, forced) {
2451                let cost = min_swaps_of_perm(&cand);
2452                match &best {
2453                    None => best = Some((cost, map.clone())),
2454                    Some((best_cost, _)) if cost < *best_cost => best = Some((cost, map.clone())),
2455                    _ => {}
2456                }
2457            }
2458            if !next_permutation(&mut map) {
2459                break;
2460            }
2461        }
2462        best
2463    }
