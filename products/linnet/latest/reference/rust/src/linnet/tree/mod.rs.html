1//! Defines core data structures and traits for representing forests (collections of trees).
2//!
3//! This module provides multiple ways to store tree node relationships, each with different
4//! trade-offs in terms of memory usage and traversal performance:
5//!
6//! *   [`ParentPointerStore`]: Each node stores only its data and a pointer to its parent.
7//!     Memory efficient, fast upward traversal, slow downward traversal (finding children).
8//! *   [`ParentChildStore`]: Uses a first-child, next-sibling representation with cyclic
9//!     sibling links. Balances memory and allows efficient child/sibling iteration.
10//! *   [`ChildVecStore`]: Each node stores its parent, data, and an explicit `Vec` of its children.
11//!     Simple child iteration, potentially higher memory usage for nodes with many children.
12//!
13//! The core components are:
14//! *   [`Forest<R, N>`]: The main struct representing a collection of trees. It stores root data (`R`)
15//!     and uses a specific node storage strategy (`N`) that implements [`ForestNodeStore`].
16//! *   [`ForestNodeStore`] trait: Defines the basic interface for node storage (adding nodes,
17//!     accessing data/parent, mapping data).
18//! *   [`ForestNodeStoreDown`] trait: Extends `ForestNodeStore` with methods for downward traversal
19//!     (iterating children and leaves).
20//! *   [`TreeNodeId`], [`RootId`], [`ParentId`]: Typed identifiers for nodes, roots, and parent links.
21//!
22//! Conversions between the different store types are provided via `From` implementations.
23

148/// Represents a forest (a collection of disjoint trees).
149///
150/// `R` is the type of data associated with each root (tree).
151/// `N` is the storage implementation for the nodes, which must implement [`ForestNodeStore`].
152/// Typically `N` will be one of [`ParentPointerStore<V>`], [`ParentChildStore<V>`], or [`ChildVecStore<V>`],
153/// where `V` is the type of data associated with each node.
154#[derive(Debug, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
155#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
156#[cfg_attr(feature = "bincode", derive(bincode::Encode, bincode::Decode))]
157#[cfg_attr(
158    feature = "rkyv",
159    derive(rkyv::Archive, rkyv::Serialize, rkyv::Deserialize)
160)]
161pub struct Forest<R, N> {
162    /// The underlying storage for all nodes in the forest.
163    pub(crate) nodes: N,
164    /// Metadata for each root/tree in the forest.
165    pub(crate) roots: Vec<RootData<R>>,
166}

272impl<R, N> Forest<R, N>
273where
274    N: ForestNodeStore
275        + Default
276        + Into<ParentPointerStore<N::NodeData>>
277        + From<ParentPointerStore<N::NodeData>>,
278{
279    pub fn reroot_many(&mut self, new_roots: impl IntoIterator<Item = TreeNodeId>) -> usize {
280        let mut nodes: ParentPointerStore<N::NodeData> = std::mem::take(&mut self.nodes).into();
281        let mut changed = 0;
282
283        for new_root in new_roots {
284            let root_id = nodes.change_root(new_root);
285            if self.roots[root_id.0].root_id != new_root {
286                self.roots[root_id.0].root_id = new_root;
287                changed += 1;
288            }
289        }
290
291        self.nodes = nodes.into();
292        changed
293    }
294}

394/// Methods for constructing/modifying the forest.
395impl<R, N> Forest<R, N> {
396    /// Adds a new root node to the forest, creating a new tree.
397    ///
398    /// Takes the data for the new node (`node_data`) and the data for the new tree (`tree_data`).
399    /// Returns the `TreeNodeId` of the new root node and the `RootId` of the new tree.
400    /// Requires `N` to implement `ForestNodeStore<NodeData = T>`.
401    pub fn add_root<T>(&mut self, node_data: T, tree_data: R) -> (TreeNodeId, RootId)
402    where
403        N: ForestNodeStore<NodeData = T>, // Constraint on the node store type
404    {
405        let root_id = RootId(self.roots.len());
406        let root_node_id = self.nodes.add_root(node_data, root_id);
407        self.roots.push(RootData {
408            data: tree_data,
409            root_id: root_node_id,
410        });
411        (root_node_id, root_id)
412    }
413
414    /// Adds a new child node to the specified parent node.
415    ///
416    /// Takes the `TreeNodeId` of the parent and the data for the new child node (`node_data`).
417    /// Returns the `TreeNodeId` of the newly added child node.
418    /// Requires `N` to implement `ForestNodeStore<NodeData = T>`.
419    pub fn add_child<T>(&mut self, parent_id: TreeNodeId, node_data: T) -> TreeNodeId
420    where
421        N: ForestNodeStore<NodeData = T>, // Constraint on the node store type
422    {
423        self.nodes.add_child(node_data, parent_id)
424    }

472    pub(crate) fn is_active_root(&self, root: RootId) -> bool {
473        let root_id = self.roots[root.0].root_id;
474        if root_id.is_empty() {
475            return root_id.is_active_empty();
476        }
477        self.nodes.root_node(root_id) == root_id && self.nodes.root(root_id) == root

480    pub fn validate_structure(&self) -> Result<(), ForestError> {
481        let roots = self.nodes.validate()?;
482
483        for (rid, nid) in roots {
484            if rid.0 >= self.roots.len() {
485                return Err(ForestError::InvalidRootId(rid));
486            }
487            if self.roots[rid.0].root_id != nid {
488                return Err(ForestError::WrongRootPointer(rid));
489            }
490        }
491
492        Ok(())
493    }

495    pub fn debug_draw<FR, FN>(
496        &self,
497        mut format_root: FR, // Closure to format root data (&R) -> String
498        mut format_node: FN, // Closure to format node data (&N::NodeData) -> String
499    ) -> String
500    where
501        FR: FnMut(&U) -> String,
502        FN: FnMut(&Option<P::NodeData>) -> String,
503        P: ForestNodeStoreDown,
504    {
505        let mut output = String::new();
506        let _ = writeln!(output, "Forest ({} roots):", self.roots.len());
507
508        // Recursive helper function - structure remains the same, handles sub-trees
509        fn draw_subtree_recursive<W: Write, N: ForestNodeStoreDown, FmtNode>(
510            f: &mut W,                 // Output writer
511            nodes: &N,                 // Node store access
512            node_id: TreeNodeId,       // Current node to draw
513            prefix: &str,              // Current line prefix (e.g., "  │   ")
514            is_last_child: bool,       // Is this the last child of its parent?
515            format_node: &mut FmtNode, // Mutable reference to node formatter closure
516        ) -> Result<(), std::fmt::Error>
517        where
518            FmtNode: FnMut(&Option<N::NodeData>) -> String, // Closure signature
519        {
520            // Determine the connector shape based on whether it's the last child
521            let connector = if is_last_child {
522                "└── "
523            } else {
524                "├── "
525            };
526            // Use Index trait which should be implemented for the store
527            let node_data_ref = &nodes[node_id]; // Assuming Index<TreeNodeId> exists for N
528            let node_data_str = format_node(node_data_ref);
529
530            // Write the line for the current node
531            writeln!(
532                f,
533                "{prefix}{connector}{node_id}:{node_data_str}", // TreeNodeId needs Debug
534            )?;
535
536            // Prepare the prefix for the children of this node
537            let child_prefix = format!("{}{}", prefix, if is_last_child { "    " } else { "│   " });
538
539            let children: Vec<_> = nodes.iter_children(node_id).collect();
540            let num_children = children.len();
541            for (i, child_id) in children.into_iter().enumerate() {
542                // Recurse: pass the new prefix and whether this child is the last one
543                draw_subtree_recursive(
544                    f,
545                    nodes,
546                    child_id,
547                    &child_prefix,
548                    i == num_children - 1,
549                    format_node,
550                )?;
551            }
552            Ok(())
553        } // End of helper fn definition
554
555        let num_roots = self.roots.len();
556        for (root_idx, root_meta) in self.roots.iter().enumerate() {
557            let root_id = RootId(root_idx);
558            let start_node_id = root_meta.root_id;
559
560            // Format data using closures
561            let root_data_str = format_root(&root_meta.data);
562            // Use Index trait for Forest to access node data via store
563            let root_node_data = &self[start_node_id];
564            let root_node_data_str = format_node(root_node_data);
565
566            // 1. Print the Root line
567            let _ = writeln!(
568                output,
569                "{root_id}:{root_data_str}", // RootId needs Debug
570            );
571
572            if start_node_id.is_empty() {
573                let _ = writeln!(output, "  empty");
574                continue;
575            }
576
577            // 2. Print the connector line
578            let _ = writeln!(output, "  │"); // Fixed indent for the connector
579
580            // 3. Define the prefix for the root node line itself and its direct children
581            let node_line_prefix = "  "; // Prefix for the root TreeNodeId line
582
583            // 4. Print the root TreeNodeId line
584            let _ = writeln!(
585                output,
586                "{node_line_prefix}{start_node_id}:{root_node_data_str}", // TreeNodeId needs Debug
587            );
588
589            // 5. Draw children, starting the recursion with the correct prefix
590            let children: Vec<_> = self.nodes.iter_children(start_node_id).collect();
591            let num_children = children.len();
592
593            // Prefix for the *children's connectors* (└── or ├──)
594            // It's based on the root node's prefix
595            // let children_connector_prefix = format!("{}{}", node_line_prefix, "    "); // Indent matching the root node + 4 spaces for connector alignment
596
597            for (i, child_id) in children.into_iter().enumerate() {
598                // Pass the prefix that aligns the connectors (└──, ├──)
599                // under the root node line.
600                let _ = draw_subtree_recursive(
601                    &mut output,
602                    &self.nodes,
603                    child_id,
604                    node_line_prefix, // Prefix aligns the connector itself
605                    i == num_children - 1,
606                    &mut format_node,
607                );
608            }
609
610            // 6. Add separation between trees (only if not the last root)
611            if root_idx < num_roots - 1 {
612                let _ = writeln!(output); // Just a blank line for separation
613            }
614        }
615
616        output
617    }
618    pub fn swap_roots(&mut self, a: RootId, b: RootId) {
619        if a == b {
620            return;
621        }
622
623        let roota = self.roots[a.0].root_id;
624        let roota_active = self.is_active_root(a);
625        let rootb = self.roots[b.0].root_id;
626        let rootb_active = self.is_active_root(b);
627
628        if roota_active && !roota.is_empty() {
629            self.nodes.set_root(roota, b);
630        }
631        if rootb_active && !rootb.is_empty() {
632            self.nodes.set_root(rootb, a);
633        }
634
635        self.roots.swap(a.0, b.0);
636    }
637
638    pub fn from_bitvec_partition<I: IntoIterator<Item = (U, SuBitGraph)>>(
639        bitvec_part: I,
640    ) -> Result<Self, ForestError> {
641        Forest::<U, ParentPointerStore<P::NodeData>>::from_bitvec_partition_impl(bitvec_part).map(
642            |a| Forest {
643                nodes: P::from_pp(a.nodes),
644                roots: a.roots,
645            },
646        )
647    }
648}

650/// Construction from BitVec partition (remains the same)
651impl<U, V> Forest<U, ParentPointerStore<V>> {
652    /// Creates a forest from a partition represented by a vector of (RootData, BitVec).
653    /// Each BitVec represents the set of node indices belonging to that tree/root.
654    /// Nodes within a set are linked arbitrarily (first encountered becomes root, others point to it).
655    /// Node data is set to `()`.
656    ///
657    /// Returns `Err(ForestError)` if the BitVecs have different lengths or overlap.
658    fn from_bitvec_partition_impl<I: IntoIterator<Item = (U, SuBitGraph)>>(
659        bitvec_part: I,
660    ) -> Result<Self, ForestError> {
661        let mut nodes = vec![];
662        let mut roots = vec![];
663        let mut cover: Option<SuBitGraph> = None;
664
665        for (d, set) in bitvec_part {
666            let len = set.size();
667            if let Some(c) = &mut cover {
668                if c.size() != len {
669                    return Err(ForestError::LengthMismatch);
670                }
671                if c.intersects(&set) {
672                    return Err(ForestError::DoesNotPartion);
673                }
674                c.union_with(&set);
675            } else {
676                cover = Some(SuBitGraph::empty(len));
677
678                nodes.resize_with(len, || None);
679                // nodes = vec![None; len];
680            }
681            let mut first = None;
682            for i in set.included_iter() {
683                if let Some(root) = first {
684                    nodes[i.0] = Some(PPNode::dataless_child(root))
685                } else {
686                    first = Some(i.into());
687                    nodes[i.0] = Some(PPNode::dataless_root(RootId(roots.len())));
688                }
689            }
690            roots.push(RootData {
691                root_id: first.unwrap_or(TreeNodeId::EMPTY),
692                data: d,
693            });
694        }
695        Ok(Forest {
696            nodes: nodes
697                .into_iter()
698                .collect::<Option<Vec<_>>>()
699                .unwrap()
700                .into_iter()
701                .collect(),
702            roots,
703        })
704    }
705}

709/// The core trait defining the interface for node storage within a `Forest`.
710/// Requires Index and IndexMut support for accessing node data and ParentId.
711pub trait ForestNodeStore:
712    for<'a> Index<&'a TreeNodeId, Output = ParentId>      // `store[&node_id]` -> ParentId
713    + Index<TreeNodeId, Output = Option<Self::NodeData>>          // `store[node_id]` -> NodeData
714    + IndexMut<TreeNodeId, Output = Option<Self::NodeData>>   // `store[node_id] = ...`
715    + Sized // Needed for some default implementations returning Self iterators
716{
717    type NodeData;
718    type Store<T>: ForestNodeStore<NodeData = T>+From<ParentPointerStore<T>>+Into<ParentPointerStore<T>>;
719
720    fn parent(&self,child:TreeNodeId)->ParentId{
721        self[&child]
722    }
723
724
725
726
727
728    fn to_store(self)->Self::Store<Self::NodeData>;
729    fn from_store(store:Self::Store<Self::NodeData>)->Self;
730    fn debug_draw(&self,node_display:impl FnMut(Option<&Self::NodeData>)->Option<String>)->String;
731    fn validate(&self)->Result<Vec<(RootId,TreeNodeId)>,ForestError>;
732
733    fn panicing_validate(&self){
734        self.validate()
735            .unwrap_or_else(|a| panic!("Err:{}\nforest:\n{}", a, self.debug_draw(|_| None)));
736    }
737
738    fn n_nodes(&self)->usize;
739    fn swap(&mut self,a:TreeNodeId,b:TreeNodeId);
740    fn set_root(&mut self,a:TreeNodeId,root:RootId);
741
742    fn from_pp(pp:ParentPointerStore<Self::NodeData>)->Self{
743        Self::from_store(Self::Store::<Self::NodeData>::from(pp))
744    }
745
746
747
748    fn split_off(&mut self,at:TreeNodeId)->Self;
749
750
751    /// Makes x's root, the root of y's root.
752    fn make_root_of(&mut self, x: TreeNodeId, y: TreeNodeId) -> TreeNodeId{
753        let rootx = self.root_node(x);
754        let rooty = self.root_node(y);
755        self.reparent(rootx,rooty);
756        rootx
757    }
758
759    fn reparent(&mut self,parent:TreeNodeId,child:TreeNodeId);
760
761    /// Finds the `RootId` by traversing upwards. Default implementation provided.
762    fn root(&self, nodeid: TreeNodeId) -> RootId { let mut current = nodeid;
763            loop {
764                match self[&current] {
765                    ParentId::PointingRoot(p)=>current=p,
766                    ParentId::Root(root_id) => return root_id,
767                    ParentId::Node(parent_node_id) => current = parent_node_id,
768                }
769            } }
770
771    fn root_node(&self,nodeid:TreeNodeId)->TreeNodeId{
772        let mut current = nodeid;
773        loop {
774            match self[&current] {
775                ParentId::PointingRoot(r)=> current=r,
776                ParentId::Root(_) => return current,
777                ParentId::Node(parent_node_id) => current = parent_node_id,
778            }
779        }
780    }
781
782    fn iter_nodes(&self) -> impl Iterator<Item = (TreeNodeId, Option<&Self::NodeData>)>;
783    fn iter_node_id(&self) -> NodeIdIter {
784        NodeIdIter {
785            range: 0..self.n_nodes(),
786            current: TreeNodeId(0),
787        }
788    }
789    /// Adds a new root node.
790    fn add_root(&mut self, data: Self::NodeData, root_id: RootId) -> TreeNodeId;
791    fn add_dataless_root(&mut self, root_id: RootId) -> TreeNodeId;
792
793    /// Adds a new child node as the *last* child of the parent.
794    fn add_child(&mut self, data: Self::NodeData, parent: TreeNodeId) -> TreeNodeId;
795
796
797    fn set_node_data(&mut self,data:Self::NodeData,node_id:TreeNodeId)->Option<Self::NodeData>;
798    /// Adds a new child node as the *last* child of the parent.
799    fn add_dataless_child(&mut self, parent: TreeNodeId) -> TreeNodeId;
800
801    fn map<F, U>(self, transform: F) -> Self::Store<U> where F: FnMut(Self::NodeData) -> U;
802    fn map_ref<F, U>(&self, transform: F) -> Self::Store<U> where F: FnMut(&Self::NodeData) -> U;
803    fn forgetful_map<U>(&self)->Self::Store<U>;
804
805    fn extend(&mut self,other:Self,shift_roots_by:RootId);
806}

848/// Default implementation for ancestor iteration for any `ForestNodeStore`.
849impl<N: ForestNodeStore> ForestNodeStoreAncestors for N {
850    fn iter_ancestors(&self, start_node: TreeNodeId) -> impl Iterator<Item = TreeNodeId> + '_ {
851        // Use the helper struct from the iterators module

856/// Default implementation for root leaves iteration for any `ForestNodeStoreDown`.
857impl<N: ForestNodeStoreDown> ForestNodeStoreRootLeaves for N {
858    fn iter_root_leaves(&self, root_id: RootId) -> impl Iterator<Item = TreeNodeId> + '_ {
859        self.iter_leaves()
860            .filter(move |&leaf_id| self.root(leaf_id) == root_id)
