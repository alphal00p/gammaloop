12/// Represents a node within a `ParentPointerStore`.
13///
14/// Contains the actual data (`V`) and a [ParentId] which points
15/// either to the parent [TreeNodeId] or identifies this node as a root via `RootId`.
16#[derive(Clone, Debug, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
17#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
18#[cfg_attr(feature = "bincode", derive(bincode::Encode, bincode::Decode))]
19#[cfg_attr(
20    feature = "rkyv",
21    derive(rkyv::Archive, rkyv::Serialize, rkyv::Deserialize)
22)]
23pub struct PPNode<V> {
24    /// Pointer to the parent node or the root ID if this is a root node.
25    pub(crate) parent: ParentId,
26    /// The data associated with this node.
27    pub(crate) data: Option<V>,
28}
29
30impl<V> PPNode<V> {
31    pub fn child(data: V, parent: TreeNodeId) -> Self {
32        PPNode {
33            parent: ParentId::Node(parent),
34            data: Some(data),
35        }
36    }

38    pub fn debug_display(
39        &self,
40        writer: &mut impl std::fmt::Write,
41        draw_data: &mut impl FnMut(Option<&V>) -> Option<String>,
42    ) {
43        write!(writer, "{}", self.parent).unwrap();
44        if let Some(root_data) = draw_data(self.data.as_ref()) {
45            writeln!(writer, "{root_data}").unwrap();
46        } else {
47            writeln!(writer).unwrap();
48        }
49    }
50
51    pub fn forget<U>(&self) -> PPNode<U> {
52        PPNode {
53            parent: self.parent,
54            data: None,
55        }
56    }
57
58    pub fn shift(&mut self, root: RootId, nodes: TreeNodeId) {
59        match &mut self.parent {
60            ParentId::Node(n) => {
61                n.0 += nodes.0;
62            }
63            ParentId::PointingRoot(n) => {
64                n.0 += nodes.0;
65            }
66            ParentId::Root(r) => {
67                r.0 += root.0;
68            }
69        }
70    }
71
72    pub fn dataless_child(parent: TreeNodeId) -> Self {
73        PPNode {
74            parent: ParentId::Node(parent),
75            data: None,
76        }
77    }
78
79    pub fn root(data: V, root_id: RootId) -> Self {
80        PPNode {
81            parent: ParentId::Root(root_id),
82            data: Some(data),
83        }
84    }
85
86    pub fn dataless_root(root_id: RootId) -> Self {
87        PPNode {
88            parent: ParentId::Root(root_id),
89            data: None,
90        }
91    }
92
93    pub fn map<F, U>(self, transform: F) -> PPNode<U>
94    where
95        F: FnMut(V) -> U,
96    {
97        PPNode {
98            parent: self.parent,
99            data: self.data.map(transform),
100        }
101    }
102
103    pub fn map_ref<F, U>(&self, transform: F) -> PPNode<U>
104    where
105        F: FnMut(&V) -> U,
106    {
107        PPNode {
108            parent: self.parent,
109            data: self.data.as_ref().map(transform),
110        }
111    }

114/// Identifies the parent of a node.
115///
116/// A node is either a child of another `Node` or it's the `Root` of a tree,
117/// identified by a `RootId`.
118#[derive(Clone, Debug, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
119#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
120#[cfg_attr(feature = "bincode", derive(bincode::Encode, bincode::Decode))]
121#[cfg_attr(
122    feature = "rkyv",
123    derive(rkyv::Archive, rkyv::Serialize, rkyv::Deserialize)
124)]
125pub enum ParentId {
126    /// This node is a root node belonging to the tree identified by `RootId`.
127    Root(RootId),
128
129    /// This node is a child of the node identified by `TreeNodeId`.
130    Node(TreeNodeId),
131    /// for use when splitting
132    PointingRoot(TreeNodeId),
133}
134
135impl Display for ParentId {
136    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
137        match self {
138            ParentId::Node(n) => write!(f, "parent node: {n}"),
139            ParentId::Root(r) => write!(f, "root id:{r}"),
140            ParentId::PointingRoot(n) => write!(f, "root node:{n}"),
141        }
142    }
143}
144
145impl ParentId {
146    pub fn is_root(&self) -> bool {
147        match self {
148            ParentId::Root(_) | ParentId::PointingRoot(_) => true,
149            ParentId::Node(_) => false,
150        }
151    }
152
153    pub fn is_node(&self) -> bool {
154        !self.is_root()
155    }
156}

158/// A forest data structure where each node only stores its data and a pointer to its parent.
159///
160/// This representation is memory-efficient, especially for sparse trees, and allows for
161/// very fast traversal *upwards* towards the root. However, traversing *downwards*
162/// (finding children) requires iterating through all nodes to find those pointing to a
163/// specific parent, which can be slow.
164///
165/// It implements the `ForestNodeStore` trait.
166#[derive(Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord, Default)]
167#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
168#[cfg_attr(feature = "bincode", derive(bincode::Encode, bincode::Decode))]
169#[cfg_attr(
170    feature = "rkyv",
171    derive(rkyv::Archive, rkyv::Serialize, rkyv::Deserialize)
172)]
173pub struct ParentPointerStore<V> {
174    /// The flat list of nodes. The index in the vector corresponds to the `TreeNodeId`.
175    pub(crate) nodes: Vec<PPNode<V>>,
176}
177
178impl<V, R> Forest<R, ParentPointerStore<V>> {
179    /// Changes the root of the tree containing `new_root`.
180    ///
181    /// All parent pointers on the path from the `new_root` to the original root
182    /// are reversed. The `new_root` becomes the root node associated with the
183    /// original tree's `RootId`. Updates the `Forest`'s root tracking accordingly.
184    ///
185    /// Returns the `RootId` of the tree that was modified.
186    pub fn change_to_root(&mut self, new_root: TreeNodeId) -> RootId {
187        let root_id = self.nodes.change_root(new_root);
188        self.roots[root_id.0].root_id = new_root;
189        root_id
190    }
191}

193impl<V> ParentPointerStore<V> {
194    /// Re–roots the tree at the given node (making it a root).
195    /// Along the chain from `new_root` to the old root, the parent pointers are reversed.
196    /// The `new_root` becomes associated with the original `RootId`.
197    ///
198    /// Returns the `RootId` of the affected tree.
199    ///
200    /// **Note:** This modifies the store directly. If using within a `Forest`,
201    /// prefer `Forest::change_to_root` which also updates the `Forest`'s root list.
202    pub fn change_root(&mut self, new_root: TreeNodeId) -> RootId {
203        let mut current = new_root;
204        let root_id = self.root(new_root);
205        let mut prev = None;
206
207        loop {
208            let orig_parent = self[&current];
209            self.nodes[current.0].parent = match prev {
210                None => ParentId::Root(root_id),
211                Some(p) => ParentId::Node(p),
212            };
213            prev = Some(current);
214
215            match orig_parent {
216                ParentId::Node(p) => current = p,
217
218                ParentId::PointingRoot(p) => current = p,
219                ParentId::Root(r) => return r,
220            }
221        }
222    }
223}

253impl<V> ForestNodeStore for ParentPointerStore<V> {

255    type Store<T> = ParentPointerStore<T>;

257    fn set_root(&mut self, a: TreeNodeId, root: RootId) {
258        let rootx = self.root_node(a);
259        self.nodes[rootx.0].parent = ParentId::Root(root);
260    }
261
262    fn validate(&self) -> Result<Vec<(RootId, TreeNodeId)>, super::ForestError> {
263        let mut roots = vec![];
264        let mut seen: SubSet<usize> = !SubSet::empty(self.n_nodes());
265
266        while let Some(next) = seen.included_iter().next() {
267            let current = TreeNodeId(next);
268            seen.sub(next);
269            self.iter_ancestors(current).for_each(|a| {
270                seen.sub(a.0);
271            });
272            let root_id = self.root_node(current);
273            let root = self.root(root_id);
274
275            roots.push((root, root_id));
276        }
277
278        Ok(roots)
279    }
280
281    fn debug_draw(
282        &self,
283        _node_display: impl FnMut(Option<&Self::NodeData>) -> Option<String>,
284    ) -> String {
285        "".to_string()
286    }

288        self.nodes.len()
289    }

291    fn set_node_data(&mut self, data: Self::NodeData, node_id: TreeNodeId) -> Option<V> {
292        let old = self.nodes[node_id.0].data.take();
293        self.nodes[node_id.0].data = Some(data);

310    // with path compression for split nodes
311    fn split_off(&mut self, at: TreeNodeId) -> Self {
312        for mut current in (0..self.nodes.len()).map(TreeNodeId) {
313            while let ParentId::Node(p) = self[&current] {
314                let mut newp = p;
315                while (newp <= at) ^ (current > at) {
316                    // the pointer crosses the splitting boundary, we need to find a new pointer.
317
318                    match self[&newp] {
319                        ParentId::Node(next) => {
320                            newp = next;
321                        }
322                        ParentId::Root(_) => {
323                            // split_root = true;
324                            break;
325                        }
326                        ParentId::PointingRoot(next) => {
327                            newp = next;
328                        }
329                    }
330                }
331
332                if let ParentId::Root(r) = self[&newp] {
333                    // set current to now be the root, and original root to point here.
334
335                    self.nodes[newp.0].parent = ParentId::PointingRoot(current);
336                    self.nodes[current.0].parent = ParentId::Root(r);
337                } else {
338                    self.reparent(newp, current);
339                }
340
341                current = p
342            }
343        }
344
345        for n in (0..self.nodes.len()).map(TreeNodeId) {
346            if let ParentId::Node(n) = &mut self.nodes[n.0].parent {
347                if *n > at {
348                    n.0 -= at.0;
349                }
350            }
351
352            if let ParentId::PointingRoot(rp) = self[&n] {
353                let root = self[&rp];
354                self.nodes[n.0].parent = root;
355            }
356        }
357        Self {
358            nodes: self.nodes.split_off(at.0),
359        }
360    }
361
362    fn extend(&mut self, other: Self, shift_roots_by: RootId) {
363        let nodeshift = TreeNodeId(self.nodes.len());
364        self.nodes.extend(other.nodes.into_iter().map(|mut r| {
365            r.shift(shift_roots_by, nodeshift);
366            r
367        }));
368    }
369
370    fn reparent(&mut self, parent: TreeNodeId, child: TreeNodeId) {
371        self.nodes[child.0].parent = ParentId::Node(parent);
372    }

374    fn swap(&mut self, a: TreeNodeId, b: TreeNodeId) {
375        println!("Swap PP");
376        for n in &mut self.nodes {
377            if let ParentId::Node(pp) = &mut n.parent {
378                if *pp == a {
379                    *pp = b;
380                } else if *pp == b {
381                    *pp = a;
382                }
383            }
384        }
385
386        self.nodes.swap(a.0, b.0);
387    }
