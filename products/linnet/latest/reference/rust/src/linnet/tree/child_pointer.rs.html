16/// Represents a node within a `ParentChildStore`.
17///
18/// Contains the parent pointer and data (`PPNode`), an optional pointer to the
19/// *first* child, and pointers to the left and right siblings. The sibling
20/// pointers form a cyclic doubly linked list among all children of the same parent.
21/// If a node is the only child, its `neighbor_left` and `neighbor_right` point to itself.
22#[derive(Clone, Debug, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
23#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
24#[cfg_attr(feature = "bincode", derive(bincode::Encode, bincode::Decode))]
25#[cfg_attr(
26    feature = "rkyv",
27    derive(rkyv::Archive, rkyv::Serialize, rkyv::Deserialize)
28)]
29pub struct PCNode<V> {
30    /// Parent pointer and node data.
31    pub(crate) parent_pointer: PPNode<V>,
32    /// Pointer to the first child of this node, if any.
33    pub(crate) child: Option<TreeNodeId>, // To get the other children, follow the sibling links of this child
34    /// Pointer to the previous (left) sibling in the cyclic list. Points to self if only child.
35    pub(crate) neighbor_left: TreeNodeId, // Will form a cyclic linked list, if equal to right, then it is the only child
36    /// Pointer to the next (right) sibling in the cyclic list. Points to self if only child.
37    pub(crate) neighbor_right: TreeNodeId, // previous sibling, if any. Will form a cyclic linked list, if equal to left, then it is the only child
38}

40impl<V> PCNode<V> {
41    pub fn debug_display(
42        &self,
43        writer: &mut impl std::fmt::Write,
44        draw_data: &mut impl FnMut(Option<&V>) -> Option<String>,
45    ) {
46        write!(writer, "{}<- ->{}", self.neighbor_left, self.neighbor_right).unwrap();
47        if let Some(child) = self.child {
48            write!(writer, "child: {child}").unwrap();
49        }
50        self.parent_pointer.debug_display(writer, draw_data);
51    }
52
53    pub fn shift(&mut self, root: RootId, nodes: TreeNodeId) {
54        self.neighbor_left.0 += nodes.0;
55        self.neighbor_right.0 += nodes.0;
56        self.parent_pointer.shift(root, nodes);
57        if let Some(child) = &mut self.child {
58            child.0 += nodes.0;
59        }
60    }
61
62    pub fn forget<U>(&self) -> PCNode<U> {
63        PCNode {
64            parent_pointer: self.parent_pointer.forget(),
65            child: self.child,
66            neighbor_left: self.neighbor_left,
67            neighbor_right: self.neighbor_right,
68        }
69    }
70
71    pub fn map<F, U>(self, transform: F) -> PCNode<U>
72    where
73        F: FnMut(V) -> U,
74    {
75        PCNode {
76            parent_pointer: self.parent_pointer.map(transform),
77            child: self.child,
78            neighbor_left: self.neighbor_left,
79            neighbor_right: self.neighbor_right,
80        }
81    }
82
83    pub fn map_ref<F, U>(&self, transform: F) -> PCNode<U>
84    where
85        F: FnMut(&V) -> U,
86    {
87        PCNode {
88            parent_pointer: self.parent_pointer.map_ref(transform),
89            child: self.child,
90            neighbor_right: self.neighbor_right,
91            neighbor_left: self.neighbor_left,
92        }
93    }
94}

96/// A forest data structure using a first-child, next-sibling representation with cyclic sibling links.
97///
98/// Each node stores its parent, data, a pointer to its first child, and pointers to its left and
99/// right siblings. Siblings under a parent form a cyclic doubly linked list.
100///
101/// This representation allows for efficient iteration over children and siblings.
102/// It implements `ForestNodeStore` and `ForestNodeStoreDown`.
103#[derive(Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord, Default)]
104#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
105#[cfg_attr(feature = "bincode", derive(bincode::Encode, bincode::Decode))]
106#[cfg_attr(
107    feature = "rkyv",
108    derive(rkyv::Archive, rkyv::Serialize, rkyv::Deserialize)
109)]
110pub struct ParentChildStore<V> {
111    /// The flat list of nodes. The index in the vector corresponds to the `TreeNodeId`.
112    pub(crate) nodes: Vec<PCNode<V>>,
113}

143/// An iterator over the siblings of a node (including the node itself).
144/// Uses the cyclic `neighbor_right` pointers.
145pub enum NeighborIter<'a, V> {
146    /// Case where the node has no children/siblings to iterate over.
147    None,
148    Some {
149        initial: TreeNodeId,
150        current: Option<TreeNodeId>,
151        first: bool,
152        store: &'a ParentChildStore<V>,
153    },
154}
155
156impl<V> Iterator for NeighborIter<'_, V> {
157    type Item = TreeNodeId;
158
159    fn next(&mut self) -> Option<Self::Item> {
160        match self {
161            NeighborIter::None => None,
162            NeighborIter::Some {
163                initial,
164                current,
165                first,
166                store,
167            } => {
168                let next = *current;
169                if let Some(c) = next {
170                    if c == *initial && !*first {
171                        return None;
172                    }
173                    *first = false;
174                    *current = store.right_neighbor_cyclic(c);
175                }
176                next
177            }
178        }
179    }
180}

182impl<V> ParentChildStore<V> {
183    /// Returns the left sibling of `node_id`.
184    /// Returns `None` if `node_id` is the only child of its parent (based on cyclic link).
185    pub fn left_neighbor_cyclic(&self, node_id: TreeNodeId) -> Option<TreeNodeId> {
186        if self.nodes[node_id.0].neighbor_left == node_id {
187            None
188        } else {
189            Some(self.nodes[node_id.0].neighbor_left)
190        }
191    }
192
193    /// Returns the left sibling of `node_id`, stopping at the "first" child.
194    ///
195    /// This differs from `left_neighbor_cyclic` by returning `None` if moving left
196    /// would wrap around from the *parent's first recorded child* back to the last child.
197    /// Useful if a non-cyclic view of siblings is needed, although the internal
198    /// representation *is* cyclic.
199    /// Returns `None` if the node is a root or the only child.
200    pub fn left_neighbor_ending(&self, node_id: TreeNodeId) -> Option<TreeNodeId> {
201        let left = self.nodes[node_id.0].neighbor_left;
202        if left == node_id {
203            None
204        } else {
205            let first_neighbor = match self[&node_id] {
206                ParentId::Node(p) => p,
207                ParentId::Root(_) => return None,
208                ParentId::PointingRoot(p) => p,
209            };
210            if left == first_neighbor {
211                None
212            } else {
213                Some(left)
214            }
215        }
216    }
217
218    /// Returns the first child of `node_id`, if one exists.
219    pub fn first_child(&self, node_id: TreeNodeId) -> Option<TreeNodeId> {
220        if node_id.is_empty() {
221            return None;
222        }
223        self.nodes[node_id.0].child
224    }
225
226    /// Returns the right sibling of `node_id`.
227    /// Returns `None` if `node_id` is the only child of its parent (based on cyclic link).
228    pub fn right_neighbor_cyclic(&self, node_id: TreeNodeId) -> Option<TreeNodeId> {
229        if self.nodes[node_id.0].neighbor_right == node_id {
230            None
231        } else {
232            Some(self.nodes[node_id.0].neighbor_right)
233        }
234    }
235
236    /// Returns the right sibling of `node_id`, stopping before wrapping around the cycle.
237    ///
238    /// This differs from `right_neighbor_cyclic` by returning `None` if moving right
239    /// would wrap around from the *last* child back to the *parent's first recorded child*.
240    /// Useful if a non-cyclic view of siblings is needed.
241    /// Returns `None` if the node is a root or the only child.
242    pub fn right_neighbor_ending(&self, node_id: TreeNodeId) -> Option<TreeNodeId> {
243        let right = self.nodes[node_id.0].neighbor_right;
244        if right == node_id {
245            None
246        } else {
247            let first_neighbor = match self[&node_id] {
248                ParentId::Node(p) => p,
249                ParentId::Root(_) => return None,
250                ParentId::PointingRoot(p) => p,
251            };
252            if right == first_neighbor {
253                None
254            } else {
255                Some(right)
256            }
257        }
258    }
259}

290    fn validate(&self) -> Result<Vec<(RootId, TreeNodeId)>, super::ForestError> {
291        let mut roots = vec![];
292        let mut seen: SubSet<TreeNodeId> = SubSet::full(self.n_nodes());
293
294        while let Some(next) = seen.included_iter().next() {
295            let current = next;
296            seen.sub(next);
297            let mut children: SubSet<TreeNodeId> = SubSet::empty(self.n_nodes());
298            for p in self.iter_ancestors(current) {
299                seen.sub(p);
300                if children.includes(&p) {
301                    return Err(ForestError::CyclicPP);
302                }
303                children.add(p);
304            }
305            for c in self.iter_children(current) {
306                if let ParentId::Node(n) = self[&c] {
307                    if n != current {
308                        return Err(ForestError::WrongPP(current));
309                    }
310                }
311            }
312            let root_id = self.root_node(current);
313            let root = self.root(root_id);
314
315            roots.push((root, root_id));
316        }
317
318        Ok(roots)
319    }

321    fn debug_draw(
322        &self,
323        mut node_display: impl FnMut(Option<&Self::NodeData>) -> Option<String>,
324    ) -> String {
325        // fn debug_draw(&self, node_display: impl FnMut()) -> String {
326        let mut output = String::new();
327
328        // Recursive helper function - structure remains the same, handles sub-trees
329        fn draw_subtree_recursive<W: Write, V>(
330            f: &mut W,                                                  // Output writer
331            nodes: &ParentChildStore<V>,                                // Node store access
332            node_id: TreeNodeId,                                        // Current node to draw
333            prefix: &str,        // Current line prefix (e.g., "  │   ")
334            is_last_child: bool, // Is this the last child of its parent?
335            format_node: &mut impl FnMut(Option<&V>) -> Option<String>, // Mutable reference to node formatter closure
336            seen: &mut SubSet<TreeNodeId>,
337        ) -> Result<(), std::fmt::Error> {
338            // Determine the connector shape based on whether it's the last child
339            let connector = if is_last_child {
340                "└── "
341            } else {
342                "├── "
343            };
344
345            write!(f, "{prefix}{connector}{node_id}:")?;
346            nodes.nodes[node_id.0].debug_display(f, format_node);
347
348            seen.sub(node_id);
349            // Prepare the prefix for the children of this node
350            let child_prefix = format!("{}{}", prefix, if is_last_child { "    " } else { "│   " });
351
352            let children: Vec<_> = nodes.iter_children(node_id).collect();
353            let num_children = children.len();
354            for (i, child_id) in children.into_iter().enumerate() {
355                seen.add(child_id);
356                // Recurse: pass the new prefix and whether this child is the last one
357                draw_subtree_recursive(
358                    f,
359                    nodes,
360                    child_id,
361                    &child_prefix,
362                    i == num_children - 1,
363                    format_node,
364                    seen,
365                )?;
366            }
367            Ok(())
368        } // End of helper fn definition
369
370        let mut seen: SubSet<TreeNodeId> = SubSet::full(self.n_nodes());
371
372        while let Some(next) = seen.included_iter().next() {
373            let root_id = self.root_node(next);
374            // println!("{root_id}");
375            seen.sub(next);
376
377            seen.sub(root_id);
378
379            let node_line_prefix = "  ";
380
381            write!(output, "{node_line_prefix}{root_id}:").unwrap();
382            self.nodes[root_id.0].debug_display(&mut output, &mut node_display);
383
384            let children: Vec<_> = self
385                .iter_children(root_id)
386                .inspect(|a| {
387                    seen.sub(*a);
388                })
389                .collect();
390            let num_children = children.len();
391
392            for (i, child_id) in children.into_iter().enumerate() {
393                // Pass the prefix that aligns the connectors (└──, ├──)
394                // under the root node line.
395                seen.sub(child_id);
396                let _ = draw_subtree_recursive(
397                    &mut output,
398                    self,
399                    child_id,
400                    node_line_prefix, // Prefix aligns the connector itself
401                    i == num_children - 1,
402                    &mut node_display,
403                    &mut seen,
404                );
405            }
406
407            let _ = writeln!(output);
408        }
409
410        output
411    }

451    fn reparent(&mut self, parent: TreeNodeId, child: TreeNodeId) {
452        let parent_node = &mut self.nodes[parent.0];
453        if let Some(first_child_id) = parent_node.child {
454            // Parent already has children. Insert new child *after* the current last child.
455            let last_child_id = self.nodes[first_child_id.0].neighbor_left; // Get current last child (left of first)
456
457            self.nodes[last_child_id.0].neighbor_right = child;
458            self.nodes[first_child_id.0].neighbor_left = child;
459            self.nodes[child.0].neighbor_left = last_child_id;
460            self.nodes[child.0].neighbor_right = first_child_id;
461        } else {
462            parent_node.child = Some(child);
463
464            self.nodes[child.0].neighbor_left = child;
465            self.nodes[child.0].neighbor_right = child;
466        }
467
468        self.nodes[child.0].parent_pointer.parent = ParentId::Node(parent);
469    }

576    use super::*;
577    pub struct PreorderIter<'a, V> {
578        store: &'a ParentChildStore<V>,
579        current: Option<TreeNodeId>,
580        // Could potentially add a stack or use the recursive logic helper
581    }
582
583    #[allow(clippy::non_canonical_clone_impl)]
584    impl<V> Clone for PreorderIter<'_, V> {
585        fn clone(&self) -> Self {
586            Self {
587                store: self.store,
588                current: self.current,
589            }
590        }
591    }
592
593    impl<V> Copy for PreorderIter<'_, V> {}
594
595    impl<'a, V> PreorderIter<'a, V> {
596        pub fn new(store: &'a ParentChildStore<V>, start: TreeNodeId) -> Self {
597            PreorderIter {
598                store,
599                current: (!start.is_empty()).then_some(start),
600            }
601        }
602
603        // Helper based on the original logic
604        fn next_node_logic(&self, node: TreeNodeId) -> Option<TreeNodeId> {
605            // 1. If the current node has a (first) child, return it.
606            if let Some(child) = self.store.first_child(node) {
607                return Some(child);
608            }
609            // 2. Otherwise, climb upward
610            let mut current = node;
611            loop {
612                // Try right sibling. Crucially, use right_neighbor_cyclic.
613                // Need to detect when we loop back to the first sibling without going up.
614                let right_neighbor = self.store.right_neighbor_cyclic(current);
615
616                // Find the parent's first child to detect cycle completion *at this level*.
617                let parent_id = self.store[&current];
618                let first_child_of_parent = match parent_id {
619                    ParentId::Node(p) | ParentId::PointingRoot(p) => self.store.nodes[p.0].child,
620                    ParentId::Root(_) => None,
621                };
622
623                // If there IS a right neighbor, AND it's NOT the first child (meaning we haven't wrapped)
624                if let Some(sibling) = right_neighbor {
625                    if Some(sibling) != first_child_of_parent {
626                        return Some(sibling); // Go to the sibling
627                    }
628                    // If sibling == first_child_of_parent, we wrapped around. Need to go up. Fall through.
629                }
630                // Only child or wrapped around siblings: move upward.
631                match parent_id {
632                    ParentId::Node(p) | ParentId::PointingRoot(p) => current = p,
633                    ParentId::Root(_) => return None, // Reached root while climbing; traversal ends
634                }
635            }
636        }
637    }
638
639    impl<V> Iterator for PreorderIter<'_, V> {
640        type Item = TreeNodeId;
641        fn next(&mut self) -> Option<Self::Item> {
642            let node_to_return = self.current?;
643            self.current = self.next_node_logic(node_to_return);
644            Some(node_to_return)
645        }
646    }
647}

652    #[derive(Clone)]
653    pub struct BfsTreeIter<'a, V> {
654        store: &'a ParentChildStore<V>,
655        queue: VecDeque<TreeNodeId>,
656    }
657    impl<'a, V> BfsTreeIter<'a, V> {
658        pub fn new(store: &'a ParentChildStore<V>, start: TreeNodeId) -> Self {
659            let mut queue = VecDeque::new();
660            if !start.is_empty() {
661                queue.push_back(start);
662            }
663            BfsTreeIter { store, queue }
664        }
665    }
666    impl<V> Iterator for BfsTreeIter<'_, V> {
667        type Item = TreeNodeId;
668        fn next(&mut self) -> Option<Self::Item> {
669            let node = self.queue.pop_front()?;
670            // Use the store's iter_children which correctly uses NeighborIter
671            for child in self.store.iter_children(node) {
672                self.queue.push_back(child);
673            }
674            Some(node)
675        }
676    }
677}
