1//! # Graph Drawing and Visualization
2//!
3//! This module is intended to provide utilities for visualizing graph structures,
4//! primarily by generating Scalable Vector Graphics (SVG) outputs. It aims to
5//! facilitate the visual inspection and analysis of graphs represented within
6//! the `linnet` library.
7//!
8//! Currently, the module's direct public API for drawing `HedgeGraph` instances
9//! is not yet fully exposed or implemented. The existing code includes examples
10//! and tests demonstrating the use of the [`kurbo`](https://crates.io/crates/kurbo)
11//! crate for creating and manipulating 2D Bézier paths, and serializing these
12//! paths into SVG format.
13//!
14//! Future development in this module will likely focus on:
15//! - Providing functions to take a `HedgeGraph` and produce an SVG representation.
16//! - Integrating layout algorithms to position nodes and edges aesthetically.
17//! - Offering customization options for styling the output (colors, stroke widths, etc.).
18//!
19//! Users interested in path drawing can look at the internal tests for examples of
20//! using `kurbo::BezPath` and its SVG export capabilities. For direct `HedgeGraph`
21//! visualization, the DOT output functionalities (see `dot_parser` module and
22//! `HedgeGraph::dot_...` methods) can be used in conjunction with external
23//! Graphviz tools.
24
25#[cfg(test)]
26pub mod test {
27
28    use kurbo::{
29        simplify::{simplify_bezpath, SimplifyOptLevel, SimplifyOptions},
30        BezPath, Point, Rect, Shape, Size,
31    };
32    use std::{f64::consts::PI, fs::File, io::Write};
33
34    #[test]
35    fn helix() {
36        // Generate helix points
37        let alpha = 0.35;
38        let radius = 10.0;
39        let steps = 1000;
40        let t_range = (0.0, 46.0 * PI);
41        let step_size = (t_range.1 - t_range.0) / steps as f64;
42
43        let mut points = Vec::new();
44        for i in 0..=steps {
45            let t = t_range.0 + i as f64 * step_size;
46            let x = (alpha * t).sin() * radius + t;
47            let y = (alpha * t).cos() * radius;
48            points.push(Point::new(x, y));
49        }
50
51        // Create path
52        let mut path = BezPath::new();
53        path.move_to(points[0]);
54        for point in points.iter().skip(1) {
55            path.line_to(*point);
56        }
57        let options = SimplifyOptions::default().opt_level(SimplifyOptLevel::Optimize);
58        let simplified = simplify_bezpath(path.clone(), 1.0, &options);
59
60        // Get the bounding box of the path
61        let bbox: Rect = path.bounding_box();
62
63        // Define target SVG dimensions and padding
64        let svg_width = 800.0;
65        let svg_height = 600.0;
66        let padding = 50.0;
67
68        // Expand the bounding box by padding (to keep a margin)
69        let padded_bbox = Rect::from_origin_size(
70            Point::new(bbox.min_x() - padding, bbox.min_y() - padding),
71            Size::new(bbox.width() + 2.0 * padding, bbox.height() + 2.0 * padding),
72        );
73
74        // Create the SVG with a viewBox set to the padded bounding box.
75        let svg_string = format!(
76        "<svg width='{width}' height='{height}' viewBox='{min_x} {min_y} {vb_width} {vb_height}' xmlns='http://www.w3.org/2000/svg'>\n  \
77         <path d='{path}' stroke='#000' fill='none' stroke-width='2'/>\n</svg>",
78        width = svg_width,
79        height = svg_height,
80        min_x = padded_bbox.min_x(),
81        min_y = padded_bbox.min_y(),
82        vb_width = padded_bbox.width(),
83        vb_height = padded_bbox.height(),
84        path = simplified.to_svg(),
85    );
86
87        // Write to file
88        let mut file = File::create("output.svg").unwrap();
89        write!(file, "{svg_string}").unwrap();
90    }
91}
