16/// Represents a cycle in a graph with a specific orientation or traversal direction.
17///
18/// While a [`Cycle`] represents the set of edges forming a cycle, a `SignedCycle`
19/// implies a particular directed path along these edges. This can be important
20/// in contexts where the direction of traversal matters.
21pub struct SignedCycle {
22    /// A bitmask indicating the set of half-edges that constitute this signed cycle,
23    /// potentially implying a specific order or direction of traversal.
24    pub filter: SuBitGraph,
25    /// An optional count, possibly representing the number of times the cycle
26    /// traverses its edges or some other property related to its structure.
27    pub loop_count: Option<usize>,
28}
29
30impl SignedCycle {
31    pub fn from_cycle<V, E, H, N: NodeStorageOps<NodeData = V>>(
32        cycle: Cycle,
33        according_to: Hedge,
34        graph: &HedgeGraph<E, V, H, N>,
35    ) -> Option<Self> {
36        if !cycle.is_circuit(graph) {
37            return None;
38        }
39
40        if !cycle.filter.includes(&according_to) {
41            return None;
42        }
43
44        let mut filter = graph.empty_subgraph::<SuBitGraph>();
45
46        let mut current_hedge = according_to;
47
48        loop {
49            if filter.includes(&current_hedge) {
50                break;
51            }
52            filter.add(current_hedge);
53
54            current_hedge = graph.inv(
55                graph
56                    .iter_crown(graph.node_id(current_hedge))
57                    .find(|h| cycle.filter.includes(h) && (*h != current_hedge))?,
58            );
59        }
60
61        Some(SignedCycle {
62            filter,
63            loop_count: cycle.loop_count,
64        })
65    }
66}

71/// Represents a cycle in a graph, defined as a set of half-edges.
72///
73/// A cycle is a path in a graph that starts and ends at the same vertex and
74/// does not revisit intermediate vertices. This struct typically stores the
75/// half-edges forming the cycle as a bitmask.
76///
77/// Operations on cycles, such as addition (symmetric difference), are common
78/// in cycle basis theory.
79pub struct Cycle {
80    /// A bitmask representing the set of half-edges that form this cycle.
81    /// The specific half-edges included define the path of the cycle.
82    pub filter: SuBitGraph,
83    /// An optional count, which might represent properties like the number of
84    /// times the cycle traverses its edges (e.g., for fundamental cycles, this
85    /// would often be 1).
86    pub loop_count: Option<usize>,
87}
88impl Cycle {
89    pub fn internal_graph<E, V, H, N: NodeStorageOps<NodeData = V>>(
90        self,
91        graph: &HedgeGraph<E, V, H, N>,
92    ) -> InternalSubGraph {
93        InternalSubGraph::cleaned_filter_pessimist(self.filter, graph)
94    }
95    pub fn new_unchecked(filter: SuBitGraph) -> Self {
96        Self {
97            filter,
98            loop_count: None,
99        }
100    }
101    pub fn is_circuit<E, V, H, N: NodeStorageOps<NodeData = V>>(
102        &self,
103        graph: &HedgeGraph<E, V, H, N>,
104    ) -> bool {
105        for (_, c, _) in graph.iter_nodes_of(&self.filter) {
106            let adgacent = c.filter(|a| self.filter.includes(a));
107            let count = adgacent.count();
108            if count != 2 {
109                return false;
110            }
111        }
112        if graph.count_connected_components(&self.filter) > 1 {
113            return false;
114        }
115        true
116    }
117    pub fn new_circuit<E, V, H, N: NodeStorageOps<NodeData = V>>(
118        filter: SuBitGraph,
119        graph: &HedgeGraph<E, V, H, N>,
120    ) -> Option<Self> {
121        let circuit = Self {
122            filter,
123            loop_count: Some(1),
124        };
125        if circuit.is_circuit(graph) {
126            Some(circuit)
127        } else {
128            None
129        }
130    }
131    pub fn new<E, V, H, N: NodeStorageOps<NodeData = V>>(
132        filter: SuBitGraph,
133        graph: &HedgeGraph<E, V, H, N>,
134    ) -> Option<Self> {
135        for (_, c, _) in graph.iter_nodes_of(&filter) {
136            let adgacent = c.filter(|a| filter.includes(a));
137            if adgacent.count() % 2 == 1 {
138                return None;
139            }
140        }
141
142        Some(Self {
143            filter,
144            loop_count: None,
145        })
146    }
147
148    pub fn all_sum_powerset_filter_map(
149        set: &[Self],
150        filter_map: &impl Fn(Self) -> Option<Self>,
151    ) -> Result<AHashSet<Self>, TryFromIntError> {
152        let mut s = AHashSet::new();
153        let mut pset = PowersetIterator::<usize>::new(set.len().try_into()?);
154
155        pset.next().unwrap(); //Skip the empty set
156
157        for i in pset {
158            let mut ones = i.included_iter();
159
160            let mut union = set[ones.next().unwrap()].clone();
161
162            for o in ones {
163                union = &union + &set[o];
164            }
165
166            if let Some(union) = filter_map(union) {
167                s.insert(union);
168            }
169        }
170
171        Ok(s)
172    }
173}
