18/// An owned trace-unfolded DAG together with the source-graph node each unfolded node came from.
19///
20/// The wrapper keeps the structural graph and the `source_nodes` mapping together across

59    /// Iterates over the leaf operations of `node` together with their dependency frontiers.
60    ///
61    /// The frontier is reconstructed structurally from the unfolded graph:
62    /// follow the unique branch that still contains the target operation, handle disjoint unions
63    /// via their relevant incoming branches, and then strip any independent prefix.
64    pub fn leaf_op_dependency_frontiers<'b, K>(
65        &'b self,
66        node: NodeIndex,
67        indep: &'b impl Independence<HiddenData<K, EdgeIndex>>,
68    ) -> impl Iterator<Item = (&'b HiddenData<K, EdgeIndex>, NodeIndex)> + 'b
69    where
70        K: Eq + Hash + Clone + Ord + 'b,
71        V: AsRef<TraceKey<K, EdgeIndex>>,
72    {
73        self.graph[node].as_ref().iter_leaf_ops().map(move |op| {
74            (
75                op,
76                self.walk_op_dependency_frontier(node, op, indep)
77                    .expect("dependency frontier must exist for leaf op"),
78            )
79        })
80    }

111/// Unfolds a graph of operations into a graph of trace keys.
112///
113/// The trait separates three concerns:
114///
115/// - [`TraceUnfold::key`] maps one traversed edge to its semantic operation label.
116/// - [`Independence`] decides when two labels commute and may live in the same Foata level.
117/// - [`TraceUnfold::realizations_for_op`] optionally expands one traversed edge into several
118///   equivalent operation sequences before canonicalization.
119///
120/// In the simplest case, each edge has exactly one realization and the unfolding merges paths
121/// that differ only by commuting independent operations.
122///
123/// # Examples
124///
125/// A minimal unfold where the two ways of reaching `AB` collapse to the same trace key:
126///
127/// ```
128/// use linnet::half_edge::{
129///     HedgeGraph, NoData, algorithms::trace_unfold::{HiddenData, Independence, TraceUnfold},
130///     builder::HedgeGraphBuilder, involution::EdgeIndex, nodestore::DefaultNodeStore,
131/// };
132///
133/// struct ExampleGraph {
134///     graph: HedgeGraph<String, &'static str, NoData, DefaultNodeStore<&'static str>>,
135/// }
136///
137/// impl Independence<HiddenData<String, EdgeIndex>> for ExampleGraph {
138///     fn independent(
139///         &self,
140///         a: &HiddenData<String, EdgeIndex>,
141///         b: &HiddenData<String, EdgeIndex>,
142///     ) -> bool {
143///         !a.order.chars().any(|c| b.order.contains(c))
144///     }
145/// }
146///
147/// impl TraceUnfold<String> for ExampleGraph {
148///     type EdgeData = String;
149///     type HedgeData = NoData;
150///     type NodeData = &'static str;
151///     type NodeStorage = DefaultNodeStore<&'static str>;
152///
153///     fn graph(
154///         &self,
155///     ) -> &HedgeGraph<Self::EdgeData, Self::NodeData, Self::HedgeData, Self::NodeStorage> {
156///         &self.graph
157///     }
158///
159///     fn key(&self, e: EdgeIndex) -> String {
160///         self.graph[e].clone()
161///     }
162/// }
163///
164/// let mut builder = HedgeGraphBuilder::<String, &'static str>::new();
165/// let empty = builder.add_node("empty");
166/// let a = builder.add_node("A");
167/// let b = builder.add_node("B");
168/// let ab = builder.add_node("AB");
169///
170/// builder.add_edge(empty, a, "A".to_string(), true);
171/// builder.add_edge(empty, b, "B".to_string(), true);
172/// builder.add_edge(a, ab, "B".to_string(), true);
173/// builder.add_edge(b, ab, "A".to_string(), true);
174///
175/// let graph = ExampleGraph {
176///     graph: builder.build::<DefaultNodeStore<&'static str>>(),
177/// };
178/// let unfolded = graph.trace_unfold::<DefaultNodeStore<usize>>(empty);
179/// let labels: Vec<_> = unfolded.iter_nodes().map(|(_, _, key)| key.to_string()).collect();
180///
181/// assert_eq!(unfolded.n_nodes(), 4);
182/// assert!(labels.iter().any(|label| label == "∅"));
183/// assert!(labels.iter().any(|label| label == "{A}"));
184/// assert!(labels.iter().any(|label| label == "{B}"));
185/// assert!(labels.iter().any(|label| label == "{A,B}"));
186/// ```

308    #[allow(clippy::type_complexity)]
309    fn trace_unfold_of<M: NodeStorageOps<NodeData = usize>, S: SubSetLike>(
310        &self,
311        subgraph: &S,
312        start: NodeIndex,
313    ) -> UnfoldedTraceGraph<Self, TraceKey<Key, EdgeIndex>, M::OpStorage<TraceKey<Key, EdgeIndex>>>
314    {
315        // `traces` is the canonical dedup table. Each unfolded node is identified by the pair
316        // `(source_graph_node, canonical_trace_key)`, so paths that differ only by commuting
317        // independent operations collapse to the same unfolded node.
318        let root = (start, TraceKey::empty());
319        let mut q = VecDeque::new();
320        let mut traces: IndexSet<(NodeIndex, TraceKey<Key, EdgeIndex>)> = IndexSet::new();
321        let mut builder: HedgeGraphBuilder<EdgeIndex, usize, NoData> = HedgeGraphBuilder::new();
322        let mut join_branches: BTreeMap<NodeIndex, BTreeMap<Key, Vec<(NodeIndex, EdgeIndex)>>> =
323            BTreeMap::new();
324        let mut added_edges = BTreeSet::new();
325        let (ind, _) = traces.insert_full(root);
326        let nid = builder.add_node(ind);
327        q.push_back((nid, start, TraceKey::empty()));
328        let g = self.graph();
329
330        while let Some((bnid, nid, key)) = q.pop_front() {
331            for hedge in g.iter_crown_in(subgraph, nid) {
332                if g.flow(hedge) == Flow::Source {
333                    if let Some(to_node) = g.involved_node_id(hedge) {
334                        let edge = g[&hedge];
335                        let edge_key = self.key(edge);
336                        let op = HiddenData {
337                            order: edge_key.clone(),
338                            data: edge,
339                        };
340                        let realizations = self.realizations_for_op(op);
341                        if realizations.is_empty() {
342                            continue;
343                        }
344
345                        if let Some(factors) = self.join_factors(to_node) {
346                            if factors.contains(&edge_key) {
347                                let branch = (bnid, edge);
348                                let branches_by_factor = join_branches.entry(to_node).or_default();
349                                let branches =
350                                    branches_by_factor.entry(edge_key.clone()).or_default();
351                                let is_new_branch = !branches.contains(&branch);
352                                if is_new_branch {
353                                    branches.push(branch);
354
355                                    let mut joins = Vec::new();
356                                    collect_factor_branch_joins(
357                                        &factors,
358                                        branches_by_factor,
359                                        &edge_key,
360                                        branch,
361                                        &mut joins,
362                                    );
363
364                                    for join in joins {
365                                        let mut branch_edges = BTreeSet::new();
366                                        if !join.iter().all(|(_, edge)| branch_edges.insert(*edge))
367                                        {
368                                            continue;
369                                        }
370
371                                        let keys = join.iter().map(|(parent, _)| {
372                                            &traces.get_index(parent.0).unwrap().1
373                                        });
374                                        let Some(join_key) = TraceKey::try_foata_join(keys, self)
375                                        else {
376                                            continue;
377                                        };
378
379                                        let (join_ind, is_new) =
380                                            traces.insert_full((to_node, join_key.clone()));
381                                        if is_new {
382                                            let join_node = builder.add_node(join_ind);
383                                            debug_assert_eq!(join_node.0, join_ind);
384                                            q.push_back((join_node, to_node, join_key));
385                                        }
386
387                                        for (parent, edge) in join {
388                                            add_unfolded_edge(
389                                                &mut builder,
390                                                &mut added_edges,
391                                                parent,
392                                                NodeIndex(join_ind),
393                                                edge,
394                                            );
395                                        }
396                                    }
397                                }
398                            }
399                        }
400
401                        for realization in realizations {
402                            let new_key = realization
403                                .into_iter()
404                                .fold(key.clone(), |acc, factor| acc.push(self, factor));
405                            let (ind, is_new) = traces.insert_full((to_node, new_key.clone()));
406                            if is_new {
407                                let bbnid = builder.add_node(ind);
408                                debug_assert_eq!(bbnid.0, ind);
409                                q.push_back((bbnid, to_node, new_key.clone()))
410                            }
411
412                            add_unfolded_edge(
413                                &mut builder,
414                                &mut added_edges,
415                                bnid,
416                                NodeIndex(ind),
417                                edge,
418                            );
419                        }
420                    }
421                }
422            }
423        }
424
425        let source_nodes = traces.iter().map(|(source_node, _)| *source_node).collect();
426        let unfolded = UnfoldedTraceGraph::new(
427            builder.build::<M>().map(
428                |_, _, v| {
429                    let mut trace: TraceKey<Key, EdgeIndex> = TraceKey::empty();
430                    let v = &mut traces.get_index_mut2(v).unwrap().1;
431                    std::mem::swap(v, &mut trace);
432                    trace
433                },
434                |_, _, _, _, a| a,
435                |_, h| h,
436            ),
437            source_nodes,
438        );
439        unfolded
440            .validate_invariant()
441            .expect("trace-unfolded graph violates non-union unique-parent invariant");
442        unfolded
443    }

467fn collect_factor_branch_joins<Key: Ord>(
468    factors: &BTreeSet<Key>,
469    branches_by_factor: &BTreeMap<Key, Vec<(NodeIndex, EdgeIndex)>>,
470    required_factor: &Key,
471    required_branch: (NodeIndex, EdgeIndex),
472    out: &mut Vec<Vec<(NodeIndex, EdgeIndex)>>,
473) {
474    let factors = factors.iter().collect::<Vec<_>>();
475    let mut selected = Vec::with_capacity(factors.len());
476    collect_factor_branch_joins_from(
477        &factors,
478        0,
479        branches_by_factor,
480        required_factor,
481        required_branch,
482        &mut selected,
483        out,
484    );
485}
486
487fn collect_factor_branch_joins_from<Key: Ord>(
488    factors: &[&Key],
489    index: usize,
490    branches_by_factor: &BTreeMap<Key, Vec<(NodeIndex, EdgeIndex)>>,
491    required_factor: &Key,
492    required_branch: (NodeIndex, EdgeIndex),
493    selected: &mut Vec<(NodeIndex, EdgeIndex)>,
494    out: &mut Vec<Vec<(NodeIndex, EdgeIndex)>>,
495) {
496    let Some(factor) = factors.get(index) else {
497        out.push(selected.clone());
498        return;
499    };
500    if *factor == required_factor {
501        selected.push(required_branch);
502        collect_factor_branch_joins_from(
503            factors,
504            index + 1,
505            branches_by_factor,
506            required_factor,
507            required_branch,
508            selected,
509            out,
510        );
511        selected.pop();
512        return;
513    }
514
515    let Some(branches) = branches_by_factor.get(*factor) else {
516        return;
517    };
518
519    for branch in branches {
520        selected.push(*branch);
521        collect_factor_branch_joins_from(
522            factors,
523            index + 1,
524            branches_by_factor,
525            required_factor,
526            required_branch,
527            selected,
528            out,
529        );
530        selected.pop();
531    }

538        self.graph
539            .iter_crown(node)
540            .filter(|hedge| self.graph.flow(*hedge) == Flow::Sink)
541            .filter_map(|hedge| {
542                let parent = self.graph.involved_node_id(hedge)?;

563    where

565        V: AsRef<TraceKey<K, EdgeIndex>>,

570    fn introduced_ops<K>(
571        &self,
572        parent: NodeIndex,
573        child: NodeIndex,
574    ) -> Vec<HiddenData<K, EdgeIndex>>
575    where
576        K: Eq + Hash + Clone + Ord,
577        V: AsRef<TraceKey<K, EdgeIndex>>,
578    {
579        let mut parent_ops: Vec<_> = self.all_ops(parent).cloned().collect();
580        let mut introduced = Vec::new();
581        for op in self.all_ops(child) {
582            // `child` is the canonicalized extension of `parent`, so the multiset difference
583            // between their operations is exactly the set of newly introduced operations on that
584            // step.
585            if let Some(index) = parent_ops.iter().position(|parent_op| parent_op == op) {
586                parent_ops.swap_remove(index);
587            } else {
588                introduced.push(op.clone());
589            }
590        }
591        introduced
592    }

594    pub fn closest_common_ancestor(&self, nodes: &[NodeIndex]) -> Option<NodeIndex> {
595        let first = *nodes.first()?;
596        // `path_to_root` follows the unique-parent spine to the empty trace, so intersecting
597        // those spines is enough to recover the closest shared prefix of the given nodes.
598        std::iter::once(first)
599            .chain(self.path_to_root(first).map(|(parent, _)| parent))
600            .find(|candidate| {
601                nodes.iter().skip(1).all(|node| {
602                    std::iter::once(*node)
603                        .chain(self.path_to_root(*node).map(|(parent, _)| parent))
604                        .any(|ancestor| ancestor == *candidate)
605                })
606            })
607    }

609    fn strip_independent_prefix<K>(
610        &self,
611        start: NodeIndex,
612        target: &HiddenData<K, EdgeIndex>,
613        indep: &impl Independence<HiddenData<K, EdgeIndex>>,
614    ) -> NodeIndex
615    where
616        K: Eq + Hash + Clone + Ord,
617        V: AsRef<TraceKey<K, EdgeIndex>>,
618    {
619        let mut current = start;
620
621        loop {
622            if self.is_disjoint_union(current) {
623                let leaf_ops: Vec<_> = self.leaf_ops(current).cloned().collect();
624                // If every operation introduced by this union is independent of `target`, the
625                // relevant reusable prefix is shared by all incoming branches, so we can replace
626                // the union by their closest common ancestor.
627                if leaf_ops.iter().all(|op| indep.independent(op, target)) {
628                    let parents = self
629                        .parents(current)
630                        .map(|(parent, _)| parent)
631                        .collect::<Vec<_>>();
632                    if let Some(prefix) = self.closest_common_ancestor(&parents) {
633                        current = prefix;
634                        continue;
635                    }
636                }
637                return current;
638            }
639
640            let Some((parent, _)) = self.unique_parent(current) else {
641                return current;
642            };
643            let introduced = self.introduced_ops(parent, current);
644            // As long as the whole step from `parent -> current` is independent of `target`, that
645            // step can be factored out of the reusable prefix and we continue with `parent`.
646            if introduced.iter().all(|op| indep.independent(op, target)) {
647                current = parent;
648            } else {
649                return current;
650            }
651        }
652    }
653
654    pub fn leaf_ops<'b, K>(
655        &'b self,
656        node: NodeIndex,
657    ) -> impl DoubleEndedIterator<Item = &'b HiddenData<K, EdgeIndex>> + 'b
658    where
659        K: Eq + Hash + Clone + Ord + 'b,
660        V: AsRef<TraceKey<K, EdgeIndex>>,
661    {
662        self.graph[node].as_ref().iter_leaf_ops()
663    }
664
665    /// Returns whether `node` represents a disjoint-union step in the unfolded graph.
666    pub fn is_disjoint_union(&self, node: NodeIndex) -> bool {
667        self.parents(node).nth(1).is_some()
668    }
669
670    /// Returns the unique parent of `node` when the node is not a disjoint union.
671    ///
672    /// The accompanying [`EdgeIndex`] is the original source-graph edge carried by that branch.
673    pub fn unique_parent(&self, node: NodeIndex) -> Option<(NodeIndex, EdgeIndex)> {
674        let mut parents = self.parents(node);
675        let parent = parents.next()?;
676        (parents.next().is_none()).then_some(parent)
677    }
678
679    /// Walks the unique-parent chain from `node` to the root.
680    ///
681    /// The iterator stops before a disjoint union, because such a node no longer has a single
682    /// canonical parent chain.
683    pub fn path_to_root(
684        &self,
685        node: NodeIndex,
686    ) -> impl Iterator<Item = (NodeIndex, EdgeIndex)> + '_ {
687        let mut current = Some(node);
688        std::iter::from_fn(move || {
689            let node = current?;
690            let parent = self.unique_parent(node)?;
691            current = Some(parent.0);
692            Some(parent)
693        })
694    }
695}

697impl<G, V, N: NodeStorageOps<NodeData = V>> UnfoldedTraceGraph<G, V, N> {
698    /// Returns the closest reusable prefix node for the contribution of `target` at `node`.
699    ///
700    /// The traversal is specific to trace-unfolded graphs:
701    ///
702    /// - it follows the unique branch that still contains `target`
703    /// - across a disjoint union, it continues with the branch containing `target`
704    /// - once `target` disappears, it strips off any prefix whose newly introduced operations are
705    ///   all independent of `target`
706    ///
707    /// This private walk is the structural core behind
708    /// [`leaf_op_dependency_frontiers`](Self::leaf_op_dependency_frontiers).
709    fn walk_op_dependency_frontier<K>(
710        &self,
711        node: NodeIndex,
712        target: &HiddenData<K, EdgeIndex>,
713        indep: &impl Independence<HiddenData<K, EdgeIndex>>,
714    ) -> Option<NodeIndex>
715    where
716        K: Eq + Hash + Clone + Ord,
717        V: AsRef<TraceKey<K, EdgeIndex>>,
718    {
719        let mut current = node;
720        loop {
721            if self.is_disjoint_union(current) {
722                let parents = self.parents(current).collect::<Vec<_>>();
723                if let Some((parent, _)) = parents
724                    .iter()
725                    .copied()
726                    .find(|(parent, _)| self.node_contains_op(*parent, target))
727                {
728                    // The target operation is still present on exactly one incoming branch, so
729                    // that branch remains the relevant prefix chain.
730                    current = parent;
731                    continue;
732                }
733                // The target no longer appears on any incoming branch, so the reusable prefix is
734                // the closest common ancestor of all incoming branches rather than any particular
735                // parent.
736                return self.closest_common_ancestor(
737                    &parents
738                        .iter()
739                        .map(|(parent, _)| *parent)
740                        .collect::<Vec<_>>(),
741                );
742            }
743
744            let Some((parent, _)) = self.unique_parent(current) else {
745                return Some(current);
746            };
747            if self.node_contains_op(parent, target) {
748                // We have not yet reached the step at which `target` was introduced.
749                current = parent;
750                continue;
751            }
752
753            return Some(self.strip_independent_prefix(parent, target, indep));
754        }
755    }
756
757    /// Checks the structural invariant used by the traversal helpers above.
758    ///
759    /// Distinct incoming branches of the same disjoint union must carry distinct source-graph
760    /// edges; otherwise helpers such as [`unique_parent`](Self::unique_parent) and
761    /// [`leaf_op_dependency_frontiers`](Self::leaf_op_dependency_frontiers) become ambiguous.
762    pub fn validate_invariant<K>(&self) -> Result<(), String>
763    where
764        K: Eq + Hash + Clone + Ord,
765        V: AsRef<TraceKey<K, EdgeIndex>>,
766    {
767        for (node, _, _) in self.graph.iter_nodes() {
768            let incoming = self.parents(node).collect::<Vec<_>>();
769            for (i, (_, edge)) in incoming.iter().enumerate() {
770                if incoming
771                    .iter()
772                    .skip(i + 1)
773                    .any(|(_, other_edge)| other_edge == edge)
774                {
775                    return Err(format!(
776                        "node {node} has duplicate incoming branches for edge {edge}"
777                    ));
778                }
779            }
780        }
781
782        Ok(())
783    }
784}

1011/// A canonical Foata-style trace normal form.
1012///
1013/// `TraceKey` stores a trace as a list of levels, where each inner vector contains operations
1014/// that are pairwise independent and therefore commute. The levels are ordered, so operations in
1015/// later levels could not be moved further left without breaking the independence relation used
1016/// when the key was constructed.
1017///
1018/// `TraceKey` is the owning form. Use [`TraceKey::view`] when you only need borrowed access to
1019/// the levels, and [`TraceKey::from_levels`] when you already have an explicit canonical level
1020/// structure and want to build an owned key directly.
1021///
1022/// The intended split is:
1023///
1024/// - [`TraceKey`] owns a canonical trace.
1025/// - [`TraceKeyView`] borrows the same trace without exposing the private storage.
1026///
1027/// # Examples
1028///
1029/// ```
1030/// use linnet::half_edge::algorithms::trace_unfold::{HiddenData, Independence, TraceKey};
1031///
1032/// struct CommutesOnlyAandC;
1033///
1034/// impl Independence<HiddenData<&'static str, ()>> for CommutesOnlyAandC {
1035///     fn independent(
1036///         &self,
1037///         a: &HiddenData<&'static str, ()>,
1038///         b: &HiddenData<&'static str, ()>,
1039///     ) -> bool {
1040///         matches!(
1041///             (a.order, b.order),
1042///             ("A", "C") | ("C", "A")
1043///         )
1044///     }
1045/// }
1046///
1047/// let trace = TraceKey::empty()
1048///     .push(&CommutesOnlyAandC, HiddenData { order: "A", data: () })
1049///     .push(&CommutesOnlyAandC, HiddenData { order: "B", data: () })
1050///     .push(&CommutesOnlyAandC, HiddenData { order: "C", data: () });
1051///
1052/// assert_eq!(trace.to_string(), "{A,C} · {B}");
1053/// assert_eq!(trace.view().to_owned(), trace);
1054/// ```
1055#[derive(Clone, Eq, PartialEq, Hash, Debug)]
1056pub struct TraceKey<O, D> {
1057    levels: Vec<Vec<HiddenData<O, D>>>,
1058}

1279    where
1280        I: Independence<HiddenData<O, D>>,
1281        O: Clone + Eq + Ord,
1282        D: Clone,
1283    {
1284        let mut levels = self.levels.clone();
1285
1286        for i in (0..levels.len()).rev() {
1287            let ok = levels[i].iter().all(|b| indep.independent(&op, b));
1288            if ok {
1289                levels[i].push(op);
1290                levels[i].sort(); // uses O only
1291                return Self { levels };
1292            }
1293        }
1294
1295        levels.push(vec![op]);
1296        Self { levels }
1297    }

1299    /// Right-aligns independent traces into their common Foata normal form.
1300    pub fn try_foata_join<'a, I>(
1301        keys: impl IntoIterator<Item = &'a Self>,
1302        indep: &I,
1303    ) -> Option<Self>
1304    where
1305        I: Independence<HiddenData<O, D>>,
1306        O: Clone + Eq + Ord + 'a,
1307        D: Clone + 'a,
1308    {
1309        let keys = keys.into_iter().collect::<Vec<_>>();
1310        let max_levels = keys.iter().map(|key| key.levels.len()).max().unwrap_or(0);
1311        let mut seen = Vec::new();
1312        let mut levels = Vec::new();
1313
1314        for level_index in 0..max_levels {
1315            let mut level_ops = Vec::new();
1316
1317            for key in &keys {
1318                let offset = max_levels - key.levels.len();
1319                if level_index < offset {
1320                    continue;
1321                }
1322                let level = &key.levels[level_index - offset];
1323
1324                for op in level {
1325                    if seen.iter().any(|seen_op| seen_op == op)
1326                        || level_ops.iter().any(|level_op| level_op == op)
1327                    {
1328                        continue;
1329                    }
1330
1331                    if !level_ops
1332                        .iter()
1333                        .all(|level_op| indep.independent(op, level_op))
1334                    {
1335                        return None;
1336                    }
1337
1338                    level_ops.push(op.clone());
1339                }
1340            }
1341
1342            level_ops.sort();
1343            seen.extend(level_ops.iter().cloned());
1344            if !level_ops.is_empty() {
1345                levels.push(level_ops);
1346            }
1347        }
1348
1349        Some(Self::from_levels(levels))
1350    }
