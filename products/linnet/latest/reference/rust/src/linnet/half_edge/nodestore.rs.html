15/// Defines a comprehensive set of operations for a node storage backend within a [`HedgeGraph`].
16///
17/// This trait extends [`NodeStorage`] by adding methods for modifying the storage (e.g.,
18/// extending, extracting, deleting, identifying nodes), building it, iterating over nodes,
19/// and mapping node data. It abstracts over the specific data structures used to store
20/// node information and their incident half-edges.
21pub trait NodeStorageOps: NodeStorage + Swap<Hedge> + Swap<NodeIndex> {
22    /// The type of storage returned after mapping operations that might change the node data type.
23    type OpStorage<N>: NodeStorageOps<NodeData = N>;
24    /// The base type of subgraph used by this node storage (e.g., a `BitVec` to represent a set of hedges).
25    type Base: BaseSubgraph;
26    // where
27    // Self: 'a;
28    /// Combines two stores using the implementation's index-space semantics.
29    ///
30    /// Callers must not assume embedded index-valued data is remapped unless the
31    /// concrete storage implementation documents that behavior.
32    fn extend(self, other: Self) -> Self;
33
34    /// Maps each node's ID, neighbors, and borrowed data into a dense [`NodeVec`].
35    fn new_nodevec<'a, V2>(
36        &'a self,
37        node_map: impl FnMut(NodeIndex, Self::NeighborsIter<'a>, &'a Self::NodeData) -> V2,
38    ) -> NodeVec<V2>;
39
40    /// Appends a store using the implementation's index-space semantics.
41    ///
42    /// Callers must not assume embedded index-valued data is remapped unless the
43    /// concrete storage implementation documents that behavior.
44    fn extend_mut(&mut self, other: Self);
45    /// Moves the selected half-edges into a new store and removes them from `self`.
46    ///
47    /// `split_node` maps data for nodes crossed by the selection boundary, while
48    /// `owned_node` maps data for nodes wholly inside the selection. Implementations may
49    /// compact both stores and thereby change hedge and node indices.
50    fn extract<S: SubSetLike<Base = Self::Base>, V2>(
51        &mut self,
52        subgraph: &S,
53        split_node: impl FnMut(&Self::NodeData) -> V2,
54        owned_node: impl FnMut(Self::NodeData) -> V2,
55    ) -> Self::OpStorage<V2>;
56
57    /// Moves complete nodes and their incident half-edges into a separate store.
58    ///
59    /// The returned subset records the selected half-edges.
60    fn extract_nodes(&mut self, nodes: impl IntoIterator<Item = NodeIndex>) -> (Self::Base, Self);
61
62    /// Removes the selected half-edges and compacts the remaining node storage.
63    fn delete<S: SubSetLike<Base = Self::Base>>(&mut self, subgraph: &S);
64
65    // fn add_node(&mut self, node_data: Self::NodeData) -> NodeIndex;
66
67    /// Identifies nodes, essentially turning them into a single node
68    /// Invalidates all previous NodeIndex values if using VecNodeStore.
69    /// Does not invalidate if using Forest as NodeStore
70    fn identify_nodes(&mut self, nodes: &[NodeIndex], node_data_merge: Self::NodeData)
71        -> NodeIndex;
72
73    /// Drops inactive nodes retained by identification operations and returns their data.
74    fn forget_identification_history(&mut self) -> NodeVec<Self::NodeData>;
75
76    /// Copies the topology into a parent-pointer forest while mapping borrowed node data.
77    fn to_forest<U, H>(
78        &self,
79        map_data: impl Fn(&Self::NodeData) -> U,
80    ) -> Forest<U, ParentPointerStore<H>>;
81
82    /// Builds a store from node descriptions that partition `n_hedges` half-edges.
83    fn build<I: IntoIterator<Item = HedgeNodeBuilder<Self::NodeData>>>(
84        nodes: I,
85        n_hedges: usize,
86    ) -> Self;
87
88    /// Builds a store from a half-edge partition while converting each node's data.
89    fn build_with_mapping<I: IntoIterator<Item = HedgeNodeBuilder<ND>>, ND>(
90        nodes: I,
91        n_hedges: usize,
92        map_data: impl FnMut(ND) -> Self::NodeData,
93    ) -> Self;
94
95    /// Appends an unpaired half-edge incident to `source`.
96    fn add_dangling_edge(self, source: NodeIndex) -> Result<Self, HedgeGraphError>;
97
98    /// Builds nodes from source and sink neighbor sets, assigning default data to each node.
99    fn random(sources: &[Self::Neighbors], sinks: &[Self::Neighbors]) -> Self
100    where
101        Self::NodeData: Default;
102
103    /// Consumes the store and yields each current node ID with its owned data.
104    fn drain(self) -> impl Iterator<Item = (NodeIndex, Self::NodeData)>;
105    /// Iterates over current node IDs and borrowed node data.
106    fn iter(&self) -> impl Iterator<Item = (NodeIndex, &Self::NodeData)>;
107
108    /// Ensures backend-maintained node mappings are ready and consistent.
109    fn check_and_set_nodes(&mut self) -> Result<(), HedgeGraphError>;
110
111    /// Checks that every half-edge belongs to exactly one consistent node.
112    fn check_nodes(&self) -> Result<(), HedgeGraphError>;
113
114    /// Maps borrowed node data while retaining topology and exposing the complete graph.
115    fn map_data_ref_graph<'a, E, V2, H>(
116        &'a self,
117        graph: &'a HedgeGraph<E, Self::NodeData, H, Self>,
118        node_map: impl FnMut(
119            &'a HedgeGraph<E, Self::NodeData, H, Self>,
120            Self::NeighborsIter<'a>,
121            &'a Self::NodeData,
122        ) -> V2,
123    ) -> Self::OpStorage<V2>;
124
125    /// Maps node data through mutable references while retaining topology.
126    fn map_data_ref_mut_graph<'a, V2>(
127        &'a mut self,
128        node_map: impl FnMut(Self::NeighborsIter<'a>, &'a mut Self::NodeData) -> V2,
129    ) -> Self::OpStorage<V2>;
130
131    /// Fallible counterpart of [`NodeStorageOps::map_data_ref_graph`].
132    fn map_data_ref_graph_result<'a, E, V2, H, Er>(
133        &'a self,
134        graph: &'a HedgeGraph<E, Self::NodeData, H, Self>,
135        node_map: impl FnMut(
136            &'a HedgeGraph<E, Self::NodeData, H, Self>,
137            Self::NeighborsIter<'a>,
138            &'a Self::NodeData,
139        ) -> Result<V2, Er>,
140    ) -> Result<Self::OpStorage<V2>, Er>;
141
142    /// Consumes the store and maps owned node data with access to the edge involution and node ID.
143    fn map_data_graph<'a, V2>(
144        self,
145        involution: &'a Involution<EdgeIndex>,
146        f: impl FnMut(&'a Involution<EdgeIndex>, NodeIndex, Self::NodeData) -> V2,
147    ) -> Self::OpStorage<V2>;
148
149    /// Fallible counterpart of [`NodeStorageOps::map_data_graph`].
150    fn map_data_graph_result<'a, V2, Err>(
151        self,
152        involution: &'a Involution<EdgeIndex>,
153        f: impl FnMut(&'a Involution<EdgeIndex>, NodeIndex, Self::NodeData) -> Result<V2, Err>,
154    ) -> Result<Self::OpStorage<V2>, Err>;
155
156    /// Iterates over the store's dense node-index range.
157    fn iter_node_id(&self) -> impl Iterator<Item = NodeIndex> {
158        (0..<Self as Swap<NodeIndex>>::len(self).0).map(NodeIndex)
159    }
160    /// Iterates over each node's ID, neighbors, and borrowed data.
161    fn iter_nodes(
162        &self,
163    ) -> impl Iterator<Item = (NodeIndex, Self::NeighborsIter<'_>, &Self::NodeData)>;
164    /// Iterates over each node's ID, neighbors, and mutably borrowed data.
165    fn iter_nodes_mut(
166        &mut self,
167    ) -> impl Iterator<Item = (NodeIndex, Self::NeighborsIter<'_>, &mut Self::NodeData)>;
168    /// Retrieves the [`NodeIndex`] that the given [`Hedge`] is incident to (or originates from).
169    fn node_id_ref(&self, hedge: Hedge) -> NodeIndex;
170    /// Returns an iterator over the half-edges incident to the specified `node_id`.
171    fn get_neighbor_iterator(&self, node_id: NodeIndex) -> Self::NeighborsIter<'_>;
172    /// Gets a reference to the data associated with `node_id`.
173    fn get_node_data(&self, node_id: NodeIndex) -> &Self::NodeData;
174    /// Gets a mutable reference to the data associated with `node_id`.
175    fn get_node_data_mut(&mut self, node_id: NodeIndex) -> &mut Self::NodeData;
176}

178/// The fundamental trait defining the interface for a node storage backend in a [`HedgeGraph`].
179///
180/// This trait abstracts the way nodes and their incident half-edges are stored. Different
181/// implementations can offer various performance trade-offs for accessing and
182/// manipulating node data and topology.
183pub trait NodeStorage: Sized {
184    /// The type of data stored for each node in the graph.
185    type NodeData;
186    /// A generic way to refer to this storage type when it holds node data of type `N`.
187    /// This is often used in `NodeStorageOps` for methods that transform node data.
188    type Storage<N>; // TODO: This might need better definition or usage clarification.
189
190    /// The type used to represent the collection of half-edges incident to a single node.
191    /// This must implement the [`SubSetLike`] trait.
192    type Neighbors: SubSetLike;
193    /// An iterator that yields the [`Hedge`] identifiers incident to a node.
194    /// It must be cloneable and provide an exact size hint.
195    type NeighborsIter<'a>: ExactSizeIterator<Item = Hedge> + Clone
196    where
197        Self: 'a;
198}
