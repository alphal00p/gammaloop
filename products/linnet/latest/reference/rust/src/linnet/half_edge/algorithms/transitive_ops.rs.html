1//! # Transitive Operations on Half-Edge Graphs
2//!
3//! This module provides algorithms for computing transitive closure and transitive reduction
4//! on directed acyclic graphs represented using the half-edge data structure.
5//!
6//! ## Half-Edge Graph Fundamentals
7//!
8//! In the half-edge representation, each directed edge is composed of two **half-edges**:
9//! - A **source half-edge** (outgoing from the source node)
10//! - A **sink half-edge** (incoming to the target node)
11//!
12//! These half-edges are paired together and can be accessed via the involution `self.inv(hedge)`.
13//!
14//! ## Critical Implementation Details
15//!
16//! When modifying edges in a half-edge graph:
17//! - **Adding edges**: Use `add_pair()` to create both half-edges simultaneously
18//! - **Removing edges**: Both half-edges must be deleted to avoid corruption
19//!
20//! ## Algorithms Provided
21//!
22//! - [`transitive_closure`](HedgeGraph::transitive_closure): Adds all transitive edges
23//! - [`transitive_reduction`](HedgeGraph::transitive_reduction): Removes redundant edges
24//!
25//! Both algorithms verify the graph is acyclic before proceeding, as transitive operations
26//! on graphs with cycles are undefined.
27

40/// Error types for transitive operations on half-edge graphs.
41///
42/// These operations require directed acyclic graphs (DAGs) to be meaningful,
43/// as transitive operations on graphs with cycles are undefined.
44#[derive(Debug, thiserror::Error)]
45pub enum TransitiveError {
46    /// The graph contains cycles, making transitive operations undefined.
47    ///
48    /// This error is returned when the topological sort fails, indicating
49    /// the presence of cycles in the directed graph.
50    #[error("Graph contains cycles: {0}")]
51    HasCycles(#[from] TopoError),
52
53    /// An error occurred in the underlying graph structure operations.
54    ///
55    /// This can happen during edge addition or removal operations.
56    #[error("Graph structure error")]
57    GraphStructure,
58}

60impl<V, S: NodeStorageOps<NodeData = V, Base = SuBitGraph>> HedgeGraph<NoData, V, NoData, S> {
61    pub fn poset<I: IntoIterator<Item = V>>(nodes: I) -> Self
62    where
63        V: PartialOrd,
64    {
65        let mut builder = HedgeGraphBuilder::new();
66        let mut nodes: Vec<Option<V>> = nodes.into_iter().map(Some).collect();
67
68        // Create nodes in the graph
69        let mut node_indices = HashMap::new();
70        for (i, _) in nodes.iter().enumerate() {
71            let idx = builder.add_node(i);
72            node_indices.insert(i, idx);
73        }
74
75        // Add ALL order relations first (complete partial order graph)
76        for (i, a) in nodes.iter().enumerate() {
77            for (j, b) in nodes.iter().enumerate() {
78                if i != j && a < b {
79                    builder.add_edge(node_indices[&i], node_indices[&j], NoData {}, true);
80                }
81            }
82        }
83        builder.build_with_map::<V, S>(|i| nodes[i].take().unwrap())
84    }
85    pub fn hasse_diagram<I: IntoIterator<Item = V>>(nodes: I) -> Self
86    where
87        V: PartialOrd + Clone,
88    {
89        use crate::half_edge::builder::HedgeGraphBuilder;
90        use crate::half_edge::involution::Orientation;
91
92        let mut builder = HedgeGraphBuilder::new();
93        let nodes: Vec<V> = nodes.into_iter().collect();
94
95        // Create nodes in the graph
96        let mut node_indices = HashMap::new();
97        for (i, node) in nodes.iter().enumerate() {
98            let idx = builder.add_node(node.clone());
99            node_indices.insert(i, idx);
100        }
101
102        // Add ALL order relations first (complete partial order graph)
103        for (i, a) in nodes.iter().enumerate() {
104            for (j, b) in nodes.iter().enumerate() {
105                if i != j && a < b {
106                    builder.add_edge(
107                        node_indices[&i],
108                        node_indices[&j],
109                        NoData {},
110                        Orientation::Default,
111                    );
112                }
113            }
114        }
115
116        // Build the complete graph and apply transitive reduction
117        let complete_graph = builder.build();
118
119        // Apply transitive reduction to get the Hasse diagram
120        complete_graph
121            .transitive_reduction()
122            .expect("Partial order should be acyclic")
123    }
124}

126impl<E, V, H, N: NodeStorageOps<NodeData = V, Base = crate::half_edge::subgraph::SuBitGraph>>

129    /// Computes the transitive closure of the directed acyclic graph.
130    ///
131    /// The transitive closure adds an edge (u, v) for every pair of nodes u, v
132    /// where there exists a directed path from u to v in the original graph.
133    ///
134    /// ## Algorithm Details
135    ///
136    /// This implementation uses a Floyd-Warshall-style algorithm optimized for DAGs:
137    /// 1. Verifies the graph is acyclic using topological sorting
138    /// 2. Builds a reachability matrix using the topological order
139    /// 3. For each missing transitive edge, adds it using `add_pair()`
140    ///
141    /// ## Half-Edge Considerations
142    ///
143    /// In HedgeGraph, each directed edge consists of two half-edges (a source and sink pair).
144    /// When adding new transitive edges, the algorithm:
145    /// - Uses `add_pair()` to create both half-edges simultaneously
146    /// - Assigns default edge data to new transitive edges
147    /// - Maintains proper half-edge pair relationships
148    ///
149    /// # Arguments
150    ///
151    /// * `self` - Takes ownership of the graph to avoid cloning
152    ///
153    /// # Returns
154    ///
155    /// The modified graph with all transitive edges added
156    ///
157    /// # Errors
158    ///
159    /// Returns `TransitiveError::HasCycles` if the graph contains cycles
160    ///
161    /// # Examples
162    ///
163    /// ```rust,ignore
164    /// // Graph: A -> B -> C
165    /// let closure = graph.transitive_closure()?;
166    /// // Result: A -> B -> C, A -> C (transitive edge added)
167    /// ```
168    pub fn transitive_closure(self) -> Result<Self, TransitiveError>
169    where
170        E: Default,
171        H: Default,
172    {
173        // First verify this is a DAG by getting topological order
174        let topo_order = self.topo_sort_kahn()?;
175
176        // Build reachability matrix using Floyd-Warshall algorithm
177        let nodes: Vec<NodeIndex> = topo_order;
178        let n = nodes.len();
179        let mut node_to_idx = HashMap::new();
180        let mut idx_to_node = Vec::new();
181
182        // Map node indices to array indices for efficiency
183        for (i, &node) in nodes.iter().enumerate() {
184            node_to_idx.insert(node, i);
185            idx_to_node.push(node);
186        }
187
188        // Initialize reachability matrix
189        let mut reachable = vec![vec![false; n]; n];
190
191        // Set diagonal to true (each node reaches itself)
192        #[allow(clippy::needless_range_loop)]
193        for i in 0..n {
194            reachable[i][i] = true;
195        }
196
197        // Initialize direct edges
198        for (i, &from_node) in nodes.iter().enumerate() {
199            for hedge in self.iter_crown(from_node) {
200                if self.flow(hedge) == Flow::Source {
201                    if let Some(to_node) = self.involved_node_id(hedge) {
202                        if let Some(&j) = node_to_idx.get(&to_node) {
203                            reachable[i][j] = true;
204                        }
205                    }
206                }
207            }
208        }
209
210        // Floyd-Warshall to compute transitive closure
211        for k in 0..n {
212            for i in 0..n {
213                for j in 0..n {
214                    reachable[i][j] = reachable[i][j] || (reachable[i][k] && reachable[k][j]);
215                }
216            }
217        }
218
219        // Add new transitive edges
220        let mut graph = self;
221        for i in 0..n {
222            for j in 0..n {
223                if i != j && reachable[i][j] {
224                    let from_node = idx_to_node[i];
225                    let to_node = idx_to_node[j];
226
227                    // Check if edge already exists
228                    // Check if edge already exists
229                    let edge_exists = graph.iter_crown(from_node).any(|hedge| {
230                        graph.flow(hedge) == crate::half_edge::involution::Flow::Source
231                            && graph.involved_node_id(hedge) == Some(to_node)
232                    });
233
234                    if !edge_exists {
235                        // Add edge using add_pair
236                        let (_, _, new_graph) = graph
237                            .add_pair(
238                                from_node,
239                                to_node,
240                                E::default(),
241                                crate::half_edge::involution::Orientation::Default,
242                            )
243                            .map_err(|_| TransitiveError::GraphStructure)?;
244                        graph = new_graph;
245                    }
246                }
247            }
248        }
249
250        Ok(graph)
251    }

253    /// Computes the transitive reduction of the directed acyclic graph.
254    ///
255    /// The transitive reduction removes all edges (u, v) such that there exists
256    /// an alternative path from u to v that doesn't use the direct edge.
257    /// This produces the minimal graph that preserves the same reachability.
258    ///
259    /// ## Algorithm Details
260    ///
261    /// This implementation:
262    /// 1. Verifies the graph is acyclic using topological sorting
263    /// 2. For each edge (u, v), checks if an alternative path u ⇝ v exists
264    /// 3. Collects all redundant edges for removal
265    /// 4. Uses `delete_hedges()` to remove redundant edges
266    ///
267    /// ## Half-Edge Considerations
268    ///
269    /// **Critical Implementation Detail**: In HedgeGraph, each directed edge consists of
270    /// two half-edges (a paired source and sink). To completely remove an edge, both
271    /// half-edges must be deleted:
272    ///
273    /// - `hedge`: The source half-edge (outgoing from source node)
274    /// - `self.inv(hedge)`: The sink half-edge (incoming to target node)
275    ///
276    /// The algorithm ensures complete edge removal by:
277    /// 1. Identifying redundant source half-edges during traversal
278    /// 2. Finding their paired sink half-edges using `self.inv()`
279    /// 3. Adding both to the removal subgraph
280    /// 4. Calling `delete_hedges()` once to remove all redundant half-edge pairs
281    ///
282    /// Failing to remove both half-edges would leave dangling half-edges and
283    /// corrupt the graph structure.
284    ///
285    /// # Arguments
286    ///
287    /// * `self` - Takes ownership of the graph to modify it in place
288    ///
289    /// # Returns
290    ///
291    /// The modified graph with all redundant edges removed
292    ///
293    /// # Errors
294    ///
295    /// Returns `TransitiveError::HasCycles` if the graph contains cycles
296    ///
297    /// # Examples
298    ///
299    /// ```rust,ignore
300    /// // Graph: A -> B -> C, A -> C (redundant)
301    /// let reduction = graph.transitive_reduction()?;
302    /// // Result: A -> B -> C (A -> C removed as redundant)
303    /// ```
304    pub fn transitive_reduction(mut self) -> Result<Self, TransitiveError> {
305        // First verify this is a DAG
306        let topo_order = self.topo_sort_kahn()?;
307
308        // Collect all redundant edges (edges that can be removed)
309        let mut redundant_hedges = Vec::new();
310        for &u in &topo_order {
311            for hedge in self.iter_crown(u) {
312                if self.flow(hedge) == Flow::Source {
313                    if let Some(v) = self.involved_node_id(hedge) {
314                        // Check if this edge is redundant (has alternative path)
315                        if self.has_alternative_path(u, v) {
316                            redundant_hedges.push(hedge);
317                        }
318                    }
319                }
320            }
321        }
322
323        // Build a subgraph with all redundant edges to remove
324        let mut edges_to_remove: crate::half_edge::subgraph::SuBitGraph = self.empty_subgraph();
325        for hedge in redundant_hedges {
326            // CRITICAL: Add both half-edges of the pair to completely remove the edge
327            // In HedgeGraph, each directed edge consists of two half-edges:
328            // - hedge: the source half-edge (outgoing from source node)
329            // - paired_hedge: the sink half-edge (incoming to target node)
330            // Both must be removed to avoid leaving dangling half-edges
331            edges_to_remove.add(hedge);
332            let paired_hedge = self.inv(hedge);
333            if paired_hedge != hedge {
334                edges_to_remove.add(paired_hedge);
335            }
336        }
337
338        // Delete all redundant hedges at once
339        self.delete_hedges(&edges_to_remove);
340
341        Ok(self)
342    }

344    /// Helper function to check if there's an alternative path from u to v
345    /// excluding the direct edge u -> v.
346    ///
347    /// This uses breadth-first search starting from all neighbors of u (except v)
348    /// to determine if v can be reached through an indirect path. This is used
349    /// to identify redundant edges in transitive reduction.
350    fn has_alternative_path(&self, u: NodeIndex, v: NodeIndex) -> bool {
351        // Use BFS to find if there's a path from u to v without using the direct edge
352        let mut queue = VecDeque::new();
353        let mut visited = HashSet::new();
354
355        // Start with all neighbors of u except v
356        for hedge in self.iter_crown(u) {
357            if self.flow(hedge) == Flow::Source {
358                if let Some(neighbor) = self.involved_node_id(hedge) {
359                    if neighbor != v && !visited.contains(&neighbor) {
360                        queue.push_back(neighbor);
361                        visited.insert(neighbor);
362                    }
363                }
364            }
365        }
366
367        // BFS to find v
368        while let Some(current) = queue.pop_front() {
369            if current == v {
370                return true;
371            }
372
373            for hedge in self.iter_crown(current) {
374                if self.flow(hedge) == Flow::Source {
375                    if let Some(neighbor) = self.involved_node_id(hedge) {
376                        if !visited.contains(&neighbor) {
377                            queue.push_back(neighbor);
378                            visited.insert(neighbor);
379                        }
380                    }
381                }
382            }
383        }
384
385        false
386    }

388    /// Checks if there is a path from `source` to `target` in the graph.
389    ///
390    /// Uses breadth-first search to determine reachability. This method
391    /// only follows source half-edges (outgoing edges) to traverse the
392    /// directed graph structure.
393    ///
394    /// # Arguments
395    ///
396    /// * `source` - The source node
397    /// * `target` - The target node
398    ///
399    /// # Returns
400    ///
401    /// `true` if there is a directed path from source to target, `false` otherwise.
402    ///
403    /// # Half-Edge Traversal
404    ///
405    /// The algorithm only follows half-edges with `Flow::Source` to ensure
406    /// it respects the directed nature of edges in the graph.
407    pub fn is_reachable(&self, source: NodeIndex, target: NodeIndex) -> bool {
408        if source == target {
409            return true;
410        }
411
412        let mut visited = HashSet::new();
413        let mut queue = VecDeque::new();
414
415        queue.push_back(source);
416        visited.insert(source);
417
418        while let Some(current) = queue.pop_front() {
419            for hedge in self.iter_crown(current) {
420                if self.flow(hedge) == Flow::Source {
421                    if let Some(neighbor) = self.involved_node_id(hedge) {
422                        if neighbor == target {
423                            return true;
424                        }
425                        if !visited.contains(&neighbor) {
426                            visited.insert(neighbor);
427                            queue.push_back(neighbor);
428                        }
429                    }
430                }
431            }
432        }
433
434        false
435    }

1108    #[test]
1109    fn test_comprehensive_reachability() {
1110        let graph: DotGraph<NodeStorageVec<DotVertexData>> = DotGraph::from_string(
1111            r#"
1112            digraph {
1113                // Create a graph with known reachability patterns
1114                root -> branch_a
1115                root -> branch_b
1116
1117                branch_a -> leaf_a1
1118                branch_a -> leaf_a2
1119                branch_a -> shared
1120
1121                branch_b -> leaf_b1
1122                branch_b -> shared
1123
1124                shared -> final_node
1125
1126                // Isolated component
1127                iso_a -> iso_b
1128                iso_b -> iso_c
1129
1130                // Single node
1131                singleton
1132            }
1133            "#,
1134        )
1135        .unwrap();
1136
1137        // Test various reachability scenarios
1138        let node_map: std::collections::HashMap<_, _> = graph
1139            .graph
1140            .iter_node_ids()
1141            .filter_map(|idx| {
1142                graph
1143                    .graph
1144                    .node_store
1145                    .get_node_data(idx)
1146                    .name
1147                    .as_ref()
1148                    .map(|id| (id.clone(), idx))
1149            })
1150            .collect();
1151
1152        // Self-reachability
1153        if let Some(&root_idx) = node_map.get("root") {
1154            assert!(graph.graph.is_reachable(root_idx, root_idx));
1155        }
1156
1157        // Direct reachability
1158        if let (Some(&root_idx), Some(&branch_a_idx)) =
1159            (node_map.get("root"), node_map.get("branch_a"))
1160        {
1161            assert!(graph.graph.is_reachable(root_idx, branch_a_idx));
1162        }
1163
1164        // Transitive reachability
1165        if let (Some(&root_idx), Some(&final_idx)) =
1166            (node_map.get("root"), node_map.get("final_node"))
1167        {
1168            assert!(graph.graph.is_reachable(root_idx, final_idx));
1169        }
1170
1171        // Non-reachability across components
1172        if let (Some(&root_idx), Some(&iso_a_idx)) = (node_map.get("root"), node_map.get("iso_a")) {
1173            assert!(!graph.graph.is_reachable(root_idx, iso_a_idx));
1174            assert!(!graph.graph.is_reachable(iso_a_idx, root_idx));
1175        }
1176
1177        // Reachability within isolated component
1178        if let (Some(&iso_a_idx), Some(&iso_c_idx)) = (node_map.get("iso_a"), node_map.get("iso_c"))
1179        {
1180            assert!(graph.graph.is_reachable(iso_a_idx, iso_c_idx));
1181            assert!(!graph.graph.is_reachable(iso_c_idx, iso_a_idx));
1182        }
1183    }
