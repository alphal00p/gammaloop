68/// A temporary structure used during the construction of a [`HedgeGraph`].
69///
70/// It holds the data for a node and a list of half-edges that are incident to it
71/// before the full graph topology (e.g., specific `NodeStorage` format) is finalized.
72pub struct HedgeNodeBuilder<V> {
73    /// The data associated with the node being built.
74    pub(crate) data: V,
75    /// A list of [`Hedge`] identifiers that are incident to this node.
76    pub(crate) hedges: Vec<Hedge>,
77}
78
79impl<V> HedgeNodeBuilder<V> {
80    pub fn to_base<S: BaseSubgraph>(&self, len: usize) -> S {
81        let mut subgraph = S::empty(len);
82
83        for hedge in &self.hedges {
84            subgraph.add(*hedge);
85        }
86
87        subgraph
88    }
89}

96/// A builder for programmatically constructing [`HedgeGraph`] instances.
97///
98/// This builder allows for the incremental addition of nodes and edges (both
99/// paired and external/dangling) before finalizing the graph structure.
100///
101/// # Type Parameters
102///
103/// - `E`: The type of data to be associated with edges.
104/// - `V`: The type of data to be associated with nodes.
105///
106/// # Example
107///
108/// ```rust,ignore
109/// use linnet::half_edge::builder::HedgeGraphBuilder;
110/// use linnet::half_edge::involution::{Flow, Orientation};
111///
112/// let mut builder = HedgeGraphBuilder::<&str, &str>::new();
113///
114/// let node0 = builder.add_node("Node_0_Data");
115/// let node1 = builder.add_node("Node_1_Data");
116/// let node2 = builder.add_node("Node_2_Data");
117///
118/// builder.add_edge(node0, node1, "Edge_01_Data", Orientation::Default);
119/// builder.add_external_edge(node2, "External_Edge_Data", Orientation::Undirected, Flow::Source);
120///
121/// // Assuming a NodeStorage type MyNodeStore is defined and implements NodeStorageOps
122/// // let graph: HedgeGraph<&str, &str, MyNodeStore> = builder.build();
123/// ```
124pub struct HedgeGraphBuilder<E, V, H = NoData> {
125    hedge_data: HedgeVec<H>,
126    /// A list of nodes currently being built, stored as [`HedgeNodeBuilder`] instances.
127    nodes: Vec<HedgeNodeBuilder<V>>,
128    /// The [`Involution`] structure managing the half-edges being added to the graph.
129    pub(crate) involution: Involution<E>,
130}
131
132impl<E, V, H> HedgeGraphBuilder<E, V, H> {
133    pub fn new() -> Self {
134        HedgeGraphBuilder {
135            hedge_data: HedgeVec::new(),
136            nodes: Vec::new(),
137            involution: Involution::new(),
138        }
139    }
140
141    pub fn build<N: NodeStorageOps<NodeData = V>>(self) -> HedgeGraph<E, V, H, N> {
142        self.into()
143    }
144
145    pub fn build_with_map<V2, N: NodeStorageOps<NodeData = V2>>(
146        self,
147        map: impl FnMut(V) -> V2,
148    ) -> HedgeGraph<E, V2, H, N> {
149        let len: Hedge = self.involution.len();
150
151        HedgeGraph {
152            node_store: N::build_with_mapping(self.nodes, len.0, map),
153            edge_store: SmartEdgeVec::new(self.involution),
154            hedge_data: self.hedge_data,
155        }
156    }
157
158    pub fn add_node(&mut self, data: V) -> NodeIndex {
159        let index = self.nodes.len();
160        self.nodes.push(HedgeNodeBuilder {
161            data,
162            hedges: Vec::new(),
163        });
164        NodeIndex(index)
165    }
166
167    pub fn add_edge(
168        &mut self,
169        source: impl Into<HedgeData<H>>,
170        sink: impl Into<HedgeData<H>>,
171        data: E,
172        directed: impl Into<Orientation>,
173    ) {
174        let source = source.into();
175        let sink = sink.into();
176        let (sourceh, sinkh) = self.involution.add_pair(data, directed);
177        self.hedge_data.push(source.data);
178        self.hedge_data.push(sink.data);
179
180        self.nodes[source.node.0].hedges.push(sourceh);
181        self.nodes[sink.node.0].hedges.push(sinkh);
182    }
183
184    pub fn add_external_edge(
185        &mut self,
186        source: impl Into<HedgeData<H>>,
187        data: E,
188        orientation: impl Into<Orientation>,
189        underlying: Flow,
190    ) {
191        let source = source.into();
192        let id = self.involution.add_identity(data, orientation, underlying);
193        self.nodes[source.node.0].hedges.push(id);
194        self.hedge_data.push(source.data);
195    }
196}

199    fn default() -> Self {
200        Self::new()
201    }
202}
