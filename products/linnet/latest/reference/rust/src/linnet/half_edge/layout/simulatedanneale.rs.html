7pub trait Energy<S> {
8    /// Return the absolute energy of `next`.
9    /// When a previous state/energy is provided, implementations may reuse or update it
10    /// instead of recomputing from scratch.
11    fn energy(&self, prev: Option<(&S, f64)>, next: &S) -> f64;
12
13    /// Called once a proposal is accepted so implementations can clear or update caches.
14    fn on_accept(&self, _state: &mut S) {}
15}
16
17pub trait Neighbor<S> {
18    /// Prepare the initial state before it is scored and stored as the best state.
19    fn prepare(&self, _state: &mut S) {}
20
21    /// Propose a new state from `s` given step scale and temperature.
22    fn propose(&self, s: &S, rng: &mut impl Rng, step: f64, temp: f64) -> S;
23}

25pub trait Schedule {
26    /// Called at the start of each epoch. May mutate step & temp. Return `false` to stop.
27    fn begin_epoch(
28        &mut self,
29        epoch: usize,
30        last_accept_ratio: f64,
31        step: &mut f64,
32        temp: &mut f64,
33    ) -> bool;
34    /// Number of Metropolis steps per epoch.
35    fn steps_per_epoch(&self) -> usize;
36    /// Total max epochs budget (upper bound; you can early stop in begin_epoch).
37    fn max_epochs(&self) -> usize;
38}

55pub fn anneal<S: Clone, N: Neighbor<S>, E: Energy<S>, Sch: Schedule, R: SeedableRng + Rng>(
56    init: S,
57    mut cfg: SAConfig,
58    neigh: &N,
59    energy: &E,
60    sched: &mut Sch,
61) -> (S, SAStats) {
62    let mut rng = R::seed_from_u64(cfg.seed);
63    let mut cur = init;
64    neigh.prepare(&mut cur);
65    let mut current_energy = energy.energy(None, &cur);
66
67    let mut best = cur.clone();
68    let mut best_energy = current_energy;
69
70    let mut total_steps = 0usize;
71    let mut last_accept_ratio = 0.0;
72
73    for ep in 0..sched.max_epochs() {
74        if !sched.begin_epoch(ep, last_accept_ratio, &mut cfg.step, &mut cfg.temp) {
75            return (
76                best,
77                SAStats {
78                    best_energy,
79                    total_steps,
80                    epochs_run: ep,
81                    last_accept_ratio,
82                    last_temp: cfg.temp,
83                    last_step: cfg.step,
84                },
85            );
86        }
87
88        let steps = sched.steps_per_epoch();
89        let mut accepted = 0usize;
90
91        for _ in 0..steps {
92            total_steps += 1;
93            let cand = neigh.propose(&cur, &mut rng, cfg.step, cfg.temp);
94            let candidate_energy = energy.energy(Some((&cur, current_energy)), &cand);
95            let delta = candidate_energy - current_energy;
96
97            let accept = delta <= 0.0 || rng.gen::<f64>() < (-delta / cfg.temp).exp();
98            if accept {
99                let mut accepted_state = cand;
100                energy.on_accept(&mut accepted_state);
101                cur = accepted_state;
102                current_energy = candidate_energy;
103                accepted += 1;
104                if candidate_energy < best_energy {
105                    best_energy = candidate_energy;
106                    best = cur.clone();
107                }
108            }
109        }
110
111        last_accept_ratio = accepted as f64 / steps as f64;
112    }
113
114    (
115        best,
116        SAStats {
117            best_energy,
118            total_steps,
119            epochs_run: sched.max_epochs(),
120            last_accept_ratio,
121            last_temp: cfg.temp,
122            last_step: cfg.step,
123        },
124    )
125}

128pub struct GeoSchedule {
129    pub steps: usize,      // e.g. 200
130    pub epochs: usize,     // e.g. 8
131    pub cool: f64,         // e.g. 0.85
132    pub accept_floor: f64, // e.g. 0.15
133    pub step_shrink: f64,  // e.g. 0.7
134    pub early_tol: f64,    // e.g. 1e-6 (relative best improvement)
135    _prev_best: Option<f64>,
136}

138impl GeoSchedule {
139    pub fn add_to_global(&self, global_data: &mut GlobalData) {
140        global_data
141            .statements
142            .insert("steps".to_string(), self.steps.to_string());
143        global_data
144            .statements
145            .insert("epochs".to_string(), self.epochs.to_string());
146        global_data
147            .statements
148            .insert("cool".to_string(), self.cool.to_string());
149        global_data
150            .statements
151            .insert("accept_floor".to_string(), self.accept_floor.to_string());
152        global_data
153            .statements
154            .insert("step_shrink".to_string(), self.step_shrink.to_string());
155        global_data
156            .statements
157            .insert("early_tol".to_string(), self.early_tol.to_string());
158    }
159
160    pub fn parse(global_data: &GlobalData) -> Self {
161        let steps = global_data
162            .statements
163            .get("steps")
164            .and_then(|s| s.parse().ok())
165            .unwrap_or(200);
166        let epochs = global_data
167            .statements
168            .get("epochs")
169            .and_then(|s| s.parse().ok())
170            .unwrap_or(8);
171        let cool = global_data
172            .statements
173            .get("cool")
174            .and_then(|s| s.parse().ok())
175            .unwrap_or(0.85);
176        let accept_floor = global_data
177            .statements
178            .get("accept_floor")
179            .and_then(|s| s.parse().ok())
180            .unwrap_or(0.15);
181        let step_shrink = global_data
182            .statements
183            .get("step_shrink")
184            .and_then(|s| s.parse().ok())
185            .unwrap_or(0.7);
186        let early_tol = global_data
187            .statements
188            .get("early_tol")
189            .and_then(|s| s.parse().ok())
190            .unwrap_or(1e-6);
191        Self::new(steps, epochs, cool, accept_floor, step_shrink, early_tol)
192    }

194    pub fn new(

220impl Schedule for GeoSchedule {
221    fn begin_epoch(
222        &mut self,
223        epoch: usize,
224        last_accept: f64,
225        step: &mut f64,
226        temp: &mut f64,
227    ) -> bool {
228        if epoch > 0 {
229            *temp *= self.cool;
230            if last_accept < self.accept_floor {
231                *step *= self.step_shrink;
232            }
233            // Early-stop hook: caller can feed best energy back by resetting prev_best, or ignore.
234        }
235        epoch < self.epochs
236    }
237    fn steps_per_epoch(&self) -> usize {
238        self.steps
239    }
240    fn max_epochs(&self) -> usize {
241        self.epochs
242    }
243}
