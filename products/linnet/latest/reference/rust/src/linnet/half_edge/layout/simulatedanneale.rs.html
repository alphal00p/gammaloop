7pub trait Energy<S> {
8    /// Return the absolute energy of `next`.
9    /// When a previous state/energy is provided, implementations may reuse or update it
10    /// instead of recomputing from scratch.
11    fn energy(&self, prev: Option<(&S, f64)>, next: &S) -> f64;
12
13    /// Called once a proposal is accepted so implementations can clear or update caches.
14    fn on_accept(&self, _state: &mut S) {}
15}

17pub trait Neighbor<S> {
18    /// Propose a new state from `s` given step scale and temperature.
19    fn propose(&self, s: &S, rng: &mut impl Rng, step: f64, temp: f64) -> S;
20}

22pub trait Schedule {
23    /// Called at the start of each epoch. May mutate step & temp. Return `false` to stop.
24    fn begin_epoch(
25        &mut self,
26        epoch: usize,
27        last_accept_ratio: f64,
28        step: &mut f64,
29        temp: &mut f64,
30    ) -> bool;
31    /// Number of Metropolis steps per epoch.
32    fn steps_per_epoch(&self) -> usize;
33    /// Total max epochs budget (upper bound; you can early stop in begin_epoch).
34    fn max_epochs(&self) -> usize;
35}

52pub fn anneal<S: Clone, N: Neighbor<S>, E: Energy<S>, Sch: Schedule, R: SeedableRng + Rng>(
53    init: S,
54    mut cfg: SAConfig,
55    neigh: &N,
56    energy: &E,
57    sched: &mut Sch,
58) -> (S, SAStats) {
59    let mut rng = R::seed_from_u64(cfg.seed);
60    let mut cur = init.clone();
61    let mut current_energy = energy.energy(None, &cur);
62
63    let mut best = cur.clone();
64    let mut best_energy = current_energy;
65
66    let mut total_steps = 0usize;
67    let mut last_accept_ratio = 0.0;
68
69    for ep in 0..sched.max_epochs() {
70        if !sched.begin_epoch(ep, last_accept_ratio, &mut cfg.step, &mut cfg.temp) {
71            return (
72                best,
73                SAStats {
74                    best_energy,
75                    total_steps,
76                    epochs_run: ep,
77                    last_accept_ratio,
78                    last_temp: cfg.temp,
79                    last_step: cfg.step,
80                },
81            );
82        }
83
84        let steps = sched.steps_per_epoch();
85        let mut accepted = 0usize;
86
87        for _ in 0..steps {
88            total_steps += 1;
89            let cand = neigh.propose(&cur, &mut rng, cfg.step, cfg.temp);
90            let candidate_energy = energy.energy(Some((&cur, current_energy)), &cand);
91            let delta = candidate_energy - current_energy;
92
93            let accept = delta <= 0.0 || rng.gen::<f64>() < (-delta / cfg.temp).exp();
94            if accept {
95                let mut accepted_state = cand;
96                energy.on_accept(&mut accepted_state);
97                cur = accepted_state;
98                current_energy = candidate_energy;
99                accepted += 1;
100                if candidate_energy < best_energy {
101                    best_energy = candidate_energy;
102                    best = cur.clone();
103                }
104            }
105        }
106
107        last_accept_ratio = accepted as f64 / steps as f64;
108    }
109
110    (
111        best,
112        SAStats {
113            best_energy,
114            total_steps,
115            epochs_run: sched.max_epochs(),
116            last_accept_ratio,
117            last_temp: cfg.temp,
118            last_step: cfg.step,
119        },
120    )
121}

124pub struct GeoSchedule {
125    pub steps: usize,      // e.g. 200
126    pub epochs: usize,     // e.g. 8
127    pub cool: f64,         // e.g. 0.85
128    pub accept_floor: f64, // e.g. 0.15
129    pub step_shrink: f64,  // e.g. 0.7
130    pub early_tol: f64,    // e.g. 1e-6 (relative best improvement)
131    _prev_best: Option<f64>,
132}
133
134impl GeoSchedule {
135    pub fn add_to_global(&self, global_data: &mut GlobalData) {
136        global_data
137            .statements
138            .insert("steps".to_string(), self.steps.to_string());
139        global_data
140            .statements
141            .insert("epochs".to_string(), self.epochs.to_string());
142        global_data
143            .statements
144            .insert("cool".to_string(), self.cool.to_string());
145        global_data
146            .statements
147            .insert("accept_floor".to_string(), self.accept_floor.to_string());
148        global_data
149            .statements
150            .insert("step_shrink".to_string(), self.step_shrink.to_string());
151        global_data
152            .statements
153            .insert("early_tol".to_string(), self.early_tol.to_string());
154    }
155
156    pub fn parse(global_data: &GlobalData) -> Self {
157        let steps = global_data
158            .statements
159            .get("steps")
160            .and_then(|s| s.parse().ok())
161            .unwrap_or(200);
162        let epochs = global_data
163            .statements
164            .get("epochs")
165            .and_then(|s| s.parse().ok())
166            .unwrap_or(8);
167        let cool = global_data
168            .statements
169            .get("cool")
170            .and_then(|s| s.parse().ok())
171            .unwrap_or(0.85);
172        let accept_floor = global_data
173            .statements
174            .get("accept_floor")
175            .and_then(|s| s.parse().ok())
176            .unwrap_or(0.15);
177        let step_shrink = global_data
178            .statements
179            .get("step_shrink")
180            .and_then(|s| s.parse().ok())
181            .unwrap_or(0.7);
182        let early_tol = global_data
183            .statements
184            .get("early_tol")
185            .and_then(|s| s.parse().ok())
186            .unwrap_or(1e-6);
187        Self::new(steps, epochs, cool, accept_floor, step_shrink, early_tol)
188    }
189
190    pub fn new(
191        steps: usize,
192        epochs: usize,
193        cool: f64,
194        accept_floor: f64,
195        step_shrink: f64,
196        early_tol: f64,
197    ) -> Self {
198        Self {

201            cool,
202            accept_floor,
203            step_shrink,
204            early_tol,

207    }
208}

216impl Schedule for GeoSchedule {
217    fn begin_epoch(
218        &mut self,
219        epoch: usize,
220        last_accept: f64,
221        step: &mut f64,
222        temp: &mut f64,
223    ) -> bool {
224        if epoch > 0 {
225            *temp *= self.cool;
226            if last_accept < self.accept_floor {
227                *step *= self.step_shrink;
228            }
229            // Early-stop hook: caller can feed best energy back by resetting prev_best, or ignore.
230        }
231        epoch < self.epochs
232    }
233    fn steps_per_epoch(&self) -> usize {
234        self.steps
235    }
236    fn max_epochs(&self) -> usize {
237        self.epochs
238    }
239}
