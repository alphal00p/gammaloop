22/// Iterator for traversing tensor fibers
23///
24/// This high-level iterator uses a core iterator to traverse fibers in tensors,
25/// returning references to tensor elements at each position.
26#[derive(Debug)]
27pub struct FiberIterator<
28    'a,
29    S: TensorStructure,
30    I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R>,
31> {
32    /// The fiber being iterated
33    pub fiber: Fiber<'a, S>,
34    /// The underlying core iterator
35    pub iter: I,
36    /// Number of indices skipped
37    pub skipped: usize,
38}

40impl<S: TensorStructure, I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R> + Clone> Clone
41    for FiberIterator<'_, S, I>
42{
43    fn clone(&self) -> Self {
44        FiberIterator {
45            fiber: self.fiber.clone(),
46            iter: self.iter.clone(),
47            skipped: self.skipped,
48        }
49    }
50}

52impl<'a, S: TensorStructure, I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R>>
53    FiberIterator<'a, S, I>
54{
55    /// Creates a new fiber iterator
56    ///
57    /// # Arguments
58    ///
59    /// * `fiber` - The fiber to iterate over
60    /// * `conj` - Whether to use conjugate iteration
61    pub fn new(fiber: Fiber<'a, S>, conj: bool) -> Self {
62        FiberIterator {
63            iter: I::new(&fiber, conj),
64            fiber,
65            skipped: 0,
66        }
67    }
68
69    /// Resets the iterator to its initial state
70    pub fn reset(&mut self) {
71        self.iter.reset();
72        self.skipped = 0;
73    }
74
75    /// Shifts the iterator by the given amount
76    ///
77    /// # Arguments
78    ///
79    /// * `shift` - The amount to shift by
80    pub fn shift(&mut self, shift: usize) {
81        self.reset();
82        self.iter.shift(shift);
83    }
84}

114impl<
115    'a,
116    I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R, Item = It>,
117    S: TensorStructure,
118    T,
119    It,
120> Iterator for FiberIterator<'a, DenseTensor<T, S>, I>
121where
122    It: FiberIteratorItem,
123{
124    type Item = (&'a T, It::OtherData);
125    fn next(&mut self) -> Option<Self::Item> {
126        self.iter.next().map(|x| {
127            // println!(
128            //     "DenseTensor: flat_idx: {}, size: {:?}",
129            //     x.flat_idx(),
130            //     self.fiber.structure.size()
131            // );
132            if let Some(t) = self.fiber.structure.get_ref_linear(x.flat_idx()) {
133                (t, x.other_data())
134            } else {
135                panic!(
136                    "DenseTensor: Out of bounds {} {}",
137                    x.flat_idx(),
138                    self.fiber.structure.size().unwrap()
139                )
140            }
141        })
142    }
143}

145impl<
146    'a,
147    I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R, Item = It>,
148    S: TensorStructure,
149    T,
150    It,
151> Iterator for FiberIterator<'a, SparseTensor<T, S>, I>
152where
153    It: FiberIteratorItem,
154{
155    type Item = (&'a T, usize, It::OtherData);
156    fn next(&mut self) -> Option<Self::Item> {
157        if let Some(i) = self.iter.next() {
158            if let Some(t) = self.fiber.structure.get_ref_linear(i.flat_idx()) {
159                let skipped = self.skipped;
160                self.skipped = 0;
161                return Some((t, skipped, i.other_data()));
162            } else {
163                self.skipped += 1;
164                return self.next();
165            }
166        }
167        None
168    }
169}

171/// Mutable iterator for traversing tensor fibers
172///
173/// Similar to `FiberIterator` but returns mutable references to tensor elements.
174pub struct MutFiberIterator<
175    'a,
176    S: TensorStructure,
177    I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R>,
178> {
179    /// The underlying core iterator
180    iter: I,
181    /// The fiber being iterated
182    fiber: FiberMut<'a, S>,
183    /// Number of indices skipped
184    skipped: usize,
185}
186
187impl<I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R, Item = It>, S: TensorStructure, T, It>
188    LendingIterator for MutFiberIterator<'_, SparseTensor<T, S>, I>
189where
190    It: FiberIteratorItem,
191{
192    type Item<'r>
193        = (&'r mut T, usize, It::OtherData)
194    where
195        Self: 'r;
196    fn next(&mut self) -> Option<Self::Item<'_>> {
197        let flat = self.iter.next()?;
198        if self.fiber.structure.is_empty_at_flat(flat.flat_idx()) {
199            let skipped = self.skipped;
200            self.skipped = 0;
201            Some((
202                self.fiber
203                    .structure
204                    .get_mut_linear(flat.flat_idx())
205                    .unwrap(),
206                skipped,
207                flat.other_data(),
208            ))
209        } else {
210            self.skipped += 1;
211            self.next()
212        }
213    }
214}
215
216impl<I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R, Item = It>, S: TensorStructure, T, It>
217    LendingIterator for MutFiberIterator<'_, DenseTensor<T, S>, I>
218where
219    It: FiberIteratorItem,
220{
221    type Item<'r>
222        = (&'r mut T, It::OtherData)
223    where
224        Self: 'r;
225    fn next(&mut self) -> Option<Self::Item<'_>> {
226        self.iter.next().map(|x| {
227            (
228                self.fiber.structure.get_mut_linear(x.flat_idx()).unwrap(),
229                x.other_data(),
230            )
231        })
232    }
233}

287/// Iterator for traversing fiber classes
288///
289/// Iterates over all fibers in a fiber class, returning an iterator for each fiber.
290pub struct FiberClassIterator<
291    'b,
292    S: TensorStructure,
293    I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R> = CoreFlatFiberIterator,
294> {
295    /// Iterator over fibers within the class
296    pub fiber_iter: FiberIterator<'b, S, I>,
297    /// Iterator over indices of fibers in the class
298    pub class_iter: CoreFlatFiberIterator,
299}
300
301impl<'b, N: TensorStructure> FiberClassIterator<'b, N, CoreFlatFiberIterator> {
302    /// Creates a new fiber class iterator
303    ///
304    /// # Arguments
305    ///
306    /// * `class` - The fiber class to iterate over
307    pub fn new(class: FiberClass<'b, N>) -> Self {
308        let (iter, iter_conj) = CoreFlatFiberIterator::new_paired_conjugates(&class);
309
310        let fiber = FiberIterator {
311            fiber: class.into(),
312            iter,
313            skipped: 0,
314        };
315
316        FiberClassIterator {
317            fiber_iter: fiber,
318            class_iter: iter_conj,
319        }
320    }
321}
322
323impl<N: TensorStructure, I: IteratesAlongFibers<<N::Slot as IsAbstractSlot>::R>>
324    FiberClassIterator<'_, N, I>
325{
326    /// Resets the iterator to its initial state
327    pub fn reset(&mut self) {
328        self.class_iter.reset();
329        self.fiber_iter.reset();
330        self.fiber_iter.shift(0);
331    }
332}

334impl<'b, N: TensorStructure, I: IteratesAlongPermutedFibers<<N::Slot as IsAbstractSlot>::R>>
335    FiberClassIterator<'b, N, I>
336{
337    /// Creates a new fiber class iterator with a permutation
338    ///
339    /// # Arguments
340    ///
341    /// * `class` - The fiber class to iterate over
342    /// * `permutation` - The permutation to apply
343    pub fn new_permuted(class: FiberClass<'b, N>, permutation: Permutation) -> Self {
344        let iter = CoreFlatFiberIterator::new(&class, false);
345
346        let fiber = FiberIterator::new_permuted(class.into(), permutation, false);
347
348        FiberClassIterator {
349            fiber_iter: fiber,
350            class_iter: iter,
351        }
352    }
353}

355impl<
356    'a,
357    S: TensorStructure + 'a,
358    I: IteratesAlongFibers<<S::Slot as IsAbstractSlot>::R> + Clone + Debug,
359> Iterator for FiberClassIterator<'a, S, I>
360{
361    type Item = FiberIterator<'a, S, I>;
362
363    fn next(&mut self) -> Option<Self::Item> {
364        let shift = self.class_iter.next()?;
365        self.fiber_iter.reset();
366        self.fiber_iter.shift(shift.into());
367        Some(self.fiber_iter.clone())
368    }
369}

381        self.fiber_iter.reset();

383        Some(&mut self.fiber_iter)
384    }
