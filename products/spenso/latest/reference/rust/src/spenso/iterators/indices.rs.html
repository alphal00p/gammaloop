11/// Abstract trait for fiber indices
12///
13/// This trait defines methods for determining whether an index is free or fixed,
14/// which is fundamental to how fibers are defined and traversed.
15pub trait AbstractFiberIndex {
16    /// Returns true if the index is free (varying during iteration)
17    fn is_free(&self) -> bool;
18
19    /// Returns true if the index is fixed (constant during iteration)
20    fn is_fixed(&self) -> bool {
21        !self.is_free()
22    }
23}
24
25/// Represents a classification of fiber indices (free or fixed)
26///
27/// Used for fiber class iterators to categorize indices.
28#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
29pub enum FiberClassIndex {
30    /// A free index that varies during iteration
31    Free,
32    /// A fixed index that remains constant during iteration
33    Fixed,
34}
35
36impl AbstractFiberIndex for FiberClassIndex {
37    fn is_free(&self) -> bool {
38        matches!(self, FiberClassIndex::Free)
39    }
40}
41
42/// Represents a specific fiber index, either free or fixed to a particular value
43///
44/// Used for precise control of which dimensions are fixed and which vary during iteration.
45#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
46pub enum FiberIndex {
47    /// A free index that varies during iteration
48    Free,
49    /// A fixed index with a specific value
50    Fixed(usize),
51}
52
53impl AbstractFiberIndex for FiberIndex {
54    fn is_free(&self) -> bool {
55        matches!(self, FiberIndex::Free)
56    }
57}
58
59impl From<usize> for FiberIndex {
60    fn from(value: usize) -> Self {
61        FiberIndex::Fixed(value)
62    }
63}
64
65impl Display for FiberIndex {
66    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
67        match self {
68            FiberIndex::Fixed(val) => {
69                write!(f, "{val}")
70            }
71            FiberIndex::Free => {
72                write!(f, ":")
73            }
74        }
75    }
76}

78/// Enumeration of different ways to specify fiber data
79///
80/// This is used as a flexible input mechanism to create fibers from various data sources.
81pub enum FiberData<'a> {
82    /// A single index position
83    Single(usize),
84    /// A flat index directly
85    Flat(FlatIndex),
86    /// A boolean filter where true indicates free indices
87    BoolFilter(&'a [bool]),
88    /// Position data where negative values mean free indices
89    Pos(&'a [isize]),
90    /// Integer filter where non-zero values indicate free indices
91    IntFilter(&'a [u8]),
92    /// A bitvec filter where true indicates free indices
93    Subset(&'a SubSet<SlotIndex>),
94}
95
96impl From<usize> for FiberData<'_> {
97    fn from(value: usize) -> Self {
98        Self::Single(value)
99    }
100}
101
102impl From<SlotIndex> for FiberData<'_> {
103    fn from(value: SlotIndex) -> Self {
104        Self::Single(value.into())
105    }
106}
107
108impl<'a> From<&'a [bool]> for FiberData<'a> {
109    fn from(value: &'a [bool]) -> Self {
110        Self::BoolFilter(value)
111    }
112}
113
114impl<'a> From<&'a SubSet<SlotIndex>> for FiberData<'a> {
115    fn from(value: &'a SubSet<SlotIndex>) -> Self {
116        Self::Subset(value)
117    }
118}
119
120impl<'a> From<&'a [u8]> for FiberData<'a> {
121    fn from(value: &'a [u8]) -> Self {
122        Self::IntFilter(value)
123    }
124}
125
126impl<'a> From<&'a [isize]> for FiberData<'a> {
127    fn from(value: &'a [isize]) -> Self {
128        Self::Pos(value)
129    }
130}
131
132impl From<FlatIndex> for FiberData<'_> {
133    fn from(value: FlatIndex) -> Self {
134        Self::Flat(value)
135    }
136}

138/// Enumeration for switching between iterator implementations
139pub enum IteratorEnum<A, B> {
140    /// First iterator type
141    A(A),
142    /// Second iterator type
143    B(B),
144}
145
146impl<A, B> Iterator for IteratorEnum<A, B>
147where
148    A: Iterator,
149    B: Iterator<Item = A::Item>,
150{
151    type Item = A::Item;
152    fn next(&mut self) -> Option<Self::Item> {
153        match self {
154            IteratorEnum::A(a) => a.next(),
155            IteratorEnum::B(b) => b.next(),
156        }
157    }
158}
