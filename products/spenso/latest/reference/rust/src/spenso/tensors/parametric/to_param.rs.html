12pub trait ToParam: HasStructure {

16pub trait ToAtom {
17    fn to_atom(self) -> Atom;
18}

20impl ToAtom for f64 {
21    fn to_atom(self) -> Atom {
22        Atom::num(self)
23    }
24}

26impl ToAtom for Complex<f64> {
27    fn to_atom(self) -> Atom {

29    }

32impl<T> ToAtom for RealOrComplex<T>
33where
34    T: ToAtom,
35{
36    fn to_atom(self) -> Atom {
37        match self {
38            RealOrComplex::Real(r) => r.to_atom(),
39            RealOrComplex::Complex(c) => c.re.to_atom() + c.im.to_atom() * Atom::i(),
40        }
41    }

44impl<I: TensorStructure + Clone, D: ToAtom> ToParam for DataTensor<D, I> {

53impl<I: TensorStructure + Clone, D: ToAtom + Clone> ToParam for RealOrComplexTensor<D, I> {
54    fn to_param(self) -> ParamTensor<Self::Structure> {
55        ParamTensor {
56            tensor: match self {
57                RealOrComplexTensor::Real(r) => r.map_data(ToAtom::to_atom),
58                RealOrComplexTensor::Complex(c) => {
59                    c.map_data(|a| a.re.to_atom() + a.im.to_atom() * Atom::i())
60                }
61            },
62            param_type: ParamOrComposite::Composite,
63        }
64    }
65}

73impl<T: ToParam + HasStructure<Structure = I>, I: ScalarStructure + TensorStructure>
74    ParamOrConcrete<T, I>
75{
76    pub fn to_param(&mut self) {
77        if matches!(self, ParamOrConcrete::Concrete(_)) {
78            let old = std::mem::replace(
79                self,
80                ParamOrConcrete::Param(ParamTensor::param(DataTensor::Sparse(
81                    SparseTensor::empty(I::scalar_structure(), Atom::Zero),
82                ))),
83            );
84
85            if let ParamOrConcrete::Concrete(r) = old {
86                *self = ParamOrConcrete::Param(r.to_param());
87            }
88        }
89    }
