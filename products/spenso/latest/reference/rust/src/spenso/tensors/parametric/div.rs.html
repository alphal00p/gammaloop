7impl<T, U, Out> Div<ConcreteOrParamRef<'_, U>> for ConcreteOrParam<T>
8where
9    Atom: Div<U, Output = Atom> + for<'a> Div<AtomView<'a>, Output = Atom>,
10    T: ToAtom + Div<U, Output = Out>,
11{
12    type Output = ConcreteOrParam<Out>;
13    fn div(self, rhs: ConcreteOrParamRef<'_, U>) -> ConcreteOrParam<Out> {
14        match (self, rhs) {
15            (ConcreteOrParam::Concrete(c1), ConcreteOrParamRef::Concrete(c2)) => {
16                ConcreteOrParam::Concrete(c1 / c2)
17            }
18            (ConcreteOrParam::Param(a), ConcreteOrParamRef::Concrete(c)) => {
19                ConcreteOrParam::Param(a / c)
20            }
21            (ConcreteOrParam::Param(a1), ConcreteOrParamRef::Param(a2)) => {
22                ConcreteOrParam::Param(a1 / a2)
23            }
24            (mut c, a) => {
25                c.to_param();
26                c / a
27            }
28        }
29    }
30}
