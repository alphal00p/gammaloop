12impl<T, I> Trace for DenseTensor<T, I>
13where
14    T: ContractableWith<T, Out = T> + Clone + RefZero + FallibleAddAssign<T> + FallibleSubAssign<T>,
15    I: TensorStructure + Clone + StructureContract,
16{
17    /// Contract the tensor with itself, i.e. trace over all matching indices.
18    fn internal_contract(&self) -> Self {
19        let mut result: DenseTensor<T, I> = self.clone();
20        // Trace positions refer to the structure being traced, so each pair is
21        // located only after the previous trace has been applied.
22        while let Some(&trace) = result.traces().first() {
23            let mut new_structure = result.structure.clone();
24            new_structure.trace(trace[0], trace[1]);
25
26            let mut new_result =
27                DenseTensor::from_storage_data_coerced(&result.data, new_structure)
28                    .unwrap_or_else(|_| unreachable!());
29            for (idx, t) in result.iter_trace(trace) {
30                new_result.set(&idx, t).unwrap_or_else(|_| unreachable!());
31            }
32            result = new_result;
33        }
34        result
35    }
36}

38impl<T, I> Trace for SparseTensor<T, I>
39where
40    T: ContractableWith<T, Out = T>
41        + Clone
42        + RefZero
43        + IsZero
44        + FallibleAddAssign<T>
45        + FallibleSubAssign<T>,
46    I: TensorStructure + Clone + StructureContract,
47{
48    /// Contract the tensor with itself, i.e. trace over all matching indices.
49    fn internal_contract(&self) -> Self {
50        let trace = if let Some(e) = self.traces().first() {
51            *e
52        } else {
53            return self.clone();
54        };
55
56        // println!("trace {:?}", trace);
57        let mut new_structure = self.structure.clone();
58        // println!("{}", new_structure);
59        new_structure.trace(trace[0], trace[1]);
60
61        let mut new_result = SparseTensor::empty(new_structure, self.zero.clone());
62        for (idx, t) in self.iter_trace(trace).filter(|(_, t)| !t.is_zero()) {
63            new_result.set(&idx, t).unwrap();
64        }
65
66        if new_result.traces().is_empty() {
67            new_result
68        } else {
69            new_result.internal_contract()
70        }
71    }
72}

74impl<T, I> Trace for DataTensor<T, I>
75where
76    T: ContractableWith<T, Out = T>,
77    T: FallibleAddAssign<T> + FallibleSubAssign<T> + Clone + RefZero + IsZero,
78    I: TensorStructure + Clone + StructureContract,
79{
80    fn internal_contract(&self) -> Self {
81        match self {
82            DataTensor::Dense(d) => DataTensor::Dense(d.internal_contract()),
83            DataTensor::Sparse(s) => DataTensor::Sparse(s.internal_contract()),
84        }

100    #[test]
101    fn dense_trace_contracts_every_pair() {
102        // Pairs of different dimensions make each trace depend on the structure
103        // left behind by the previous one.
104        let euclidean = Euclidean {}.new_slot(2, 0).to_lib();
105        let lorentz = Lorentz {}.new_slot(3, 1).to_lib();
106        let structure: OrderedStructure<LibraryRep> =
107            Canonicalized::from_iter([euclidean, euclidean, lorentz, lorentz.dual()])
108                .into_canonical();
109        let tensor =
110            DenseTensor::from_storage_data((1..=36).map(f64::from).collect(), structure).unwrap();
111
112        // The row-major diagonal components T[e, e, l, l] are 1 + 27 e + 4 l.
113        let expected = (0..2)
114            .flat_map(|e| (0..3).map(move |l| f64::from(1 + 27 * e + 4 * l)))
115            .sum::<f64>();
116        assert_eq!(tensor.internal_contract().scalar(), Some(expected));
117        assert_eq!(
118            tensor.to_sparse().internal_contract().scalar(),
119            Some(expected)
120        );
121    }

123    #[test]
124    fn sparse_trace_preserves_dual_partitions_for_subsequent_contraction() {
125        let [up0, up1, up2] = [0, 1, 2].map(|index| Lorentz {}.new_slot(2, index).to_lib());
126        let structure: OrderedStructure<LibraryRep> =
127            OrderedStructure::new(vec![up0, up2, up0.dual(), up1.dual()]).into_canonical();
128        let mut b = SparseTensor::<i32, _>::empty(structure, 0);
129        for (indices, value) in [
130            ([0, 0, 0, 0], 2),
131            ([1, 0, 1, 0], 3),
132            ([0, 0, 0, 1], 7),
133            ([1, 1, 1, 0], -11),
134            ([0, 1, 0, 1], 13),
135            ([1, 1, 1, 1], 17),
136            // Unequal coordinates on the traced slots must not contribute.
137            ([0, 1, 1, 0], 101),
138        ] {
139            b.set(&indices, value).unwrap();
140        }
141
142        let traced = b.internal_contract();
143        assert_eq!(
144            traced.external_structure_iter().collect::<Vec<_>>(),
145            [up2, up1.dual()]
146        );
147        assert_eq!(traced.structure.n_base(), 1);
148        assert_eq!(traced.structure.n_dual(), 1);
149        for (indices, value) in [([0, 0], 5), ([0, 1], 7), ([1, 0], -11), ([1, 1], 30)] {
150            assert_eq!(*traced.get_ref(indices).unwrap(), value);
151        }
152
153        let structure: OrderedStructure<LibraryRep> =
154            OrderedStructure::new(vec![up1, up2.dual()]).into_canonical();
155        let mut a = SparseTensor::<i32, _>::empty(structure, 0);
156        for (indices, value) in [([0, 0], 19), ([0, 1], 23), ([1, 0], -29), ([1, 1], 31)] {
157            a.set(&indices, value).unwrap();
158        }
159
160        let result = a.contract(&traced).unwrap();
161        assert!(result.is_scalar());
162        // A's slot order is the reverse of the remaining B coordinates.
163        assert_eq!(result.scalar(), Some(19 * 5 + 23 * -11 - 29 * 7 + 31 * 30));
164    }
