1//! Materialization helpers for shorthand syntax before ordinary network parsing.
2//!
3//! Network-level shorthand materialization lives on the parser implementation:
4//! it handles chain and trace topology and returns parsed networks. The helper
5//! in this module is deliberately narrower than algebraic simplification. It
6//! takes compact Schoonschip syntax that cannot be parsed as an ordinary leaf
7//! yet and rewrites it into explicit slots plus ordinary tensor factors. The
8//! parser can then recurse on the resulting expression and build the same graph
9//! it would have built from fully expanded syntax.
10//!
11//! The main Schoonschip convention is:
12//! 1. a tensor with one compact axis, such as `p(rep)`, used as a function
13//!    argument becomes a fresh slot in that argument position; any explicit
14//!    spectator slots on the tensor are preserved;
15//! 2. the tensor `p(slot)` is multiplied next to the rebuilt function;
16//! 3. compact scalar products `g(p(rep), q(rep.dual()))` and
17//!    `dot(p(rep), q(rep.dual()))` share one fresh abstract index and become
18//!    `p(rep(index)) * q(rep.dual()(index))`.
19//!
20//! Additional factors are accumulated beside the current atom. Compact scalar
21//! factors nested inside a compact vector wrapper are recursively materialized
22//! before ordinary parsing so no shorthand is left hidden inside a tensor leaf.
23//!
24//! Chain and trace materialization chooses the symbolic `in`/`out` replacements,
25//! then lets this Schoonschip helper expand compact arguments inside each factor.
26

56struct SchoonschipMaterialization {
57    /// Expression that remains at the original syntactic position.
58    current: Atom,
59    /// Extra factors to multiply beside `current`.
60    ///
61    /// These factors are explicit tensor syntax after any nested compact scalar
62    /// shorthand has been recursively materialized.
63    additional_factors: Vec<Atom>,
64}
65
66struct ChainEndpoint<Aind> {
67    slot: Slot<LibraryRep, Aind>,
68    additional_factors: Vec<Atom>,
69}

71impl SchoonschipMaterialization {
72    /// Merge the current atom and accumulated factors into parser input.
73    fn into_expression(self) -> Atom {
74        self.additional_factors
75            .into_iter()
76            .fold(self.current, |expression, factor| expression * factor)
77    }
78}

80/// Lowers expandable shorthand atoms into ordinary parser syntax.
81///
82/// The materializer owns no parse state. It borrows the parser's dummy allocator
83/// so every fresh slot it creates shares the same abstract-index namespace as
84/// the surrounding network parse.
85pub(super) struct SchoonschipMaterializer<'a, Aind> {
86    state: &'a ParseState<Aind>,
87    mode: SchoonschipExpansionMode,
88}

90impl<'a, Aind: AbsInd + DummyAind + ParseableAind> SchoonschipMaterializer<'a, Aind> {
91    /// Build a materializer with explicit Schoonschip expansion controls.
92    pub(super) fn with_mode(state: &'a ParseState<Aind>, mode: SchoonschipExpansionMode) -> Self {
93        Self { state, mode }
94    }

96    /// Return true when this function root contains compact Schoonschip syntax.
97    ///
98    /// This is the non-allocating counterpart to `materialize_shorthand`: compact
99    /// scalar products are shorthand at the root, while compact vectors become
100    /// shorthand only when they occur as arguments of another function.
101    pub(super) fn contains_schoonschip_shorthand(value: AtomView<'_>) -> bool {
102        let AtomView::Fun(fun) = value else {
103            return false;
104        };
105
106        Self::is_compact_scalar_product(fun)
107            || fun.iter().any(Self::contains_schoonschip_shorthand_arg)
108    }
109
110    /// Materialize an expandable shorthand expression, or return it unchanged.
111    ///
112    /// The root must be a function for rewriting to occur. When rewriting is
113    /// possible, the result is one expression where the rebuilt root is
114    /// multiplied by all tensor factors introduced while replacing compact
115    /// vector arguments. If no shorthand is present, callers still get a valid
116    /// parser input: the original atom.
117    pub(super) fn materialize_shorthand(&self, value: AtomView<'_>) -> Atom {
118        self.materialize_shorthand_root(value)
119            .map(SchoonschipMaterialization::into_expression)
120            .unwrap_or_else(|| value.to_owned())
121    }
122
123    /// Materialize shorthand at a function root.
124    ///
125    /// Compact scalar products get their special lowering first. Everything else
126    /// is rebuilt by recursively scanning function arguments.
127    fn materialize_shorthand_root(
128        &self,
129        value: AtomView<'_>,
130    ) -> Option<SchoonschipMaterialization> {
131        let AtomView::Fun(fun) = value else {
132            return None;
133        };
134
135        if Self::is_chain_like_head(fun) && !self.mode.expand_inside_chains {
136            return None;
137        }
138
139        if Self::is_inner_product_head(fun) {
140            if !self.mode.inner_products {
141                return None;
142            }
143            if let Some(materialized) = self.compact_scalar_product(fun) {
144                return Some(materialized);
145            }
146        }
147
148        self.materialize_shorthand_function(fun)
149    }
150
151    /// Materialize one function argument according to shorthand position rules.
152    ///
153    /// A compact vector has special meaning only in argument position: it
154    /// contributes a fresh slot at that position and an extra rank-one tensor
155    /// factor. Non-compact functions are scanned recursively.
156    fn materialize_shorthand_arg(&self, value: AtomView<'_>) -> Option<SchoonschipMaterialization> {
157        if self.mode.expand_schoonship
158            && let Some(materialized) = self.materialize_compact_vector_arg(value)
159        {
160            return Some(materialized);
161        }
162
163        self.materialize_shorthand_root(value)
164    }
165
166    fn contains_schoonschip_shorthand_arg(value: AtomView<'_>) -> bool {
167        Self::compact_vector_rep(value).is_some() || Self::contains_schoonschip_shorthand(value)
168    }

170    /// Materialize one compact vector argument as `slot` plus `vector(slot)`.
171    ///
172    /// The compact vector may be a single function or a sum of functions, but
173    /// every visible compact representation must be the same representation.
174    fn materialize_compact_vector_arg(
175        &self,
176        value: AtomView<'_>,
177    ) -> Option<SchoonschipMaterialization> {
178        let rep = Self::compact_vector_rep(value)?;
179        let slot = self.state.slot(&rep).to_atom();
180        let factor = self.materialize_compact_vector_with_slot(value, &rep, &slot)?;
181        tracing::debug!(
182            target: "spenso::network::parsing",
183            spenso_parser = true,
184            generation = true,
185            compile = true,
186            inspect = true,
187            stage = "schoonschip_compact_vector_slot",
188            slot = %slot.to_plain_string(),
189            representation = %rep,
190            file.value = %value.to_plain_string(),
191            file.factor = %factor.to_plain_string(),
192            "Spenso parser allocated Schoonschip compact vector slot"
193        );
194
195        Some(SchoonschipMaterialization {
196            current: slot,
197            additional_factors: vec![factor],
198        })
199    }
200
201    /// Rebuild a function after materializing any shorthand arguments.
202    ///
203    /// Introduced factors are accumulated next to the rebuilt function. If the
204    /// rebuilt function is `dot(slot_i, slot_j)`, it is normalized to the metric
205    /// spelling expected by the tensor library.
206    fn materialize_shorthand_function(
207        &self,
208        fun: FunView<'_>,
209    ) -> Option<SchoonschipMaterialization> {
210        let mut changed = false;
211        let mut additional_factors = Vec::new();
212        let mut rebuilt = FunctionBuilder::new(fun.get_symbol());
213
214        for arg in fun.iter() {
215            if let Some(materialized) = self.materialize_shorthand_arg(arg) {
216                changed = true;
217                rebuilt = rebuilt.add_arg(&materialized.current);
218                additional_factors.extend(materialized.additional_factors);
219            } else {
220                rebuilt = rebuilt.add_arg(arg);
221            }
222        }
223
224        changed.then(|| {
225            let replacement = rebuilt.finish();
226            SchoonschipMaterialization {
227                current: Self::compact_dot_as_metric(replacement.as_view()).unwrap_or(replacement),
228                additional_factors,
229            }
230        })
231    }
232
233    /// Materialize a compact metric or dot product into two tensor factors.
234    ///
235    /// Both arguments must have one compact axis with matching representations.
236    /// Assign one fresh abstract index with each argument's orientation. Explicit
237    /// spectator slots remain unchanged, including spectators in that representation.
238    fn compact_scalar_product(&self, value: FunView<'_>) -> Option<SchoonschipMaterialization> {
239        let (lhs, rhs, lhs_rep, rhs_rep) = Self::compact_scalar_product_parts(value)?;
240
241        let lhs_slot = self.state.slot(&lhs_rep);
242        let rhs_slot = rhs_rep.slot::<Aind, _>(lhs_slot.aind());
243        let lhs = self.materialize_compact_vector_with_slot(lhs, &lhs_rep, &lhs_slot.to_atom())?;
244        let rhs = self.materialize_compact_vector_with_slot(rhs, &rhs_rep, &rhs_slot.to_atom())?;
245        tracing::debug!(
246            target: "spenso::network::parsing",
247            spenso_parser = true,
248            generation = true,
249            compile = true,
250            inspect = true,
251            stage = "schoonschip_compact_scalar_product_slot",
252            lhs_slot = %lhs_slot.to_atom().to_plain_string(),
253            rhs_slot = %rhs_slot.to_atom().to_plain_string(),
254            lhs_representation = %lhs_rep,
255            rhs_representation = %rhs_rep,
256            file.value = %value.as_view().to_plain_string(),
257            file.lhs = %lhs.to_plain_string(),
258            file.rhs = %rhs.to_plain_string(),
259            "Spenso parser allocated Schoonschip scalar-product slot"
260        );
261
262        Some(SchoonschipMaterialization {
263            current: Atom::num(1),
264            additional_factors: vec![lhs, rhs],
265        })
266    }

268    fn is_compact_scalar_product(value: FunView<'_>) -> bool {
269        Self::compact_scalar_product_parts(value).is_some()
270    }

272    fn is_inner_product_head(value: FunView<'_>) -> bool {
273        let symbol = value.get_symbol();
274        symbol == ETS.metric || symbol == SPENSO_TAG.dot
275    }

277    fn is_chain_like_head(value: FunView<'_>) -> bool {
278        let symbol = value.get_symbol();
279        symbol == SPENSO_TAG.chain || symbol == SPENSO_TAG.trace
280    }

282    fn is_structured_scalar(value: AtomView<'_>) -> bool {
283        if let AtomView::Fun(fun) = value
284            && Self::is_compact_scalar_product(fun)
285        {
286            return true;
287        }
288
289        value
290            .infer_structure::<OrderedStructure<LibraryRep, Aind>>(StructureInferenceMode::Fast)
291            .is_ok_and(|structure| structure.canonical().is_scalar())
292    }
293
294    fn compact_scalar_product_parts(
295        value: FunView<'_>,
296    ) -> Option<(
297        AtomView<'_>,
298        AtomView<'_>,
299        Representation<LibraryRep>,
300        Representation<LibraryRep>,
301    )> {
302        if !Self::is_inner_product_head(value) {
303            return None;
304        }
305
306        let args = value.iter().collect::<Vec<_>>();
307        let [lhs, rhs] = args.as_slice() else {
308            return None;
309        };
310
311        let lhs_rep = Self::compact_vector_rep(*lhs)?;
312        let rhs_rep = Self::compact_vector_rep(*rhs)?;
313        if !lhs_rep.matches(&rhs_rep) {
314            return None;
315        }
316
317        Some((*lhs, *rhs, lhs_rep, rhs_rep))
318    }

320    /// Infer the unique compact axis carried by a shorthand atom.
321    ///
322    /// Tensor-tagged functions expose a compact representation through exactly
323    /// one direct representation argument. Scalar products, unary broadcasts,
324    /// projectors, and sums preserve it when they contain one compatible vector.
325    /// Every other product factor must be syntactically scalar.
326    fn compact_vector_rep(value: AtomView<'_>) -> Option<Representation<LibraryRep>> {
327        match value {
328            AtomView::Fun(fun) if fun.get_symbol().has_tag(&SPENSO_TAG.broadcast) => {
329                let args = fun.iter().collect::<Vec<_>>();
330                let [argument] = args.as_slice() else {
331                    return None;
332                };
333                Self::compact_vector_rep(*argument)
334            }
335            AtomView::Fun(fun)
336                if fun.get_symbol() == *shadowing::SYM
337                    || fun.get_symbol() == *shadowing::ANTISYM
338                    || fun.get_symbol() == *shadowing::CYCLIC =>
339            {
340                let mut candidates = fun.iter().filter_map(|argument| {
341                    Self::compact_vector_rep(argument).map(|rep| (argument, rep))
342                });
343                let (argument, rep) = candidates.next()?;
344                if candidates.next().is_some()
345                    || fun.iter().any(|candidate| {
346                        candidate != argument
347                            && candidate.is_tensorial(StrictTensorFilter::Tagged)
348                            && !Self::is_structured_scalar(candidate)
349                    })
350                {
351                    None
352                } else {
353                    Some(rep)
354                }
355            }
356            AtomView::Fun(fun) => Self::compact_tensor_rep_arg(fun).map(|(_, rep)| rep),
357            AtomView::Add(add) => {
358                let mut reps = add.iter().map(Self::compact_vector_rep);
359                let rep = reps.next()??;
360                reps.all(|candidate| candidate == Some(rep)).then_some(rep)
361            }
362            AtomView::Mul(product) => Self::compact_vector_product_parts(product)
363                .map(|(_, representation)| representation),
364            _ => None,
365        }
366    }
367
368    /// Locate the only compact vector in a scalar-weighted product.
369    ///
370    /// Failure to identify a compact vector does not prove that a factor is
371    /// scalar: explicit-slot tensors also have no compact representation. The
372    /// ordinary syntactic structure inference is therefore the authority for
373    /// every remaining factor.
374    fn compact_vector_product_parts(
375        product: MulView<'_>,
376    ) -> Option<(usize, Representation<LibraryRep>)> {
377        let mut compact_vector = None;
378
379        for (position, factor) in product.iter().enumerate() {
380            if let Some(representation) = Self::compact_vector_rep(factor) {
381                if compact_vector.is_some() {
382                    return None;
383                }
384                compact_vector = Some((position, representation));
385            } else if !OrderedStructure::<LibraryRep, Aind>::syntactic_structure_from_atom(factor)
386                .ok()?
387                .is_scalar()
388            {
389                return None;
390            }
391        }
392
393        compact_vector
394    }

396    /// Locate the compact representation argument of one tensor function.
397    ///
398    /// A compact tensor function is tensor-tagged, is not itself a representation,
399    /// metric or dot product, and has exactly one direct representation argument.
400    /// Explicit slots are spectators: only that unique unindexed axis is replaced.
401    fn compact_tensor_rep_arg(value: FunView<'_>) -> Option<(usize, Representation<LibraryRep>)> {
402        if !value.get_symbol().has_tag(&SPENSO_TAG.tensor)
403            || value.get_symbol() == ETS.metric
404            || value.get_symbol() == SPENSO_TAG.dot
405        {
406            return None;
407        }
408
409        if Representation::<LibraryRep>::try_from(value.as_view()).is_ok() {
410            return None;
411        }
412
413        let args = value.iter().collect::<Vec<_>>();
414        let rep_args = args
415            .iter()
416            .enumerate()
417            .filter_map(|(position, arg)| {
418                if Slot::<LibraryRep, Aind>::try_from(*arg).is_ok() {
419                    None
420                } else {
421                    Self::compact_rep_pattern_match(*arg).map(|rep| (position, rep))
422                }
423            })
424            .collect::<Vec<_>>();
425
426        let [(position, rep)] = rep_args.as_slice() else {
427            return None;
428        };
429
430        Some((*position, *rep))
431    }

433    /// Match one argument against the symbolic representation wildcard.
434    ///
435    /// Matching is restricted to the argument itself (`max_level = 0`) so a
436    /// nested representation inside metadata does not accidentally become the
437    /// tensor's compact slot.
438    fn compact_rep_pattern_match(arg: AtomView<'_>) -> Option<Representation<LibraryRep>> {
439        if let Ok(representation) = Representation::<LibraryRep>::try_from(arg) {
440            return Some(representation);
441        }
442
443        let rep_pattern = Atom::var(SPENSO_TAG.rep_).to_pattern();
444        let settings = MatchSettings::new().max_level(0).partial(false);
445        let mut matches = arg.pattern_match(&rep_pattern, None, Some(&settings));
446        let matched = matches.next_detailed()?;
447        let rep = rep_pattern.replace_wildcards_with_matches(matched.match_stack);
448        Representation::<LibraryRep>::try_from(rep.as_view()).ok()
449    }

451    /// Replace a compact representation with a concrete slot.
452    ///
453    /// For a function, this rebuilds the function with the matched compact
454    /// representation argument replaced by `slot`. For a sum, every summand is
455    /// rebuilt with the same slot so the expansion keeps a single dummy edge. A
456    /// scalar-weighted product preserves its scalar factors around that rebuilt
457    /// vector.
458    fn materialize_compact_vector_with_slot(
459        &self,
460        value: AtomView<'_>,
461        rep: &Representation<LibraryRep>,
462        slot: &Atom,
463    ) -> Option<Atom> {
464        match value {
465            AtomView::Fun(fun) if fun.get_symbol().has_tag(&SPENSO_TAG.broadcast) => {
466                let args = fun.iter().collect::<Vec<_>>();
467                let [argument] = args.as_slice() else {
468                    return None;
469                };
470                Some(
471                    FunctionBuilder::new(fun.get_symbol())
472                        .add_arg(self.materialize_compact_vector_with_slot(*argument, rep, slot)?)
473                        .finish(),
474                )
475            }
476            AtomView::Fun(fun)
477                if fun.get_symbol() == *shadowing::SYM
478                    || fun.get_symbol() == *shadowing::ANTISYM
479                    || fun.get_symbol() == *shadowing::CYCLIC =>
480            {
481                let mut changed = false;
482                let mut rebuilt = FunctionBuilder::new(fun.get_symbol());
483                for argument in fun.iter() {
484                    if Self::compact_vector_rep(argument) == Some(*rep) {
485                        if changed {
486                            return None;
487                        }
488                        changed = true;
489                        rebuilt = rebuilt.add_arg(
490                            self.materialize_compact_vector_with_slot(argument, rep, slot)?,
491                        );
492                    } else {
493                        rebuilt = rebuilt.add_arg(argument);
494                    }
495                }
496                changed.then(|| rebuilt.finish())
497            }
498            AtomView::Fun(fun) => {
499                let (position, matched_rep) = Self::compact_tensor_rep_arg(fun)?;
500                if matched_rep != *rep {
501                    return None;
502                }
503
504                let mut tensor = FunctionBuilder::new(fun.get_symbol());
505                for (arg_position, arg) in fun.iter().enumerate() {
506                    if arg_position == position {
507                        tensor = tensor.add_arg(slot);
508                    } else {
509                        tensor = tensor.add_arg(arg);
510                    }
511                }
512                Some(tensor.finish())
513            }
514            AtomView::Add(add) => {
515                let mut terms = add
516                    .iter()
517                    .map(|term| self.materialize_compact_vector_with_slot(term, rep, slot));
518                let first = terms.next()??;
519                let rest = terms.collect::<Option<Vec<_>>>()?;
520                Some(rest.into_iter().fold(first, |sum, term| sum + term))
521            }
522            AtomView::Mul(product) => {
523                let (vector_position, matched_rep) = Self::compact_vector_product_parts(product)?;
524                if matched_rep != *rep {
525                    return None;
526                }
527
528                product
529                    .iter()
530                    .enumerate()
531                    .try_fold(Atom::num(1), |product, (position, factor)| {
532                        if position == vector_position {
533                            self.materialize_compact_vector_with_slot(factor, rep, slot)
534                                .map(|factor| product * factor)
535                        } else {
536                            let scalar = self
537                                .materialize_shorthand_root(factor)
538                                .map(SchoonschipMaterialization::into_expression)
539                                .unwrap_or_else(|| factor.to_owned());
540                            Some(product * scalar)
541                        }
542                    })
543            }
544            _ => None,
545        }
546    }
547
548    /// Normalize `dot(slot_i, slot_j)` to `g(slot_i, slot_j)`.
549    ///
550    /// This only applies after arguments have already been materialized to
551    /// concrete slots. Non-slot dot products are left untouched.
552    fn compact_dot_as_metric(value: AtomView<'_>) -> Option<Atom> {
553        let AtomView::Fun(fun) = value else {
554            return None;
555        };
556
557        if fun.get_symbol() != SPENSO_TAG.dot || fun.get_nargs() != 2 {
558            return None;
559        }
560
561        let args = fun.iter().collect::<Vec<_>>();
562        if args
563            .iter()
564            .all(|arg| Slot::<LibraryRep, Aind>::try_from(*arg).is_ok())
565        {
566            Some(
567                FunctionBuilder::new(ETS.metric)
568                    .add_arg(args[0])
569                    .add_arg(args[1])
570                    .finish(),
571            )
572        } else {
573            None
574        }
575    }
576}

590    #[allow(clippy::result_large_err)]
591    pub(super) fn is_shorthand_function(value: FunView<'a>) -> bool {
592        let symbol = value.get_symbol();
593        symbol == SPENSO_TAG.chain
594            || symbol == SPENSO_TAG.trace
595            || symbol == SPENSO_TAG.dot
596            || SchoonschipMaterializer::<Aind>::contains_schoonschip_shorthand(value.as_view())
597    }
598
599    #[allow(clippy::result_large_err)]
600    pub(super) fn materialize_shorthand<S, Lib, FunLib>(
601        value: FunView<'a>,
602        state: ParseState<Aind>,
603        library: &Lib,
604        function_library: &FunLib,
605        settings: &ParseSettings,
606    ) -> Result<Self, TensorNetworkError<K, Symbol>>
607    where
608        S: TensorStructure + ScalarStructure + Clone + StructureFromAtom,
609        TensorShell<S>: Concretize<T>,
610        S::Slot: IsAbstractSlot<Aind = Aind>,
611        T::Slot: IsAbstractSlot<Aind = Aind>,
612        T: TensorFromExpression<S, Sc, K, Symbol, Aind, Lib, FunLib>,
613        Lib: TensorLibraryFor<S, T, Key = K>,
614        FunLib: FunctionLibrary<T, Sc, Key = Symbol>,
615    {
616        let symbol = value.get_symbol();
617        let root_chain_disabled =
618            symbol == SPENSO_TAG.chain && !settings.shorthand_parsing.expands_chain();
619        let root_trace_disabled =
620            symbol == SPENSO_TAG.trace && !settings.shorthand_parsing.expands_trace();
621
622        if ((symbol == SPENSO_TAG.chain && !root_chain_disabled)
623            || (symbol == SPENSO_TAG.trace && !root_trace_disabled))
624            && let Some(expanded) = shadowing::expand_chain_like_projector(value.as_view())
625        {
626            return Self::try_from_view_impl(
627                expanded.as_view(),
628                state,
629                library,
630                function_library,
631                settings,
632            );
633        }
634
635        if symbol == SPENSO_TAG.chain && !root_chain_disabled {
636            return Self::materialize_chain_shorthand(
637                value,
638                state,
639                library,
640                function_library,
641                settings,
642            );
643        }
644
645        if symbol == SPENSO_TAG.trace && !root_trace_disabled {
646            return Self::materialize_trace_shorthand(
647                value,
648                state,
649                library,
650                function_library,
651                settings,
652            );
653        }
654
655        let has_schoonschip_shorthand =
656            SchoonschipMaterializer::<Aind>::contains_schoonschip_shorthand(value.as_view());
657        let schoonschip_mode = settings
658            .shorthand_parsing
659            .schoonschip_expansion()
660            .unwrap_or_else(SchoonschipExpansionMode::none);
661        let effective_schoonschip_mode = if root_chain_disabled || root_trace_disabled {
662            schoonschip_mode.for_chain_like_root()
663        } else {
664            schoonschip_mode
665        };
666        let materialized = if effective_schoonschip_mode.any() {
667            SchoonschipMaterializer::<Aind>::with_mode(&state, effective_schoonschip_mode)
668                .materialize_shorthand(value.as_view())
669        } else {
670            value.as_view().to_owned()
671        };
672
673        if materialized == value.as_view().to_owned() {
674            // The atom rewriter is at a fixed point; recurse only after an actual rewrite.
675            if root_chain_disabled || root_trace_disabled || has_schoonschip_shorthand {
676                return Self::as_inferred_leaf::<S, Lib, FunLib>(
677                    value.as_view(),
678                    super::StructureInferenceMode::Fast,
679                    library,
680                    function_library,
681                    settings,
682                );
683            }
684            return Self::parse_regular_function_leaf::<S, Lib>(value, library);
685        }
686
687        Self::try_from_view_impl(
688            materialized.as_view(),
689            state,
690            library,
691            function_library,
692            settings,
693        )
694    }

696    fn materialize_chain_endpoint(
697        value: AtomView<'a>,
698        label: &str,
699        state: &ParseState<Aind>,
700        schoonschip_mode: SchoonschipExpansionMode,
701    ) -> Result<ChainEndpoint<Aind>, TensorNetworkError<K, Symbol>> {
702        match Slot::<LibraryRep, Aind>::try_from(value) {
703            Ok(slot) => Ok(ChainEndpoint {
704                slot,
705                additional_factors: Vec::new(),
706            }),
707            Err(slot_err) => {
708                if schoonschip_mode.any()
709                    && let Some(materialized) =
710                        SchoonschipMaterializer::<Aind>::with_mode(state, schoonschip_mode)
711                            .materialize_shorthand_arg(value)
712                {
713                    let slot =
714                        match Slot::<LibraryRep, Aind>::try_from(materialized.current.as_view()) {
715                            Ok(slot) => slot,
716                            Err(err) => {
717                                return Err(eyre!(
718                                    "invalid materialized chain {label} `{}` from `{}`: {err}",
719                                    materialized.current,
720                                    value
721                                )
722                                .into());
723                            }
724                        };
725                    return Ok(ChainEndpoint {
726                        slot,
727                        additional_factors: materialized.additional_factors,
728                    });
729                }
730
731                Err(eyre!("invalid chain {label} `{}`: {slot_err}", value).into())
732            }
733        }
734    }

736    #[allow(clippy::result_large_err)]
737    fn materialize_chain_shorthand<S, Lib, FunLib>(
738        value: FunView<'a>,
739        state: ParseState<Aind>,
740        library: &Lib,
741        function_library: &FunLib,
742        settings: &ParseSettings,
743    ) -> Result<Self, TensorNetworkError<K, Symbol>>
744    where
745        S: TensorStructure + ScalarStructure + Clone + StructureFromAtom,
746        TensorShell<S>: Concretize<T>,
747        S::Slot: IsAbstractSlot<Aind = Aind>,
748        T::Slot: IsAbstractSlot<Aind = Aind>,
749        T: TensorFromExpression<S, Sc, K, Symbol, Aind, Lib, FunLib>,
750        Lib: TensorLibraryFor<S, T, Key = K>,
751        FunLib: FunctionLibrary<T, Sc, Key = Symbol>,
752    {
753        let args = value.iter().collect::<Vec<_>>();
754        if args.len() < 2 {
755            return Err(TensorNetworkError::TooManyArgsFunction(
756                value.as_view().to_plain_string(),
757            ));
758        }
759
760        let schoonschip_mode = settings
761            .shorthand_parsing
762            .schoonschip_expansion()
763            .unwrap_or_else(SchoonschipExpansionMode::none);
764        let ChainEndpoint {
765            slot: start,
766            additional_factors: start_factors,
767        } = Self::materialize_chain_endpoint(args[0], "start", &state, schoonschip_mode)?;
768        let ChainEndpoint {
769            slot: end,
770            additional_factors: end_factors,
771        } = Self::materialize_chain_endpoint(args[1], "end", &state, schoonschip_mode)?;
772        let factors = &args[2..];
773
774        let factor_schoonschip_mode = schoonschip_mode.for_chain_like_root();
775        let factor_settings = settings
776            .clone()
777            .with_schoonschip_expansion(factor_schoonschip_mode);
778
779        let mut factor_networks = Vec::new();
780        for factor in start_factors.into_iter().chain(end_factors) {
781            factor_networks.extend(Self::parse_chain_like_factor_networks::<S, Lib, FunLib>(
782                factor,
783                state.clone(),
784                library,
785                function_library,
786                settings,
787            )?);
788        }
789
790        if factors.is_empty() {
791            let metric = FunctionBuilder::new(ETS.metric)
792                .add_arg(start.to_atom())
793                .add_arg(end.to_atom())
794                .finish();
795            factor_networks.extend(Self::parse_chain_like_factor_networks::<S, Lib, FunLib>(
796                metric,
797                state,
798                library,
799                function_library,
800                settings,
801            )?);
802            return Ok(Self::chain_like_network_product(factor_networks));
803        }
804
805        let mut left = start;
806        for (position, factor) in factors.iter().enumerate() {
807            let fresh_right = position + 1 != factors.len();
808            let right = if position + 1 == factors.len() {
809                end.dual()
810            } else {
811                state.slot(&left.rep)
812            };
813            if fresh_right {
814                tracing::debug!(
815                    target: "spenso::network::parsing",
816                    spenso_parser = true,
817                    generation = true,
818                    compile = true,
819                    inspect = true,
820                    stage = "chain_shorthand_link_slot",
821                    position,
822                    factor_count = factors.len(),
823                    left = %left,
824                    right = %right,
825                    right_atom = %right.to_atom().to_plain_string(),
826                    start = %start,
827                    end = %end,
828                    file.factor = %factor.to_plain_string(),
829                    "Spenso parser allocated chain shorthand link slot"
830                );
831            }
832            let factor = ChainExpansion::replace_placeholders(
833                *factor,
834                &left.to_atom(),
835                &right.dual().to_atom(),
836            );
837            let factor = if factor_schoonschip_mode.any() {
838                SchoonschipMaterializer::<Aind>::with_mode(&state, factor_schoonschip_mode)
839                    .materialize_shorthand(factor.as_view())
840            } else {
841                factor
842            };
843            factor_networks.extend(Self::parse_chain_like_factor_networks::<S, Lib, FunLib>(
844                factor,
845                state.clone(),
846                library,
847                function_library,
848                &factor_settings,
849            )?);
850            left = right;
851        }
852
853        Ok(Self::chain_like_network_product(factor_networks))
854    }

856    #[allow(clippy::result_large_err)]
857    fn materialize_trace_shorthand<S, Lib, FunLib>(
858        value: FunView<'a>,
859        state: ParseState<Aind>,
860        library: &Lib,
861        function_library: &FunLib,
862        settings: &ParseSettings,
863    ) -> Result<Self, TensorNetworkError<K, Symbol>>
864    where
865        S: TensorStructure + ScalarStructure + Clone + StructureFromAtom,
866        TensorShell<S>: Concretize<T>,
867        S::Slot: IsAbstractSlot<Aind = Aind>,
868        T::Slot: IsAbstractSlot<Aind = Aind>,
869        T: TensorFromExpression<S, Sc, K, Symbol, Aind, Lib, FunLib>,
870        Lib: TensorLibraryFor<S, T, Key = K>,
871        FunLib: FunctionLibrary<T, Sc, Key = Symbol>,
872    {
873        let args = value.iter().collect::<Vec<_>>();
874        let Some(rep_view) = args.first() else {
875            return Err(TensorNetworkError::TooManyArgsFunction(
876                value.as_view().to_plain_string(),
877            ));
878        };
879
880        let rep = Representation::<LibraryRep>::try_from(*rep_view)
881            .map_err(|err| eyre!("invalid trace representation `{rep_view}`: {err}"))?;
882        let factors = shadowing::trace_factor_views(&args[1..]);
883
884        if factors.is_empty() {
885            return Self::try_from_view_impl(
886                rep.dim.to_symbolic().as_view(),
887                state,
888                library,
889                function_library,
890                settings,
891            );
892        }
893
894        let links = (0..factors.len())
895            .map(|position| {
896                let slot = state.slot(&rep);
897                tracing::debug!(
898                    target: "spenso::network::parsing",
899                    spenso_parser = true,
900                    generation = true,
901                    compile = true,
902                    inspect = true,
903                    stage = "trace_shorthand_link_slot",
904                    position,
905                    factor_count = factors.len(),
906                    slot = %slot,
907                    slot_atom = %slot.to_atom().to_plain_string(),
908                    representation = %rep,
909                    file.factor = %factors[position].to_plain_string(),
910                    "Spenso parser allocated trace shorthand link slot"
911                );
912                slot
913            })
914            .collect::<Vec<_>>();
915        let factor_schoonschip_mode = settings
916            .shorthand_parsing
917            .schoonschip_expansion()
918            .unwrap_or_else(SchoonschipExpansionMode::none)
919            .for_chain_like_root();
920        let factor_settings = settings
921            .clone()
922            .with_schoonschip_expansion(factor_schoonschip_mode);
923
924        let materialized_factors = factors
925            .iter()
926            .enumerate()
927            .map(|(position, factor)| {
928                let left = links[position].to_atom();
929                let right = links[(position + 1) % factors.len()].dual().to_atom();
930                let factor = ChainExpansion::replace_placeholders(*factor, &left, &right);
931                if factor_schoonschip_mode.any() {
932                    SchoonschipMaterializer::<Aind>::with_mode(&state, factor_schoonschip_mode)
933                        .materialize_shorthand(factor.as_view())
934                } else {
935                    factor
936                }
937            })
938            .collect::<Vec<_>>();
939
940        let mut factor_networks = Vec::new();
941        for factor in materialized_factors {
942            factor_networks.extend(Self::parse_chain_like_factor_networks::<S, Lib, FunLib>(
943                factor,
944                state.clone(),
945                library,
946                function_library,
947                &factor_settings,
948            )?);
949        }
950        Ok(Self::chain_like_network_product(factor_networks))
951    }

953    #[allow(clippy::result_large_err)]
954    fn parse_chain_like_factor_networks<S, Lib, FunLib>(
955        factor: Atom,
956        state: ParseState<Aind>,
957        library: &Lib,
958        function_library: &FunLib,
959        settings: &ParseSettings,
960    ) -> Result<Vec<Self>, TensorNetworkError<K, Symbol>>
961    where
962        S: TensorStructure + ScalarStructure + Clone + StructureFromAtom,
963        TensorShell<S>: Concretize<T>,
964        S::Slot: IsAbstractSlot<Aind = Aind>,
965        T::Slot: IsAbstractSlot<Aind = Aind>,
966        T: TensorFromExpression<S, Sc, K, Symbol, Aind, Lib, FunLib>,
967        Lib: TensorLibraryFor<S, T, Key = K>,
968        FunLib: FunctionLibrary<T, Sc, Key = Symbol>,
969    {
970        let AtomView::Mul(product) = factor.as_view() else {
971            return Ok(vec![Self::try_from_view_impl(
972                factor.as_view(),
973                state.clone(),
974                library,
975                function_library,
976                settings,
977            )?]);
978        };
979
980        let mut scalars = Vec::new();
981        let mut tensors = Vec::new();
982        for arg in product.iter() {
983            let network =
984                Self::try_from_view_impl(arg, state.clone(), library, function_library, settings)?;
985            if network.state == NetworkState::PureScalar {
986                scalars.push(network);
987            } else {
988                tensors.push(network);
989            }
990        }
991
992        if scalars.is_empty() || tensors.len() != 1 {
993            return Ok(vec![Self::try_from_view_impl(
994                factor.as_view(),
995                state.clone(),
996                library,
997                function_library,
998                settings,
999            )?]);
1000        }
1001
1002        scalars.extend(tensors);
1003        Ok(scalars)
1004    }

1006    fn chain_like_network_product(networks: Vec<Self>) -> Self {
1007        let mut networks = networks.into_iter();
1008        let first = networks.next().unwrap();
1009        let rest = networks.collect::<Vec<_>>();
1010        if rest.is_empty() {
1011            first
1012        } else {
1013            first.n_mul(rest)
1014        }
1015    }

1018/// Utilities for lowering chain and trace shorthand factors.
1019///
1020/// Chain parsing chooses the concrete left and right slots for each factor.
1021/// These helpers rewrite the symbolic placeholders in the factor expression
1022/// before normal parsing resumes.
1023pub(super) struct ChainExpansion;
1024
1025impl ChainExpansion {
1026    /// Replace `in` and `out` placeholders recursively.
1027    ///
1028    /// Nested chain/trace heads are rejected before lowering, so the complete
1029    /// factor belongs to the current placeholder consumer. This walk only
1030    /// substitutes its selected endpoints and never allocates dummies.
1031    pub(super) fn replace_placeholders(
1032        value: AtomView<'_>,
1033        chain_in: &Atom,
1034        chain_out: &Atom,
1035    ) -> Atom {
1036        match value {
1037            AtomView::Var(var) if var.get_symbol() == SPENSO_TAG.chain_in => chain_in.clone(),
1038            AtomView::Var(var) if var.get_symbol() == SPENSO_TAG.chain_out => chain_out.clone(),
1039            AtomView::Fun(fun) => {
1040                let mut rebuilt = FunctionBuilder::new(fun.get_symbol());
1041                for arg in fun.iter() {
1042                    rebuilt = rebuilt.add_arg(Self::replace_placeholders(arg, chain_in, chain_out));
1043                }
1044                rebuilt.finish()
1045            }
1046            AtomView::Add(add) => add.iter().fold(Atom::Zero, |sum, term| {
1047                sum + Self::replace_placeholders(term, chain_in, chain_out)
1048            }),
1049            AtomView::Mul(mul) => mul.iter().fold(Atom::num(1), |product, factor| {
1050                product * Self::replace_placeholders(factor, chain_in, chain_out)
1051            }),
1052            AtomView::Pow(pow) => {
1053                let (base, exponent) = pow.get_base_exp();
1054                Self::replace_placeholders(base, chain_in, chain_out).pow(exponent.to_owned())
1055            }
1056            _ => value.to_owned(),
1057        }
1058    }
1059}
