1//! Materialization helpers for shorthand syntax before ordinary network parsing.
2//!
3//! Network-level shorthand materialization lives on the parser implementation:
4//! it handles chain and trace topology and returns parsed networks. The helper
5//! in this module is deliberately narrower than algebraic simplification. It
6//! takes compact Schoonschip syntax that cannot be parsed as an ordinary leaf
7//! yet and rewrites it into explicit slots plus ordinary tensor factors. The
8//! parser can then recurse on the resulting expression and build the same graph
9//! it would have built from fully expanded syntax.
10//!
11//! The main Schoonschip convention is:
12//! 1. a compact rank-one tensor `p(rep)` used as a function argument becomes a
13//!    fresh slot in that argument position;
14//! 2. the tensor `p(slot)` is multiplied next to the rebuilt function;
15//! 3. compact scalar products `g(p(rep), q(rep))` and `dot(p(rep), q(rep))`
16//!    share one fresh self-dual slot and become the product `p(slot) * q(slot)`.
17//!
18//! Additional factors are accumulated beside the current atom and are not
19//! recursively inspected by this materialization pass.
20//!
21//! Chain and trace materialization chooses the symbolic `in`/`out` replacements,
22//! then lets this Schoonschip helper expand compact arguments inside each factor.

53struct SchoonschipMaterialization {
54    /// Expression that remains at the original syntactic position.
55    current: Atom,
56    /// Extra factors to multiply beside `current`.
57    ///
58    /// These factors are already explicit tensor syntax, so the materializer
59    /// does not run its shorthand pattern search on them during this pass.
60    additional_factors: Vec<Atom>,
61}
62
63struct ChainEndpoint<Aind> {
64    slot: Slot<LibraryRep, Aind>,
65    additional_factors: Vec<Atom>,
66}
67
68impl SchoonschipMaterialization {
69    /// Merge the current atom and accumulated factors into parser input.
70    fn into_expression(self) -> Atom {
71        self.additional_factors
72            .into_iter()
73            .fold(self.current, |expression, factor| expression * factor)
74    }
75}

77/// Lowers expandable shorthand atoms into ordinary parser syntax.
78///
79/// The materializer owns no parse state. It borrows the parser's dummy allocator
80/// so every fresh slot it creates shares the same abstract-index namespace as
81/// the surrounding network parse.
82pub(super) struct SchoonschipMaterializer<'a, Aind> {
83    state: &'a ParseState<Aind>,
84    mode: SchoonschipExpansionMode,
85}

87impl<'a, Aind: AbsInd + DummyAind + ParseableAind> SchoonschipMaterializer<'a, Aind> {
88    /// Build a materializer with explicit Schoonschip expansion controls.
89    pub(super) fn with_mode(state: &'a ParseState<Aind>, mode: SchoonschipExpansionMode) -> Self {
90        Self { state, mode }
91    }
92
93    /// Return true when this function root contains compact Schoonschip syntax.
94    ///
95    /// This is the non-allocating counterpart to `materialize_shorthand`: compact
96    /// scalar products are shorthand at the root, while compact vectors become
97    /// shorthand only when they occur as arguments of another function.
98    pub(super) fn contains_schoonschip_shorthand(value: AtomView<'_>) -> bool {
99        let AtomView::Fun(fun) = value else {
100            return false;
101        };
102
103        Self::is_compact_scalar_product(fun)
104            || fun.iter().any(Self::contains_schoonschip_shorthand_arg)
105    }
106
107    /// Materialize an expandable shorthand expression, or return it unchanged.
108    ///
109    /// The root must be a function for rewriting to occur. When rewriting is
110    /// possible, the result is one expression where the rebuilt root is
111    /// multiplied by all tensor factors introduced while replacing compact
112    /// vector arguments. If no shorthand is present, callers still get a valid
113    /// parser input: the original atom.
114    pub(super) fn materialize_shorthand(&self, value: AtomView<'_>) -> Atom {
115        self.materialize_shorthand_root(value)
116            .map(SchoonschipMaterialization::into_expression)
117            .unwrap_or_else(|| value.to_owned())
118    }
119
120    /// Materialize shorthand at a function root.
121    ///
122    /// Compact scalar products get their special lowering first. Everything else
123    /// is rebuilt by recursively scanning function arguments.
124    fn materialize_shorthand_root(
125        &self,
126        value: AtomView<'_>,
127    ) -> Option<SchoonschipMaterialization> {
128        let AtomView::Fun(fun) = value else {
129            return None;
130        };
131
132        if Self::is_chain_like_head(fun) && !self.mode.expand_inside_chains {
133            return None;
134        }
135
136        if Self::is_inner_product_head(fun) {
137            if !self.mode.inner_products {
138                return None;
139            }
140            if let Some(materialized) = self.compact_scalar_product(fun) {
141                return Some(materialized);
142            }
143        }
144
145        self.materialize_shorthand_function(fun)
146    }
147
148    /// Materialize one function argument according to shorthand position rules.
149    ///
150    /// A compact vector has special meaning only in argument position: it
151    /// contributes a fresh slot at that position and an extra rank-one tensor
152    /// factor. Non-compact functions are scanned recursively.
153    fn materialize_shorthand_arg(&self, value: AtomView<'_>) -> Option<SchoonschipMaterialization> {
154        if self.mode.expand_schoonship
155            && let Some(materialized) = self.materialize_compact_vector_arg(value)
156        {
157            return Some(materialized);
158        }
159
160        self.materialize_shorthand_root(value)
161    }
162
163    fn contains_schoonschip_shorthand_arg(value: AtomView<'_>) -> bool {
164        Self::compact_vector_rep(value).is_some() || Self::contains_schoonschip_shorthand(value)
165    }
166
167    /// Materialize one compact vector argument as `slot` plus `vector(slot)`.
168    ///
169    /// The compact vector may be a single function or a sum of functions, but
170    /// every visible compact representation must be the same representation.
171    fn materialize_compact_vector_arg(
172        &self,
173        value: AtomView<'_>,
174    ) -> Option<SchoonschipMaterialization> {
175        let rep = Self::compact_vector_rep(value)?;
176        let slot = self.state.slot(&rep).to_atom();
177        let factor = Self::materialize_compact_vector_with_slot(value, &rep, &slot)?;
178        tracing::debug!(
179            target: "spenso::network::parsing",
180            spenso_parser = true,
181            generation = true,
182            compile = true,
183            inspect = true,
184            stage = "schoonschip_compact_vector_slot",
185            slot = %slot.to_plain_string(),
186            representation = %rep,
187            file.value = %value.to_plain_string(),
188            file.factor = %factor.to_plain_string(),
189            "Spenso parser allocated Schoonschip compact vector slot"
190        );
191
192        Some(SchoonschipMaterialization {
193            current: slot,
194            additional_factors: vec![factor],
195        })
196    }
197
198    /// Rebuild a function after materializing any shorthand arguments.
199    ///
200    /// Introduced factors are accumulated next to the rebuilt function. If the
201    /// rebuilt function is `dot(slot_i, slot_j)`, it is normalized to the metric
202    /// spelling expected by the tensor library.
203    fn materialize_shorthand_function(
204        &self,
205        fun: FunView<'_>,
206    ) -> Option<SchoonschipMaterialization> {
207        let mut changed = false;
208        let mut additional_factors = Vec::new();
209        let mut rebuilt = FunctionBuilder::new(fun.get_symbol());
210
211        for arg in fun.iter() {
212            if let Some(materialized) = self.materialize_shorthand_arg(arg) {
213                changed = true;
214                rebuilt = rebuilt.add_arg(&materialized.current);
215                additional_factors.extend(materialized.additional_factors);
216            } else {
217                rebuilt = rebuilt.add_arg(arg);
218            }
219        }
220
221        changed.then(|| {
222            let replacement = rebuilt.finish();
223            SchoonschipMaterialization {
224                current: Self::compact_dot_as_metric(replacement.as_view()).unwrap_or(replacement),
225                additional_factors,
226            }
227        })
228    }
229
230    /// Materialize a compact metric or dot product into two tensor factors.
231    ///
232    /// Both arguments must be compact vectors with the same self-dual
233    /// representation. They are assigned the same fresh slot, so
234    /// `g(p(rep), q(rep))` becomes `p(slot) * q(slot)`.
235    fn compact_scalar_product(&self, value: FunView<'_>) -> Option<SchoonschipMaterialization> {
236        let (lhs, rhs, rep) = Self::compact_scalar_product_parts(value)?;
237
238        let slot = self.state.slot(&rep).to_atom();
239        let lhs = Self::materialize_compact_vector_with_slot(lhs, &rep, &slot)?;
240        let rhs = Self::materialize_compact_vector_with_slot(rhs, &rep, &slot)?;
241        tracing::debug!(
242            target: "spenso::network::parsing",
243            spenso_parser = true,
244            generation = true,
245            compile = true,
246            inspect = true,
247            stage = "schoonschip_compact_scalar_product_slot",
248            slot = %slot.to_plain_string(),
249            representation = %rep,
250            file.value = %value.as_view().to_plain_string(),
251            file.lhs = %lhs.to_plain_string(),
252            file.rhs = %rhs.to_plain_string(),
253            "Spenso parser allocated Schoonschip scalar-product slot"
254        );
255
256        Some(SchoonschipMaterialization {
257            current: Atom::num(1),
258            additional_factors: vec![lhs, rhs],
259        })
260    }
261
262    fn is_compact_scalar_product(value: FunView<'_>) -> bool {
263        Self::compact_scalar_product_parts(value).is_some()
264    }
265
266    fn is_inner_product_head(value: FunView<'_>) -> bool {
267        let symbol = value.get_symbol();
268        symbol == ETS.metric || symbol == SPENSO_TAG.dot
269    }
270
271    fn is_chain_like_head(value: FunView<'_>) -> bool {
272        let symbol = value.get_symbol();
273        symbol == SPENSO_TAG.chain || symbol == SPENSO_TAG.trace
274    }
275
276    fn compact_scalar_product_parts(
277        value: FunView<'_>,
278    ) -> Option<(AtomView<'_>, AtomView<'_>, Representation<LibraryRep>)> {
279        if !Self::is_inner_product_head(value) {
280            return None;
281        }
282
283        let args = value.iter().collect::<Vec<_>>();
284        let [lhs, rhs] = args.as_slice() else {
285            return None;
286        };
287
288        let rep = Self::compact_vector_rep(*lhs)?;
289        if rep != Self::compact_vector_rep(*rhs)? || !rep.rep.is_self_dual() {
290            return None;
291        }
292
293        Some((*lhs, *rhs, rep))
294    }
295
296    /// Infer the compact representation carried by a rank-one shorthand atom.
297    ///
298    /// Functions expose a compact representation through exactly one direct
299    /// representation argument. Sums are accepted only when every summand exposes
300    /// the same representation.
301    fn compact_vector_rep(value: AtomView<'_>) -> Option<Representation<LibraryRep>> {
302        match value {
303            AtomView::Fun(fun) => Self::compact_tensor_rep_arg(fun).map(|(_, rep)| rep),
304            AtomView::Add(add) => {
305                let mut reps = add.iter().map(Self::compact_vector_rep);
306                let rep = reps.next()??;
307                reps.all(|candidate| candidate == Some(rep)).then_some(rep)
308            }
309            _ => None,
310        }
311    }
312
313    /// Locate the compact representation argument of one tensor function.
314    ///
315    /// A compact vector function is not itself a representation, is not a metric
316    /// or dot product, has no explicit slot argument, and has exactly one direct
317    /// argument matching the representation wildcard convention.
318    fn compact_tensor_rep_arg(value: FunView<'_>) -> Option<(usize, Representation<LibraryRep>)> {
319        if value.get_symbol() == ETS.metric || value.get_symbol() == SPENSO_TAG.dot {
320            return None;
321        }
322
323        if Representation::<LibraryRep>::try_from(value.as_view()).is_ok() {
324            return None;
325        }
326
327        let args = value.iter().collect::<Vec<_>>();
328        if args
329            .iter()
330            .any(|arg| Slot::<LibraryRep, Aind>::try_from(*arg).is_ok())
331        {
332            return None;
333        }
334
335        let rep_args = args
336            .iter()
337            .enumerate()
338            .filter_map(|(position, arg)| {
339                Self::compact_rep_pattern_match(*arg).map(|rep| (position, rep))
340            })
341            .collect::<Vec<_>>();
342
343        let [(position, rep)] = rep_args.as_slice() else {
344            return None;
345        };
346
347        Some((*position, *rep))
348    }
349
350    /// Match one argument against the symbolic representation wildcard.
351    ///
352    /// Matching is restricted to the argument itself (`level_range = 0`) so a
353    /// nested representation inside metadata does not accidentally become the
354    /// tensor's compact slot.
355    fn compact_rep_pattern_match(arg: AtomView<'_>) -> Option<Representation<LibraryRep>> {
356        let rep_pattern = Atom::var(SPENSO_TAG.rep_).to_pattern();
357        let settings = MatchSettings::new()
358            .level_range((0, Some(0)))
359            .partial(false);
360        let mut matches = arg.pattern_match(&rep_pattern, None, Some(&settings));
361        let matched = matches.next_detailed()?;
362        let rep = rep_pattern.replace_wildcards_with_matches(matched.match_stack);
363        Representation::<LibraryRep>::try_from(rep.as_view()).ok()
364    }
365
366    /// Replace a compact representation with a concrete slot.
367    ///
368    /// For a function, this rebuilds the function with the matched compact
369    /// representation argument replaced by `slot`. For a sum, every summand is
370    /// rebuilt with the same slot so the expansion keeps a single dummy edge.
371    fn materialize_compact_vector_with_slot(
372        value: AtomView<'_>,
373        rep: &Representation<LibraryRep>,
374        slot: &Atom,
375    ) -> Option<Atom> {
376        match value {
377            AtomView::Fun(fun) => {
378                let (position, matched_rep) = Self::compact_tensor_rep_arg(fun)?;
379                if matched_rep != *rep {
380                    return None;
381                }
382
383                let mut tensor = FunctionBuilder::new(fun.get_symbol());
384                for (arg_position, arg) in fun.iter().enumerate() {
385                    if arg_position == position {
386                        tensor = tensor.add_arg(slot);
387                    } else {
388                        tensor = tensor.add_arg(arg);
389                    }
390                }
391                Some(tensor.finish())
392            }
393            AtomView::Add(add) => {
394                let mut terms = add
395                    .iter()
396                    .map(|term| Self::materialize_compact_vector_with_slot(term, rep, slot));
397                let first = terms.next()??;
398                let rest = terms.collect::<Option<Vec<_>>>()?;
399                Some(rest.into_iter().fold(first, |sum, term| sum + term))
400            }
401            _ => None,
402        }
403    }
404
405    /// Normalize `dot(slot_i, slot_j)` to `g(slot_i, slot_j)`.
406    ///
407    /// This only applies after arguments have already been materialized to
408    /// concrete slots. Non-slot dot products are left untouched.
409    fn compact_dot_as_metric(value: AtomView<'_>) -> Option<Atom> {
410        let AtomView::Fun(fun) = value else {
411            return None;
412        };
413
414        if fun.get_symbol() != SPENSO_TAG.dot || fun.get_nargs() != 2 {
415            return None;
416        }
417
418        let args = fun.iter().collect::<Vec<_>>();
419        if args
420            .iter()
421            .all(|arg| Slot::<LibraryRep, Aind>::try_from(*arg).is_ok())
422        {
423            Some(
424                FunctionBuilder::new(ETS.metric)
425                    .add_arg(args[0])
426                    .add_arg(args[1])
427                    .finish(),
428            )
429        } else {
430            None
431        }
432    }
433}

447    #[allow(clippy::result_large_err)]
448    pub(super) fn is_shorthand_function(value: FunView<'a>) -> bool {
449        let symbol = value.get_symbol();
450        symbol == SPENSO_TAG.chain
451            || symbol == SPENSO_TAG.trace
452            || symbol == SPENSO_TAG.dot
453            || SchoonschipMaterializer::<Aind>::contains_schoonschip_shorthand(value.as_view())
454    }
455
456    #[allow(clippy::result_large_err)]
457    pub(super) fn materialize_shorthand<S, Lib, FunLib>(
458        value: FunView<'a>,
459        state: ParseState<Aind>,
460        library: &Lib,
461        function_library: &FunLib,
462        settings: &ParseSettings,
463    ) -> Result<Self, TensorNetworkError<K, Symbol>>
464    where
465        S: TensorStructure + ScalarStructure + Clone + StructureFromAtom,
466        TensorShell<S>: Concretize<T>,
467        S::Slot: IsAbstractSlot<Aind = Aind>,
468        T::Slot: IsAbstractSlot<Aind = Aind>,
469        T: TensorFromExpression<S, Sc, K, Symbol, Aind, Lib, FunLib>,
470        Lib: TensorLibraryFor<S, T, Key = K>,
471        FunLib: FunctionLibrary<T, Sc, Key = Symbol>,
472    {
473        let symbol = value.get_symbol();
474        let root_chain_disabled =
475            symbol == SPENSO_TAG.chain && !settings.shorthand_parsing.expands_chain();
476        let root_trace_disabled =
477            symbol == SPENSO_TAG.trace && !settings.shorthand_parsing.expands_trace();
478
479        if symbol == SPENSO_TAG.chain && !root_chain_disabled {
480            return Self::materialize_chain_shorthand(
481                value,
482                state,
483                library,
484                function_library,
485                settings,
486            );
487        }
488
489        if symbol == SPENSO_TAG.trace && !root_trace_disabled {
490            return Self::materialize_trace_shorthand(
491                value,
492                state,
493                library,
494                function_library,
495                settings,
496            );
497        }
498
499        let has_schoonschip_shorthand =
500            SchoonschipMaterializer::<Aind>::contains_schoonschip_shorthand(value.as_view());
501        let schoonschip_mode = settings
502            .shorthand_parsing
503            .schoonschip_expansion()
504            .unwrap_or_else(SchoonschipExpansionMode::none);
505        let effective_schoonschip_mode = if root_chain_disabled || root_trace_disabled {
506            schoonschip_mode.for_chain_like_root()
507        } else {
508            schoonschip_mode
509        };
510        let materialized = if effective_schoonschip_mode.any() {
511            SchoonschipMaterializer::<Aind>::with_mode(&state, effective_schoonschip_mode)
512                .materialize_shorthand(value.as_view())
513        } else {
514            value.as_view().to_owned()
515        };
516
517        if materialized == value.as_view().to_owned() {
518            // The atom rewriter is at a fixed point; recurse only after an actual rewrite.
519            if root_chain_disabled || root_trace_disabled || has_schoonschip_shorthand {
520                return Self::as_inferred_leaf::<S, Lib, FunLib>(
521                    value.as_view(),
522                    super::StructureInferenceMode::Fast,
523                    library,
524                    function_library,
525                    settings,
526                );
527            }
528            return Self::parse_regular_function_leaf::<S, Lib>(value, library);
529        }
530
531        Self::try_from_view_impl(
532            materialized.as_view(),
533            state,
534            library,
535            function_library,
536            settings,
537        )
538    }

540    fn materialize_chain_endpoint(
541        value: AtomView<'a>,
542        label: &str,
543        state: &ParseState<Aind>,
544        schoonschip_mode: SchoonschipExpansionMode,
545    ) -> Result<ChainEndpoint<Aind>, TensorNetworkError<K, Symbol>> {
546        match Slot::<LibraryRep, Aind>::try_from(value) {
547            Ok(slot) => Ok(ChainEndpoint {
548                slot,
549                additional_factors: Vec::new(),
550            }),
551            Err(slot_err) => {
552                if schoonschip_mode.any()
553                    && let Some(materialized) =
554                        SchoonschipMaterializer::<Aind>::with_mode(state, schoonschip_mode)
555                            .materialize_shorthand_arg(value)
556                {
557                    let slot =
558                        match Slot::<LibraryRep, Aind>::try_from(materialized.current.as_view()) {
559                            Ok(slot) => slot,
560                            Err(err) => {
561                                return Err(eyre!(
562                                    "invalid materialized chain {label} `{}` from `{}`: {err}",
563                                    materialized.current,
564                                    value
565                                )
566                                .into());
567                            }
568                        };
569                    return Ok(ChainEndpoint {
570                        slot,
571                        additional_factors: materialized.additional_factors,
572                    });
573                }
574
575                Err(eyre!("invalid chain {label} `{}`: {slot_err}", value).into())
576            }
577        }
578    }

580    #[allow(clippy::result_large_err)]
581    fn materialize_chain_shorthand<S, Lib, FunLib>(
582        value: FunView<'a>,
583        state: ParseState<Aind>,
584        library: &Lib,
585        function_library: &FunLib,
586        settings: &ParseSettings,
587    ) -> Result<Self, TensorNetworkError<K, Symbol>>
588    where
589        S: TensorStructure + ScalarStructure + Clone + StructureFromAtom,
590        TensorShell<S>: Concretize<T>,
591        S::Slot: IsAbstractSlot<Aind = Aind>,
592        T::Slot: IsAbstractSlot<Aind = Aind>,
593        T: TensorFromExpression<S, Sc, K, Symbol, Aind, Lib, FunLib>,
594        Lib: TensorLibraryFor<S, T, Key = K>,
595        FunLib: FunctionLibrary<T, Sc, Key = Symbol>,
596    {
597        let args = value.iter().collect::<Vec<_>>();
598        if args.len() < 2 {
599            return Err(TensorNetworkError::TooManyArgsFunction(
600                value.as_view().to_plain_string(),
601            ));
602        }
603
604        let schoonschip_mode = settings
605            .shorthand_parsing
606            .schoonschip_expansion()
607            .unwrap_or_else(SchoonschipExpansionMode::none);
608        let ChainEndpoint {
609            slot: start,
610            additional_factors: start_factors,
611        } = Self::materialize_chain_endpoint(args[0], "start", &state, schoonschip_mode)?;
612        let ChainEndpoint {
613            slot: end,
614            additional_factors: end_factors,
615        } = Self::materialize_chain_endpoint(args[1], "end", &state, schoonschip_mode)?;
616        let factors = &args[2..];
617
618        let factor_schoonschip_mode = schoonschip_mode.for_chain_like_root();
619        let factor_settings = settings
620            .clone()
621            .with_schoonschip_expansion(factor_schoonschip_mode);
622
623        let mut factor_networks = Vec::new();
624        for factor in start_factors.into_iter().chain(end_factors) {
625            factor_networks.extend(Self::parse_chain_like_factor_networks::<S, Lib, FunLib>(
626                factor,
627                state.clone(),
628                library,
629                function_library,
630                settings,
631            )?);
632        }
633
634        if factors.is_empty() {
635            let metric = FunctionBuilder::new(ETS.metric)
636                .add_arg(start.to_atom())
637                .add_arg(end.to_atom())
638                .finish();
639            factor_networks.extend(Self::parse_chain_like_factor_networks::<S, Lib, FunLib>(
640                metric,
641                state,
642                library,
643                function_library,
644                settings,
645            )?);
646            return Ok(Self::chain_like_network_product(factor_networks));
647        }
648
649        let mut left = start;
650        for (position, factor) in factors.iter().enumerate() {
651            let fresh_right = position + 1 != factors.len();
652            let right = if position + 1 == factors.len() {
653                end.dual()
654            } else {
655                state.slot(&left.rep)
656            };
657            if fresh_right {
658                tracing::debug!(
659                    target: "spenso::network::parsing",
660                    spenso_parser = true,
661                    generation = true,
662                    compile = true,
663                    inspect = true,
664                    stage = "chain_shorthand_link_slot",
665                    position,
666                    factor_count = factors.len(),
667                    left = %left,
668                    right = %right,
669                    right_atom = %right.to_atom().to_plain_string(),
670                    start = %start,
671                    end = %end,
672                    file.factor = %factor.to_plain_string(),
673                    "Spenso parser allocated chain shorthand link slot"
674                );
675            }
676            let factor = ChainExpansion::replace_placeholders(
677                *factor,
678                &left.to_atom(),
679                &right.dual().to_atom(),
680            );
681            let factor = if factor_schoonschip_mode.any() {
682                SchoonschipMaterializer::<Aind>::with_mode(&state, factor_schoonschip_mode)
683                    .materialize_shorthand(factor.as_view())
684            } else {
685                factor
686            };
687            factor_networks.extend(Self::parse_chain_like_factor_networks::<S, Lib, FunLib>(
688                factor,
689                state.clone(),
690                library,
691                function_library,
692                &factor_settings,
693            )?);
694            left = right;
695        }
696
697        Ok(Self::chain_like_network_product(factor_networks))
698    }

700    #[allow(clippy::result_large_err)]
701    fn materialize_trace_shorthand<S, Lib, FunLib>(
702        value: FunView<'a>,
703        state: ParseState<Aind>,
704        library: &Lib,
705        function_library: &FunLib,
706        settings: &ParseSettings,
707    ) -> Result<Self, TensorNetworkError<K, Symbol>>
708    where
709        S: TensorStructure + ScalarStructure + Clone + StructureFromAtom,
710        TensorShell<S>: Concretize<T>,
711        S::Slot: IsAbstractSlot<Aind = Aind>,
712        T::Slot: IsAbstractSlot<Aind = Aind>,
713        T: TensorFromExpression<S, Sc, K, Symbol, Aind, Lib, FunLib>,
714        Lib: TensorLibraryFor<S, T, Key = K>,
715        FunLib: FunctionLibrary<T, Sc, Key = Symbol>,
716    {
717        let args = value.iter().collect::<Vec<_>>();
718        let Some(rep_view) = args.first() else {
719            return Err(TensorNetworkError::TooManyArgsFunction(
720                value.as_view().to_plain_string(),
721            ));
722        };
723
724        let rep = Representation::<LibraryRep>::try_from(*rep_view)
725            .map_err(|err| eyre!("invalid trace representation `{rep_view}`: {err}"))?;
726        let factors = shadowing::trace_factor_views(&args[1..]);
727
728        if factors.is_empty() {
729            return Self::try_from_view_impl(
730                rep.dim.to_symbolic().as_view(),
731                state,
732                library,
733                function_library,
734                settings,
735            );
736        }
737
738        let links = (0..factors.len())
739            .map(|position| {
740                let slot = state.slot(&rep);
741                tracing::debug!(
742                    target: "spenso::network::parsing",
743                    spenso_parser = true,
744                    generation = true,
745                    compile = true,
746                    inspect = true,
747                    stage = "trace_shorthand_link_slot",
748                    position,
749                    factor_count = factors.len(),
750                    slot = %slot,
751                    slot_atom = %slot.to_atom().to_plain_string(),
752                    representation = %rep,
753                    file.factor = %factors[position].to_plain_string(),
754                    "Spenso parser allocated trace shorthand link slot"
755                );
756                slot
757            })
758            .collect::<Vec<_>>();
759        let factor_schoonschip_mode = settings
760            .shorthand_parsing
761            .schoonschip_expansion()
762            .unwrap_or_else(SchoonschipExpansionMode::none)
763            .for_chain_like_root();
764        let factor_settings = settings
765            .clone()
766            .with_schoonschip_expansion(factor_schoonschip_mode);
767
768        let materialized_factors = factors
769            .iter()
770            .enumerate()
771            .map(|(position, factor)| {
772                let left = links[position].to_atom();
773                let right = links[(position + 1) % factors.len()].dual().to_atom();
774                let factor = ChainExpansion::replace_placeholders(*factor, &left, &right);
775                if factor_schoonschip_mode.any() {
776                    SchoonschipMaterializer::<Aind>::with_mode(&state, factor_schoonschip_mode)
777                        .materialize_shorthand(factor.as_view())
778                } else {
779                    factor
780                }
781            })
782            .collect::<Vec<_>>();
783
784        let mut factor_networks = Vec::new();
785        for factor in materialized_factors {
786            factor_networks.extend(Self::parse_chain_like_factor_networks::<S, Lib, FunLib>(
787                factor,
788                state.clone(),
789                library,
790                function_library,
791                &factor_settings,
792            )?);
793        }
794        Ok(Self::chain_like_network_product(factor_networks))
795    }

797    #[allow(clippy::result_large_err)]
798    fn parse_chain_like_factor_networks<S, Lib, FunLib>(
799        factor: Atom,
800        state: ParseState<Aind>,
801        library: &Lib,
802        function_library: &FunLib,
803        settings: &ParseSettings,
804    ) -> Result<Vec<Self>, TensorNetworkError<K, Symbol>>
805    where
806        S: TensorStructure + ScalarStructure + Clone + StructureFromAtom,
807        TensorShell<S>: Concretize<T>,
808        S::Slot: IsAbstractSlot<Aind = Aind>,
809        T::Slot: IsAbstractSlot<Aind = Aind>,
810        T: TensorFromExpression<S, Sc, K, Symbol, Aind, Lib, FunLib>,
811        Lib: TensorLibraryFor<S, T, Key = K>,
812        FunLib: FunctionLibrary<T, Sc, Key = Symbol>,
813    {
814        let AtomView::Mul(product) = factor.as_view() else {
815            return Ok(vec![Self::try_from_view_impl(
816                factor.as_view(),
817                state.clone(),
818                library,
819                function_library,
820                settings,
821            )?]);
822        };
823
824        let mut scalars = Vec::new();
825        let mut tensors = Vec::new();
826        for arg in product.iter() {
827            let network =
828                Self::try_from_view_impl(arg, state.clone(), library, function_library, settings)?;
829            if network.state == NetworkState::PureScalar {
830                scalars.push(network);
831            } else {
832                tensors.push(network);
833            }
834        }
835
836        if scalars.is_empty() || tensors.len() != 1 {
837            return Ok(vec![Self::try_from_view_impl(
838                factor.as_view(),
839                state.clone(),
840                library,
841                function_library,
842                settings,
843            )?]);
844        }
845
846        scalars.extend(tensors);
847        Ok(scalars)
848    }
849
850    fn chain_like_network_product(networks: Vec<Self>) -> Self {
851        let mut networks = networks.into_iter();
852        let first = networks.next().unwrap();
853        let rest = networks.collect::<Vec<_>>();
854        if rest.is_empty() {
855            first
856        } else {
857            first.n_mul(rest)
858        }
859    }
860}

862/// Utilities for lowering chain and trace shorthand factors.
863///
864/// Chain parsing chooses the concrete left and right slots for each factor.
865/// These helpers rewrite the symbolic placeholders in the factor expression
866/// before normal parsing resumes.
867pub(super) struct ChainExpansion;
868
869impl ChainExpansion {
870    /// Replace `in` and `out` placeholders recursively.
871    ///
872    /// The recursion is purely syntactic over functions, sums, products, and
873    /// powers. It does not allocate dummies or inspect tensor structure.
874    pub(super) fn replace_placeholders(
875        value: AtomView<'_>,
876        chain_in: &Atom,
877        chain_out: &Atom,
878    ) -> Atom {
879        match value {
880            AtomView::Var(var) if var.get_symbol() == SPENSO_TAG.chain_in => chain_in.clone(),
881            AtomView::Var(var) if var.get_symbol() == SPENSO_TAG.chain_out => chain_out.clone(),
882            AtomView::Fun(fun) => {
883                let mut rebuilt = FunctionBuilder::new(fun.get_symbol());
884                for arg in fun.iter() {
885                    rebuilt = rebuilt.add_arg(Self::replace_placeholders(arg, chain_in, chain_out));
886                }
887                rebuilt.finish()
888            }
889            AtomView::Add(add) => add.iter().fold(Atom::Zero, |sum, term| {
890                sum + Self::replace_placeholders(term, chain_in, chain_out)
891            }),
892            AtomView::Mul(mul) => mul.iter().fold(Atom::num(1), |product, factor| {
893                product * Self::replace_placeholders(factor, chain_in, chain_out)
894            }),
895            AtomView::Pow(pow) => {
896                let (base, exponent) = pow.get_base_exp();
897                Self::replace_placeholders(base, chain_in, chain_out).pow(exponent.to_owned())
898            }
899            _ => value.to_owned(),
900        }
901    }
902}
