203impl<T, S> NetworkStoreAccess for NetworkStore<T, S> {
204    type Tensor = T;
205    type Scalar = S;
206
207    fn tensor(&self, index: usize) -> &Self::Tensor {
208        &self.tensors[index]
209    }
210
211    fn scalar(&self, index: usize) -> &Self::Scalar {
212        &self.scalar[index]
213    }
214
215    fn scalar_ref(&self, scalar: ScalarRef) -> &Self::Scalar {
216        match scalar {
217            ScalarRef::Store(index) => &self.scalar[index],
218            ScalarRef::Alias(index) => self
219                .scalar_aliases
220                .get(index)
221                .and_then(Option::as_ref)
222                .unwrap_or(&self.scalar[index]),
223        }
224    }
225
226    fn push_tensor(&mut self, tensor: Self::Tensor) -> usize {
227        let index = self.tensors.len();
228        self.tensors.push(tensor);
229        index
230    }
231
232    fn push_scalar(&mut self, scalar: Self::Scalar) -> usize {
233        let index = self.scalar.len();
234        self.scalar.push(scalar);
235        self.scalar_aliases.push(None);
236        index
237    }
238
239    fn retain_graph_tensors<K: Debug, FK: Debug, Aind: AbsInd>(
240        &mut self,
241        graph: &mut NetworkGraph<K, FK, Aind>,
242    ) {
243        if self.tensors.is_empty() {
244            return;
245        }
246        // MAX marks an unreferenced slot, never a graph index. Mark every
247        // reference, including shared IDs, before moving any tensor.
248        let mut indices = vec![usize::MAX; self.tensors.len()];
249        graph.map_tensor_refs(|index| {
250            indices[index] = index;
251            index
252        });
253        let mut retained = 0;
254        let mut old_indices = indices.iter_mut();
255        self.tensors.retain(|_| {
256            let index = old_indices.next().unwrap();
257            if *index == usize::MAX {
258                return false;
259            }
260            *index = retained;
261            retained += 1;
262            true
263        });
264        graph.map_tensor_refs(|index| indices[index]);
265    }
266}

268impl<T, S> NetworkStoreAccess for NetworkStoreOverlay<'_, T, S> {
269    type Tensor = T;
270    type Scalar = S;
271
272    fn tensor(&self, index: usize) -> &Self::Tensor {
273        if index < self.base_tensors.len() {
274            &self.base_tensors[index]
275        } else {
276            &self.tensors[index - self.base_tensors.len()]
277        }
278    }
279
280    fn scalar(&self, index: usize) -> &Self::Scalar {
281        if index < self.base_scalars.len() {
282            &self.base_scalars[index]
283        } else {
284            &self.scalar[index - self.base_scalars.len()]
285        }
286    }
287
288    fn scalar_ref(&self, scalar: ScalarRef) -> &Self::Scalar {
289        let index = scalar.index();
290        match scalar {
291            ScalarRef::Store(_) => self.scalar(index),
292            ScalarRef::Alias(_) => {
293                if index < self.base_scalars.len() {
294                    self.base_scalar_aliases
295                        .get(index)
296                        .and_then(Option::as_ref)
297                        .unwrap_or(&self.base_scalars[index])
298                } else {
299                    let local = index - self.base_scalars.len();
300                    self.scalar_aliases
301                        .get(local)
302                        .and_then(Option::as_ref)
303                        .unwrap_or(&self.scalar[local])
304                }
305            }
306        }
307    }
308
309    fn push_tensor(&mut self, tensor: Self::Tensor) -> usize {
310        let index = self.base_tensors.len() + self.tensors.len();
311        self.tensors.push(tensor);
312        index
313    }
314
315    fn push_scalar(&mut self, scalar: Self::Scalar) -> usize {
316        let index = self.base_scalars.len() + self.scalar.len();
317        self.scalar.push(scalar);
318        self.scalar_aliases.push(None);
319        index
320    }
321
322    fn retain_graph_tensors<K: Debug, FK: Debug, Aind: AbsInd>(
323        &mut self,
324        _graph: &mut NetworkGraph<K, FK, Aind>,
325    ) {
326        // Workers borrow the shared base and can hold references outside their
327        // operation graph. Reclamation belongs to the joined, merged wave.
328    }
329}

756    #[test]
757    fn completed_execution_waves_reclaim_tensor_inputs_in_every_strategy() {
758        use crate::{
759            network::{
760                ExecutionResult, Network, Sequential, SequentialExtract, SequentialRef,
761                SmallestDegree,
762                library::{DummyKey, DummyLibrary, DummyLibraryTensor, panicing::ErroringLibrary},
763            },
764            structure::{
765                OrderedStructure,
766                representation::{Euclidean, RepName},
767            },
768            tensors::data::DenseTensor,
769        };
770
771        #[cfg(feature = "shadowing")]
772        let _parallel = crate::symbolic_parallelism::scoped_symbolica_rayon_setting_for_test(
773            crate::symbolic_parallelism::SymbolicParallelism::Parallel,
774            || true,
775        );
776        type Tensor = DenseTensor<f64, OrderedStructure<Euclidean>>;
777        type Net = Network<NetworkStore<Tensor, f64>, DummyKey, DummyKey>;
778        type LibTensor = DummyLibraryTensor<Tensor>;
779        type Lib = DummyLibrary<Tensor, DummyKey>;
780        type FnLib = ErroringLibrary<DummyKey>;
781        let lib = Lib::new();
782        let fn_lib = FnLib::new();
783        let structure = OrderedStructure::new(vec![Euclidean {}.new_slot(2, 1)]).into_canonical();
784        let tensor =
785            |data| Net::from_tensor(Tensor::from_storage_data(data, structure.clone()).unwrap());
786        // The two inner sums can run in separate overlays before later waves
787        // negate and combine them. Every stage must use the remapped indices.
788        let original = -(tensor(vec![1.0, 2.0]) + tensor(vec![3.0, 4.0]))
789            + -(tensor(vec![5.0, 6.0]) + tensor(vec![7.0, 8.0]));
790        for strategy in 0..4 {
791            let mut network = original.clone();
792            match strategy {
793                0 => network
794                    .execute::<Sequential, SmallestDegree, LibTensor, Lib, FnLib>(&lib, &fn_lib),
795                1 => network
796                    .execute::<SequentialRef, SmallestDegree, LibTensor, Lib, FnLib>(&lib, &fn_lib),
797                2 => network.execute::<SequentialExtract, SmallestDegree, LibTensor, Lib, FnLib>(
798                    &lib, &fn_lib,
799                ),
800                _ => {
801                    network.execute_parallel::<SmallestDegree, LibTensor, Lib, FnLib>(&lib, &fn_lib)
802                }
803            }
804            .unwrap();
805            let ExecutionResult::Val(actual) =
806                network.result_tensor::<LibTensor, Lib>(&lib).unwrap()
807            else {
808                panic!("expected a tensor result")
809            };
810            assert_eq!(actual.data, [-16.0, -20.0]);
811        }
812
813        // A self-loop-only network finishes before scheduling an operation wave.
814        let structure = OrderedStructure::new(vec![
815            Euclidean {}.new_slot(2, 1),
816            Euclidean {}.new_slot(2, 1),
817        ])
818        .into_canonical();
819        let mut trace = Net::from_tensor(
820            Tensor::from_storage_data(vec![1.0, 2.0, 3.0, 4.0], structure).unwrap(),
821        );
822        trace
823            .execute::<Sequential, SmallestDegree, LibTensor, Lib, FnLib>(&lib, &fn_lib)
824            .unwrap();
825        let ExecutionResult::Val(value) = trace.result_scalar().unwrap() else {
826            panic!("expected a trace")
827        };
828        assert_eq!(*value, 5.0);
829    }
