25/// Builds a tensor leaf from an opaque expression once its structure is known.
26///
27/// Opaque shorthand parsing separates topology from realization: structure
28/// inference decides the exposed slots, then this trait decides how the target
29/// tensor type represents the original expression. Symbolic tensors can keep
30/// the expression directly, while concretized tensor types may parse and
31/// execute an expanded sub-network here.
32pub trait TensorFromExpression<S, Sc, K, FK, Aind, Lib, FunLib>: Sized
33where
34    S: TensorStructure,
35    Self: HasStructure,
36    Lib: TensorLibraryFor<S, Self, Key = K>,
37    FunLib: FunctionLibrary<Self, Sc, Key = FK>,
38{
39    #[allow(clippy::result_large_err)]
40    fn tensor_from_expression(
41        expression: AtomView<'_>,
42        structure: PermutedStructure<S>,
43        tensor_library: &Lib,
44        function_library: &FunLib,
45        settings: &ParseSettings,
46    ) -> Result<Self, TensorNetworkError<K, FK>>
47    where
48        K: Display,
49        FK: Display;
50}

52/// A tensor library whose stored tensor can be indexed into `T`.
53pub trait TensorLibraryFor<S, T>:

72/// Marker for tensor types whose opaque expression can be realized by
73/// expanding and executing a nested network.

84impl<S, Sc, T, K, Aind, Lib, FunLib>
85    TensorFromExpression<S, Sc, K, symbolica::atom::Symbol, Aind, Lib, FunLib> for T
86where
87    S: TensorStructure + ScalarStructure + Clone + StructureFromAtom,
88    TensorShell<S>: Concretize<T>,
89    S::Slot: IsAbstractSlot<Aind = Aind>,
90    T::Slot: IsAbstractSlot<Aind = Aind>,
91    T: ExpandedTensorFromExpression
92        + HasStructure<Structure = S>
93        + TensorStructure
94        + Clone
95        + ScalarTensor
96        + PermuteTensor<Permuted = T>,
97    Sc: for<'r> TryFrom<AtomView<'r>> + Clone + Into<T::Scalar>,
98    for<'r> TensorNetworkError<K, symbolica::atom::Symbol>:
99        From<<Sc as TryFrom<AtomView<'r>>>::Error>,
100    K: Display + Debug + Clone,
101    Aind: crate::structure::slot::AbsInd
102        + crate::structure::slot::DummyAind
103        + crate::structure::slot::ParseableAind,
104    Lib: TensorLibraryFor<S, T, Key = K> + Sync,
105    FunLib: FunctionLibrary<T, Sc, Key = symbolica::atom::Symbol>,
106    NetworkStore<T, Sc>: ExecuteOp<FunLib, Lib, K, symbolica::atom::Symbol, Aind>,
107    SmallestDegree: ContractionStrategy<NetworkStore<T, Sc>, Lib, K, symbolica::atom::Symbol, Aind>,
108{
109    fn tensor_from_expression(
110        expression: AtomView<'_>,
111        _structure: PermutedStructure<S>,
112        tensor_library: &Lib,
113        function_library: &FunLib,
114        settings: &ParseSettings,
115    ) -> Result<Self, TensorNetworkError<K, symbolica::atom::Symbol>>
116    where
117        K: Display,
118        symbolica::atom::Symbol: Display,
119    {
120        let mut expanded_settings = settings.clone();
121        expanded_settings.shorthand_parsing = ShorthandParsing::expand_all();
122
123        let mut network =
124            Network::<NetworkStore<T, Sc>, K, symbolica::atom::Symbol, Aind>::try_from_view_with_function_library::<
125                S,
126                Lib,
127                FunLib,
128            >(
129                expression,
130                tensor_library,
131                function_library,
132                &expanded_settings,
133            )?;
134        network.execute::<Sequential, SmallestDegree, Lib::LibraryTensor, Lib, FunLib>(
135            tensor_library,
136            function_library,
137        )?;
138
139        match network.result()? {
140            ExecutionResult::Val(TensorOrScalarOrKey::Tensor { tensor, .. }) => Ok(tensor.clone()),
141            ExecutionResult::Val(TensorOrScalarOrKey::Scalar(scalar)) => {
142                Ok(T::new_scalar(scalar.clone().into()))
143            }
144            ExecutionResult::Val(TensorOrScalarOrKey::Key { nodeid, .. }) => {
145                Ok(network
146                    .graph
147                    .get_lib_data::<S, Lib::LibraryTensor, Lib>(tensor_library, nodeid)?)
148            }
149            ExecutionResult::One => {
150                let one: Sc = Atom::num(1).as_view().try_into()?;
151                Ok(T::new_scalar(one.into()))
152            }
153            ExecutionResult::Zero => {
154                let zero: Sc = Atom::num(0).as_view().try_into()?;
155                Ok(T::new_scalar(zero.into()))
156            }
157        }
158    }
159}

169    use crate::{

181    #[test]
182    fn param_tensor_from_expression_uses_inferred_structure() {
183        type Structure = NamedStructure<Symbol, Vec<Atom>, LibraryRep, AbstractIndex>;
184
185        let rep = Minkowski {}.new_rep(4);
186        let slot = rep.slot(AbstractIndex::from(1));
187        let structure: PermutedStructure<Structure> =
188            NamedStructure::from_iter([slot], symbol!("f"), None::<Vec<Atom>>);
189        let expression = function!(tensor_symbol!(opaque), slot.to_atom());
190        type TensorLib = DummyLibrary<ParamTensor<Structure>, DummyKey>;
191        type FunLib = ErroringLibrary<Symbol>;
192        let tensor = <ParamTensor<Structure> as TensorFromExpression<
193            Structure,
194            Atom,
195            DummyKey,
196            Symbol,
197            AbstractIndex,
198            TensorLib,
199            FunLib,
200        >>::tensor_from_expression(
201            expression.as_view(),
202            structure,
203            &TensorLib::new(),
204            &FunLib::new(),
205            &ParseSettings::default(),
206        )
207        .unwrap();
208
209        assert_eq!(tensor.structure().order(), 1);
210    }
