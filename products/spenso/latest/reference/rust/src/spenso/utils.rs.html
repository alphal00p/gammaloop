4fn to_unicode(number: isize, digits: &[char; 10], minus_sign: char) -> String {
5    if number == 0 {
6        return digits[0].to_string();
7    }
8    let mut num = number;
9    let mut digit_stack = Vec::new();
10    while num != 0 {
11        let digit = (num % 10).unsigned_abs();
12        digit_stack.push(digits[digit]);
13        num /= 10;
14    }
15    let mut result = String::new();
16    if number < 0 {
17        result.push(minus_sign);
18    }
19    result.extend(digit_stack.drain(..).rev());
20    result
21}
