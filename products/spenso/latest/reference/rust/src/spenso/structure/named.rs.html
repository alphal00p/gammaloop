24/// A named structure is a structure with a global name, and a list of slots
25///
26/// It is useful when you want to shadow tensors, to nest tensor network contraction operations.
27#[derive(
28    Clone, PartialEq, Eq, Default, Hash, bincode_trait_derive::Encode, bincode_trait_derive::Decode,
29)]
30#[cfg_attr(not(feature = "shadowing"), derive(Serialize, Deserialize))]
31#[cfg_attr(
32    feature = "shadowing",
33    trait_decode(trait = symbolica::state::HasStateMap),
34)]
35pub struct NamedStructure<
36    Name = String,
37    Args = usize,
38    R: RepName = LibraryRep,
39    Aind = AbstractIndex,
40> {
41    pub structure: OrderedStructure<R, Aind>,
42    pub global_name: Option<Name>,
43    pub additional_args: Option<Args>,
44}

49impl<Name, Args, R: RepName, Aind: AbsInd> NamedStructure<Name, Args, R, Aind> {
50    #[must_use]
51    pub fn from_iter<I, T>(iter: T, name: Name, args: Option<Args>) -> Canonicalized<Self>
52    where
53        R: From<I>,
54        I: RepName,
55        T: IntoIterator<Item = Slot<I, Aind>>,
56    {
57        iter.into_iter()
58            .map(|a| a.cast())
59            .collect::<Canonicalized<_>>()
60            .map_canonical(move |structure| Self {
61                structure,
62                global_name: Some(name),
63                additional_args: args,
64            })
65    }
66}

250impl<N: std::fmt::Display, A: ArgDisplay, R: RepName, Aind: AbsInd> std::fmt::Display
251    for NamedStructure<N, A, R, Aind>
252{
253    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
254        let mut table = Builder::new();
255
256        table.push_record(&[
257            self.global_name
258                .as_ref()
259                .map(|a| format!("{a}"))
260                .unwrap_or("NO NAME".to_string()),
261            self.additional_args
262                .as_ref()
263                .map(|a| a.arg_display())
264                .unwrap_or("".to_string()),
265        ]);
266        for item in self.structure.structure.iter() {
267            if item.rep.rep.is_self_dual() {
268                table.push_record(&[item.rep.to_string(), format!("{}", item.aind)]);
269            } else if item.rep.rep.is_base() {
270                table.push_record(&[item.rep.to_string(), format!("{:+}", item.aind)]);
271            } else {
272                table.push_record(&[item.rep.to_string(), format!("{:-}", item.aind)]);
273            }
274        }
275        writeln!(f)?;
276        table.build().with(Style::rounded()).fmt(f)
277    }
278}
279impl<N: std::fmt::Debug, A: ArgDisplay, R: RepName, Aind: AbsInd> std::fmt::Debug
280    for NamedStructure<N, A, R, Aind>
281{
282    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
283        let mut table = Builder::new();
284
285        table.push_record(&[
286            self.global_name
287                .as_ref()
288                .map(|a| format!("{:?}", a))
289                .unwrap_or("NO NAME".to_string()),
290            self.additional_args
291                .as_ref()
292                .map(|a| a.arg_debug())
293                .unwrap_or("".to_string()),
294        ]);
295        for (index, item) in self.structure.structure.iter().enumerate() {
296            table.push_record(&[
297                index.to_string(),
298                format!("{:?}", item.rep.rep),
299                format!("{:?}", item.rep.dim),
300                format!("{:?}", item.aind),
301            ]);
302        }
303        writeln!(f)?;
304        write!(f, "{}", table.build().with(Style::rounded()))
305    }
