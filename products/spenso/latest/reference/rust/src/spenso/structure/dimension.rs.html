39impl Dimension {

46        match self {

69#[derive(Error, Debug)]
70pub enum DimensionError {
71    #[error("Dimension is not concrete")]
72    NotConcrete,
73    #[error("Dimension too large")]
74    TooLarge,
75    #[error("Dimension negative")]
76    Negative,
77    #[error("Dimension is not natural")]
78    NotNatural,
79
80    #[cfg(feature = "shadowing")]
81    #[error("Dimension is not a var or int: {0}")]

110#[cfg(feature = "shadowing")]
111impl<'a> TryFrom<AtomView<'a>> for Dimension {
112    type Error = DimensionError;
113    fn try_from(value: AtomView<'a>) -> Result<Self, Self::Error> {
114        match value {
115            AtomView::Var(a) => Ok(Dimension::Symbolic(a.get_symbol().into())),
116            AtomView::Num(n) => match n.get_coeff_view() {
117                CoefficientView::Natural(n, 1, _, _) => {
118                    if n < 0 {
119                        return Err(DimensionError::Negative);
120                    }
121                    Ok(Dimension::Concrete(n as usize))
122                }
123                _ => {
124                    let mut err = Atom::new();
125                    err.set_from_view(&value);
126                    Err(DimensionError::NotVarOrInt(err))
127                }
128            },
129            _ => {
130                let mut err = Atom::new();
131                err.set_from_view(&value);
132                Err(DimensionError::NotVarOrInt(err))
133            }
134        }
135    }
136}

159#[cfg(feature = "shadowing")]
160mod shadowing_tests {
161    use symbolica::{atom::Atom, function, symbol};
162
163    use super::Dimension;
164
165    #[test]
166    fn dimension_from_view() {
167        let a = Atom::num(5);
168        let b = Atom::var(symbol!("b"));
169        let c = function!(symbol!("a"), symbol!("b"));
170        let d = Atom::num(-1);
171        let e = Atom::num((1, 2));
172
173        let dima = Dimension::try_from(a.as_view()).unwrap();
174
175        assert_eq!(dima, Dimension::Concrete(5));
176        let dimb = Dimension::try_from(b.as_view()).unwrap();
177        assert_eq!(dimb, Dimension::Symbolic(symbol!("b").into()));
178        let dimc = Dimension::try_from(c.as_view());
179        assert!(dimc.is_err());
180        let dimd = Dimension::try_from(d.as_view());
181        assert!(dimd.is_err());
182        let dime = Dimension::try_from(e.as_view());
183        assert!(dime.is_err());
184    }
