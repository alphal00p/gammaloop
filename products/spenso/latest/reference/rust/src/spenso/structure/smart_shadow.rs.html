24/// A structure to enable smart shadowing of tensors in a tensor network contraction algorithm.
25#[derive(
26    Clone, PartialEq, Eq, Debug, Hash, bincode_trait_derive::Encode, bincode_trait_derive::Decode,
27)]
28#[cfg_attr(not(feature = "shadowing"), derive(Serialize, Deserialize))]
29#[cfg_attr(
30feature = "shadowing",
31trait_decode(trait = symbolica::state::HasStateMap),
32)]
33pub struct SmartShadowStructure<
34    Name = String,
35    Args = usize,
36    R: RepName = LibraryRep,
37    Aind: AbsInd = AbstractIndex,
38> {
39    pub structure: OrderedStructure<R, Aind>,
40    pub contractions: usize,
41    pub global_name: Option<Name>,
42    additional_args: Option<Args>,
43}

45impl<Name, Args, R: RepName, Aind: AbsInd> SmartShadowStructure<Name, Args, R, Aind> {
46    pub fn map_underlying_structure(
47        self,
48        mut f: impl FnMut(OrderedStructure<R, Aind>) -> OrderedStructure<R, Aind>,
49    ) -> Self {
50        Self {
51            structure: f(self.structure),
52            contractions: self.contractions,
53            global_name: self.global_name,
54            additional_args: self.additional_args,
55        }
56    }
57
58    /// Constructs a new [`SmartShadowStructure`] from a list of indices and
59    /// dimensions (assumed to be Euclidean), together with a name.
60    #[must_use]
61    pub fn from_iter<I, T>(iter: T, name: Option<Name>, args: Option<Args>) -> Canonicalized<Self>
62    where
63        I: Into<Slot<R, Aind>>,
64        T: IntoIterator<Item = I>,
65    {
66        let res = iter.into_iter().map(I::into).collect::<Canonicalized<_>>();
67        res.map_canonical(|structure| Self {
68            structure,
69            global_name: name,
70            additional_args: args,
71            contractions: 0,
72        })
73    }
74}

96impl<N: IntoSymbol, A: IntoArgs, R: RepName, Aind: AbsInd> std::fmt::Display
97    for SmartShadowStructure<N, A, R, Aind>
98{
99    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
100        if let Some(ref name) = self.global_name {
101            write!(f, "{}", name.ref_into_symbol())?
102        }
103        write!(f, "(")?;
104        if let Some(ref args) = self.additional_args {
105            let args: Vec<std::string::String> =
106                args.ref_into_args().map(|s| s.to_string()).collect();
107            write!(f, "{},", args.join(","))?
108        }
109
110        write!(f, "{})", self.structure)?;
111        Result::Ok(())
112    }
113}

202impl<N, A, R: RepName<Dual = R>> StructureContract for SmartShadowStructure<N, A, R> {
203    fn merge(
204        &self,
205        other: &Self,
206    ) -> Result<(Self, SubSet<SlotIndex>, SubSet<SlotIndex>, MergeInfo), StructureError> {
207        let contractions = self.contractions + other.contractions;
208        let (structure, pos_self, pos_other, mergeinfo) = self.structure.merge(&other.structure)?;
209        Ok((
210            Self {
211                contractions,
212                structure,
213                global_name: None,
214                additional_args: None,
215            },
216            pos_self,
217            pos_other,
218            mergeinfo,
219        ))
220    }
221
222    delegate! {
223        to self.structure{
224            fn trace_out(&mut self);
225            fn trace(&mut self, i: usize, j: usize);
226        }
227    }
228}
