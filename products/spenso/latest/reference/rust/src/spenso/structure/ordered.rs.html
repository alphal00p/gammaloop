79    fn permute_inds(self, permutation: &Permutation) -> Self::Permuted {
80        let mut dummy_structure = Vec::new();
81        let mut ids = Vec::new();
82
83        if permutation.is_identity() {
84            return (
85                OrderedStructure {
86                    structure: self.structure.into_iter().map(|s| s.to_lib()).collect(),
87                    dual_start: self.dual_start,
88                    base_start: self.base_start,
89                },
90                ids,
91            );
92        }
93        // println!("{}", permutation);
94        for s in permutation.iter_slice(&self.structure) {
95            let d = s.to_dummy_ind();
96            let ogs = s.to_lib();
97            // println!("{ogs}");
98            // println!("{d}");
99            dummy_structure.push(d);
100            ids.push(OrderedStructure::id(d.dual(), ogs));
101        }
102        let strct = OrderedStructure::new(dummy_structure);
103        debug_assert!(
104            strct.index_permutation.is_identity(),
105            "should be identity but is: {}",
106            strct.index_permutation
107        );
108
109        (strct.structure, ids)
110    }
111
112    fn permute_reps(self, rep_perm: &Permutation) -> Self::Permuted {
113        let mut dummy_structure = Vec::new();
114        let mut og_reps = Vec::new();
115        let mut ids = Vec::new();
116
117        if rep_perm.is_identity() {
118            return (
119                OrderedStructure {
120                    structure: self.structure.into_iter().map(|s| s.to_lib()).collect(),
121                    dual_start: self.dual_start,
122                    base_start: self.base_start,
123                },
124                ids,
125            );
126        }
127        // println!("{rep_perm}");
128        for s in rep_perm.iter_slice(&self.structure) {
129            og_reps.push(s.rep.to_lib());
130            let d = s.to_dummy_rep().to_dummy_ind();
131            // println!("{d}");
132            dummy_structure.push(d);
133        }
134
135        for (i, s) in rep_perm.iter_slice(&self.structure).enumerate() {
136            let d = dummy_structure[i];
137            let new_slot = og_reps[i].slot(s.aind);
138            // println!("{new_slot}");
139            // println!("{d}");
140            ids.push(OrderedStructure::id(d.dual(), new_slot));
141        }
142        let strct = OrderedStructure::new(dummy_structure);
143        if !strct.index_permutation.is_identity() {
144            panic!(
145                "should be identity but is: {} for {}",
146                strct.index_permutation, strct.structure
147            );
148        }
149        (strct.structure, ids)
150    }

330impl<R: RepName, Aind: AbsInd> From<Vec<Slot<R, Aind>>>
331    for PermutedStructure<OrderedStructure<R, Aind>>
332{
333    fn from(mut structure: Vec<Slot<R, Aind>>) -> Self {
334        let rep_permutation = Permutation::sort_by_key(&structure, |a| a.rep);
335        rep_permutation.apply_slice_in_place(&mut structure);
336
337        if structure.is_empty() {
338            return PermutedStructure {
339                structure: OrderedStructure {
340                    structure,
341                    base_start: usize::MAX,
342                    dual_start: usize::MAX,
343                },
344                rep_permutation,
345                index_permutation: Permutation::id(0),
346            };
347        }
348
349        let mut base_start =
350            if structure[0].rep.rep.is_base() && !structure[0].rep.rep.is_self_dual() {
351                0
352            } else {
353                usize::MAX
354            };
355        let mut dual_start =
356            if structure[0].rep.rep.is_dual() && !structure[0].rep.rep.is_self_dual() {
357                base_start = 0;
358                0
359            } else {
360                usize::MAX
361            };
362
363        for i in 0..(structure.len() - 1) {
364            if structure[i].rep.rep.is_self_dual() && !structure[i + 1].rep.rep.is_self_dual() {
365                base_start = i + 1;
366            }
367
368            if structure[i].rep.rep.is_base()
369                && !structure[i + 1].rep.rep.is_self_dual()
370                && structure[i + 1].rep.rep.is_dual()
371            {
372                dual_start = i + 1;
373            }
374        }
375
376        let index_permutation = Permutation::sort(&structure);
377        index_permutation.apply_slice_in_place(&mut structure);
378
379        PermutedStructure {
380            structure: OrderedStructure {
381                structure,
382                base_start,
383                dual_start,
384            },
385            rep_permutation,
386            index_permutation,
387        }
388    }
389}

479        table.build().with(Style::rounded()).fmt(f)
480    }

543    ) -> Result<DenseTensor<Atom, Self>>
544    where
545        Self: Sized,
546        T: TensorCoefficient,
547    {

551            let im = index_to_atom(&self, index.into()).to_atom_im().unwrap();

554        }
555
556        Ok(DenseTensor {

558            structure: self,
559        })

577impl<R: RepName<Dual = R>, Aind: AbsInd> StructureContract for OrderedStructure<R, Aind> {
578    fn trace(&mut self, i: usize, j: usize) {
579        if i < j {
580            self.trace(j, i);
581            return;
582        }
583        let a = self.structure.remove(i);
584        let b = self.structure.remove(j);
585        assert!(a.matches(&b), "cannot trace unmatched slots {a} and {b}");
586    }
587
588    fn trace_out(&mut self) {
589        while let Some([i, j]) = self.traces().first().copied() {
590            self.trace(i, j);
591        }
592    }
593
594    fn merge(
595        &self,
596        other: &Self,
597    ) -> Result<(Self, SubSet<SlotIndex>, SubSet<SlotIndex>, MergeInfo), StructureError> {
598        // println!("Merge called with self: {} and other: {}", self, other);
599        debug_assert!(
600            self.self_dual_slice().windows(3).all(|w| w[0] <= w[1]
601                && w[1] <= w[2]
602                && !(w[0].matches(&w[1]) && w[1].matches(&w[2]))),
603            "Input vector 'self' to merge must be sorted in self dual part: {:?}",
604            self.self_dual_slice()
605        );
606        debug_assert!(
607            self.dual_slice().windows(2).all(|w| w[0] < w[1]),
608            "Input vector 'self' to merge must be sorted in dual part: {:?}",
609            self.dual_slice()
610        );
611        debug_assert!(
612            self.base_slice().windows(2).all(|w| w[0] < w[1]),
613            "Input vector 'self' to merge must be sorted in base part: {:?}",
614            self.base_slice()
615        );
616
617        debug_assert!(
618            other.self_dual_slice().windows(3).all(|w| w[0] <= w[1]
619                && w[1] <= w[2]
620                && !(w[0].matches(&w[1]) && w[1].matches(&w[2]))),
621            "Input vector 'other' to merge must be sorted in self dual part: {:?}",
622            other.self_dual_slice()
623        );
624        debug_assert!(
625            other.dual_slice().windows(2).all(|w| w[0] < w[1]),
626            "Input vector 'other' to merge must be sorted in dual part: {:?}",
627            other.dual_slice()
628        );
629        debug_assert!(
630            other.base_slice().windows(2).all(|w| w[0] < w[1]),
631            "Input vector 'other' to merge must be sorted in base part: {:?}",
632            other.base_slice()
633        );
634
635        let mut common_indices_self = SubSet::empty(self.order());
636        let mut common_indices_other = SubSet::empty(other.order());
637
638        if self.is_scalar() || other.is_scalar() {
639            // If either is empty, there are no common slots
640            if self.is_scalar() {
641                return Ok((
642                    other.clone(),
643                    common_indices_self,
644                    common_indices_other,
645                    MergeInfo::SecondBeforeFirst,
646                ));
647            } else {
648                return Ok((
649                    self.clone(),
650                    common_indices_self,
651                    common_indices_other,
652                    MergeInfo::FirstBeforeSecond,
653                ));
654            }
655        }
656
657        let mut partition: SubSet<SlotIndex> = SubSet::empty(0); // For interleaved mergeinfo
658        // partition.reserve(self.order() + other.order());
659
660        let mut i = 0;
661        let mut j = 0;
662
663        // Merge self dual parts:
664
665        let mut resulting_structure = Vec::new();
666
667        while i < self.n_self_dual() && j < other.n_self_dual() {
668            match self.structure[i].cmp(&other.structure[j]) {
669                std::cmp::Ordering::Less => {
670                    resulting_structure.push(self.structure[i]);
671                    partition.push(true);
672                    i += 1;
673                }
674                std::cmp::Ordering::Greater => {
675                    resulting_structure.push(other.structure[j]);
676                    partition.push(false);
677                    j += 1;
678                }
679                std::cmp::Ordering::Equal => {
680                    if self.structure[i].matches(&other.structure[j]) {
681                        // Common item
682                        common_indices_self.add(SlotIndex(i)); //, true); // Set bit for index in first vector
683                        common_indices_other.add(SlotIndex(j)); // Set bit for index in second vector
684                        i += 1;
685                        j += 1;
686                        continue; // Skip transition check for common items since they're not added to result_non_common
687                    } else {
688                        panic!("Matching and equal items must be identical")
689                    }
690                }
691            }
692        }
693
694        while i < self.n_self_dual() {
695            resulting_structure.push(self.structure[i]);
696            partition.push(true);
697            i += 1;
698        }
699
700        while j < other.n_self_dual() {
701            resulting_structure.push(other.structure[j]);
702            partition.push(false);
703            j += 1;
704        }
705
706        let base_start = partition.len().0;
707
708        //Merge base with dual
709
710        let mut ibase = 0;
711        let mut idual = 0;
712        let mut jbase = 0;
713        let mut jdual = 0;
714
715        let snbase = self.n_base();
716        let onbase = other.n_base();
717
718        let find_match_in_duals = |base_slot: &Slot<R, Aind>,
719                                   base_slot_index: usize,
720                                   common_indices_base: &mut SubSet<SlotIndex>,
721                                   dual_structure: &Self,
722                                   dual_cursor: &mut usize,
723                                   common_indices_dual: &mut SubSet<SlotIndex>,
724                                   dual_offset: usize|
725         -> bool {
726            let mut found_match = false;
727            let dual_count = dual_structure.n_dual();
728
729            // println!("dual_count: {}", dual_count);
730            // println!("dual_cursor: {}", dual_cursor);
731            while *dual_cursor < dual_count {
732                let dual_slot_index = dual_offset + *dual_cursor;
733                let dual_slot = &dual_structure.structure[dual_slot_index];
734                // println!("{}vs{}", base_slot, dual_slot);
735                match base_slot.match_cmp(dual_slot) {
736                    std::cmp::Ordering::Less => {
737                        // print!("less");
738                        break;
739                    }
740                    std::cmp::Ordering::Greater => {
741                        // print!("more");
742                        *dual_cursor += 1;
743                    }
744                    std::cmp::Ordering::Equal => {
745                        if base_slot.matches(dual_slot) {
746                            common_indices_base.add(SlotIndex(base_slot_index));
747                            common_indices_dual.add(SlotIndex(dual_slot_index));
748                            found_match = true;
749                            *dual_cursor += 1;
750                            break;
751                        } else {
752                            // return Err(StructureError::SlotError(SlotError));
753                            panic!("Matching and equal items must be identical");
754                        }
755                    }
756                }
757            }
758            found_match
759        };
760
761        while ibase < snbase && jbase < onbase {
762            match self.structure[i + ibase].cmp(&other.structure[j + jbase]) {
763                std::cmp::Ordering::Less => {
764                    let base_slot = &self.structure[i + ibase];
765                    let found_match = find_match_in_duals(
766                        base_slot,
767                        i + ibase,
768                        &mut common_indices_self,
769                        other,
770                        &mut jdual,
771                        &mut common_indices_other,
772                        j + onbase,
773                    );
774                    if !found_match {
775                        resulting_structure.push(*base_slot);
776                        partition.push(true);
777                    }
778                    ibase += 1;
779                }
780                std::cmp::Ordering::Greater => {
781                    let base_slot = &other.structure[j + jbase];
782                    let found_match = find_match_in_duals(
783                        base_slot,
784                        j + jbase,
785                        &mut common_indices_other,
786                        self,
787                        &mut idual,
788                        &mut common_indices_self,
789                        i + snbase,
790                    );
791                    if !found_match {
792                        resulting_structure.push(*base_slot);
793                        partition.push(false);
794                    }
795                    jbase += 1;
796                }
797                std::cmp::Ordering::Equal => {
798                    panic!(
799                        "cannot merge ordered structures with equal base slots at self[{left}] and other[{right}]: {slot}\nself: {self:?}\nother: {other:?}",
800                        left = i + ibase,
801                        right = j + jbase,
802                        slot = self.structure[i + ibase],
803                    )
804                }
805            }
806        }
807
808        while ibase < snbase {
809            let base_slot = &self.structure[i + ibase];
810            let found_match = find_match_in_duals(
811                base_slot,
812                i + ibase,
813                &mut common_indices_self,
814                other,
815                &mut jdual,
816                &mut common_indices_other,
817                j + onbase,
818            );
819            if !found_match {
820                resulting_structure.push(*base_slot);
821                partition.push(true);
822            }
823            ibase += 1;
824        }
825
826        while jbase < onbase {
827            let base_slot = &other.structure[j + jbase];
828            let found_match = find_match_in_duals(
829                base_slot,
830                j + jbase,
831                &mut common_indices_other,
832                self,
833                &mut idual,
834                &mut common_indices_self,
835                i + snbase,
836            );
837            if !found_match {
838                resulting_structure.push(*base_slot);
839                partition.push(false);
840            }
841            jbase += 1;
842        }
843
844        let dual_start = partition.len().0;
845
846        let mut idual = 0;
847        let mut jdual = 0;
848
849        while idual < self.n_dual() && jdual < other.n_dual() {
850            match self.structure[i + ibase + idual].cmp(&other.structure[j + jbase + jdual]) {
851                Ordering::Less => {
852                    if !common_indices_self[SlotIndex(i + ibase + idual)] {
853                        resulting_structure.push(self.structure[i + ibase + idual]);
854                        partition.push(true);
855                    }
856                    idual += 1;
857                }
858                Ordering::Greater => {
859                    if !common_indices_other[SlotIndex(j + jbase + jdual)] {
860                        resulting_structure.push(other.structure[j + jbase + jdual]);
861                        partition.push(false);
862                    }
863                    jdual += 1;
864                }
865                Ordering::Equal => {
866                    panic!(
867                        "cannot merge ordered structures with equal dual slots at self[{left}] and other[{right}]: {slot}\nself: {self:?}\nother: {other:?}",
868                        left = i + ibase + idual,
869                        right = j + jbase + jdual,
870                        slot = self.structure[i + ibase + idual],
871                    )
872                }
873            }
874        }
875
876        while idual < self.n_dual() {
877            if !common_indices_self[SlotIndex(i + ibase + idual)] {
878                resulting_structure.push(self.structure[i + ibase + idual]);
879                partition.push(true);
880            }
881            idual += 1;
882        }
883
884        while jdual < other.n_dual() {
885            if !common_indices_other[SlotIndex(j + jbase + jdual)] {
886                resulting_structure.push(other.structure[j + jbase + jdual]);
887                partition.push(false);
888            }
889            jdual += 1;
890        }
891
892        assert!(
893            Permutation::sort(&resulting_structure).is_identity(),
894            "Permutation is not identity for  {}\n{:?}\nself:{:?}\n other:{:?}",
895            Permutation::sort(&resulting_structure),
896            OrderedStructure {
897                structure: resulting_structure,
898                dual_start,
899                base_start,
900            },
901            self,
902            other
903        );
904
905        // println!(
906        //     "Merged structure: {}",
907        //     OrderedStructure {
908        //         structure: resulting_structure.clone(),
909        //         dual_start,
910        //         base_start,
911        //     }
912        // );
913
914        // println!(
915        //     "Common indices self: {:?}\nCommon indices other: {:?}",
916        //     common_indices_self, common_indices_other
917        // );
918        // println!("Partition: {:?}", partition);
919
920        Ok((
921            OrderedStructure {
922                structure: resulting_structure,
923                dual_start,
924                base_start,
925            },
926            common_indices_self,
927            common_indices_other,
928            partition.into(),
929        ))
930    }
