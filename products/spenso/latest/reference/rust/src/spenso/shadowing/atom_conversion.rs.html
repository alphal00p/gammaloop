7/// Converts common symbolic Spenso values into owned Symbolica atoms.
8///
9/// This is used by the variadic chain and trace macros so callers can pass
10/// representation slots, stripped representations, symbols, atoms, or atom
11/// views without spelling out the conversion at every call site.
12pub trait IntoAtom {
13    fn into_atom(self) -> Atom;
14}
15
16macro_rules! impl_into_atom_via_atom_or_view {
17    (<$lt:lifetime> $($ty:ty),+ $(,)?) => {
18        $(
19            impl<$lt> IntoAtom for $ty {
20                fn into_atom(self) -> Atom {
21                    let atom: AtomOrView<$lt> = self.into();
22                    atom.into_owned()
23                }
24            }
25        )+
26    };
27    ($($ty:ty),+ $(,)?) => {
28        $(
29            impl IntoAtom for $ty {
30                fn into_atom(self) -> Atom {
31                    let atom: AtomOrView<'_> = self.into();
32                    atom.into_owned()
33                }
34            }
35        )+
36    };
37}
38
39impl_into_atom_via_atom_or_view!(Atom, Symbol);
40impl_into_atom_via_atom_or_view!(<'a> &'a Atom, AtomView<'a>, AtomOrView<'a>);
41
42macro_rules! impl_into_atom_integer {
43    ($($ty:ty),+ $(,)?) => {
44        $(
45            impl IntoAtom for $ty {
46                fn into_atom(self) -> Atom {
47                    Atom::num(self)
48                }
49            }
50        )+
51    };
52}
53
54macro_rules! impl_into_atom_integer_via {
55    ($target:ty; $($ty:ty),+ $(,)?) => {
56        $(
57            impl IntoAtom for $ty {
58                fn into_atom(self) -> Atom {
59                    Atom::num(<$target>::from(self))
60                }
61            }
62        )+
63    };
64}

67impl_into_atom_integer_via!(i64; i8, i16);
68impl_into_atom_integer_via!(u64; u8, u16);
