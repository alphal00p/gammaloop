67fn normalize_cyclic_projector(view: AtomView<'_>, out: &mut Settable<Atom>) {
68    let AtomView::Fun(fun) = view else {
69        return;
70    };
71    let args = fun.iter().collect::<Vec<_>>();
72    if let [AtomView::Fun(inner)] = args.as_slice()
73        && inner.get_symbol() == *SYM
74    {
75        out.set_from_view(&args[0]);
76    }
77}

80where
81    F: IntoAtom,

107fn expand_projectors_impl(expression: AtomView) -> Atom {
108    let mut current = expression.to_owned();
109    loop {
110        let next = current
111            .replace_map(|arg, _context, out| {
112                if let Some(expanded) = expand_chain_like_projector(arg) {
113                    **out = expanded;
114                }
115            })
116            .collect_tensors();
117
118        if next == current {
119            return next;
120        }
121        current = next;
122    }
123}

125fn expand_chain_like_projector(arg: AtomView) -> Option<Atom> {
126    let AtomView::Fun(f) = arg else {
127        return None;
128    };
129    let offset = if f.get_symbol() == SPENSO_TAG.chain {
130        2
131    } else if f.get_symbol() == SPENSO_TAG.trace {
132        1
133    } else {
134        return None;
135    };
136    if f.get_nargs() < offset {
137        return None;
138    }
139
140    let args = f.iter().collect::<Vec<_>>();
141    let (prefix, raw_factors) = args.split_at(offset);
142    let factors = if f.get_symbol() == SPENSO_TAG.trace {
143        trace_factor_views(raw_factors)
144    } else {
145        raw_factors.to_vec()
146    };
147    for (position, factor) in factors.iter().enumerate() {
148        let Some((prefactor, projector_symbol, projector_factors)) = projector_factor(*factor)
149        else {
150            continue;
151        };
152
153        let mut sum = Atom::Zero;
154        let is_trace = f.get_symbol() == SPENSO_TAG.trace;
155        for (coefficient, expanded_factors) in
156            projector_expansion_terms(&projector_factors, projector_symbol)?
157        {
158            let new_factors = factors[..position]
159                .iter()
160                .map(|factor| factor.to_owned())
161                .chain(expanded_factors)
162                .chain(
163                    factors[position + 1..]
164                        .iter()
165                        .map(|factor| factor.to_owned()),
166                );
167            let rebuilt = if is_trace {
168                trace_with_cyclic_factors(prefix, new_factors)
169            } else {
170                let prefix = prefix.iter().map(|arg| arg.to_owned()).collect::<Vec<_>>();
171                chain_like_with_factors(f.get_symbol(), &prefix, new_factors)
172            };
173            sum += prefactor.clone() * coefficient * rebuilt;
174        }
175        return Some(sum);
176    }
177
178    None
179}

181fn projector_factor(factor: AtomView) -> Option<(Atom, Symbol, Vec<Atom>)> {
182    if let Some((projector_symbol, factors)) = projector_parts(factor) {
183        return Some((Atom::num(1), projector_symbol, factors));
184    }
185
186    let AtomView::Mul(product) = factor else {
187        return None;
188    };
189
190    let mut prefactor = Atom::num(1);
191    let mut projector = None;
192    for factor in product.iter() {
193        if let Some(parts) = projector_parts(factor) {
194            if projector.is_some() {
195                return None;
196            }
197            projector = Some(parts);
198        } else if matches!(factor, AtomView::Num(_)) {
199            prefactor *= factor.to_owned();
200        } else {
201            return None;
202        }
203    }
204
205    let (projector_symbol, factors) = projector?;
206    Some((prefactor, projector_symbol, factors))
207}

209fn projector_parts(projector: AtomView) -> Option<(Symbol, Vec<Atom>)> {
210    let AtomView::Fun(f) = projector else {
211        return None;
212    };
213    if f.get_symbol() != *SYM && f.get_symbol() != *ANTISYM && f.get_symbol() != *CYCLIC {
214        return None;
215    }
216
217    Some((f.get_symbol(), f.iter().map(|arg| arg.to_owned()).collect()))
218}

220fn projector_expansion_terms(
221    factors: &[Atom],
222    projector_symbol: Symbol,
223) -> Option<Vec<(Atom, Vec<Atom>)>> {
224    if projector_symbol == *CYCLIC {
225        return cyclic_projector_terms(factors);
226    }
227
228    let denominator = Atom::num(factorial_i64(factors.len())?);
229    let mut permutation = (0..factors.len()).collect::<Vec<_>>();
230    let mut terms = Vec::new();
231    collect_projector_terms(
232        factors,
233        projector_symbol == *ANTISYM,
234        0,
235        1,
236        &mut permutation,
237        &denominator,
238        &mut terms,
239    );
240    Some(terms)
241}

243fn cyclic_projector_terms(factors: &[Atom]) -> Option<Vec<(Atom, Vec<Atom>)>> {
244    if factors.is_empty() {
245        return Some(vec![(Atom::num(1), Vec::new())]);
246    }
247
248    let denominator = Atom::num(i64::try_from(factors.len()).ok()?);
249    Some(
250        (0..factors.len())
251            .map(|shift| {
252                let rotated = factors
253                    .iter()
254                    .cycle()
255                    .skip(shift)
256                    .take(factors.len())
257                    .cloned()
258                    .collect();
259                (Atom::num(1) / denominator.clone(), rotated)
260            })
261            .collect(),
262    )
263}

265fn collect_projector_terms(
266    factors: &[Atom],
267    antisymmetric: bool,
268    position: usize,
269    sign: i64,
270    permutation: &mut [usize],
271    denominator: &Atom,
272    terms: &mut Vec<(Atom, Vec<Atom>)>,
273) {
274    if position == permutation.len() {
275        let coefficient = if antisymmetric {
276            Atom::num(sign) / denominator.clone()
277        } else {
278            Atom::num(1) / denominator.clone()
279        };
280        let permuted = permutation
281            .iter()
282            .map(|index| factors[*index].clone())
283            .collect();
284        terms.push((coefficient, permuted));
285        return;
286    }
287
288    for i in position..permutation.len() {
289        permutation.swap(position, i);
290        let next_sign = if i == position { sign } else { -sign };
291        collect_projector_terms(
292            factors,
293            antisymmetric,
294            position + 1,
295            next_sign,
296            permutation,
297            denominator,
298            terms,
299        );
300        permutation.swap(position, i);
301    }
302}

304fn factorial_i64(n: usize) -> Option<i64> {
305    (1..=n).try_fold(1_i64, |acc, value| {
306        acc.checked_mul(i64::try_from(value).ok()?)
307    })
308}
