9pub fn trace<R, F>(rep: R, factors: impl IntoIterator<Item = F>) -> Atom
10where
11    R: IntoAtom,
12    F: IntoAtom,
13{
14    let rep = rep.into_atom();
15    let factors = factors
16        .into_iter()
17        .map(IntoAtom::into_atom)
18        .collect::<Vec<_>>();
19
20    if factors.is_empty() {
21        return chain_like_with_factors(SPENSO_TAG.trace, &[rep], std::iter::empty());
22    }
23
24    let trace_factor = if let [factor] = factors.as_slice() {
25        if is_cyclic_projector(factor.as_view()) {
26            factor.clone()
27        } else {
28            cyclic(factors)
29        }
30    } else {
31        cyclic(factors)
32    };
33
34    chain_like_with_factors(SPENSO_TAG.trace, &[rep], std::iter::once(trace_factor))
35}

37/// Builds a trace whose payload is a single symmetric projector.
38///
39/// This is the compact form for fully symmetric trace invariants such as color
40/// `d` tensors: `trace(rep, sym(factors...))`. Ordinary ordered closed traces
41/// should use [`trace`], which wraps factors in `cyclic(...)` instead.
42pub fn trace_sym<R, F>(rep: R, factors: impl IntoIterator<Item = F>) -> Atom
43where
44    R: IntoAtom,
45    F: IntoAtom,
46{
47    let rep = rep.into_atom();
48    let projector = sym(factors);
49    chain_like_with_factors(SPENSO_TAG.trace, &[rep], std::iter::once(projector))
50}

52/// Return the representation and actual factor sequence of a trace.
53///
54/// Canonical traces store their factors under a single `cyclic(...)` argument;
55/// this helper unwraps that representation for algorithms that need the
56/// ordered factor list. The non-canonical raw shape `trace(rep, factors...)` is
57/// accepted on input so readers can normalize externally constructed atoms, but
58/// all builders emit the cyclic form.
59pub fn trace_parts<'a>(fun: FunView<'a>) -> Option<(AtomView<'a>, Vec<AtomView<'a>>)> {
60    if fun.get_symbol() != SPENSO_TAG.trace {
61        return None;
62    }
63
64    let args = fun.iter().collect::<Vec<_>>();
65    let [rep, factors @ ..] = args.as_slice() else {
66        return None;
67    };
68
69    Some((*rep, trace_factor_views(factors)))
70}
71
72/// Flatten the canonical cyclic trace factor wrapper.
73pub fn trace_factor_views<'a>(factors: &[AtomView<'a>]) -> Vec<AtomView<'a>> {
74    if let [factor] = factors
75        && let AtomView::Fun(fun) = *factor
76        && fun.get_symbol() == *CYCLIC
77    {
78        return fun.iter().collect();
79    }
80
81    factors.to_vec()
82}
83
84fn is_cyclic_projector(factor: AtomView<'_>) -> bool {
85    matches!(factor, AtomView::Fun(fun) if fun.get_symbol() == *CYCLIC)
86}

88pub(crate) fn chain_like_with_factors(
89    symbol: Symbol,
90    prefix: &[Atom],
91    factors: impl IntoIterator<Item = Atom>,
92) -> Atom {
93    prefix
94        .iter()
95        .cloned()
96        .chain(factors)
97        .fold(FunctionBuilder::new(symbol), |builder, arg| {
98            builder.add_arg(arg)
99        })
100        .finish()
101}
