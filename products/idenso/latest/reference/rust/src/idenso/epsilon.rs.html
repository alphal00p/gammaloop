9spenso::symbolica_init_lazy_static! {
10    /// Symbolica-level Levi-Civita symbol. The `Antisymmetric` attribute lets
11    /// Symbolica canonicalize argument order and annihilate repeated arguments.
12    pub static EPSILON_SYMBOL, EPSILON_SYMBOL_INNER: Symbol =
13        || tensor_symbol!("spenso::epsilon"; Antisymmetric);
14}
15
16/// Builds an antisymmetric epsilon tensor with arbitrary rank.
17pub fn epsilon<'a, F>(factors: impl IntoIterator<Item = F>) -> Atom
18where
19    F: Into<AtomOrView<'a>>,
20{
21    factors
22        .into_iter()
23        .fold(FunctionBuilder::new(*EPSILON_SYMBOL), |builder, factor| {
24            builder.add_arg(factor)
25        })
26        .finish()
27}

29/// Builds an antisymmetric epsilon tensor with arbitrary rank.
30///
31/// Arguments are converted through `spenso::shadowing::IntoAtom`, so tests
32/// and rewrite code can pass slots, atoms, atom views, symbols, and integers
33/// without spelling out the conversion.
34#[macro_export]
35macro_rules! epsilon {
36    (; $factors:expr $(,)?) => {
37        $crate::epsilon::epsilon(($factors).into_iter())
38    };
39    ($($arg:expr),* $(,)?) => {{
40        let builder = symbolica::atom::FunctionBuilder::new(*$crate::epsilon::EPSILON_SYMBOL);
41        $(
42            let builder = builder.add_arg(spenso::shadowing::IntoAtom::into_atom($arg));
43        )*
44        builder.finish()
45    }};
46}
47
48/// Builds a four-dimensional Levi-Civita tensor.
49pub fn epsilon4<'a, 'b, 'c, 'd>(
50    mu: impl Into<AtomOrView<'a>>,
51    nu: impl Into<AtomOrView<'b>>,
52    rho: impl Into<AtomOrView<'c>>,
53    sigma: impl Into<AtomOrView<'d>>,
54) -> Atom {
55    FunctionBuilder::new(*EPSILON_SYMBOL)
56        .add_arg(mu)
57        .add_arg(nu)
58        .add_arg(rho)
59        .add_arg(sigma)
60        .finish()
61}

63/// Simplifies epsilon-metric contractions and epsilon-pair contractions.
64pub trait EpsilonSimplifier {
65    fn simplify_epsilon(&self) -> Atom;
66}
67
68impl EpsilonSimplifier for Atom {
69    fn simplify_epsilon(&self) -> Atom {
70        self.as_view().simplify_epsilon()
71    }
72}
73
74impl EpsilonSimplifier for AtomView<'_> {
75    fn simplify_epsilon(&self) -> Atom {
76        EpsilonSimplifierPass::run(*self)
77    }
78}
79
80struct EpsilonSimplifierPass;
81
82impl EpsilonSimplifierPass {
83    /// Runs the epsilon identities to a fixed point, then lets metric
84    /// simplification consume any determinants that collapsed to traces.
85    fn run(expr: AtomView<'_>) -> Atom {
86        // Force Symbolica to register epsilon as antisymmetric before any
87        // builders below create new epsilon nodes.
88        let _ = *EPSILON_SYMBOL;
89        if !Self::contains_epsilon(expr) {
90            return expr.to_owned();
91        }
92
93        let mut current = expr.schoonschip();
94
95        loop {
96            let next = current
97                .replace_map(|term, _context, out| {
98                    if let Some(rewritten) = Self::simplify_power(term.schoonschip().as_view())
99                        .or_else(|| Self::simplify_pair_product(term))
100                    {
101                        **out = rewritten;
102                    }
103                })
104                .schoonschip();
105
106            if next == current {
107                return next;
108            }
109
110            current = next;
111        }
112    }
113
114    fn contains_epsilon(expr: AtomView<'_>) -> bool {
115        match expr {
116            AtomView::Fun(f) => {
117                f.get_symbol() == *EPSILON_SYMBOL || f.iter().any(Self::contains_epsilon)
118            }
119            AtomView::Add(add) => add.iter().any(Self::contains_epsilon),
120            AtomView::Mul(mul) => mul.iter().any(Self::contains_epsilon),
121            AtomView::Pow(pow) => pow.iter().any(Self::contains_epsilon),
122            AtomView::Num(_) | AtomView::Var(_) => false,
123        }
124    }
125
126    /// Applies the repeated-epsilon identity to powers:
127    /// `epsilon(a_1,...,a_n)^k -> det(g(a_i,a_j)) epsilon(a_1,...,a_n)^(k-2)`.
128    ///
129    /// This handles the common canonical form where Symbolica has already
130    /// merged an epsilon product into a positive integer power.
131    fn simplify_power(expr: AtomView<'_>) -> Option<Atom> {
132        let AtomView::Pow(pow) = expr else {
133            return None;
134        };
135        let (base, exponent) = pow.get_base_exp();
136        let epsilon_args = Self::epsilon_args(base)?;
137        let exponent = Self::positive_integer(exponent)?;
138        if exponent < 2 {
139            return None;
140        }
141
142        Some(
143            Self::metric_determinant(&epsilon_args, &epsilon_args)
144                * Self::power(base, exponent - 2),
145        )
146    }
147
148    /// Applies the epsilon-pair contraction:
149    /// `epsilon(a_1,...,a_n) epsilon(b_1,...,b_n) -> det(g(a_i,b_j))`.
150    ///
151    /// Extra multiplicative factors are preserved around the determinant.
152    fn simplify_pair_product(expr: AtomView<'_>) -> Option<Atom> {
153        let factors = Self::multiplicative_factors(expr);
154        if factors.len() < 2 {
155            return None;
156        }
157
158        for (left_index, left) in factors.iter().enumerate() {
159            let Some(left_args) = Self::epsilon_args(*left) else {
160                continue;
161            };
162
163            for (right_index, right) in factors.iter().enumerate().skip(left_index + 1) {
164                let Some(right_args) = Self::epsilon_args(*right) else {
165                    continue;
166                };
167                if left_args.len() != right_args.len() || left_args.is_empty() {
168                    continue;
169                }
170
171                let determinant = Self::metric_determinant(&left_args, &right_args);
172                let mut excluded = vec![false; factors.len()];
173                excluded[left_index] = true;
174                excluded[right_index] = true;
175                return Some(Self::product_excluding(&factors, &excluded) * determinant);
176            }
177        }
178
179        None
180    }
181
182    fn positive_integer(expr: AtomView<'_>) -> Option<i64> {
183        let AtomView::Num(number) = expr else {
184            return None;
185        };
186        let CoefficientView::Natural(value, 1, 0, 1) = number.get_coeff_view() else {
187            return None;
188        };
189
190        (value > 0).then_some(value)
191    }
192
193    fn power(base: AtomView<'_>, exponent: i64) -> Atom {
194        match exponent {
195            0 => Atom::num(1),
196            1 => base.to_owned(),
197            _ => base.to_owned().pow(Atom::num(exponent)),
198        }
199    }
200
201    /// Emits the determinant `det(g(left_i,right_j))` by summing over all
202    /// signed permutations of the right indices.
203    fn metric_determinant(left: &[Atom], right: &[Atom]) -> Atom {
204        let mut permutation = (0..right.len()).collect::<Vec<_>>();
205        let mut sum = Atom::Zero;
206        Self::determinant_terms(left, right, 0, &mut permutation, 1, &mut sum);
207        sum
208    }
209
210    fn determinant_terms(
211        left: &[Atom],
212        right: &[Atom],
213        position: usize,
214        permutation: &mut [usize],
215        sign: i64,
216        sum: &mut Atom,
217    ) {
218        if position == permutation.len() {
219            let term = left
220                .iter()
221                .enumerate()
222                .fold(Atom::num(sign), |product, (i, arg)| {
223                    product * g!(arg.clone(), right[permutation[i]].clone())
224                });
225            *sum += term;
226            return;
227        }
228
229        for i in position..permutation.len() {
230            permutation.swap(position, i);
231            let next_sign = if i == position { sign } else { -sign };
232            Self::determinant_terms(left, right, position + 1, permutation, next_sign, sum);
233            permutation.swap(position, i);
234        }
235    }
236
237    fn epsilon_args(epsilon: AtomView<'_>) -> Option<Vec<Atom>> {
238        let AtomView::Fun(f) = epsilon else {
239            return None;
240        };
241        if f.get_symbol() != *EPSILON_SYMBOL {
242            return None;
243        }
244
245        Some(f.iter().map(|arg| arg.to_owned()).collect())
246    }
247
248    fn multiplicative_factors<'a>(expr: AtomView<'a>) -> Vec<AtomView<'a>> {
249        match expr {
250            AtomView::Mul(mul) => mul.iter().collect(),
251            _ => vec![expr],
252        }
253    }
254
255    fn product_excluding(factors: &[AtomView<'_>], excluded: &[bool]) -> Atom {
256        factors
257            .iter()
258            .enumerate()
259            .filter(|(index, _)| !excluded[*index])
260            .fold(Atom::num(1), |product, (_, factor)| {
261                product * factor.to_owned()
262            })
263    }
264}
