46    pub(crate) fn run(&self, expression: AtomView<'_>) -> Atom {
47        let mut current = expression.simplify_metrics();
48
49        loop {
50            let next = self.apply_once(current.as_view());
51            if next == current {
52                // Every antisymmetric color tensor carries adjoint slots, so this
53                // isolates its complete network without expanding other sectors.
54                let mut pruned = false;
55                let canonicalized =
56                    next.collect_rep_with_map(ColorAdjoint {}.into(), |factor, _context, out| {
57                        let AtomView::Fun(collected) = factor else {
58                            return;
59                        };
60                        let Some(color) = collected.iter().next() else {
61                            return;
62                        };
63                        let reduced = remove_antisymmetric_zero_terms::<AbstractIndex>(color);
64                        pruned |= reduced.as_view() != color;
65                        **out = reduced;
66                    });
67                if pruned {
68                    current = canonicalized;
69                    continue;
70                }
71                let simplified =
72                    restore_explicit_default_generator_chains(canonicalized).simplify_metrics();
73                return if self.settings.substitute_cof_dimension_invariants {
74                    simplified.to_cof_dimension_invariants()
75                } else {
76                    simplified
77                };
78            }
79            current = next;
80        }
81    }

83    fn apply_once(&self, expression: AtomView<'_>) -> Atom {
84        let collected = self.collect_lines(expression);
85        self.rewrite_terms(collected.as_view())
86            .collect_color()
87            .simplify_metrics()
88    }

90    fn rewrite_terms(&self, expr: AtomView<'_>) -> Atom {
91        // Terminal trace rules can create sums; product rules such as f*f -> CA*g
92        // then need to run on each generated term instead of on the whole Add.
93        if let AtomView::Add(add) = expr {
94            return add
95                .iter()
96                .map(|term| self.rewrite_terms(term))
97                .fold(Atom::Zero, |sum, term| sum + term);
98        }
99
100        if let Some(rewritten) = self.rewrite_node(expr) {
101            return rewritten;
102        }
103
104        // Try product-level rewrites first so trace*f contractions can fire before the
105        // trace terminal expands into symmetric trace and f terms.
106        expr.to_owned().replace_map(|arg, _context, out| {
107            if let Some(rewritten) = self.rewrite_node(arg) {
108                **out = rewritten;
109            }
110        })
111    }

113    fn collect_lines(&self, expr: AtomView<'_>) -> Atom {
114        let rep = ColorFundamental {}.into();
115        expr.to_owned()

118            .replace_map(&Self::close_trace_chain)
119            .replace_map(&Self::collapse_identity_chain)

122    fn close_trace_chain(arg: AtomView, _context: &Context, out: &mut Settable<'_, Atom>) {
123        let Some((start, end, factors)) = chain_parts(arg) else {
124            return;
125        };
126        let Some((dimension, start_index, false)) = color_fundamental_slot(start.as_view()) else {
127            return;
128        };
129        let Some((end_dimension, end_index, true)) = color_fundamental_slot(end.as_view()) else {
130            return;
131        };
132        if dimension != end_dimension || start_index != end_index {
133            return;
134        }
135
136        **out = trace!(ColorFundamental {}.to_symbolic([dimension]); factors);
137    }

140        let Some((start, end, factors)) = chain_parts(arg) else {

151    fn rewrite_node(&self, arg: AtomView<'_>) -> Option<Atom> {
152        self.simplify_product(arg)
153            .or_else(|| Self::simplify_chain_node(arg))
154            .or_else(|| {
155                self.settings
156                    .evaluate_traces
157                    .then(|| Self::simplify_trace_node(arg))
158                    .flatten()
159            })
160            .or_else(|| self.simplify_power(arg))

163    fn simplify_chain_node(chain: AtomView) -> Option<Atom> {
164        let (start, end, factors) = chain_parts(chain)?;
165        if factors.is_empty()
166            || factors
167                .iter()
168                .all(|factor| is_chain_identity_factor(factor.as_view()))
169        {
170            return Some(color_metric(start.clone(), end.clone()));
171        }
172
173        if let Some(identity_index) = factors
174            .iter()
175            .position(|factor| is_chain_identity_factor(factor.as_view()))
176        {
177            return Some(chain_with_factors(
178                start,
179                end,
180                factors_excluding_indices(&factors, &[identity_index]),
181            ));
182        }
183
184        if let Some(rewritten) =
185            Self::simplify_antisymmetric_chain_projector(&start, &end, &factors)
186        {
187            return Some(rewritten);
188        }
189
190        for i in 0..factors.len().saturating_sub(1) {
191            let Some(left) = color_generator_adjoint(factors[i].as_view()) else {
192                continue;
193            };
194            let Some(right) = color_generator_adjoint(factors[i + 1].as_view()) else {
195                continue;
196            };
197            if left != right {
198                continue;
199            }
200
201            return Some(
202                fundamental_casimir(fundamental_chain_dimension(&start, &end)?)
203                    * chain_with_removed_range(&start, &end, &factors, i, i + 2),
204            );
205        }
206
207        if factors.len() > 2
208            && let Some(rewritten) = Self::simplify_separated_chain_casimir(&start, &end, &factors)
209        {
210            return Some(rewritten);
211        }
212
213        None
214    }

216    fn simplify_trace_node(trace: AtomView) -> Option<Atom> {
217        let (rep, factors) = trace_parts(trace)?;
218
219        if factors.is_empty()
220            || factors
221                .iter()
222                .all(|factor| is_chain_identity_factor(factor.as_view()))
223        {
224            // Tr(1) and Tr(identity-line factors) collapse to the traced
225            // representation dimension.
226            return trace_terminal_dimension(rep.as_view());
227        }
228
229        if let Some(identity_index) = factors
230            .iter()
231            .position(|factor| is_chain_identity_factor(factor.as_view()))
232        {
233            // An identity line inside a longer trace is neutral:
234            // Tr(... 1 ... ) -> Tr(...).
235            return Some(trace_with_factors(
236                rep,
237                factors_excluding_indices(&factors, &[identity_index]),
238            ));
239        }
240
241        if let Some(rewritten) = Self::simplify_antisymmetric_trace_projector(&rep, &factors) {
242            return Some(rewritten);
243        }
244
245        if factors.len() > 2
246            && let Some(rewritten) = Self::simplify_adjacent_trace_casimir(&rep, &factors)
247        {
248            return Some(rewritten);
249        }
250
251        if factors.len() > 3
252            && let Some(rewritten) = Self::simplify_separated_trace_casimir(&rep, &factors)
253        {
254            return Some(rewritten);
255        }
256
257        let generators = factors
258            .iter()
259            .map(|factor| color_generator_adjoint(factor.as_view()))
260            .collect::<Option<Vec<_>>>()?;
261
262        match generators.as_slice() {
263            // Tr(T^a) -> 0.
264            [_] => Some(Atom::Zero),
265            // Tr(T^a T^b) -> idx(2,rep) g^{ab}.
266            [a, b] => Some(quadratic_index(rep.clone()) * color_metric(a.clone(), b.clone())),
267            // Tr(T^a T^b T^c) ->
268            //   Tr(sym(T^a,T^b,T^c)) + i/2 idx(2,rep) f^{abc}.
269            [a, b, c] => Some(
270                color_symmetric_trace(&rep, [a.clone(), b.clone(), c.clone()])
271                    + Atom::i() * Atom::num(1) / Atom::num(2)
272                        * quadratic_index(rep.clone())
273                        * color_f!(a.clone(), b.clone(), c.clone()),
274            ),
275            [a, b, c, d] => Self::simplify_four_generator_trace_terminal(&rep, a, b, c, d),
276            _ => None,
277        }
278    }

280    fn simplify_adjacent_trace_casimir(rep: &Atom, factors: &[Atom]) -> Option<Atom> {
281        for i in 0..factors.len().saturating_sub(1) {
282            let Some(left) = color_generator_adjoint(factors[i].as_view()) else {
283                continue;
284            };
285            let Some(right) = color_generator_adjoint(factors[i + 1].as_view()) else {
286                continue;
287            };
288            if left != right {
289                continue;
290            }
291
292            // Adjacent equal generators inside a fundamental trace:
293            // Tr(... T^a T^a ...) -> cas(2,rep) Tr(...).
294            return Some(
295                quadratic_casimir(rep.clone())
296                    * trace_with_factors(rep.clone(), factors_excluding_range(factors, i, i + 2)),
297            );
298        }
299
300        None
301    }

303    fn simplify_antisymmetric_chain_projector(
304        start: &Atom,
305        end: &Atom,
306        factors: &[Atom],
307    ) -> Option<Atom> {
308        // `antisym(T^a,T^b)` is the normalized commutator: i/2 f^{abx} T^x.
309        for (position, factor) in factors.iter().enumerate() {
310            let Some((prefactor, args)) = color_antisymmetric_generator_args(factor.as_view())
311            else {
312                continue;
313            };
314            let [a, b] = args.as_slice() else {
315                continue;
316            };
317            let x = color_adjoint_dummy_for_pair(a, b)?;
318
319            let mut replacement_factors = factors.to_vec();
320            replacement_factors[position] = color_t!(x.clone());
321            return Some(
322                prefactor * Atom::i() / Atom::num(2)
323                    * color_f!(a.clone(), b.clone(), x.clone())
324                    * chain_with_factors(start.clone(), end.clone(), replacement_factors),
325            );
326        }
327
328        None
329    }

331    fn simplify_antisymmetric_trace_projector(rep: &Atom, factors: &[Atom]) -> Option<Atom> {
332        if let [factor] = factors {
333            let (prefactor, args) = color_antisymmetric_generator_args(factor.as_view())?;
334            return match args.as_slice() {
335                // Tr(antisym(T^a,T^b)) is the trace of a commutator.
336                [_, _] => Some(Atom::Zero),
337                // Tr(antisym(T^a,T^b,T^c)) -> i/2 idx(2,rep) f^{abc}.
338                [a, b, c] => Some(
339                    prefactor
340                        * Atom::i()
341                        * quadratic_index(rep.clone())
342                        * color_f!(a.clone(), b.clone(), c.clone())
343                        / Atom::num(2),
344                ),
345                _ => None,
346            };
347        }
348
349        for (position, factor) in factors.iter().enumerate() {
350            let Some((prefactor, args)) = color_antisymmetric_generator_args(factor.as_view())
351            else {
352                continue;
353            };
354            let [a, b] = args.as_slice() else {
355                continue;
356            };
357            let x = color_adjoint_dummy_for_pair(a, b)?;
358
359            let mut replacement_factors = factors.to_vec();
360            replacement_factors[position] = color_t!(x.clone());
361            // In a longer trace, antisym(T^a,T^b) is the normalized
362            // commutator: i/2 f^{abx} T^x.
363            return Some(
364                prefactor * Atom::i() / Atom::num(2)
365                    * color_f!(a.clone(), b.clone(), x.clone())
366                    * trace_with_factors(rep.clone(), replacement_factors),
367            );
368        }
369
370        None
371    }

373    fn simplify_separated_trace_casimir(rep: &Atom, factors: &[Atom]) -> Option<Atom> {
374        for i in 0..factors.len().saturating_sub(2) {
375            let Some(left) = color_generator_adjoint(factors[i].as_view()) else {
376                continue;
377            };
378            let Some(right) = color_generator_adjoint(factors[i + 2].as_view()) else {
379                continue;
380            };
381            if left != right {
382                continue;
383            }
384
385            // Separated equal generators with one generator between them:
386            // Tr(... T^a T^b T^a ...) -> (cas(2,rep) - cas(2,adj)/2) Tr(... T^b ...).
387            return Some(
388                (quadratic_casimir(rep.clone())
389                    - adjoint_casimir_for_dimension(color_adjoint_dimension(&left)?)
390                        / Atom::num(2))
391                    * trace_with_factors(
392                        rep.clone(),
393                        factors_excluding_indices(factors, &[i, i + 2]),
394                    ),
395            );
396        }
397
398        None
399    }

401    fn simplify_separated_chain_casimir(
402        start: &Atom,
403        end: &Atom,
404        factors: &[Atom],
405    ) -> Option<Atom> {
406        for i in 0..factors.len().saturating_sub(2) {
407            let Some(left) = color_generator_adjoint(factors[i].as_view()) else {
408                continue;
409            };
410            let Some(right) = color_generator_adjoint(factors[i + 2].as_view()) else {
411                continue;
412            };
413            if left != right {
414                continue;
415            }
416
417            let coefficient = fundamental_casimir(fundamental_chain_dimension(start, end)?)
418                - adjoint_casimir_for_dimension(color_adjoint_dimension(&left)?) / Atom::num(2);
419
420            return Some(
421                coefficient
422                    * chain_with_factors(
423                        start.clone(),
424                        end.clone(),
425                        factors_excluding_indices(factors, &[i, i + 2]),
426                    ),
427            );
428        }
429
430        None
431    }

433    fn simplify_four_generator_trace_terminal(
434        rep: &Atom,
435        a: &Atom,
436        b: &Atom,
437        c: &Atom,
438        d: &Atom,
439    ) -> Option<Atom> {
440        let x = color_adjoint_dummy_like(a)?;
441
442        // Four-generator terminal decomposition:
443        // Tr(T^a T^b T^c T^d) is split into a fully symmetric trace, two
444        // f * symmetric-trace terms, and two f*f terms.
445        Some(
446            color_symmetric_trace(rep, [a.clone(), b.clone(), c.clone(), d.clone()])
447                + Atom::i() / Atom::num(2)
448                    * color_symmetric_trace(rep, [a.clone(), b.clone(), x.clone()])
449                    * color_f!(c.clone(), d.clone(), x.clone())
450                + Atom::i() / Atom::num(2)
451                    * color_symmetric_trace(rep, [c.clone(), d.clone(), x.clone()])
452                    * color_f!(a.clone(), b.clone(), x.clone())
453                - quadratic_index(rep.clone()) / Atom::num(6)
454                    * color_f!(a.clone(), c.clone(), x.clone())
455                    * color_f!(b.clone(), d.clone(), x.clone())
456                + quadratic_index(rep.clone()) / Atom::num(3)
457                    * color_f!(a.clone(), d.clone(), x.clone())
458                    * color_f!(b.clone(), c.clone(), x.clone()),
459        )
460    }

462    fn simplify_product(&self, product: AtomView) -> Option<Atom> {
463        let product = ProductView::parse(product);
464        if product.len() < 2 {
465            return None;
466        }
467
468        product
469            .distribute_color_sum_factor()
470            .or_else(|| Self::join_color_chain_product(&product))
471            .or_else(|| {
472                self.settings
473                    .evaluate_traces
474                    .then(|| self.simplify_trace_structure_product(&product))
475                    .flatten()
476            })
477            .or_else(|| self.simplify_chain_structure_product(&product))
478            .or_else(|| Self::simplify_symmetric_structure_product(&product))
479            .or_else(|| self.simplify_two_f_loop_product(&product))
480            .or_else(|| self.simplify_three_f_loop_product(&product))
481            .or_else(|| Self::simplify_symmetric_invariant_product(&product))
482            .or_else(|| self.simplify_embedded_color_node(&product))
483            .or_else(|| {
484                self.settings
485                    .expand_cross_chain_fierz
486                    .then(|| Self::simplify_cross_chain_fierz_product(&product))
487                    .flatten()
488            })
489    }

491    fn join_color_chain_product(product: &ProductView) -> Option<Atom> {
492        for (left_index, left_factor) in product.factors.iter().enumerate() {
493            let Some(left_chain) = &left_factor.chain else {
494                continue;
495            };
496            let Some((left_end_dim, left_end_index, true)) = color_fundamental_slot(left_chain.end)
497            else {
498                continue;
499            };
500
501            for (right_index, right_factor) in product.factors.iter().enumerate() {
502                if right_index == left_index {
503                    continue;
504                }
505                let Some(right_chain) = &right_factor.chain else {
506                    continue;
507                };
508                let Some((right_start_dim, right_start_index, false)) =
509                    color_fundamental_slot(right_chain.start)
510                else {
511                    continue;
512                };
513                if left_end_dim != right_start_dim || left_end_index != right_start_index {
514                    continue;
515                }
516
517                let replacement = chain!(
518                    left_chain.start,
519                    right_chain.end;
520                    left_chain.factors.iter().cloned().chain(right_chain.factors.iter().cloned())
521                );
522                return Some(product.replacing_pair(left_index, right_index, replacement));
523            }
524        }
525
526        None
527    }

529    fn simplify_embedded_color_node(&self, product: &ProductView) -> Option<Atom> {
530        for (index, factor) in product.factors.iter().enumerate() {
531            let rewritten = Self::simplify_chain_node(factor.atom)
532                .or_else(|| {
533                    self.settings
534                        .evaluate_traces
535                        .then(|| Self::simplify_trace_node(factor.atom))
536                        .flatten()
537                })
538                .or_else(|| self.simplify_power(factor.atom));
539            let Some(rewritten) = rewritten else {
540                continue;
541            };
542
543            return Some(product.replacing_one(index, rewritten));
544        }

546        None

549    fn simplify_cross_chain_fierz_product(product: &ProductView) -> Option<Atom> {
550        for (left_index, left_factor) in product.factors.iter().enumerate() {
551            let Some(left_chain) = &left_factor.chain else {
552                continue;
553            };
554            let Some(left_dimension) =
555                fundamental_chain_dimension_view(left_chain.start, left_chain.end)
556            else {
557                continue;
558            };
559
560            for (right_index, right_factor) in
561                product.factors.iter().enumerate().skip(left_index + 1)
562            {
563                let Some(right_chain) = &right_factor.chain else {
564                    continue;
565                };
566                let Some(right_dimension) =
567                    fundamental_chain_dimension_view(right_chain.start, right_chain.end)
568                else {
569                    continue;
570                };
571                if left_dimension != right_dimension {
572                    continue;
573                }
574
575                for (left_generator_index, left_generator) in left_chain.factors.iter().enumerate()
576                {
577                    let Some(left_adjoint) = color_generator_adjoint(*left_generator) else {
578                        continue;
579                    };
580                    for (right_generator_index, right_generator) in
581                        right_chain.factors.iter().enumerate()
582                    {
583                        let Some(right_adjoint) = color_generator_adjoint(*right_generator) else {
584                            continue;
585                        };
586                        if left_adjoint != right_adjoint {
587                            continue;
588                        }
589
590                        let left_before = &left_chain.factors[..left_generator_index];
591                        let left_after = &left_chain.factors[left_generator_index + 1..];
592                        let right_before = &right_chain.factors[..right_generator_index];
593                        let right_after = &right_chain.factors[right_generator_index + 1..];
594
595                        let crossed_left = chain_with_factor_view_slices(
596                            left_chain.start,
597                            right_chain.end,
598                            &[left_before, right_after],
599                        );
600                        let crossed_right = chain_with_factor_view_slices(
601                            right_chain.start,
602                            left_chain.end,
603                            &[right_before, left_after],
604                        );
605                        let uncrossed_left = chain_with_factor_view_slices(
606                            left_chain.start,
607                            left_chain.end,
608                            &[left_before, left_after],
609                        );
610                        let uncrossed_right = chain_with_factor_view_slices(
611                            right_chain.start,
612                            right_chain.end,
613                            &[right_before, right_after],
614                        );
615
616                        let dimension = left_dimension.to_owned();
617                        let replacement = fundamental_index(dimension.clone())
618                            * (crossed_left * crossed_right
619                                - uncrossed_left * uncrossed_right / dimension);
620
621                        return Some(product.replacing_pair(left_index, right_index, replacement));
622                    }
623                }
624            }
625        }
626
627        None
628    }

630    fn simplify_power(&self, power: AtomView) -> Option<Atom> {
631        let AtomView::Pow(pow) = power else {
632            return None;
633        };
634        let (base, exponent) = pow.get_base_exp();
635        if positive_integer(exponent)? != 2 {
636            return None;
637        }
638
639        if let Some(invariant) = color_symmetric_invariant(base)
640            && invariant.args.len() >= 3
641        {
642            return Some(color_symmetric_product(
643                invariant.args.len(),
644                invariant.rep.clone(),
645                invariant.rep,
646            ));
647        }
648
649        let args = structure_constant_args(base)?;
650        let dimension = color_structure_dimension(&args)?;
651        Some(adjoint_casimir_for_dimension(dimension.clone()) * dimension)
652    }

654    fn simplify_trace_structure_product(&self, product: &ProductView) -> Option<Atom> {
655        for (trace_index, trace_factor) in product.factors.iter().enumerate() {
656            let Some(trace) = &trace_factor.trace else {
657                continue;
658            };
659            if trace.factors.len() < 2 {
660                continue;
661            }
662
663            for (f_index, f_factor) in product.factors.iter().enumerate() {
664                if f_index == trace_index {
665                    continue;
666                }
667                let Some(structure) = &f_factor.structure else {
668                    continue;
669                };
670
671                for pair_index in 0..trace.factors.len() {
672                    let first = trace.factors[pair_index];
673                    let second = trace.factors[(pair_index + 1) % trace.factors.len()];
674                    let Some(a) = color_generator_adjoint_view(first) else {
675                        continue;
676                    };
677                    let Some(b) = color_generator_adjoint_view(second) else {
678                        continue;
679                    };
680                    let Some((target, structure_prefactor)) =
681                        Self::structure_target_for_generator_pair(&structure.args, &a, &b)
682                    else {
683                        continue;
684                    };
685
686                    let rest = (0..trace.factors.len() - 2).map(|offset| {
687                        trace.factors[(pair_index + 2 + offset) % trace.factors.len()].to_owned()
688                    });
689                    let replacement_factors = std::iter::once(color_t!(target.clone()))
690                        .chain(rest)
691                        .collect::<Vec<_>>();
692
693                    // f^{abx} Tr(T^a T^b rest) -> i CA/2 Tr(T^x rest).
694                    let replacement = structure_prefactor
695                        * Atom::i()
696                        * adjoint_casimir_for_dimension(
697                            color_adjoint_dimension(&target)
698                                .unwrap_or_else(default_adjoint_dimension),
699                        )
700                        / Atom::num(2)
701                        * trace_with_factors(trace.rep.to_owned(), replacement_factors);
702                    return Some(product.replacing_pair(trace_index, f_index, replacement));
703                }
704            }
705        }
706
707        None
708    }

710    fn simplify_chain_structure_product(&self, product: &ProductView) -> Option<Atom> {
711        for (chain_index, chain_factor) in product.factors.iter().enumerate() {
712            let Some(chain) = &chain_factor.chain else {
713                continue;
714            };
715
716            for pair_index in 0..chain.factors.len().saturating_sub(1) {
717                let Some(left) = color_generator_adjoint_view(chain.factors[pair_index]) else {
718                    continue;
719                };
720                let Some(right) = color_generator_adjoint_view(chain.factors[pair_index + 1])
721                else {
722                    continue;
723                };
724
725                for (f_index, f_factor) in product.factors.iter().enumerate() {
726                    if f_index == chain_index {
727                        continue;
728                    }
729                    let Some(structure) = &f_factor.structure else {
730                        continue;
731                    };
732                    let Some((target, structure_prefactor)) =
733                        Self::structure_target_for_generator_pair(&structure.args, &left, &right)
734                    else {
735                        continue;
736                    };
737
738                    let coefficient = structure_prefactor
739                        * Atom::i()
740                        * adjoint_casimir_for_dimension(
741                            color_adjoint_dimension(&target)
742                                .unwrap_or_else(default_adjoint_dimension),
743                        )
744                        / Atom::num(2);
745                    let chain_factors = chain
746                        .factors
747                        .iter()
748                        .map(|factor| factor.to_owned())
749                        .collect::<Vec<_>>();
750                    let replacement = coefficient
751                        * chain_replacing_factor_pair(
752                            &chain.start.to_owned(),
753                            &chain.end.to_owned(),
754                            &chain_factors,
755                            pair_index,
756                            color_t!(target.clone()),
757                        );
758                    return Some(product.replacing_pair(chain_index, f_index, replacement));
759                }
760            }
761        }
762
763        None
764    }

766    fn structure_target_for_generator_pair(
767        args: &[AtomView<'_>; 3],
768        left: &AtomView<'_>,
769        right: &AtomView<'_>,
770    ) -> Option<(Atom, Atom)> {
771        let actual = color_f!(args[0].to_owned(), args[1].to_owned(), args[2].to_owned());
772
773        args.iter().find_map(|target| {
774            let oriented = color_f!(target.to_owned(), left.to_owned(), right.to_owned());
775            let prefactor = oriented.coefficient(actual.as_view());
776            (!prefactor.is_zero()).then_some((target.to_owned(), prefactor))
777        })
778    }

835    fn simplify_three_f_loop_product(&self, product: &ProductView) -> Option<Atom> {
836        for indices in (0..product.factors.len()).combinations(3) {
837            let Some(f_args) = indices
838                .iter()
839                .map(|index| {
840                    product.factors[*index]
841                        .structure
842                        .as_ref()
843                        .map(|structure| structure.args.map(|arg| arg.to_owned()))
844                })
845                .collect::<Option<Vec<_>>>()
846            else {
847                continue;
848            };
849
850            let common01 = common_structure_positions(&f_args[0], &f_args[1]);
851            let common12 = common_structure_positions(&f_args[1], &f_args[2]);
852            let common20 = common_structure_positions(&f_args[2], &f_args[0]);
853            let ([edge01], [edge12], [edge20]) = (
854                common01.as_slice(),
855                common12.as_slice(),
856                common20.as_slice(),
857            ) else {
858                continue;
859            };
860            if edge01.left == edge20.right
861                || edge01.right == edge12.left
862                || edge12.right == edge20.left
863            {
864                continue;
865            }
866
867            let Some(external0) =
868                (0..3).find(|position| *position != edge01.left && *position != edge20.right)
869            else {
870                continue;
871            };
872            let Some(external1) =
873                (0..3).find(|position| *position != edge01.right && *position != edge12.left)
874            else {
875                continue;
876            };
877            let Some(external2) =
878                (0..3).find(|position| *position != edge12.right && *position != edge20.left)
879            else {
880                continue;
881            };
882
883            let internal01 = f_args[0][edge01.left].clone();
884            let internal12 = f_args[1][edge12.left].clone();
885            let internal20 = f_args[2][edge20.left].clone();
886            let externals = [
887                f_args[0][external0].clone(),
888                f_args[1][external1].clone(),
889                f_args[2][external2].clone(),
890            ];
891            let actual = f_args
892                .iter()
893                .map(|args| color_f!(args[0].clone(), args[1].clone(), args[2].clone()))
894                .collect::<Vec<_>>();
895            let oriented = [
896                color_f!(internal20.clone(), internal01.clone(), externals[0].clone()),
897                color_f!(internal01, internal12.clone(), externals[1].clone()),
898                color_f!(internal12, internal20, externals[2].clone()),
899            ];
900            // Orient all three structures around the cycle; their coefficient
901            // product carries the permutation parity into the reduced f.
902            let prefactor = oriented
903                .iter()
904                .zip(actual)
905                .fold(Atom::num(1), |prefactor, (oriented, actual)| {
906                    prefactor * oriented.coefficient(actual.as_view())
907                });
908            if prefactor.is_zero() {
909                continue;
910            }
911
912            let replacement = prefactor
913                * adjoint_casimir_for_dimension(color_structure_dimension(&f_args[0])?)
914                / Atom::num(2)
915                * color_f!(
916                    externals[0].clone(),
917                    externals[1].clone(),
918                    externals[2].clone()
919                );
920            let mut excluded = vec![false; product.factors.len()];
921            for index in indices {
922                excluded[index] = true;
923            }
924            return Some(product.excluding(&excluded) * replacement);
925        }
926
927        None
928    }

930    fn simplify_symmetric_invariant_product(product: &ProductView) -> Option<Atom> {
931        for (left_index, left_factor) in product.factors.iter().enumerate() {
932            let Some(left) = &left_factor.symmetric_invariant else {
933                continue;
934            };
935            for (right_index, right_factor) in
936                product.factors.iter().enumerate().skip(left_index + 1)
937            {
938                let Some(right) = &right_factor.symmetric_invariant else {
939                    continue;
940                };
941                if left.args.len() != right.args.len() || left.args.len() < 3 {
942                    continue;
943                };
944
945                let left_args = left
946                    .args
947                    .iter()
948                    .map(|arg| arg.to_owned())
949                    .collect::<Vec<_>>();
950                let right_args = right
951                    .args
952                    .iter()
953                    .map(|arg| arg.to_owned())
954                    .collect::<Vec<_>>();
955                // Contract equal-rank symmetric traces into the corresponding
956                // scalar invariant family, leaving a metric for one open pair.
957                let (common, left_open, right_open) =
958                    symmetric_common_and_open_args(&left_args, &right_args);
959                if common.len() == left_args.len() {
960                    let replacement = color_symmetric_product(
961                        left_args.len(),
962                        left.rep.to_owned(),
963                        right.rep.to_owned(),
964                    );
965                    return Some(product.replacing_pair(left_index, right_index, replacement));
966                }
967                let ([left_open], [right_open]) = (left_open.as_slice(), right_open.as_slice())
968                else {
969                    continue;
970                };
971                let dimension = color_adjoint_dimension(left_open)
972                    .or_else(|| color_adjoint_dimension(right_open))
973                    .or_else(|| common.iter().find_map(color_adjoint_dimension))?;
974                let replacement = color_symmetric_product(
975                    left_args.len(),
976                    left.rep.to_owned(),
977                    right.rep.to_owned(),
978                ) * color_metric(left_open.clone(), right_open.clone())
979                    / dimension;
980                return Some(product.replacing_pair(left_index, right_index, replacement));
981            }
982        }
983
984        None
985    }
986}

1117impl<'a> ProductView<'a> {
1118    fn parse(expr: AtomView<'a>) -> Self {
1119        Self {
1120            factors: multiplicative_factor_views(expr)
1121                .into_iter()
1122                .map(ProductFactor::parse)
1123                .collect(),
1124        }
1125    }

1127    fn len(&self) -> usize {
1128        self.factors.len()
1129    }

1132        self.factors
1133            .iter()
1134            .enumerate()
1135            .filter(|(index, _)| *index != left_index && *index != right_index)
1136            .fold(replacement, |product, (_, factor)| {
1137                product * factor.atom.to_owned()
1138            })
1139    }

1142        self.factors
1143            .iter()
1144            .enumerate()
1145            .fold(Atom::num(1), |product, (index, factor)| {
1146                if index == target_index {
1147                    product * replacement.clone()
1148                } else {
1149                    product * factor.atom.to_owned()
1150                }
1151            })
1152    }

1155        self.factors
1156            .iter()
1157            .enumerate()
1158            .filter(|(index, _)| !excluded[*index])
1159            .fold(Atom::num(1), |product, (_, factor)| {
1160                product * factor.atom.to_owned()
1161            })
1162    }
1163
1164    fn distribute_color_sum_factor(&self) -> Option<Atom> {
1165        let (sum_index, sum) = self
1166            .factors
1167            .iter()
1168            .enumerate()
1169            .find_map(|(index, factor)| match factor.atom {
1170                AtomView::Add(add) if add.iter().any(atom_contains_color_node) => {
1171                    Some((index, add))
1172                }
1173                _ => None,
1174            })?;
1175
1176        Some(
1177            sum.iter()
1178                .map(|term| self.replacing_one(sum_index, term.to_owned()))
1179                .fold(Atom::Zero, |sum, term| sum + term),
1180        )
1181    }
1182}

1308fn projector_factor(factor: AtomView) -> Option<(Atom, Symbol, Vec<Atom>)> {
1309    if let Some((projector_symbol, factors)) = projector_parts(factor) {
1310        return Some((Atom::num(1), projector_symbol, factors));
1311    }
1312
1313    let AtomView::Mul(product) = factor else {
1314        return None;
1315    };
1316
1317    let mut prefactor = Atom::num(1);
1318    let mut projector = None;
1319    for factor in product.iter() {
1320        if let Some(parts) = projector_parts(factor) {
1321            if projector.is_some() {
1322                return None;
1323            }
1324            projector = Some(parts);
1325        } else if matches!(factor, AtomView::Num(_)) {
1326            prefactor *= factor.to_owned();
1327        } else {
1328            return None;
1329        }
1330    }
1331
1332    let (projector_symbol, factors) = projector?;
1333    Some((prefactor, projector_symbol, factors))
1334}

1524fn symmetric_common_and_open_args(
1525    left: &[Atom],
1526    right: &[Atom],
1527) -> (Vec<Atom>, Vec<Atom>, Vec<Atom>) {
1528    let mut right_used = vec![false; right.len()];
1529    let mut common = Vec::new();
1530    let mut left_open = Vec::new();
1531
1532    for left_arg in left {
1533        if let Some((right_index, _)) = right
1534            .iter()
1535            .enumerate()
1536            .find(|(index, right_arg)| !right_used[*index] && *right_arg == left_arg)
1537        {
1538            right_used[right_index] = true;
1539            common.push(left_arg.clone());
1540        } else {
1541            left_open.push(left_arg.clone());
1542        }
1543    }
1544
1545    let right_open = right
1546        .iter()
1547        .enumerate()
1548        .filter(|(index, _)| !right_used[*index])
1549        .map(|(_, right_arg)| right_arg.clone())
1550        .collect();
1551
1552    (common, left_open, right_open)
1553}

1635fn atom_contains_color_node(expr: AtomView<'_>) -> bool {
1636    match expr {
1637        AtomView::Fun(f) => {
1638            let symbol = f.get_symbol();
1639            symbol == CS.f
1640                || symbol == CS.d
1641                || symbol == CS.t
1642                || symbol == CS.gram
1643                || symbol == CS.cas
1644                || symbol == CS.idx
1645                || symbol == T.chain
1646                || shadowing::trace_parts(f).is_some()
1647                || f.iter().any(atom_contains_color_node)
1648        }
1649        AtomView::Add(add) => add.iter().any(atom_contains_color_node),
1650        AtomView::Mul(mul) => mul.iter().any(atom_contains_color_node),
1651        AtomView::Pow(pow) => {

1653            atom_contains_color_node(base) || atom_contains_color_node(exponent)
1654        }
1655        AtomView::Num(_) | AtomView::Var(_) => false,
1656    }

1659fn two_structure_loop_contraction(
1660    left: &[Atom; 3],
1661    right: &[Atom; 3],
1662    adjoint_casimir: Atom,
1663) -> Option<Atom> {
1664    let common = common_structure_positions(left, right);
1665    match common.as_slice() {
1666        [first, second] => {
1667            let left_open = (0..3).find(|index| *index != first.left && *index != second.left)?;
1668            let right_open =
1669                (0..3).find(|index| *index != first.right && *index != second.right)?;
1670            let common = [left[first.left].clone(), left[second.left].clone()];
1671            let left_actual = color_f!(left[0].clone(), left[1].clone(), left[2].clone());
1672            let right_actual = color_f!(right[0].clone(), right[1].clone(), right[2].clone());
1673            let left_oriented = color_f!(
1674                left[left_open].clone(),
1675                common[0].clone(),
1676                common[1].clone()
1677            );
1678            let right_oriented = color_f!(
1679                right[right_open].clone(),
1680                common[0].clone(),
1681                common[1].clone()
1682            );
1683
1684            let left_prefactor = left_oriented.coefficient(left_actual.as_view());
1685            let right_prefactor = right_oriented.coefficient(right_actual.as_view());
1686            if left_prefactor.is_zero() || right_prefactor.is_zero() {
1687                return None;
1688            }
1689            Some(
1690                left_prefactor
1691                    * right_prefactor
1692                    * adjoint_casimir
1693                    * color_metric(left[left_open].clone(), right[right_open].clone()),
1694            )
1695        }
1696        [_, _, _] => Some(adjoint_casimir * color_structure_dimension(left)?),
1697        _ => None,
1698    }
1699}

1702struct CommonStructurePosition {
1703    left: usize,
1704    right: usize,
1705}

1707fn common_structure_positions(left: &[Atom; 3], right: &[Atom; 3]) -> Vec<CommonStructurePosition> {
1708    let mut right_used = [false; 3];
1709    let mut common = Vec::new();
1710
1711    for (left_index, left_arg) in left.iter().enumerate() {
1712        if let Some((right_index, _)) = right
1713            .iter()
1714            .enumerate()
1715            .find(|(right_index, right_arg)| !right_used[*right_index] && *right_arg == left_arg)
1716        {
1717            right_used[right_index] = true;
1718            common.push(CommonStructurePosition {
1719                left: left_index,
1720                right: right_index,
1721            });
1722        }
1723    }
1724
1725    common
1726}
