42pub struct ReferenceCase {
43    pub name: &'static str,
44    pub domain: ReferenceDomain,
45    pub expression: fn() -> Atom,
46    pub simplification: ReferenceSimplification,
47    pub validation: ReferenceValidation,
48}
49
50impl ReferenceCase {
51    pub fn expression(&self) -> Atom {
52        (self.expression)()
53    }
54
55    pub fn simplify(&self, expression: &Atom) -> Atom {
56        match self.simplification {
57            ReferenceSimplification::GammaDefault => expression.simplify_gamma(),
58            ReferenceSimplification::GammaCanonical => {
59                expression.simplify_gamma_with(GammaSimplifySettings::canonical())
60            }
61            ReferenceSimplification::ColorDefault => expression.simplify_color(),
62        }
63    }
64
65    pub fn simplified(&self) -> Atom {
66        let expression = self.expression();
67        self.simplify(&expression)
68    }
69}

107fn color_chain(
108    r: &ConcreteReps,
109    start: symbolica::atom::Symbol,
110    end: symbolica::atom::Symbol,
111    indices: &[symbolica::atom::Symbol],
112) -> Atom {
113    let factors = indices
114        .iter()
115        .copied()
116        .map(|index| color_t(r, index))
117        .collect::<Vec<_>>();
118    chain!(
119        r.cof3.slot::<AbstractIndex, _>(start),
120        r.cof3.dual().slot::<AbstractIndex, _>(end);
121        factors
122    )
123}

455#[cfg(test)]
456mod tests {
457    use insta::assert_snapshot;
458    use symbolica::atom::AtomCore;
459    use symbolica_utils::AtomPrintExt;
460
461    use super::*;
462
463    #[test]
464    fn reference_case_simplification_snapshots() {
465        crate::test_support::test_initialize();
466        let snapshot = reference_cases()
467            .iter()
468            .map(|case| {
469                let validation = match case.validation {
470                    ReferenceValidation::Enabled => "enabled".to_string(),
471                    ReferenceValidation::Blocked(reason) => format!("blocked: {reason}"),
472                };
473                let simplified = case.simplified();
474                let simplified = match case.domain {
475                    ReferenceDomain::Dirac4 => simplified.expand_num(),
476                    ReferenceDomain::ColorSu3 => simplified,
477                };
478                format!(
479                    "{} [{:?}, {:?}, {}]\n{}",
480                    case.name,
481                    case.domain,
482                    case.simplification,
483                    validation,
484                    simplified.to_bare_ordered_string(),
485                )
486            })
487            .collect::<Vec<_>>()
488            .join("\n\n");
489
490        assert_snapshot!("reference_case_simplifications", snapshot);
491    }
