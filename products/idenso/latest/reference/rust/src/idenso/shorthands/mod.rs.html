67impl<'a> UndoShorthands for AtomView<'a> {
68    fn undo_all<Aind: AbsInd + DummyAind + ParseableAind>(
69        &self,
70    ) -> Result<Atom, NetworkToolingError> {
71        let net = self
72            .parse_to_symbolic_net::<Aind>(&ParseSettings {
73                shorthand_parsing: ShorthandParsing::Expand {
74                    schoonschip: SchoonschipExpansionMode::full(),
75                    trace: true,
76                    chain: true,
77                },
78                ..Default::default()
79            })
80            .map_err(|error| NetworkToolingError::Parse {
81                reason: error.to_string(),
82            })?;
83
84        net.simple_execute::<()>()
85    }

188    fn undo_schoonschip_across_chain() {
189        let _ = test_initialize();
190        let expr = chain!(
191            mink!(4, i),
192            mink!(4, j),
193            chain_factor!(
194                schoonschip_only_factor,
195                in,
196                out,
197                (vector!(compact_p, mink!(4)))

201        insta::assert_snapshot!( expr.undo_schoonschip::<AbstractIndex>().unwrap().to_bare_ordered_string(), @"chain(mink(4,i),mink(4,j),schoonschip_only_factor(in,out,mink(4,d_1000000)))*compact_p(mink(4,d_1000000))");
202    }
