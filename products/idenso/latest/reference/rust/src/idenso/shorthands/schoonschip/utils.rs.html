37fn distribute_expanded_left_sum(expanded_left: &Atom, right: &Atom) -> Atom {
38    expanded_left.terms().fold(Atom::Zero, |sum, term| {
39        let term = term.to_owned();
40        sum + &term * right
41    })
42}
43
44fn distribute_expanded_right_sum(left: &Atom, expanded_right: &Atom) -> Atom {
45    expanded_right.terms().fold(Atom::Zero, |sum, term| {
46        let term = term.to_owned();
47        sum + left * &term
48    })
49}

51pub(super) fn distribute_smallest_expanded_sum_side(left: &Atom, right: &Atom) -> Atom {
52    match (is_sum(left), is_sum(right)) {
53        (true, true) => {
54            if expression_size(left) <= expression_size(right) {
55                distribute_expanded_left_sum(&left.collect_tensors(), right)
56            } else {
57                distribute_expanded_right_sum(left, &right.collect_tensors())
58            }
59        }
60        (true, false) => distribute_expanded_left_sum(&left.collect_tensors(), right),
61        (false, true) => distribute_expanded_right_sum(left, &right.collect_tensors()),
62        (false, false) => left * right,
63    }
64}
