14static ASYMMETRIC_SCHOONSCHIP_VECTOR_IN_VECTOR: LazyLock<[Replacement; 1]> = LazyLock::new(|| {
15    let stripped = rep_!(0; W_.d_);
16
17    [
18        //  p(...,q(..,rep)) is asymmetric, so we replace it with a dot product, using a schoonschiped metric:
19        //  p(...,q(..,rep)) => g(p(...,rep), q(..,rep))
20        Replacement::new(
21            rank1_!(0; W_.c___, rank1_!(1; W_.a___, &stripped)).to_pattern(),
22            g!(
23                rank1_!(0; W_.c___, &stripped),
24                rank1_!(1; W_.a___, &stripped),
25            ),
26        ),
27    ]
28});

30static REDUNDANT_METRIC_SCHOONSCHIPS: LazyLock<[Replacement; 4]> = LazyLock::new(|| {
31    let self_dual = self_dual_!(0; W_.d_, W_.i_);
32    let self_dual_stripped = self_dual_!(0; W_.d_);
33    let dualizable = dualizable_!(0; W_.d_, W_.i_);
34    let dualizable_stripped = dualizable_!(0; W_.d_);
35    let dualizable_dual = dualizable_dual_!(0; W_.d_, W_.i_);
36
37    [
38        // g(mu,p(...,rep)) is redundant:
39        // g(mu,p(...,rep)) => p(...,mu)
40        Replacement::new(
41            g!(&self_dual, rank1_!(0; W_.c___, self_dual_stripped)).to_pattern(),
42            rank1_!(0; W_.c___, self_dual.clone()),
43        ),
44        // Same thing but for dualizable
45        Replacement::new(
46            g!(
47                &dualizable,
48                rank1_!(0; W_.c___, dualizable_stripped.clone())
49            )
50            .to_pattern(),
51            rank1_!(0; W_.c___, dualizable),
52        ),
53        // Same thing but for dual dualizable
54        Replacement::new(
55            g!(&dualizable_dual, rank1_!(0; W_.c___, dualizable_stripped)).to_pattern(),
56            rank1_!(0; W_.c___, dualizable_dual),
57        ),
58        // g(mu,p(...)) is also allowed (although a bit weird), here we normalize it to p(...,mu)
59        Replacement::new(
60            g!(&self_dual, rank1_!(0; W_.c___)).to_pattern(),
61            rank1_!(0; W_.c___, self_dual),
62        ),
63    ]
64});

66static METRIC_DOT_PRODUCT: LazyLock<[Replacement; 2]> = LazyLock::new(|| {
67    let self_dual_stripped1 = self_dual_!(0; W_.d_);
68    let self_dual_stripped2 = self_dual_!(0; W_.e_);
69    let dualizable_stripped1 = dualizable_!(0; W_.d_);
70    let dualizable_stripped2 = dualizable_!(0; W_.e_);
71
72    [
73        Replacement::new(
74            g!(
75                rank1_!(0; W_.d___, &self_dual_stripped1),
76                rank1_!(1; W_.c___, &self_dual_stripped2)

78            .to_pattern(),
79            dot!(
80                rank1_!(0; W_.d___, self_dual_stripped1),
81                rank1_!(1; W_.c___, self_dual_stripped2)

86                rank1_!(0; W_.d___, &dualizable_stripped1),
87                rank1_!(1; W_.c___, &dualizable_stripped2)

89            .to_pattern(),

91                rank1_!(0; W_.d___, dualizable_stripped1),
92                rank1_!(1; W_.c___, dualizable_stripped2)

98static VECTOR_POWER_NORMALIZATIONS: LazyLock<[Replacement; 2]> = LazyLock::new(|| {
99    let self_dual = T.self_dual_::<0, _>([W_.d_, W_.i_]);
100    let self_dual_stripped = T.self_dual_::<0, _>([W_.d_]);
101    let self_dual_vector = T.rank1_::<0, _>([&Atom::var(W_.c___), &self_dual]);
102    let self_dual_square = g!(
103        T.rank1_::<0, _>([&Atom::var(W_.c___), &self_dual_stripped]),
104        T.rank1_::<0, _>([&Atom::var(W_.c___), &self_dual_stripped]),
105    );
106
107    [
108        // Normalize even powers of a vector p(...,mu)^2n -> g(p(...,rep),p(...,rep))^(n/2)
109        Replacement::new(
110            self_dual_vector.clone().pow(Atom::var(W_.n_)).to_pattern(),
111            self_dual_square.pow(Atom::var(W_.n_) / 2),
112        )
113        .when(W_.n_.filter(DotNormalizer::even_power)),
114        // Normalize odd powers of a vector p(...,mu)^(2n+1) -> p(...,mu) * g(p(...,rep),p(...,rep))^(n/2)
115        Replacement::new(
116            self_dual_vector.clone().pow(Atom::var(W_.n_)).to_pattern(),
117            self_dual_square.pow((Atom::var(W_.n_) - 1) / 2) * self_dual_vector,
118        )
119        .when(W_.n_.filter(DotNormalizer::odd_power)),
120    ]
121});

123static METRIC_POWER_NORMALIZATIONS: LazyLock<[Replacement; 2]> = LazyLock::new(|| {
124    let self_dual = T.self_dual_::<0, _>([W_.d_, W_.i_]);
125    let self_dual_j = T.self_dual_::<0, _>([W_.d_, W_.j_]);
126    let self_dual_metric = g!(&self_dual, &self_dual_j);
127
128    [
129        // Normalize even powers of a metric g(rep(dim,i),rep(dim,j))^2n -> dim^(n/2)
130        Replacement::new(
131            self_dual_metric.pow(Atom::var(W_.n_)).to_pattern(),
132            Atom::var(W_.d_).pow(Atom::var(W_.n_) / 2),
133        )
134        .when(W_.n_.filter(DotNormalizer::even_power)),
135        // Normalize odd powers of a metric g(rep(dim,i),rep(dim,j))^(2n+1) -> dim^(n/2) * g(rep(dim,i),rep(dim,j))
136        Replacement::new(
137            self_dual_metric.pow(Atom::var(W_.n_)).to_pattern(),
138            Atom::var(W_.d_).pow((Atom::var(W_.n_) - 1) / 2) * &self_dual_metric,
139        )
140        .when(W_.n_.filter(DotNormalizer::odd_power)),
141    ]
142});

144static METRIC_TRACE_NORMALIZATIONS: LazyLock<[Replacement; 2]> = LazyLock::new(|| {
145    let self_dual = T.self_dual_::<0, _>([W_.d_, W_.i_]);
146    let dualizable = T.dualizable_::<0, _>([W_.d_, W_.i_]);
147    let dualizable_dual = T.dualizable_dual_::<0, _>([W_.d_, W_.i_]);
148
149    [
150        // g(i,i) -> d
151        Replacement::new(g!(&self_dual, &self_dual).to_pattern(), Atom::var(W_.d_)),
152        // g(i,dind(i)) -> d
153        Replacement::new(
154            g!(&dualizable, &dualizable_dual).to_pattern(),
155            Atom::var(W_.d_),
156        ),
157    ]
158});

160pub(crate) struct DotNormalizer;
161
162impl DotNormalizer {
163    pub(super) fn run(view: AtomView<'_>) -> Atom {
164        view.to_owned()
165            .replace_multiple(&*ASYMMETRIC_SCHOONSCHIP_VECTOR_IN_VECTOR)
166            .replace_multiple_repeat(&*REDUNDANT_METRIC_SCHOONSCHIPS)
167            .replace_multiple(&*VECTOR_POWER_NORMALIZATIONS)
168            .replace_multiple(&*METRIC_POWER_NORMALIZATIONS)
169            .replace_multiple(&*METRIC_TRACE_NORMALIZATIONS)
170    }
171
172    pub(crate) fn metric_shorthand_to_dot(view: AtomView<'_>) -> Atom {
173        view.to_owned().replace_multiple(&*METRIC_DOT_PRODUCT)
174    }
175
176    fn even_power(exp: AtomView<'_>) -> bool {
177        matches!(i64::try_from(exp), Ok(exp) if exp % 2 == 0)
178    }
179
180    fn odd_power(exp: AtomView<'_>) -> bool {
181        matches!(i64::try_from(exp), Ok(exp) if exp % 2 == 1)
182    }
183}
