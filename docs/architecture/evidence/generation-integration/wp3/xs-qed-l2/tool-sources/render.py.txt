class Parse(HTMLParser):
    def __init__(self):super().__init__(convert_charrefs=True);self.text=[];self.active=[]
    def handle_data(self,text):self.text.append(text)
    def handle_starttag(self,t,a):
        if t=='script' or any(k.startswith('on') for k,v in a):self.active.append(t)
def semantic(path):
    p=Parse();p.feed(Path(path).read_text());text=' '.join(' '.join(p.text).split())
    return EXPECTED in text and not p.active,text
